18pub async fn stake_table_info(
19    l1_url: Url,
20    stake_table_address: Address,
21    l1_block_number: u64,
22) -> Result<Vec<RegisteredValidator<BLSPubKey>>> {
23    let block_range = configured_block_range()?;
24    let l1 = L1Client::new(vec![l1_url])?;
25    let validators =
26        fetch_validators_adaptively(l1, stake_table_address, l1_block_number, block_range).await?;
27
28    Ok(validators
29        .into_iter()
30        .map(|(_address, validator)| validator)
31        .collect())
32}

34/// Fetch the stake table in one request, falling back to the chunked fetcher if that is refused.
35///
36/// Providers pull in opposite directions: the gateways this CLI defaults to serve the whole range
37/// in one request but rate limit the hundreds of requests chunking needs, while others cap the
38/// range and only work chunked. The single request is tried without retrying, so a provider that
39/// refuses it costs one round trip rather than the chunked fetcher's retry budget.
40async fn fetch_validators_adaptively(
41    l1: L1Client,
42    stake_table_address: Address,
43    l1_block_number: u64,
44    block_range: Option<u64>,
45) -> Result<RegisteredValidatorMap> {
46    // A pinned range means the provider caps it, so don't spend a request on a rejection.
47    if block_range.is_none() {
48        match fetch_validators_in_one_request(&l1, stake_table_address, l1_block_number).await {
49            Ok(validators) => return Ok(validators),
50            Err(err) => tracing::warn!(
51                %err,
52                "could not fetch the stake table in one request, retrying in smaller ranges"
53            ),
54        }
55    }
56
57    Ok(
58        Fetcher::fetch_all_validators_from_contract(l1, stake_table_address, l1_block_number)
59            .await?
60            .0,
61    )
62}

64/// The whole stake table in one request. The chunked fetcher reads the initialization block
65/// itself, so failing to read it here is just another reason to fall back to it.
66async fn fetch_validators_in_one_request(
67    l1: &L1Client,
68    stake_table_address: Address,
69    l1_block_number: u64,
70) -> Result<RegisteredValidatorMap> {
71    let stake_table = StakeTableV3::new(stake_table_address, &l1.provider);
72    let from_block = stake_table.initializedAtBlock().call().await?.to::<u64>();
73    let events = Fetcher::try_fetch_events_from_contract(
74        l1.clone(),
75        stake_table_address,
76        from_block,
77        l1_block_number,
78    )
79    .await?;
80    Ok(validators_from_l1_events(events.into_iter().map(|(_, e)| e))?.0)
81}

83pub fn display_stake_table(
84    stake_table: Vec<RegisteredValidator<BLSPubKey>>,
85    compact: bool,
86) -> Result<()> {
87    let mut stake_table = stake_table.clone();
88    stake_table.sort_by_key(|a| a.stake);
89
90    let shorten = |s: String| {
91        if compact {
92            let end = s.chars().map(|c| c.len_utf8()).take(40).sum();
93            format!("{}..", &s[..end])
94        } else {
95            s
96        }
97    };
98    for validator in stake_table.iter() {
99        let comm: Commission = validator.commission.try_into()?;
100        let key_str = shorten(match &validator.stake_table_key {
101            Some(key) => key.to_string(),
102            None => "<no_bls_key>".to_string(),
103        });
104        output_success(format!(
105            "Validator {}: {key_str} comm={comm} stake={} ESP",
106            validator.account,
107            format_ether(validator.stake),
108        ));
109        if let Some(x25519) = &validator.x25519_key {
110            output_success(format!(" - {}", shorten(x25519.to_string())));
111        }
112        if let Some(p2p_addr) = &validator.p2p_addr {
113            output_success(format!(" - p2p_addr={p2p_addr}"));
114        }
115
116        if validator.delegators.is_empty() {
117            output_success(" - No delegators");
118            continue;
119        }
120
121        // sort delegators by address for easier reading
122        let mut delegators = validator.delegators.iter().collect::<Vec<_>>();
123        delegators.sort_by(|a, b| a.0.cmp(b.0));
124        for (delegator, stake) in delegators {
125            output_success(format!(
126                " - Delegator {delegator}: stake={} ESP",
127                format_ether(*stake)
128            ));
129        }
130    }
131    Ok(())
132}

134pub async fn fetch_token_address(rpc_url: Url, stake_table_address: Address) -> Result<Address> {
135    let provider = ProviderBuilder::new().connect_http(rpc_url);
136    StakeTableV3::new(stake_table_address, provider)
137        .token()
138        .call()
139        .await
140        .with_context(|| {
141            format!(
142                "Failed to fetch token address from stake table contract at {stake_table_address}"
143            )
144        })
145}
