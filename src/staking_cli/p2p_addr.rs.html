1//! Validation of the p2p address a validator registers in the stake table.
2
3use hotshot_types::addr::NetAddr;
4use tokio::net::lookup_host;
5
6use crate::output::{output_error, output_warn};
7
8/// The longest address the stake table contract accepts.
9const MAX_ADDR_LEN: usize = 512;
10
11/// Parse a p2p address to register in the stake table.
12///
13/// [`NetAddr::validate`] holds the rules for the address itself; what is added here is
14/// what the stake table asks of one: it fits the contract's length limit, and it names a
15/// port a peer can connect to, which a missing port does not. An address that prints
16/// differently to how it was written is registered in the form it prints in, and the
17/// caller is told so, since that is the string that lands on chain.
18pub fn parse_p2p_addr(s: &str) -> Result<NetAddr, String> {
19    let addr: NetAddr = s.parse().map_err(|e| format!("`{s}`: {e}"))?;
20    addr.validate().map_err(|e| format!("`{s}`: {e}"))?;
21
22    if addr.port() == 0 {
23        return Err(format!("`{s}`: port 0 is not a port a peer can connect to"));
24    }
25
26    let registered = addr.to_string();
27    if registered.len() > MAX_ADDR_LEN {
28        return Err(format!(
29            "`{s}` is {} bytes long, the stake table contract accepts at most {MAX_ADDR_LEN}",
30            registered.len()
31        ));
32    }
33
34    if registered != s {
35        output_warn(format!("`{s}` will be registered as `{registered}`"));
36    }
37
38    Ok(addr)
39}

41pub(crate) async fn check_if_reachable(addr: &NetAddr) {
42    if !addr.is_probably_global() {
43        output_error(format!(
44            "`{addr}` is not publicly routable, other validators will not be able to connect to it"
45        ));
46    }
47
48    if addr.is_ip() {
49        return;
50    }
51
52    let resolved = match lookup_host(addr.to_string()).await {
53        Ok(addrs) => addrs.collect::<Vec<_>>(),
54        Err(err) => {
55            output_error(format!("`{addr}` does not resolve: {err}"));
56        },
57    };
58
59    if resolved.is_empty() {
60        output_error(format!("`{addr}` does not resolve to any address"));
61    }
62
63    if !resolved
64        .iter()
65        .any(|a| NetAddr::from((a.ip().to_canonical(), a.port())).is_probably_global())
66    {
67        output_error(format!(
68            "`{addr}` resolves to addresses that are not publicly routable, other validators will \
69             not be able to connect to it"
70        ))
71    }
72}
