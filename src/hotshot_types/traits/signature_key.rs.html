38/// Trait for abstracting private signature key
39pub trait PrivateSignatureKey:
40    Send + Sync + Sized + Clone + Debug + Eq + Hash + for<'a> TryFrom<&'a TaggedBase64>
41{
42    /// Serialize the private key into bytes
43    fn to_bytes(&self) -> Vec<u8>;
44
45    /// Deserialize the private key from bytes
46    /// # Errors
47    /// If deserialization fails.
48    fn from_bytes(bytes: &[u8]) -> anyhow::Result<Self>;
49
50    /// Serialize the private key into TaggedBase64 blob.
51    /// # Errors
52    /// If serialization fails.
53    fn to_tagged_base64(&self) -> Result<TaggedBase64, Tb64Error>;
54}

56/// Trait for abstracting public key signatures
57/// Self is the public key type
58pub trait SignatureKey:
59    Send
60    + Sync
61    + Clone
62    + Sized
63    + Debug
64    + Hash
65    + Serialize
66    + for<'a> Deserialize<'a>
67    + PartialEq
68    + Eq
69    + PartialOrd
70    + Ord
71    + Display
72    + ToFieldsLightClientCompat
73    + for<'a> TryFrom<&'a TaggedBase64>
74    + Into<TaggedBase64>
75{
76    /// The private key type for this signature algorithm
77    type PrivateKey: PrivateSignatureKey;
78    /// The type of the entry that contain both public key and stake value
79    type StakeTableEntry: StakeTableEntryType<Self>
80        + Send
81        + Sync
82        + Sized
83        + Clone
84        + Debug
85        + Hash
86        + Eq
87        + Serialize
88        + for<'a> Deserialize<'a>;
89    /// The verification key type
90    type VerificationKeyType: Debug
91        + Hash
92        + Clone
93        + Send
94        + Sync
95        + for<'a> Deserialize<'a>
96        + Serialize
97        + PartialEq
98        + Eq;
99    /// The type of the quorum certificate parameters used for assembled signature
100    type QcParams<'a>: Send + Sync + Sized + Clone + Debug + Hash;
101    /// The type of the assembled signature, without `BitVec`
102    type PureAssembledSignatureType: Send
103        + Sync
104        + Sized
105        + Clone
106        + Debug
107        + Hash
108        + PartialEq
109        + Eq
110        + Serialize
111        + for<'a> Deserialize<'a>
112        + Into<TaggedBase64>
113        + for<'a> TryFrom<&'a TaggedBase64>;
114    /// The type of the assembled qc: assembled signature + `BitVec`
115    type QcType: Send
116        + Sync
117        + Sized
118        + Clone
119        + Debug
120        + Hash
121        + PartialEq
122        + Eq
123        + Serialize
124        + for<'a> Deserialize<'a>;
125
126    /// Type of error that can occur when signing data
127    type SignError: std::error::Error + Send + Sync;
128
129    // Signature type represented as a vec/slice of bytes to let the implementer handle the nuances
130    // of serialization, to avoid Cryptographic pitfalls
131    /// Validate a signature
132    fn validate(&self, signature: &Self::PureAssembledSignatureType, data: &[u8]) -> bool;
133
134    /// Produce a signature
135    /// # Errors
136    /// If unable to sign the data with the key
137    fn sign(
138        private_key: &Self::PrivateKey,
139        data: &[u8],
140    ) -> Result<Self::PureAssembledSignatureType, Self::SignError>;
141
142    /// Produce a public key from a private key
143    fn from_private(private_key: &Self::PrivateKey) -> Self;
144    /// Serialize a public key to bytes
145    fn to_bytes(&self) -> Vec<u8>;
146    /// Deserialize a public key from bytes
147    /// # Errors
148    ///
149    /// Will return `Err` if deserialization fails
150    fn from_bytes(bytes: &[u8]) -> Result<Self, SerializationError>;
151
152    /// Generate a new key pair
153    fn generated_from_seed_indexed(seed: [u8; 32], index: u64) -> (Self, Self::PrivateKey);
154
155    /// get the stake table entry from the public key and stake value
156    fn stake_table_entry(&self, stake: U256) -> Self::StakeTableEntry;
157
158    /// only get the public key from the stake table entry
159    fn public_key(entry: &Self::StakeTableEntry) -> Self;
160
161    /// get the public parameter for the assembled signature checking
162    fn public_parameter(
163        stake_entries: &[Self::StakeTableEntry],
164        threshold: U256,
165    ) -> Self::QcParams<'_>;
166
167    /// check the quorum certificate for the assembled signature, returning `Ok(())` if it is valid.
168    ///
169    /// # Errors
170    /// Returns an error if the signature key fails to validate
171    fn check(
172        real_qc_pp: &Self::QcParams<'_>,
173        data: &[u8],
174        qc: &Self::QcType,
175    ) -> Result<(), SignatureError>;
176
177    /// Get the list of signers given a qc.
178    ///
179    /// # Errors
180    /// Returns an error if the inputs mismatch (e.g. wrong verifier parameter or original message).
181    fn signers(
182        real_qc_pp: &Self::QcParams<'_>,
183        qc: &Self::QcType,
184    ) -> Result<Vec<Self::VerificationKeyType>, SignatureError>;
185
186    /// get the assembled signature and the `BitVec` separately from the assembled signature
187    fn sig_proof(signature: &Self::QcType) -> (Self::PureAssembledSignatureType, BitVec);
188
189    /// assemble the signature from the partial signature and the indication of signers in `BitVec`
190    fn assemble(
191        real_qc_pp: &Self::QcParams<'_>,
192        signers: &BitSlice,
193        sigs: &[Self::PureAssembledSignatureType],
194    ) -> Self::QcType;
195
196    /// generates the genesis public key. Meant to be dummy/filler
197    #[must_use]
198    fn genesis_proposer_pk() -> Self;
199
200    /// convert the public key to verification key
201    fn to_verification_key(&self) -> Self::VerificationKeyType;
202}

204/// Builder Signature Key trait with minimal requirements
205pub trait BuilderSignatureKey:
206    Send
207    + Sync
208    + Clone
209    + Sized
210    + Debug
211    + Hash
212    + Serialize
213    + DeserializeOwned
214    + PartialEq
215    + Eq
216    + PartialOrd
217    + Ord
218    + Display
219{
220    /// The type of the keys builder would use to sign its messages
221    type BuilderPrivateKey: PrivateSignatureKey;
222
223    /// The type of the signature builder would use to sign its messages
224    type BuilderSignature: Send
225        + Sync
226        + Sized
227        + Clone
228        + Debug
229        + Eq
230        + Serialize
231        + for<'a> Deserialize<'a>
232        + Hash;
233
234    /// Type of error that can occur when signing data
235    type SignError: std::error::Error + Send + Sync;
236
237    /// validate the message with the builder's public key
238    fn validate_builder_signature(&self, signature: &Self::BuilderSignature, data: &[u8]) -> bool;
239
240    /// validate signature over sequencing fee information
241    /// with the builder's public key
242    fn validate_fee_signature<Metadata: EncodeBytes>(
243        &self,
244        signature: &Self::BuilderSignature,
245        fee_amount: u64,
246        metadata: &Metadata,
247    ) -> bool {
248        self.validate_builder_signature(signature, &aggregate_fee_data(fee_amount, metadata))
249    }
250
251    /// validate signature over sequencing fee information
252    /// with the builder's public key, including vid commitment
253    fn validate_fee_signature_with_vid_commitment<Metadata: EncodeBytes>(
254        &self,
255        signature: &Self::BuilderSignature,
256        fee_amount: u64,
257        metadata: &Metadata,
258        vid_commitment: &VidCommitment,
259    ) -> bool {
260        self.validate_builder_signature(
261            signature,
262            &aggregate_fee_data_with_vid_commitment(fee_amount, metadata, vid_commitment),
263        )
264    }
265
266    /// sign the message with the builder's private key
267    /// # Errors
268    /// If unable to sign the data with the key
269    fn sign_builder_message(
270        private_key: &Self::BuilderPrivateKey,
271        data: &[u8],
272    ) -> Result<Self::BuilderSignature, Self::SignError>;
273
274    /// sign sequencing fee offer
275    /// # Errors
276    /// If unable to sign the data with the key
277    fn sign_fee<Metadata: EncodeBytes>(
278        private_key: &Self::BuilderPrivateKey,
279        fee_amount: u64,
280        metadata: &Metadata,
281    ) -> Result<Self::BuilderSignature, Self::SignError> {
282        Self::sign_builder_message(private_key, &aggregate_fee_data(fee_amount, metadata))
283    }
284
285    /// Generate a new key pair
286    fn generated_from_seed_indexed(seed: [u8; 32], index: u64) -> (Self, Self::BuilderPrivateKey);
287}

289/// Aggregate all inputs used for signature over fee data
290fn aggregate_fee_data<Metadata: EncodeBytes>(fee_amount: u64, metadata: &Metadata) -> Vec<u8> {
291    let mut fee_info = Vec::new();
292
293    fee_info.extend_from_slice(fee_amount.to_be_bytes().as_ref());
294    fee_info.extend_from_slice(metadata.encode().as_ref());
295
296    fee_info
297}
298
299/// Aggregate all inputs used for signature over fee data, including the vid commitment
300fn aggregate_fee_data_with_vid_commitment<Metadata: EncodeBytes>(
301    fee_amount: u64,
302    metadata: &Metadata,
303    vid_commitment: &VidCommitment,
304) -> Vec<u8> {
305    let mut fee_info = Vec::new();
306
307    fee_info.extend_from_slice(fee_amount.to_be_bytes().as_ref());
308    fee_info.extend_from_slice(metadata.encode().as_ref());
309    fee_info.extend_from_slice(vid_commitment.as_ref());
310
311    fee_info
312}
