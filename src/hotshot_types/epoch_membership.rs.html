40/// One in-flight attempt, shared by every caller that asks for the same epoch
41/// while it runs. Whoever polls it drives it, so it outlives the task that
42/// started it as long as somebody is still waiting.
43type Attempt = Shared<BoxFuture<'static, Result<()>>>;

45/// The in-flight DRB computations, each with the token that stops it. An entry
46/// is the claim on its epoch: one computation runs per epoch, and the next may
47/// start only once the entry is gone.
48///
49/// When an external source supplies the DRB result for `epoch` (e.g. a decided
50/// leaf carrying `next_drb_result`), `supply_drb` fires the token so the local
51/// computation can stop early instead of grinding to completion.
52type DrbComputations = Arc<Mutex<HashMap<EpochNumber, CancellationToken>>>;

233    /// Return the union of the stake table and DA committee for `epoch`,
234    /// keyed by signature key. Each entry's `Option<PeerConnectInfo>`
235    /// reflects whether the peer has connection info registered.
236    ///
237    /// For a peer in both, `Some` connect info wins over `None`: the snapshot's DA
238    /// committee may be a stale bootstrap copy lacking connect info while the stake
239    /// table has fresh L1-derived info (e.g. during CLIQUENET cutover), so plain
240    /// last-write-wins would clobber the live data.
241    ///
242    /// Returns `None` if the stake table for `epoch` is unavailable
243    /// (e.g. catchup is still in progress).
244    pub fn epoch_peers(
245        &self,
246        e: Option<EpochNumber>,
247    ) -> Option<HashMap<TYPES::SignatureKey, Option<PeerConnectInfo>>> {
248        let membership = self.stake_table_for_epoch(e).ok()?;
249        let mut out: HashMap<TYPES::SignatureKey, Option<PeerConnectInfo>> = HashMap::new();
250        let mut merge =
251            |key: TYPES::SignatureKey, info: Option<PeerConnectInfo>| match out.entry(key) {
252                Entry::Vacant(slot) => {
253                    slot.insert(info);
254                },
255                Entry::Occupied(mut slot) => {
256                    if slot.get().is_none() && info.is_some() {
257                        slot.insert(info);
258                    }
259                },
260            };
261        if let Some(snap) = membership.snapshot() {
262            for m in snap.stake_table().chain(snap.da_stake_table()) {
263                merge(m.stake_table_entry.public_key(), m.connect_info.clone());
264            }
265        } else {
266            let snap = membership.non_epoch_snapshot()?;
267            for m in snap.stake_table().chain(snap.da_stake_table()) {
268                merge(m.stake_table_entry.public_key(), m.connect_info.clone());
269            }
270        }
271        Some(out)
272    }

274    /// Collect the union of `epoch-1`, `epoch`, and `epoch+1` stake tables
275    /// (each merged with its DA committee) as a flat map of peers to dial.
276    ///
277    /// Newest-wins ordering for `connect_info`: next overrides curr overrides
278    /// prev. Entries with no `connect_info` are filtered out.
279    ///
280    /// Used to seed networks like cliquenet with the same window
281    /// `on_epoch_change` would build for `epoch`.
282    pub fn window_peers(&self, e: EpochNumber) -> HashMap<TYPES::SignatureKey, PeerConnectInfo> {
283        let curr = self.epoch_peers(Some(e)).unwrap_or_default();
284        let prev = if *e > 0 {
285            self.epoch_peers(Some(e - 1)).unwrap_or_default()
286        } else {
287            HashMap::new()
288        };
289        let next = self.epoch_peers(Some(e + 1)).unwrap_or_default();
290
291        // Newest-wins merge: start from prev, overlay curr and next.
292        let mut merged: HashMap<TYPES::SignatureKey, Option<PeerConnectInfo>> = prev;
293        for (k, v) in curr.into_iter().chain(next) {
294            merged.insert(k, v);
295        }
296
297        merged
298            .into_iter()
299            .filter_map(|(k, v)| v.map(|info| (k, info)))
300            .collect()
301    }

303    /// Start a catchup task for `epoch` unless one is already running.
304    ///
305    /// Returns `true` if this call started one.
306    ///
307    /// The work runs in its own task. The task spawned here holds the claim on
308    /// the epoch and reports a failure nobody happened to be waiting for, so
309    /// killing it releases the claim and lets the next caller start again,
310    /// while the work itself runs on until its steps time out.
311    fn spawn_catchup(&self, epoch: EpochNumber) -> bool {
312        if self.stake_table_map.contains_attempt(epoch) {
313            return false;
314        }
315        let (attempt, entry) = self.catchup_map.get_or_insert(epoch, || {
316            let coordinator = self.clone();
317            async move { coordinator.catchup(epoch).await }.boxed()
318        });
319        let Some(entry) = entry else {
320            return false;
321        };
322        spawn(async move {
323            let _entry = entry; // Dropping the entry removes the attempt.
324            if let Err(err) = attempt.await {
325                error!(%epoch, %err, "catchup failed");
326            }
327        });
328        true
329    }

331    /// Bring `epoch` to a usable membership: its stake table, then its DRB.
332    ///
333    /// Walks from the latest consecutive pair of stake tables this node holds
334    /// to `epoch`, fetching what is missing one epoch at a time. `epoch` is
335    /// where the walk stops, not where it starts, so an epoch nobody can serve
336    /// costs one fetch attempt at the first gap rather than a probe of every
337    /// earlier epoch.
338    ///
339    /// Concurrent catchups walk the same epochs and share one attempt per
340    /// epoch, so whichever reaches an epoch first fetches it while the others
341    /// wait for its result. Every walk awaits attempts in increasing epoch
342    /// order and an attempt awaits none, so no two can wait on each other.
343    async fn catchup(&self, epoch: EpochNumber) -> Result<()> {
344        let seeded = self.membership.first_epoch() + 1;
345
346        if epoch <= seeded {
347            return Err(anytrace::error!(
348                "catchup requested for epoch {epoch}, which the membership seeds at construction \
349                 rather than deriving it from an epoch root"
350            ));
351        }
352
353        let known = self.membership.highest_known_epoch().unwrap_or(seeded);
354
355        let mut start = known
356            .min(EpochNumber::new(epoch.saturating_sub(1)))
357            .max(seeded);
358
359        while start > seeded {
360            if self.stake_table_present(start).await? && self.stake_table_present(start - 1).await?
361            {
362                break;
363            }
364            start = start - 1;
365        }
366
367        info!(target: "announce::catchup", %epoch, %start, "catching up to epoch");
368
369        let mut try_epoch = start + 1;
370        while try_epoch <= epoch {
371            let (attempt, _entry) = self.stake_table_map.get_or_insert(try_epoch, || {
372                let coordinator = self.clone();
373                async move {
374                    if coordinator.stake_table_present(try_epoch).await? {
375                        return Ok(());
376                    }
377                    coordinator.fetch_stake_table(try_epoch).await
378                }
379                .boxed()
380            });
381            attempt.await.map_err(|err| {
382                anytrace::error!("catchup for epoch {epoch} failed at epoch {try_epoch}: {err:?}")
383            })?;
384            try_epoch += 1;
385        }
386
387        self.drb_ready(epoch).await
388    }

401    /// Whether `epoch`'s stake table is in memory.
402    ///
403    /// It is read from storage if it is not already in RAM.
404    async fn stake_table_present(&self, epoch: EpochNumber) -> Result<bool> {
405        if self.membership.snapshot(epoch).is_some() {
406            return Ok(true);
407        }
408        debug!(%epoch, "loading stake table from storage");
409        timeout(
410            self.catchup_timeout,
411            self.membership.load_stake_table(epoch),
412        )
413        .await
414        .map_err(|_| {
415            anytrace::error!(
416                "loading the stake table for epoch {epoch} from storage exceeded {:?}",
417                self.catchup_timeout
418            )
419        })
420    }

422    /// Make `epoch`'s DRB result present.
423    ///
424    /// From peers if they have it, by computing it locally if they do not.
425    async fn drb_ready(&self, epoch: EpochNumber) -> Result<()> {
426        if self
427            .membership
428            .snapshot(epoch)
429            .is_some_and(|snapshot| snapshot.has_drb())
430        {
431            return Ok(());
432        }
433        debug!(%epoch, "fetching drb result from peers");
434        let fetched = timeout(self.catchup_timeout, self.get_epoch_drb(epoch))
435            .await
436            .map_err(|_| {
437                anytrace::error!(
438                    "fetching the DRB result for epoch {epoch} from peers exceeded {:?}",
439                    self.catchup_timeout
440                )
441            })?;
442        match fetched {
443            Ok(drb_result) => {
444                info!(
445                    target: "announce::catchup",
446                    %epoch,
447                    ?drb_result,
448                    "drb result retrieved from peers"
449                );
450                self.membership.add_drb_result(epoch, drb_result);
451            },
452            Err(err) => {
453                info!(
454                    target: "announce::catchup",
455                    %epoch,
456                    %err,
457                    "recalculating the drb result the peers lack"
458                );
459                let root_leaf = self.epoch_root_leaf(epoch).await?;
460                self.compute_drb_result(epoch, root_leaf).await?;
461            },
462        }
463
464        if self
465            .membership
466            .snapshot(epoch)
467            .is_some_and(|snapshot| snapshot.has_drb())
468        {
469            Ok(())
470        } else {
471            Err(anytrace::error!(
472                "the drb result for epoch {epoch} was not stored; its stake table is gone"
473            ))
474        }
475    }

490    /// Call this method if you think catchup is in progress for a given epoch
491    /// and you want to wait for it to finish and get the stake table.
492    /// If it's not, it will try to return the stake table if already available.
493    /// Returns an error if the catchup failed or the catchup is not in progress
494    /// and the stake table is not available.
495    pub async fn wait_for_catchup(&self, epoch: EpochNumber) -> Result<EpochMembership<TYPES>> {
496        let failed = |err| anytrace::error!("Catchup for epoch {epoch} failed: {err:?}");
497        if let Some(attempt) = self.catchup_map.get_attempt(epoch) {
498            attempt.await.map_err(failed)?;
499        } else if let Some(attempt) = self.stake_table_map.get_attempt(epoch) {
500            attempt.await.map_err(failed)?;
501        } else if let Some(membership) = self.epoch_membership(epoch) {
502            return Ok(membership);
503        } else {
504            return Err(anytrace::error!(
505                "No catchup in progress for epoch {epoch} and we don't have a stake table for it"
506            ));
507        }
508        self.epoch_membership(epoch).ok_or_else(|| {
509            anytrace::error!("Catchup for epoch {epoch} reported success but left no stake table")
510        })
511    }

513    /// The epoch root leaf `epoch`'s stake table is derived from.
514    ///
515    /// Fetched from peers and verified against the stake table of `epoch - 2`,
516    /// which must already be present. Also seeds the local DRB computation, so
517    /// it is fetched on that path too, without a stake table to install.
518    async fn epoch_root_leaf(&self, epoch: EpochNumber) -> Result<Leaf2<TYPES>> {
519        let root_epoch = EpochNumber::new(epoch.saturating_sub(2));
520        // Snapshot eviction can drop the root epoch from memory while a catchup
521        // is in progress, so load it back from local storage. Deliberately not
522        // `stake_table_for_epoch`, which on a miss claims the root epoch and
523        // spawns a nested catchup for it: that is how one step turns into a
524        // cascade.
525        self.stake_table_present(root_epoch).await?;
526        let Some(root_membership) = self.epoch_membership(root_epoch) else {
527            return Err(anytrace::error!(
528                "the stake table for epoch {epoch} is derived from epoch {root_epoch}, which is \
529                 in neither memory nor local storage"
530            ));
531        };
532
533        // Verification of the root is handled in get_epoch_root_and_drb
534        debug!(%epoch, %root_epoch, "fetching epoch root");
535        let fetched = timeout(self.catchup_timeout, root_membership.get_epoch_root())
536            .await
537            .map_err(|_| {
538                anytrace::error!(
539                    "fetching the epoch root of epoch {epoch} exceeded {:?}",
540                    self.catchup_timeout
541                )
542            })?;
543        fetched.map_err(|err| {
544            anytrace::error!("get epoch root leaf failed for epoch {root_epoch:?}: {err:#}")
545        })
546    }

569    /// Compute `epoch`'s DRB result locally, claiming the epoch for the
570    /// duration so that only one computation runs per epoch.
571    ///
572    /// Bounded by progress rather than by time: the hash chain is
573    /// purposefully long, so it is given as long as it keeps hashing and
574    /// abandoned once it stops, which a chain queued behind a saturated
575    /// blocking pool does.
576    pub async fn compute_drb_result(
577        &self,
578        epoch: EpochNumber,
579        root_leaf: Leaf2<TYPES>,
580    ) -> Result<DrbResult> {
581        let cancel_token = {
582            let mut computations = self.drb_computations.lock();
583            if computations.contains_key(&epoch) {
584                return Err(anytrace::debug!(
585                    "DRB calculation for epoch {epoch} already in progress"
586                ));
587            }
588            let token = CancellationToken::new();
589            computations.insert(epoch, token.clone());
590            token
591        };
592        // The single owner of the bookkeeping inserted above: cleared on drop.
593        let _drb_state = DrbStateGuard {
594            coordinator: self.clone(),
595            epoch,
596        };
597
598        let Ok(drb_seed_input_vec) = bincode::serialize(&root_leaf.justify_qc().signatures) else {
599            return Err(anytrace::error!(
600                "Failed to serialize the QC signature for leaf {root_leaf:?}"
601            ));
602        };
603
604        let Some(drb_difficulty_selector) = self.drb_difficulty_selector.read().clone() else {
605            return Err(anytrace::error!(
606                "The DRB difficulty selector is missing from the epoch membership coordinator. \
607                 This node will not be able to spawn any DRB calculation tasks from catchup."
608            ));
609        };
610
611        let drb_difficulty = drb_difficulty_selector(root_leaf.block_header().version()).await;
612
613        let mut drb_seed_input = [0u8; 32];
614
615        if root_leaf.block_header().version() >= DRB_FIX_VERSION {
616            drb_seed_input = Sha256::digest(&drb_seed_input_vec).into();
617        } else {
618            let len = drb_seed_input_vec.len().min(32);
619            drb_seed_input[..len].copy_from_slice(&drb_seed_input_vec[..len]);
620        }
621
622        let drb_input = DrbInput {
623            epoch: *epoch,
624            iteration: 0,
625            value: drb_seed_input,
626            difficulty_level: drb_difficulty,
627        };
628
629        let store_drb_progress_fn = self.store_drb_progress_fn.clone();
630        let load_drb_progress_fn = self.load_drb_progress_fn.clone();
631
632        let heartbeat = Heartbeat::default();
633        let Some(computed) = heartbeat
634            .while_alive(
635                self.catchup_timeout,
636                compute_drb_result(
637                    drb_input,
638                    store_drb_progress_fn,
639                    load_drb_progress_fn,
640                    DRB_PROGRESS_LOAD_TIMEOUT,
641                    heartbeat.clone(),
642                    cancel_token,
643                ),
644            )
645            .await
646        else {
647            return Err(anytrace::error!(
648                "the DRB computation for epoch {epoch} reported no progress for {:?}",
649                self.catchup_timeout
650            ));
651        };
652        let drb = match computed {
653            Some(drb) => drb,
654            None => {
655                info!(
656                    %epoch,
657                    "the local drb computation was cancelled; fetching the externally supplied \
658                     result"
659                );
660                let fetched = timeout(self.catchup_timeout, self.get_epoch_drb(epoch))
661                    .await
662                    .map_err(|_| {
663                        anytrace::error!(
664                            "fetching the DRB result for epoch {epoch} from peers after \
665                             cancellation exceeded {:?}",
666                            self.catchup_timeout
667                        )
668                    })?;
669                return fetched.map_err(|e| {
670                    anytrace::error!(
671                        "DRB calculation for epoch {epoch} was cancelled but no externally \
672                         supplied result is available: {e}"
673                    )
674                });
675            },
676        };
677
678        // Publish the result in memory before persisting it, mirroring
679        // `supply_drb`.
680        self.membership.add_drb_result(epoch, drb);
681
682        // The result is already published in memory, so the epoch resolves
683        // whether or not the write gets through.
684        info!(%epoch, ?drb, "writing the computed drb result to storage");
685        match timeout(self.catchup_timeout, (self.store_drb_result_fn)(epoch, drb)).await {
686            Ok(Err(err)) => warn!(%epoch, %err, "failed to store the drb result"),
687            Ok(Ok(())) => {},
688            Err(_) => warn!(
689                %epoch,
690                timeout = ?self.catchup_timeout,
691                "storing the drb result timed out"
692            ),
693        }
694
695        Ok(drb)
696    }

698    /// Supply a DRB result obtained from an external source (e.g. a decided
699    /// leaf carrying `next_drb_result`). Adds the result to membership,
700    /// persists it to storage, and cancels any in-flight local computation
701    /// for `epoch`.
702    ///
703    /// If the stake table for `epoch` has not yet been loaded (e.g. the async
704    /// catchup that registers it is still in flight), this logs an error and
705    /// returns; the in-flight catchup will compute the DRB itself once it
706    /// completes.
707    pub fn supply_drb(&self, epoch: EpochNumber, drb: DrbResult) {
708        if self.membership.snapshot(epoch).is_none() {
709            error!(
710                %epoch,
711                "supply_drb called before the stake table is loaded; dropping the externally \
712                 supplied drb and relying on the in-flight catchup"
713            );
714            return;
715        }
716        self.membership.add_drb_result(epoch, drb);
717        // Left in place: the claim belongs to the computation, which clears it
718        // on its way out.
719        let maybe_token = self.drb_computations.lock().get(&epoch).cloned();
720        if let Some(token) = maybe_token {
721            token.cancel();
722        }
723        let store_drb_result_fn = self.store_drb_result_fn.clone();
724        spawn(async move {
725            info!(%epoch, ?drb, "writing the supplied drb result to storage");
726            if let Err(e) = store_drb_result_fn(epoch, drb).await {
727                warn!(%epoch, %e, "failed to store the supplied drb result");
728            }
729        });
730    }
731
732    /// The cancel token of the in-flight DRB computation for `epoch`, if one
733    /// is running. Test observability: lets a test assert that dropping a
734    /// computation's future fires its token.
735    pub fn drb_cancel_token(&self, epoch: EpochNumber) -> Option<CancellationToken> {
736        self.drb_computations.lock().get(&epoch).cloned()
737    }
738
739    /// Cancel all in-flight DRB calculations (e.g. on shutdown).
740    pub fn cancel_all_drb(&self) {
741        // Claims are left for the computations to release as they unwind, so
742        // that none of them is replaced by a fresh one on the way out.
743        let tokens: Vec<_> = self.drb_computations.lock().values().cloned().collect();
744        for token in tokens {
745            token.cancel();
746        }
747    }
748}

750/// Owns one computation's entry in `drb_computations`. Dropping it releases
751/// the claim and fires the token it held.
752struct DrbStateGuard<TYPES: NodeType> {
753    coordinator: EpochMembershipCoordinator<TYPES>,
754    epoch: EpochNumber,
755}
756
757impl<TYPES: NodeType> Drop for DrbStateGuard<TYPES> {
758    fn drop(&mut self) {
759        let token = self.coordinator.drb_computations.lock().remove(&self.epoch);
760        if let Some(token) = token {
761            token.cancel();
762        }
763    }
764}

769/// The in-flight attempts of one kind of work, keyed by the epoch they serve.
770#[derive(Clone, Default)]
771struct Attempts {
772    map: Arc<Mutex<HashMap<EpochNumber, Attempt>>>,
773}
774
775impl Attempts {
776    fn get_or_insert(
777        &self,
778        epoch: EpochNumber,
779        action: impl FnOnce() -> BoxFuture<'static, Result<()>>,
780    ) -> (Attempt, Option<AttemptEntry>) {
781        let mut map = self.map.lock();
782        if let Some(running) = map.get(&epoch) {
783            return (running.clone(), None);
784        }
785        // The work runs in its own task so that a panic in it is contained by
786        // the runtime: a `Shared` whose future panics poisons itself, and every
787        // task that later polls a handle panics too, so one panicking step
788        // would take down every caller waiting on the epoch. What the `Shared`
789        // holds is the join, which cannot panic.
790        //
791        // TODO: Remove explicit panic calls in drb.rs and committee.rs.
792        let task = spawn(action());
793        let attempt = async move {
794            task.await.unwrap_or_else(|err| {
795                Err(anytrace::error!(
796                    "the attempt for epoch {epoch} died: {err}"
797                ))
798            })
799        }
800        .boxed()
801        .shared();
802        map.insert(epoch, attempt.clone());
803        (
804            attempt,
805            Some(AttemptEntry {
806                attempts: self.clone(),
807                epoch,
808            }),
809        )
810    }
811
812    fn contains_attempt(&self, epoch: EpochNumber) -> bool {
813        self.map.lock().contains_key(&epoch)
814    }
815
816    fn get_attempt(&self, epoch: EpochNumber) -> Option<Attempt> {
817        self.map.lock().get(&epoch).cloned()
818    }
819}
820
821/// Holds an epoch's place in an [`Attempts`] map for as long as its owner is driving it.
822struct AttemptEntry {
823    attempts: Attempts,
824    epoch: EpochNumber,
825}
826
827impl Drop for AttemptEntry {
828    fn drop(&mut self) {
829        self.attempts.map.lock().remove(&self.epoch);
830    }
831}

833/// Wrapper around a membership that holds a captured snapshot for a given
834/// epoch (or the pre-epoch state). All accessors observe one consistent
835/// view because the snapshot is held inline.
836pub struct EpochMembership<TYPES: NodeType> {
837    /// The captured snapshot, either per-epoch or pre-epoch.
838    snapshot: EpochMembershipSnapshot<TYPES>,
839    /// Underlying coordinator, retained so navigation methods like
840    /// `next_epoch` can construct fresh snapshots.
841    pub coordinator: EpochMembershipCoordinator<TYPES>,
842}
843
844enum EpochMembershipSnapshot<TYPES: NodeType> {
845    Epoch {
846        epoch: EpochNumber,
847        snapshot: <TYPES::Membership as Membership<TYPES>>::Snapshot,
848    },
849    NonEpoch(<TYPES::Membership as Membership<TYPES>>::NonEpochSnapshot),
850}
851
852impl<TYPES: NodeType> Clone for EpochMembershipSnapshot<TYPES> {
853    fn clone(&self) -> Self {
854        match self {
855            Self::Epoch { epoch, snapshot } => Self::Epoch {
856                epoch: *epoch,
857                snapshot: snapshot.clone(),
858            },
859            Self::NonEpoch(s) => Self::NonEpoch(s.clone()),
860        }
861    }
862}
863
864impl<TYPES: NodeType> Clone for EpochMembership<TYPES> {
865    fn clone(&self) -> Self {
866        Self {
867            coordinator: self.coordinator.clone(),
868            snapshot: self.snapshot.clone(),
869        }
870    }
871}
