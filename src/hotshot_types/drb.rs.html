45pub type DrbDifficultySelectorFn =
46    Arc<dyn Fn(Version) -> BoxFuture<'static, u64> + Send + Sync + 'static>;

48pub fn drb_difficulty_selector<TYPES: NodeType>(
49    config: &HotShotConfig<TYPES>,
50) -> DrbDifficultySelectorFn {
51    let base_difficulty = config.drb_difficulty;
52    let upgrade_difficulty = config.drb_upgrade_difficulty;
53    Arc::new(move |version| {
54        Box::pin(async move {
55            if version >= DRB_AND_HEADER_UPGRADE_VERSION {
56                upgrade_difficulty
57            } else {
58                base_difficulty
59            }
60        })
61    })
62}

109/// Hash `hash` repeatedly `count` times, checking `cancel` between batches.
110///
111/// Returns `None` if cancelled, `Some(hash)` otherwise.
112fn hash_batches(mut hash: [u8; 32], count: u64, cancel: CancellationToken) -> Option<[u8; 32]> {
113    let mut done = 0u64;
114    while done < count {
115        if cancel.is_cancelled() {
116            return None;
117        }
118        let n = DRB_CANCEL_BATCH.min(count - done);
119        for _ in 0..n {
120            hash = Sha256::digest(hash).into();
121        }
122        done += n;
123    }
124    Some(hash)
125}

127/// Raises a flag for as long as it is alive.
128struct HashingGuard(Option<Arc<AtomicBool>>);
129
130impl HashingGuard {
131    fn new(flag: Option<Arc<AtomicBool>>) -> Self {
132        if let Some(flag) = &flag {
133            flag.store(true, Ordering::Release);
134        }
135        Self(flag)
136    }
137}
138
139impl Drop for HashingGuard {
140    fn drop(&mut self) {
141        if let Some(flag) = &self.0 {
142            flag.store(false, Ordering::Release);
143        }
144    }
145}

147/// Default upper bound on loading previously stored DRB progress at the
148/// start of a computation. Sized like the stake-table storage bound
149/// (espresso's `DEFAULT_STORAGE_READ_TIMEOUT`).
150pub const DRB_PROGRESS_LOAD_TIMEOUT: Duration = Duration::from_secs(60);
151
152/// Compute the DRB result for the leader rotation.
153///
154/// This is to be started two epochs in advance and spawned in a non-blocking thread.
155/// Returns `None` if the computation was cancelled via `cancel`.
156///
157/// # Arguments
158/// * `drb_seed_input` - Serialized QC signature.
159/// * `progress_load_timeout` - Bound on the initial stored-progress load
160///   (see [`DRB_PROGRESS_LOAD_TIMEOUT`]).
161/// * `hashing` - Raised while the hash chain itself is running — after the
162///   initial progress load, until return.
163/// * `cancel` - Token that stops the hash loop when fired.
164#[must_use]
165pub async fn compute_drb_result(
166    drb_input: DrbInput,
167    store_drb_progress: StoreDrbProgressFn,
168    load_drb_progress: LoadDrbProgressFn,
169    progress_load_timeout: Duration,
170    hashing: Option<Arc<AtomicBool>>,
171    cancel: CancellationToken,
172) -> Option<DrbResult> {
173    info!(target: "announce::drb", ?drb_input, "beginning drb calculation");
174    let mut drb_input = drb_input;
175
176    match tokio::time::timeout(progress_load_timeout, load_drb_progress(drb_input.epoch)).await {
177        Ok(Ok(loaded_drb_input)) => {
178            if loaded_drb_input.difficulty_level != drb_input.difficulty_level {
179                error!(
180                    ?drb_input,
181                    ?loaded_drb_input,
182                    "we are calculating the drb result with input that has a different difficulty \
183                     level for this epoch than a previously stored one => discarding the value \
184                     from storage"
185                );
186            } else if loaded_drb_input.iteration >= drb_input.iteration {
187                drb_input = loaded_drb_input;
188            }
189        },
190        Ok(Err(e)) => {
191            warn!(
192                epoch = drb_input.epoch,
193                "loading stored drb progress failed; computing from the given input: {e}"
194            );
195        },
196        Err(_) => {
197            warn!(
198                epoch = drb_input.epoch,
199                ?progress_load_timeout,
200                "loading stored drb progress timed out; computing from the given input"
201            );
202        },
203    }
204
205    // From here on the work is the hash chain; the progress stores below are
206    // fire-and-forget, so the flag stays raised until return.
207    let _hashing = HashingGuard::new(hashing);
208
209    let mut hash: [u8; 32] = drb_input.value;
210    let mut iteration = drb_input.iteration;
211    let remaining_iterations = drb_input
212        .difficulty_level
213        .checked_sub(iteration)
214        .unwrap_or_else(|| {
215            panic!(
216                "DRB difficulty level {} exceeds the iteration {} of the input we were given. \
217                 This is a fatal error",
218                drb_input.difficulty_level, iteration
219            )
220        });
221
222    let final_checkpoint = remaining_iterations / DRB_CHECKPOINT_INTERVAL;
223
224    let mut last_time = Instant::now();
225    let mut last_iteration = iteration;
226
227    // loop up to, but not including, the `final_checkpoint`
228    for _ in 0..final_checkpoint {
229        let c = cancel.clone();
230        hash = tokio::task::spawn_blocking(move || hash_batches(hash, DRB_CHECKPOINT_INTERVAL, c))
231            .await
232            .expect("completes")?;
233
234        iteration += DRB_CHECKPOINT_INTERVAL;
235
236        let updated_drb_input = DrbInput {
237            epoch: drb_input.epoch,
238            iteration,
239            value: hash,
240            difficulty_level: drb_input.difficulty_level,
241        };
242
243        let elapsed_time = last_time.elapsed().as_millis();
244
245        let store_drb_progress = store_drb_progress.clone();
246        tokio::spawn(async move {
247            info!(
248                target: "announce::drb",
249                ?updated_drb_input,
250                %last_iteration,
251                %elapsed_time,
252                "storing partial drb progress"
253            );
254            if let Err(err) = store_drb_progress(updated_drb_input).await {
255                warn!(%err, "failed to store drb progress during calculation");
256            }
257        });
258
259        last_time = Instant::now();
260        last_iteration = iteration;
261    }
262
263    let final_checkpoint_iteration = iteration;
264    // Holds by construction: `iteration` only ever reaches multiples of the checkpoint
265    // interval not exceeding `difficulty_level`.
266    let remainder = drb_input
267        .difficulty_level
268        .checked_sub(final_checkpoint_iteration)
269        .expect("sufficient iterations");
270
271    // perform the remaining iterations
272    let c = cancel.clone();
273    let drb_result = tokio::task::spawn_blocking(move || hash_batches(hash, remainder, c))
274        .await
275        .expect("DRB spawn_blocking panicked")?;
276
277    let final_drb_input = DrbInput {
278        epoch: drb_input.epoch,
279        iteration: drb_input.difficulty_level,
280        value: drb_result,
281        difficulty_level: drb_input.difficulty_level,
282    };
283
284    info!(target: "announce::drb", ?final_drb_input, "completed drb calculation");
285
286    let store_drb_progress = store_drb_progress.clone();
287    tokio::spawn(async move {
288        if let Err(err) = store_drb_progress(final_drb_input).await {
289            warn!(%err, "failed to store drb progress during calculation");
290        }
291    });
292
293    Some(drb_result)
294}

296/// Seeds for DRB computation and computed results.
297#[derive(Clone, Debug)]
298pub struct DrbResults {
299    /// Stored results from computations
300    pub results: BTreeMap<EpochNumber, DrbResult>,
301}
302
303impl DrbResults {
304    #[must_use]
305    /// Constructor with initial values for epochs 1 and 2.
306    pub fn new() -> Self {
307        Self {
308            results: BTreeMap::from([
309                (EpochNumber::new(1), INITIAL_DRB_RESULT),
310                (EpochNumber::new(2), INITIAL_DRB_RESULT),
311            ]),
312        }
313    }
314
315    pub fn store_result(&mut self, epoch: EpochNumber, result: DrbResult) {
316        self.results.insert(epoch, result);
317    }
318
319    /// Garbage collects internal data structures
320    pub fn garbage_collect(&mut self, epoch: EpochNumber) {
321        if epoch.u64() < KEEP_PREVIOUS_RESULT_COUNT {
322            return;
323        }
324
325        let retain_epoch = epoch - KEEP_PREVIOUS_RESULT_COUNT;
326        // N.B. x.split_off(y) returns the part of the map where key >= y
327
328        // Remove result entries older than EPOCH
329        self.results = self.results.split_off(&retain_epoch);
330    }
331}
332
333impl Default for DrbResults {
334    fn default() -> Self {
335        Self::new()
336    }

339/// Functions for leader selection based on the DRB.
340///
341/// The algorithm we use is:
342///
343/// Initialization:
344/// - obtain `drb: [u8; 32]` from the DRB calculation
345/// - sort the stake table for a given epoch by `xor(drb, public_key)`
346/// - generate a cdf of the cumulative stake using this newly-sorted table,
347///   along with a hash of the stake table entries
348///
349/// Selecting a leader:
350/// - calculate the SHA512 hash of the `drb_result`, `view_number` and `stake_table_hash`
351/// - find the first index in the cdf for which the remainder of this hash modulo the `total_stake`
352///   is strictly smaller than the cdf entry
353/// - return the corresponding node as the leader for that view
354pub mod election {
355    use alloy_primitives::{U256, U512};
356    use sha2::{Digest, Sha256, Sha512};
357
358    use crate::traits::signature_key::{SignatureKey, StakeTableEntryType};
359
360    /// Calculate `xor(drb.cycle(), public_key)`, returning the result as a vector of bytes
361    fn cyclic_xor(drb: [u8; 32], public_key: Vec<u8>) -> Vec<u8> {
362        let drb: Vec<u8> = drb.to_vec();
363
364        let mut result: Vec<u8> = vec![];
365
366        for (drb_byte, public_key_byte) in public_key.iter().zip(drb.iter().cycle()) {
367            result.push(drb_byte ^ public_key_byte);
368        }
369
370        result
371    }
372
373    /// Generate the stake table CDF, as well as a hash of the resulting stake table
374    pub fn generate_stake_cdf<Key: SignatureKey, Entry: StakeTableEntryType<Key>>(
375        mut stake_table: Vec<Entry>,
376        drb: [u8; 32],
377    ) -> RandomizedCommittee<Entry> {
378        // sort by xor(public_key, drb_result)
379        stake_table.sort_by(|a, b| {
380            cyclic_xor(drb, a.public_key().to_bytes())
381                .cmp(&cyclic_xor(drb, b.public_key().to_bytes()))
382        });
383
384        let mut hasher = Sha256::new();
385
386        let mut cumulative_stake = U256::from(0);
387        let mut cdf = vec![];
388
389        for entry in stake_table {
390            cumulative_stake += entry.stake();
391            hasher.update(entry.public_key().to_bytes());
392
393            cdf.push((entry, cumulative_stake));
394        }
395
396        RandomizedCommittee {
397            cdf,
398            stake_table_hash: hasher.finalize().into(),
399            drb,
400        }
401    }
402
403    /// select the leader for a view
404    ///
405    /// # Panics
406    /// Panics if `cdf` is empty. Results in undefined behaviour if `cdf` is not ordered.
407    ///
408    /// Note that we try to downcast a U512 to a U256,
409    /// but this should never panic because the U512 should be strictly smaller than U256::MAX by construction.
410    pub fn select_randomized_leader<
411        SignatureKey,
412        Entry: StakeTableEntryType<SignatureKey> + Clone,
413    >(
414        randomized_committee: &RandomizedCommittee<Entry>,
415        view: u64,
416    ) -> Entry {
417        let RandomizedCommittee {
418            cdf,
419            stake_table_hash,
420            drb,
421        } = randomized_committee;
422        // We hash the concatenated drb, view and stake table hash.
423        let mut hasher = Sha512::new();
424        hasher.update(drb);
425        hasher.update(view.to_le_bytes());
426        hasher.update(stake_table_hash);
427        let raw_breakpoint: [u8; 64] = hasher.finalize().into();
428
429        // then calculate the remainder modulo the total stake as a U512
430        let remainder: U512 =
431            U512::from_le_bytes(raw_breakpoint) % U512::from(cdf.last().unwrap().1);
432
433        // and drop the top 32 bytes, downcasting to a U256
434        let breakpoint: U256 = U256::from_le_slice(&remainder.to_le_bytes_vec()[0..32]);
435
436        // now find the first index where the breakpoint is strictly smaller than the cdf
437        //
438        // in principle, this may result in an index larger than `cdf.len()`.
439        // however, we have ensured by construction that `breakpoint < total_stake`
440        // and so the largest index we can actually return is `cdf.len() - 1`
441        let index = cdf.partition_point(|(_, cumulative_stake)| breakpoint >= *cumulative_stake);
442
443        // and return the corresponding entry
444        cdf[index].0.clone()
445    }
446
447    #[derive(Clone, Debug)]
448    pub struct RandomizedCommittee<Entry> {
449        /// cdf of nodes by cumulative stake
450        cdf: Vec<(Entry, U256)>,
451        /// Hash of the stake table
452        stake_table_hash: [u8; 32],
453        /// DRB result
454        drb: [u8; 32],
455    }
456
457    impl<Entry> RandomizedCommittee<Entry> {
458        pub fn drb_result(&self) -> [u8; 32] {
459            self.drb
460        }
461    }
462}

587    #[test]
588    fn test_randomized_leader() {
589        let mut rng = rand::thread_rng();
590        // use an arbitrary Sha256 output.
591        let drb: [u8; 32] = Sha256::digest(b"drb").into();
592        // a stake table with 10 nodes, each with a stake of 1-100
593        let stake_table_entries: Vec<_> = (0..10)
594            .map(|i| StakeTableEntry {
595                stake_key: BLSPubKey::generated_from_seed_indexed([0u8; 32], i).0,
596                stake_amount: U256::from(rng.next_u64() % 100 + 1),
597            })
598            .collect();
599        let randomized_committee = generate_stake_cdf(stake_table_entries.clone(), drb);
600
601        // Number of views to test
602        let num_views = 100000;
603        let mut selected = HashMap::<_, u64>::new();
604        // Test the leader election for 100000 views.
605        for i in 0..num_views {
606            let leader = select_randomized_leader(&randomized_committee, i);
607            *selected.entry(leader).or_insert(0) += 1;
608        }
609
610        // Total variation distance
611        let mut tvd = 0.;
612        let total_stakes = stake_table_entries
613            .iter()
614            .map(|e| e.stake())
615            .sum::<U256>()
616            .to::<u64>() as f64;
617        for entry in stake_table_entries {
618            let expected = entry.stake().to::<u64>() as f64 / total_stakes;
619            let actual = *selected.get(&entry).unwrap_or(&0) as f64 / num_views as f64;
620            tvd += (expected - actual).abs();
621        }
622
623        // sanity check
624        assert!(tvd >= 0.0);
625        // Allow a small margin of error
626        assert!(tvd < 0.03);
627    }
