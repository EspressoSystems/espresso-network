28/// Build a full request URL from a server base URL and a path produced by one of the
29/// `routes::v1::*` (or `routes::v2::*`) builders.
30///
31/// Use this from test/CLI sites that have a `url::Url` pointing at the API server and want
32/// the absolute URL for a single request. Internally this is just `base.join(path)`; the
33/// helper exists so the (path-const, builder, joiner) trio reads as one chain.
34pub fn url(base: &::url::Url, path: impl AsRef<str>) -> ::url::Url {
35    base.join(path.as_ref())
36        .expect("path produced by routes::*::path_fn is always a valid relative URL")
37}

39/// Start Axum HTTP server with combined v1 and v2 APIs
40///
41/// This serves both APIs at /v1/* and /v2/* from a single state implementation.
42///
43/// `catchup`, like the query-service modules (`status`, `availability`, `node`, `token`,
44/// `block-state`, `fee-state`, `reward-state`, `database`) and `v2`, is always on: tide-disco's
45/// SQL mode registered it unconditionally. `submit`, `config`, `explorer`, `light-client`, and
46/// `hotshot-events` follow `Options`, matching `Options::init_with_query_module_sql`.
47pub async fn serve_axum<S>(
48    port: u16,
49    state: S,
50    modules: OptionalModules,
51    max_connections: Option<usize>,
52) -> anyhow::Result<()>
53where
54    S: v1::RewardApi
55        + v1::AvailabilityApi
56        + v1::HotShotAvailabilityApi
57        + v1::BlockStateApi
58        + v1::FeeStateApi
59        + v1::StatusApi
60        + v1::ConfigApi
61        + v1::NodeApi
62        + v1::CatchupApi
63        + v1::SubmitApi
64        + v1::StateSignatureApi
65        + v1::HotShotEventsApi
66        + v1::LightClientApi
67        + v1::ExplorerApi
68        + v1::TokenApi
69        + v1::DatabaseApi
70        + v2::RewardApi
71        + v2::DataApi
72        + v2::ConsensusApi
73        + Clone
74        + Send
75        + Sync
76        + 'static,
77{
78    let listener = bind_api(port).await?;
79    let mut router = axum::router_reward(state.clone())
80        .merge(axum::router_availability(state.clone()))
81        .merge(axum::router_block_state(state.clone()))
82        .merge(axum::router_fee_state(state.clone()))
83        .merge(axum::router_status(state.clone()))
84        .merge(axum::router_node(state.clone()))
85        .merge(axum::router_catchup(state.clone()))
86        .merge(axum::router_state_signature(state.clone()))
87        .merge(axum::router_token(state.clone()))
88        .merge(axum::router_database(state.clone()));
89    if modules.submit {
90        router = router.merge(axum::router_submit(state.clone()));
91    }
92    if modules.config {
93        router = router.merge(axum::router_config(state.clone()));
94    }
95    if modules.explorer {
96        router = router.merge(axum::router_explorer(state.clone()));
97    }
98    if modules.light_client {
99        router = router.merge(axum::router_light_client(state.clone()));
100    }
101    if modules.hotshot_events {
102        router = router.merge(axum::router_hotshot_events(state.clone()));
103    }
104    let router = axum::finish_v1_docs(router).merge(axum::create_router_v2(state));
105    serve_router(listener, "v1 and v2", router, max_connections).await
106}

121/// Serve the query API used by the filesystem-backed storage mode: status, availability, node,

123/// submit, config, and hotshot-events follow `Options`. Filesystem storage doesn't implement the
124/// reward/merklized-state/explorer/database traits, so those modules aren't served (a request to
125/// one of their routes 404s, matching tide).
126pub async fn serve_axum_fs<S>(
127    port: u16,
128    state: S,
129    modules: OptionalModules,
130    max_connections: Option<usize>,
131) -> anyhow::Result<()>
132where
133    S: v1::StatusApi
134        + v1::AvailabilityApi
135        + v1::HotShotAvailabilityApi
136        + v1::NodeApi
137        + v1::TokenApi
138        + v1::CatchupApi
139        + v1::SubmitApi
140        + v1::StateSignatureApi
141        + v1::ConfigApi
142        + v1::HotShotEventsApi
143        + Clone
144        + Send
145        + Sync
146        + 'static,
147{
148    let listener = bind_api(port).await?;
149    let mut router = axum::router_status(state.clone())
150        .merge(axum::router_availability(state.clone()))
151        .merge(axum::router_node(state.clone()))
152        .merge(axum::router_token(state.clone()))
153        .merge(axum::router_catchup(state.clone()))
154        .merge(axum::router_state_signature(state.clone()));
155    if modules.submit {
156        router = router.merge(axum::router_submit(state.clone()));
157    }
158    if modules.config {
159        router = router.merge(axum::router_config(state.clone()));
160    }
161    if modules.hotshot_events {
162        router = router.merge(axum::router_hotshot_events(state));
163    }
164    serve_router(
165        listener,
166        "fs",
167        axum::finish_v1_docs(router),
168        max_connections,
169    )
170    .await
171}

173/// Serve the status-only API: no availability/node/token data source is available, so only
174/// status and the HotShot modules (submit, catchup, state-signature, config, hotshot-events) can
175/// be served. State-signature is always on; the rest follow `Options`.
176pub async fn serve_axum_status<S>(
177    port: u16,
178    state: S,
179    modules: OptionalModules,
180    max_connections: Option<usize>,
181) -> anyhow::Result<()>
182where
183    S: v1::StatusApi
184        + v1::SubmitApi
185        + v1::CatchupApi
186        + v1::StateSignatureApi
187        + v1::ConfigApi
188        + v1::HotShotEventsApi
189        + Clone
190        + Send
191        + Sync
192        + 'static,
193{
194    let listener = bind_api(port).await?;
195    let router =
196        axum::router_status(state.clone()).merge(axum::router_state_signature(state.clone()));
197    let router = merge_hotshot_modules(router, &state, modules);
198    serve_router(
199        listener,
200        "status",
201        axum::finish_v1_docs(router),
202        max_connections,
203    )
204    .await
205}

269/// Add the reserved top-level routes, apply the optional concurrency limit, rewrite legacy URIs,
270/// and bind/serve the router. Shared by all `serve_axum*` entry points.
271/// Bind before composing routers: OpenAPI generation takes ~0.5s in debug builds, and clients
272/// connecting during it should queue in the accept backlog rather than get refused.
273async fn bind_api(port: u16) -> anyhow::Result<tokio::net::TcpListener> {
274    let addr = format!("0.0.0.0:{}", port);
275    tracing::info!("Binding to {}", addr);
276    Ok(tokio::net::TcpListener::bind(&addr).await?)
277}

279async fn serve_router(
280    listener: tokio::net::TcpListener,
281    mode: &str,
282    router: ::axum::Router,
283    max_connections: Option<usize>,
284) -> anyhow::Result<()> {
285    let mut router = axum::with_top_level_routes(router);
286    if let Some(limit) = max_connections {
287        router = apply_connection_limit(router, limit);
288    }
289    // CORS goes on last so it wraps the connection limit, whose 429 would otherwise skip it.
290    let router = router.layer(axum::cors_layer());
291    // `Router::layer` middleware runs after routing, so it can't rewrite a URI to match a
292    // different route. Wrapping the whole router with `MapRequestLayer` instead runs the
293    // rewrite before routing, per the axum-documented pattern for this case.
294    let router = tower::util::MapRequestLayer::new(axum::rewrite_legacy_uri).layer(router);
295
296    tracing::info!(
297        "Axum API server listening on {:?} ({} mode)",
298        listener.local_addr()?,
299        mode
300    );
301    ::axum::serve(listener, ::axum::ServiceExt::into_make_service(router)).await?;
302
303    tracing::info!("Axum server stopped");
304    Ok(())
305}
306
307/// Shared budget: plain requests hold a slot while in flight, streaming sockets for their
308/// lifetime; excess gets 429.
309fn apply_connection_limit(router: ::axum::Router, limit: usize) -> ::axum::Router {
310    let semaphore = std::sync::Arc::new(tokio::sync::Semaphore::new(limit));
311    router
312        .layer(::axum::middleware::from_fn(axum::limit_requests))
313        .layer(::axum::Extension(axum::RequestLimit(semaphore)))
314}

316/// Start Tonic gRPC server
317pub async fn serve_tonic<S>(port: u16, state: S) -> anyhow::Result<()>
318where
319    S: v2::RewardApi + Clone + Send + Sync + 'static,
320{
321    use ::tonic::transport::Server;
322
323    let addr = std::net::SocketAddr::from(([0, 0, 0, 0], port));
324
325    let reward_service = create_reward_service(state);
326
327    // Enable gRPC reflection for tools like grpcurl
328    let reflection_service = tonic_reflection::server::Builder::configure()
329        .register_encoded_file_descriptor_set(include_bytes!(concat!(
330            env!("OUT_DIR"),
331            "/reflection_descriptor.bin"
332        )))
333        .build_v1()?;
334
335    tracing::info!("gRPC server listening on {}", addr);
336    Server::builder()
337        .add_service(reward_service)
338        .add_service(reflection_service)
339        .serve(addr)
340        .await?;
341
342    Ok(())
