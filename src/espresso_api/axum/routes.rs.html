1/// Generate a path-builder function from a route constant.
2///
3/// Substitutes each `{placeholder}` segment with the corresponding argument
4/// (formatted via `Display`). Used to keep request URLs in sync with the
5/// route definitions registered with the Axum router.
6macro_rules! path_fn {
7    ($name:ident, $route:expr $(,)?) => {
8        pub fn $name() -> String {
9            ::std::string::String::from($route)
10        }
11    };
12    ($name:ident, $route:expr, $($param:ident),+ $(,)?) => {
13        pub fn $name($($param: impl ::std::fmt::Display),+) -> String {
14            let mut path = ::std::string::String::from($route);
15            $(
16                path = path.replace(
17                    concat!("{", stringify!($param), "}"),
18                    &$param.to_string(),
19                );
20            )+
21            path
22        }
23    };
24}
