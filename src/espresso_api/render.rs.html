52impl From<ResolvableChainConfig> for proto::ResolvableChainConfig {
53    fn from(chain_config: ResolvableChainConfig) -> Self {
54        // `commit` would hash a full config too; only `resolve` tells the two apart.
55        let resolved = match chain_config.resolve() {
56            Some(config) => ChainConfig::Full(proto::ChainConfig {
57                chain_id: config.chain_id.to_string(),
58                max_block_size: *config.max_block_size,
59                base_fee: config.base_fee.to_string(),
60                // v1 renders addresses as lowercase hex through `ethers_core::H160` and
61                // `FixedBytes`; alloy's `Display` prints them EIP-55 checksummed instead.
62                fee_contract: config.fee_contract.map(|address| format!("{address:#x}")),
63                fee_recipient: format!("{:#x}", config.fee_recipient.0),
64                stake_table_contract: config
65                    .stake_table_contract
66                    .map(|address| format!("{address:#x}")),
67            }),
68            None => ChainConfig::Commitment(chain_config.commit().to_string()),
69        };
70        proto::ResolvableChainConfig {
71            chain_config: Some(resolved),
72        }
73    }
74}

106/// The proto message per protocol version, mirroring the `Header` enum. Versions sharing a shape
107/// share a message, so only the arm distinguishes 0.1 from 0.2 and 0.5 from 0.6 and 0.7.
108impl From<&Header> for proto::HeaderResponse {
109    fn from(header: &Header) -> Self {
110        let shape = match header {
111            Header::V1(_) => Shape::V1(header.into()),
112            Header::V2(_) => Shape::V2(header.into()),
113            Header::V3(_) => Shape::V3(header.into()),
114            Header::V4(_) => Shape::V4(header_v4(header)),
115            Header::V5(_) => Shape::V5(header_v5(header)),
116            Header::V6(_) => Shape::V6(header_v5(header)),
117            Header::V7(_) => Shape::V7(header_v5(header)),
118        };
119        proto::HeaderResponse {
120            header: Some(shape),
121        }
122    }
123}

169/// Private, unlike the 0.1 and 0.3 shapes above, because the two `expect`s are sound only under

173    proto::HeaderV4 {
174        chain_config: Some(header.chain_config().into()),

176        timestamp: header.timestamp_internal(),
177        timestamp_millis: header.timestamp_millis_internal(),
178        l1_head: header.l1_head(),
179        l1_finalized: header.l1_finalized().map(Into::into),
180        payload_commitment: header.payload_commitment().to_string(),
181        builder_commitment: header.builder_commitment().to_string(),
182        ns_table: Some(proto::NsTable {
183            bytes: header.ns_table().encode().to_vec(),
184        }),
185        block_merkle_tree_root: header.block_merkle_tree_root().to_string(),
186        fee_merkle_tree_root: header.fee_merkle_tree_root().to_string(),
187        fee_info: header.fee_info().first().map(Into::into),
188        builder_signature: header.builder_signature().first().map(Into::into),
189        reward_merkle_tree_root: reward_merkle_tree_root(header),
190        total_reward_distributed: header

192            .expect("0.4 and later headers carry total_reward_distributed")
193            .to_string(),
194        next_stake_table_hash: header.next_stake_table_hash().map(|hash| hash.to_string()),
195    }

365/// v1's serialized form is the source for the ADVZ and AvidM arms: jellyfish keeps the ADVZ
366/// share's fields private, and the AvidM payload is ark-serialized into single TaggedBase64
367/// strings that only its serde impl produces. The gf2 arm maps from the share's own accessors,
368/// because its payload is raw bytes that a JSON round trip would inflate into one number per byte.
369impl TryFrom<&VidShare> for proto::VidShareResponse {
370    type Error = tonic::Status;
371
372    fn try_from(share: &VidShare) -> Result<Self, Self::Error> {
373        // Matched on the enum rather than sniffed from the JSON, so a fourth scheme fails to
374        // compile instead of surfacing as a 500.
375        let arm = match share {
376            VidShare::V0(_) => {
377                let json = to_json(share)?;
378                let share = json_entry(&json, "V0")?;
379                let evals_proof = json_entry(share, "evals_proof")?;
380                proto::vid_share_response::Share::V0(proto::AdvzVidShare {
381                    index: json_field(share, "index")?,
382                    aggregate_proofs: json_field(share, "aggregate_proofs")?,
383                    evals: json_field(share, "evals")?,
384                    evals_proof: Some(proto::AdvzMerkleProof {
385                        pos: json_field(evals_proof, "pos")?,
386                        proof: json_array(evals_proof, "proof")?
387                            .iter()
388                            .map(advz_merkle_node)
389                            .collect::<Result<_, _>>()?,
390                    }),
391                })
392            },
393            VidShare::V1(_) => {
394                let json = to_json(share)?;
395                let share = json_entry(&json, "V1")?;
396                proto::vid_share_response::Share::V1(proto::AvidmVidShare {
397                    index: json_field(share, "index")?,
398                    ns_commits: json_field(share, "ns_commits")?,
399                    ns_lens: json_field(share, "ns_lens")?,
400                    content: json_array(share, "content")?
401                        .iter()
402                        .map(|content| {
403                            let range = json_entry(content, "range")?;
404                            Ok(proto::AvidmShareContent {
405                                range: Some(proto::ShardRange {
406                                    start: json_field(range, "start")?,
407                                    end: json_field(range, "end")?,
408                                }),
409                                payload: json_field(content, "payload")?,
410                                mt_proofs: json_field(content, "mt_proofs")?,
411                            })
412                        })
413                        .collect::<Result<_, tonic::Status>>()?,
414                })
415            },
416            VidShare::V2(gf2) => proto::vid_share_response::Share::V2(proto::AvidmGf2VidShare {
417                namespaces: gf2
418                    .ns_shares()
419                    .iter()
420                    .map(|namespace| proto::AvidmGf2Namespace {
421                        range: Some(proto::ShardRange {
422                            start: namespace.range().start as u64,
423                            end: namespace.range().end as u64,
424                        }),
425                        payload: namespace.payload().to_vec(),
426                        mt_proofs: namespace
427                            .mt_proofs()
428                            .iter()
429                            .map(ToString::to_string)
430                            .collect(),
431                    })
432                    .collect(),
433            }),
434        };
435        Ok(proto::VidShareResponse { share: Some(arm) })
436    }
437}

439fn advz_merkle_node(value: &serde_json::Value) -> Result<proto::AdvzMerkleNode, tonic::Status> {
440    let node = if let Some(leaf) = value.get("Leaf") {
441        Node::Leaf(proto::AdvzMerkleNodeLeaf {
442            elem: json_field(leaf, "elem")?,
443            pos: json_field(leaf, "pos")?,
444            value: json_field(leaf, "value")?,
445        })
446    } else if let Some(branch) = value.get("Branch") {
447        Node::Branch(proto::AdvzMerkleNodeBranch {
448            children: json_array(branch, "children")?
449                .iter()
450                .map(advz_merkle_node)
451                .collect::<Result<_, _>>()?,
452            value: json_field(branch, "value")?,
453        })
454    } else if let Some(subtree) = value.get("ForgettenSubtree") {
455        // Upstream's spelling, which the proto field name corrects.
456        Node::ForgottenSubtree(proto::AdvzMerkleNodeForgottenSubtree {
457            value: json_field(subtree, "value")?,
458        })
459    } else if value.as_str() == Some("Empty") {
460        // A unit variant, so v1 writes it as a bare string.
461        Node::Empty(proto::AdvzMerkleNodeEmpty {})
462    } else {
463        let arms: Vec<&str> = value
464            .as_object()
465            .map(|node| node.keys().map(String::as_str).collect())
466            .unwrap_or_default();
467        return Err(tonic::Status::internal(format!(
468            "VID share JSON has an unknown Merkle node arm: {arms:?}"
469        )));
470    };
471    Ok(proto::AdvzMerkleNode { node: Some(node) })
472}

554impl<E, I, T> From<&MerkleNode<E, I, T>> for proto::AdvzMerkleNode
555where
556    E: Element + CanonicalSerialize,
557    I: Index + CanonicalSerialize,
558    T: NodeValue,
559{
560    fn from(node: &MerkleNode<E, I, T>) -> Self {
561        let node = match node {
562            MerkleNode::Empty => Node::Empty(proto::AdvzMerkleNodeEmpty {}),
563            MerkleNode::Branch { value, children } => Node::Branch(proto::AdvzMerkleNodeBranch {
564                value: field_tb64(value),
565                children: children
566                    .iter()
567                    .map(|child| proto::AdvzMerkleNode::from(&**child))
568                    .collect(),
569            }),
570            MerkleNode::Leaf { value, pos, elem } => Node::Leaf(proto::AdvzMerkleNodeLeaf {
571                value: field_tb64(value),
572                pos: field_tb64(pos),
573                elem: field_tb64(elem),
574            }),
575            MerkleNode::ForgettenSubtree { value } => {
576                Node::ForgottenSubtree(proto::AdvzMerkleNodeForgottenSubtree {
577                    value: field_tb64(value),
578                })
579            },
580        };
581        proto::AdvzMerkleNode { node: Some(node) }
582    }
583}

585impl From<RewardAccountQueryDataV2> for proto::RewardAccountProofResponse {
586    fn from(query: RewardAccountQueryDataV2) -> proto::RewardAccountProofResponse {
587        let RewardAccountQueryDataV2 {
588            balance,
589            proof: RewardAccountProofV2 { account, proof },
590        } = query;
591        let proof = match proof {
592            RewardMerkleProofV2::Presence(proof) => {
593                proto::reward_merkle_proof::Proof::Presence(proto::MerklePathResponse::from(&proof))

609/// The encoding jellyfish's `canonical` serde helper gives every hash, index and element of a
610/// proof: ark-compressed bytes under the `FIELD` tag, whatever the underlying type is.
611/// `block_state_path_mirrors_its_v1_rendering` pins the two to the same bytes.
612fn field_tb64<T>(value: &T) -> String
613where
614    T: CanonicalSerialize,
615{
616    let mut bytes = Vec::new();
617    value
618        .serialize_compressed(&mut bytes)
619        .expect("serializing to a Vec cannot fail");
620    TaggedBase64::new("FIELD", &bytes)
621        .expect("FIELD is a valid tag")
622        .to_string()
623}
624
625/// jellyfish keeps the fields of the VID shares and common, the range proofs and the bad-encoding
626/// proof private, so v1's serde encoding is their one public view.
627fn to_json(value: &impl serde::Serialize) -> Result<serde_json::Value, tonic::Status> {
628    serde_json::to_value(value)
629        .map_err(|err| tonic::Status::internal(format!("v1 encoding failed: {err}")))
630}
631
632/// A missing or mistyped field means the upstream type changed, which is a 500 rather than an
633/// empty value. v1 writes byte fields as integer arrays, which read back as `Vec<u8>`.
634fn json_field<T>(value: &serde_json::Value, field: &str) -> Result<T, tonic::Status>
635where
636    T: serde::de::DeserializeOwned,
637{
638    T::deserialize(json_entry(value, field)?)
639        .map_err(|err| tonic::Status::internal(format!("v1 JSON {field}: {err}")))
640}
641
642fn json_entry<'a>(
643    value: &'a serde_json::Value,
644    field: &str,
645) -> Result<&'a serde_json::Value, tonic::Status> {
646    value
647        .get(field)
648        .ok_or_else(|| tonic::Status::internal(format!("v1 JSON has no {field}")))
649}
650
651fn json_array<'a>(
652    value: &'a serde_json::Value,
653    field: &str,
654) -> Result<&'a [serde_json::Value], tonic::Status> {
655    json_entry(value, field)?
656        .as_array()
657        .map(Vec::as_slice)
658        .ok_or_else(|| tonic::Status::internal(format!("v1 JSON {field}: not an array")))
659}
