44impl From<ResolvableChainConfig> for proto::ResolvableChainConfig {
45    fn from(chain_config: ResolvableChainConfig) -> Self {
46        // `commit` would hash a full config too; only `resolve` tells the two apart.
47        let resolved = match chain_config.resolve() {
48            Some(config) => ChainConfig::Full(proto::ChainConfig {
49                chain_id: config.chain_id.to_string(),
50                max_block_size: *config.max_block_size,
51                base_fee: config.base_fee.to_string(),
52                // v1 renders addresses as lowercase hex through `ethers_core::H160` and
53                // `FixedBytes`; alloy's `Display` prints them EIP-55 checksummed instead.
54                fee_contract: config.fee_contract.map(|address| format!("{address:#x}")),
55                fee_recipient: format!("{:#x}", config.fee_recipient.0),
56                stake_table_contract: config
57                    .stake_table_contract
58                    .map(|address| format!("{address:#x}")),
59            }),
60            None => ChainConfig::Commitment(chain_config.commit().to_string()),
61        };
62        Self {
63            chain_config: Some(resolved),
64        }
65    }
66}

98/// The proto message per protocol version, mirroring the `Header` enum. Versions sharing a shape
99/// share a message, so only the arm distinguishes 0.1 from 0.2 and 0.5 from 0.6 and 0.7.
100impl From<&Header> for proto::HeaderResponse {
101    fn from(header: &Header) -> Self {
102        let shape = match header {
103            Header::V1(_) => Shape::V1(header.into()),
104            Header::V2(_) => Shape::V2(header.into()),
105            Header::V3(_) => Shape::V3(header.into()),
106            Header::V4(_) => Shape::V4(header_v4(header)),
107            Header::V5(_) => Shape::V5(header_v5(header)),
108            Header::V6(_) => Shape::V6(header_v5(header)),
109            Header::V7(_) => Shape::V7(header_v5(header)),
110        };
111        Self {
112            header: Some(shape),
113        }
114    }
115}

357/// v1's serialized form is the source for the ADVZ and AvidM arms: jellyfish keeps the ADVZ
358/// share's fields private, and the AvidM payload is ark-serialized into single TaggedBase64
359/// strings that only its serde impl produces. The gf2 arm maps from the share's own accessors,
360/// because its payload is raw bytes that a JSON round trip would inflate into one number per byte.
361impl TryFrom<&VidShare> for proto::VidShareResponse {
362    type Error = tonic::Status;
363
364    fn try_from(share: &VidShare) -> Result<Self, Self::Error> {
365        // Matched on the enum rather than sniffed from the JSON, so a fourth scheme fails to
366        // compile instead of surfacing as a 500.
367        let arm = match share {
368            VidShare::V0(_) => {
369                let json = to_json(share)?;
370                let share = json_entry(&json, "V0")?;
371                let evals_proof = json_entry(share, "evals_proof")?;
372                proto::vid_share_response::Share::V0(proto::AdvzVidShare {
373                    index: json_field(share, "index")?,
374                    aggregate_proofs: json_field(share, "aggregate_proofs")?,
375                    evals: json_field(share, "evals")?,
376                    evals_proof: Some(proto::AdvzMerkleProof {
377                        pos: json_field(evals_proof, "pos")?,
378                        proof: json_array(evals_proof, "proof")?
379                            .iter()
380                            .map(advz_merkle_node)
381                            .collect::<Result<_, _>>()?,
382                    }),
383                })
384            },
385            VidShare::V1(_) => {
386                let json = to_json(share)?;
387                let share = json_entry(&json, "V1")?;
388                proto::vid_share_response::Share::V1(proto::AvidmVidShare {
389                    index: json_field(share, "index")?,
390                    ns_commits: json_field(share, "ns_commits")?,
391                    ns_lens: json_field(share, "ns_lens")?,
392                    content: json_array(share, "content")?
393                        .iter()
394                        .map(|content| {
395                            let range = json_entry(content, "range")?;
396                            Ok(proto::AvidmShareContent {
397                                range: Some(proto::ShardRange {
398                                    start: json_field(range, "start")?,
399                                    end: json_field(range, "end")?,
400                                }),
401                                payload: json_field(content, "payload")?,
402                                mt_proofs: json_field(content, "mt_proofs")?,
403                            })
404                        })
405                        .collect::<Result<_, tonic::Status>>()?,
406                })
407            },
408            VidShare::V2(gf2) => proto::vid_share_response::Share::V2(proto::AvidmGf2VidShare {
409                namespaces: gf2
410                    .ns_shares()
411                    .iter()
412                    .map(|namespace| proto::AvidmGf2Namespace {
413                        range: Some(proto::ShardRange {
414                            start: namespace.range().start as u64,
415                            end: namespace.range().end as u64,
416                        }),
417                        payload: namespace.payload().to_vec(),
418                        mt_proofs: namespace
419                            .mt_proofs()
420                            .iter()
421                            .map(ToString::to_string)
422                            .collect(),
423                    })
424                    .collect(),
425            }),
426        };
427        Ok(Self { share: Some(arm) })
428    }
429}

431fn advz_merkle_node(value: &serde_json::Value) -> Result<proto::AdvzMerkleNode, tonic::Status> {
432    let node = if let Some(leaf) = value.get("Leaf") {
433        Node::Leaf(proto::AdvzMerkleNodeLeaf {
434            elem: json_field(leaf, "elem")?,
435            pos: json_field(leaf, "pos")?,
436            value: json_field(leaf, "value")?,
437        })
438    } else if let Some(branch) = value.get("Branch") {
439        Node::Branch(proto::AdvzMerkleNodeBranch {
440            children: json_array(branch, "children")?
441                .iter()
442                .map(advz_merkle_node)
443                .collect::<Result<_, _>>()?,
444            value: json_field(branch, "value")?,
445        })
446    } else if let Some(subtree) = value.get("ForgettenSubtree") {
447        // Upstream's spelling, which the proto field name corrects.
448        Node::ForgottenSubtree(proto::AdvzMerkleNodeForgottenSubtree {
449            value: json_field(subtree, "value")?,
450        })
451    } else if value.as_str() == Some("Empty") {
452        // A unit variant, so v1 writes it as a bare string.
453        Node::Empty(proto::AdvzMerkleNodeEmpty {})
454    } else {
455        let arms: Vec<&str> = value
456            .as_object()
457            .map(|node| node.keys().map(String::as_str).collect())
458            .unwrap_or_default();
459        return Err(tonic::Status::internal(format!(
460            "VID share JSON has an unknown Merkle node arm: {arms:?}"
461        )));
462    };
463    Ok(proto::AdvzMerkleNode { node: Some(node) })
464}

532/// jellyfish keeps the fields of the VID shares and common, the range proofs and the bad-encoding
533/// proof private, so v1's serde encoding is their one public view.
534fn to_json(value: &impl serde::Serialize) -> Result<serde_json::Value, tonic::Status> {
535    serde_json::to_value(value)
536        .map_err(|err| tonic::Status::internal(format!("v1 encoding failed: {err}")))
537}

539/// A missing or mistyped field means the upstream type changed, which is a 500 rather than an
540/// empty value. v1 writes byte fields as integer arrays, which read back as `Vec<u8>`.
541fn json_field<T>(value: &serde_json::Value, field: &str) -> Result<T, tonic::Status>
542where
543    T: serde::de::DeserializeOwned,
544{
545    T::deserialize(json_entry(value, field)?)
546        .map_err(|err| tonic::Status::internal(format!("v1 JSON {field}: {err}")))
547}
548
549fn json_entry<'a>(
550    value: &'a serde_json::Value,
551    field: &str,
552) -> Result<&'a serde_json::Value, tonic::Status> {
553    value
554        .get(field)
555        .ok_or_else(|| tonic::Status::internal(format!("v1 JSON has no {field}")))
556}
557
558fn json_array<'a>(
559    value: &'a serde_json::Value,
560    field: &str,
561) -> Result<&'a [serde_json::Value], tonic::Status> {
562    json_entry(value, field)?
563        .as_array()
564        .map(Vec::as_slice)
565        .ok_or_else(|| tonic::Status::internal(format!("v1 JSON {field}: not an array")))
566}

672where
673    V: Voteable<SeqTypes>,

679        .map(|(signature, signers)| proto::QuorumSignatures {
680            signature: signature.to_string(),
681            signers: signers.iter().by_vals().collect(),
682        })
