1//! Typed API errors for proper HTTP/gRPC status code mapping
2
3use std::fmt;
4
5use thiserror::Error;
6
7/// Marker errors for availability endpoint failures. These are wrapped in `anyhow::Error` by the
8/// state implementation and downcasted in the Axum handlers to select the right HTTP status code.
9#[derive(Debug, Error)]
10pub enum AvailabilityError {
11    #[error("{0}")]
12    NotFound(String),
13    #[error("{0}")]
14    RangeExceeded(String),
15    #[error("{0}")]
16    BadRequest(String),
17}
18
19/// API error types that can be downcast at the HTTP/gRPC boundary
20#[derive(Debug)]
21pub enum ApiError {
22    /// Client provided invalid input (maps to 400 Bad Request / INVALID_ARGUMENT)
23    BadRequest(anyhow::Error),
24    /// Requested resource does not exist (maps to 404 Not Found)
25    NotFound(anyhow::Error),
26    /// Handler failed for any reason (maps to 500 Internal Server Error / INTERNAL)
27    Internal(anyhow::Error),
28}
29
30impl fmt::Display for ApiError {
31    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
32        match self {
33            ApiError::BadRequest(err) | ApiError::NotFound(err) | ApiError::Internal(err) => {
34                // Both transports render handler errors through this impl (v1 via the axum
35                // handlers, v2 via `to_status`), so a provider credential in the message is
36                // removed once, here.
37                f.write_str(&espresso_utils::redact::scrub(&err.to_string()))
38            },
39        }
40    }
41}
42
43impl std::error::Error for ApiError {
44    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
45        match self {
46            ApiError::BadRequest(err) | ApiError::NotFound(err) | ApiError::Internal(err) => {
47                err.source()
48            },
49        }
50    }
51}
52
53/// Errors raised as [`AvailabilityError`] by a state implementation carry semantic meaning;
54/// anything else is a failure the client cannot act on. Both transports classify through here,
55/// so v1 and v2 cannot drift on what counts as a 404.
56pub fn classify(err: anyhow::Error) -> ApiError {
57    match err.downcast_ref::<AvailabilityError>() {
58        Some(AvailabilityError::NotFound(_)) => ApiError::NotFound(err),
59        Some(_) => ApiError::BadRequest(err),
60        None => ApiError::Internal(err),
61    }
62}
63
64/// Render a handler error for the v2 transports: the gRPC `grpc-message` trailer and, via
65/// `tonic_rest::RestError`, the REST error body. Classification and the credential scrub happen
66/// in [`classify`] and [`ApiError`]'s `Display`, shared with the v1 path.
67pub fn to_status(err: anyhow::Error) -> tonic::Status {
68    let err = classify(err);
69    let message = err.to_string();
70    match err {
71        ApiError::NotFound(_) => tonic::Status::not_found(message),
72        ApiError::BadRequest(_) => tonic::Status::invalid_argument(message),
73        ApiError::Internal(_) => tonic::Status::internal(message),
74    }
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80
81    /// Ported from the deleted `api/src/tonic.rs` when the v2 adapter stopped rendering through
82    /// [`ApiError`]. The token endpoints inline the L1 provider's error, URL and API key included,
83    /// and both the `grpc-message` trailer and the REST body reach unauthenticated callers.
84    #[test]
85    fn to_status_scrubs_provider_credentials() {
86        let err = anyhow::anyhow!(
87            r#"failed to get total supply: reqwest::Error {{ url: "https://u:p@rpc.invalid/v1/FAKEKEY" }}"#
88        );
89
90        let status = to_status(err);
91
92        assert_eq!(status.code(), tonic::Code::Internal);
93        assert!(
94            !status.message().contains("FAKEKEY"),
95            "{}",
96            status.message()
97        );
98        assert!(!status.message().contains("u:p"), "{}", status.message());
99        assert!(
100            status.message().contains("rpc.invalid"),
101            "{}",
102            status.message()
103        );
104    }
105
106    /// A 404 has to stay a 404 through the classifier, since the scrub sits on the same path.
107    #[test]
108    fn availability_not_found_maps_to_not_found() {
109        let status = to_status(anyhow::Error::new(AvailabilityError::NotFound(
110            "leaf 7".to_string(),
111        )));
112        assert_eq!(status.code(), tonic::Code::NotFound);
113    }
