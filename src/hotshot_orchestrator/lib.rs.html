64/// Generate an keypair based on a `seed` and an `index`
65/// # Panics
66/// This panics if libp2p is unable to generate a secret key from the seed
67#[must_use]
68pub fn libp2p_generate_indexed_identity(seed: [u8; 32], index: u64) -> Keypair {
69    let mut hasher = blake3::Hasher::new();
70    hasher.update(&seed);
71    hasher.update(&index.to_le_bytes());
72    let new_seed = *hasher.finalize().as_bytes();
73    let sk_bytes = SecretKey::try_from_bytes(new_seed).unwrap();
74    <EdKeypair as From<SecretKey>>::from(sk_bytes).into()
75}

77/// The state of the orchestrator
78#[derive(Default, Clone)]
79#[allow(clippy::struct_excessive_bools)]
80struct OrchestratorState<TYPES: NodeType> {
81    /// Tracks the latest node index we have generated a configuration for
82    latest_index: u16,
83    /// Tracks the latest temporary index we have generated for init validator's key pair
84    tmp_latest_index: u16,
85    /// The network configuration
86    config: NetworkConfig<TYPES>,
87    /// Whether the network configuration has been updated with all the peer's public keys/configs
88    peer_pub_ready: bool,
89    /// A map from public keys to `(node_index, is_da)`.
90    pub_posted: HashMap<Vec<u8>, (u64, bool)>,
91    /// Whether nodes should start their HotShot instances
92    /// Will be set to true once all nodes post they are ready to start
93    start: bool,
94    /// The total nodes that have posted they are ready to start
95    nodes_connected: HashSet<PeerConfig<TYPES>>,
96    /// The results of the benchmarks
97    bench_results: BenchResults,
98    /// The number of nodes that have posted their results
99    nodes_post_results: u64,
100    /// Whether the orchestrator can be started manually
101    manual_start_allowed: bool,
102    /// Whether we are still accepting new keys for registration
103    accepting_new_keys: bool,
104    /// Builder address pool
105    builders: Vec<Url>,
106    /// whether we are using a fixed stake table, disabling public key registration
107    fixed_stake_table: bool,
108}

110impl<TYPES: NodeType> OrchestratorState<TYPES> {
111    /// create a new [`OrchestratorState`]
112    pub fn new(network_config: NetworkConfig<TYPES>) -> Self {
113        let mut peer_pub_ready = false;
114        let mut fixed_stake_table = false;
115
116        if network_config.config.known_nodes_with_stake.is_empty() {
117            println!(
118                "No nodes were loaded from the config file. Nodes will be allowed to register \
119                 dynamically."
120            );
121        } else {
122            println!("Initializing orchestrator with fixed stake table.");
123            peer_pub_ready = true;
124            fixed_stake_table = true;
125        }
126
127        let builders = if matches!(network_config.builder, BuilderType::External) {
128            network_config.config.builder_urls.clone().into()
129        } else {
130            vec![]
131        };
132
133        OrchestratorState {
134            latest_index: 0,
135            tmp_latest_index: 0,
136            config: network_config,
137            peer_pub_ready,
138            pub_posted: HashMap::new(),
139            nodes_connected: HashSet::new(),
140            start: false,
141            bench_results: BenchResults::default(),
142            nodes_post_results: 0,
143            manual_start_allowed: true,
144            accepting_new_keys: true,
145            builders,
146            fixed_stake_table,
147        }
148    }
149
150    /// Output the results to a csv file according to orchestrator state
151    pub fn output_to_csv(&self) {
152        let output_csv = BenchResultsDownloadConfig {
153            commit_sha: self.config.commit_sha.clone(),
154            total_nodes: self.config.config.num_nodes_with_stake.into(),
155            da_committee_size: self.config.config.da_staked_committee_size,
156            fixed_leader_for_gpuvid: self.config.config.fixed_leader_for_gpuvid,
157            transactions_per_round: self.config.transactions_per_round,
158            transaction_size: self.bench_results.transaction_size_in_bytes,
159            rounds: self.config.rounds,
160            partial_results: self.bench_results.partial_results.clone(),
161            avg_latency_in_sec: self.bench_results.avg_latency_in_sec,
162            minimum_latency_in_sec: self.bench_results.minimum_latency_in_sec,
163            maximum_latency_in_sec: self.bench_results.maximum_latency_in_sec,
164            throughput_bytes_per_sec: self.bench_results.throughput_bytes_per_sec,
165            total_transactions_committed: self.bench_results.total_transactions_committed,
166            total_time_elapsed_in_sec: self.bench_results.total_time_elapsed_in_sec,
167            total_num_views: self.bench_results.total_num_views,
168            failed_num_views: self.bench_results.failed_num_views,
169            committee_type: self.bench_results.committee_type.clone(),
170        };
171        // Open the CSV file in append mode
172        let results_csv_file = OpenOptions::new()
173            .create(true)
174            .append(true) // Open in append mode
175            .open("scripts/benchmarks_results/results.csv")
176            .unwrap();
177        // Open a file for writing
178        let mut wtr = Writer::from_writer(results_csv_file);
179        let _ = wtr.serialize(output_csv);
180        let _ = wtr.flush();
181        println!("Results successfully saved in scripts/benchmarks_results/results.csv");
182    }

248impl<TYPES: NodeType> OrchestratorState<TYPES>
249where
250    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
251{
252    /// register a node with an unknown public key.
253    /// this method should be used when we don't have a fixed stake table
254    fn register_unknown(
255        &mut self,
256        pubkey: &mut Vec<u8>,
257        da_requested: bool,
258        libp2p_address: Option<Multiaddr>,
259        libp2p_public_key: Option<PeerId>,
260    ) -> Result<(u64, bool), ServerError> {
261        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
262            return Ok((*node_index, *is_da));
263        }
264
265        if !self.accepting_new_keys {
266            return Err(ServerError {
267                status: StatusCode::FORBIDDEN,
268                message: "Network has been started manually, and is no longer registering new \
269                          keys."
270                    .to_string(),
271            });
272        }
273
274        let node_index = self.pub_posted.len() as u64;
275
276        // Deserialize the public key
277        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
278
279        self.config
280            .config
281            .known_nodes_with_stake
282            .push(staked_pubkey.clone());
283
284        let mut added_to_da = false;
285
286        let da_full =
287            self.config.config.known_da_nodes.len() >= self.config.config.da_staked_committee_size;
288
289        #[allow(clippy::nonminimal_bool)]
290        // We add the node to the DA committee depending on either its node index or whether it requested membership.
291        //
292        // Since we issue `node_index` incrementally, if we are deciding DA membership by node_index
293        // we only need to check that the DA committee is not yet full.
294        //
295        // Note: this logically simplifies to (self.config.indexed_da || da_requested) && !da_full,
296        // but writing it that way makes it a little less clear to me.
297        if (self.config.indexed_da || (!self.config.indexed_da && da_requested)) && !da_full {
298            self.config.config.known_da_nodes.push(staked_pubkey);
299            added_to_da = true;
300        }
301
302        self.pub_posted
303            .insert(pubkey.clone(), (node_index, added_to_da));
304
305        // If the orchestrator is set up for libp2p and we have supplied the proper
306        // Libp2p data, add our node to the list of bootstrap nodes.
307        if self.config.libp2p_config.clone().is_some()
308            && let (Some(libp2p_public_key), Some(libp2p_address)) =
309                (libp2p_public_key, libp2p_address)
310        {
311            // Push to our bootstrap nodes
312            self.config
313                .libp2p_config
314                .as_mut()
315                .unwrap()
316                .bootstrap_nodes
317                .push((libp2p_public_key, libp2p_address));
318        }
319
320        tracing::error!("Posted public key for node_index {node_index}");
321
322        // node_index starts at 0, so once it matches `num_nodes_with_stake`
323        // we will have registered one node too many. hence, we want `node_index + 1`.
324        if node_index + 1 >= (self.config.config.num_nodes_with_stake.get() as u64) {
325            self.peer_pub_ready = true;
326            self.accepting_new_keys = false;
327        }
328        Ok((node_index, added_to_da))
329    }
330
331    /// register a node on the fixed stake table, which was loaded at startup
332    fn register_from_list(
333        &mut self,
334        pubkey: &mut Vec<u8>,
335        da_requested: bool,
336        libp2p_address: Option<Multiaddr>,
337        libp2p_public_key: Option<PeerId>,
338    ) -> Result<(u64, bool), ServerError> {
339        // if we've already registered this node before, we just retrieve its info from `pub_posted`
340        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
341            return Ok((*node_index, *is_da));
342        }
343
344        // Deserialize the public key
345        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
346
347        // Check if the node is allowed to connect, returning its index and config entry if so.
348        let Some((node_index, node_config)) =
349            self.config.public_keys.iter().enumerate().find(|keys| {
350                keys.1.stake_table_key == staked_pubkey.stake_table_entry.public_key()
351            })
352        else {
353            return Err(ServerError {
354                status: StatusCode::FORBIDDEN,
355                message: "You are unauthorized to register with the orchestrator".to_string(),
356            });
357        };
358
359        // Check that our recorded DA status for the node matches what the node actually requested
360        if node_config.da != da_requested {
361            return Err(ServerError {
362                status: StatusCode::BAD_REQUEST,
363                message: format!(
364                    "Mismatch in DA status in registration for node {}. DA requested: {}, \
365                     expected: {}",
366                    node_index, da_requested, node_config.da
367                ),
368            });
369        }
370
371        let added_to_da = node_config.da;
372
373        self.pub_posted
374            .insert(pubkey.clone(), (node_index as u64, added_to_da));
375
376        // If the orchestrator is set up for libp2p and we have supplied the proper
377        // Libp2p data, add our node to the list of bootstrap nodes.
378        if self.config.libp2p_config.clone().is_some()
379            && let (Some(libp2p_public_key), Some(libp2p_address)) =
380                (libp2p_public_key, libp2p_address)
381        {
382            // Push to our bootstrap nodes
383            self.config
384                .libp2p_config
385                .as_mut()
386                .unwrap()
387                .bootstrap_nodes
388                .push((libp2p_public_key, libp2p_address));
389        }
390
391        tracing::error!("Node {node_index} has registered.");
392
393        Ok((node_index as u64, added_to_da))
394    }
395}

397impl<TYPES: NodeType> OrchestratorApi<TYPES> for OrchestratorState<TYPES>
398where
399    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
400{
401    /// Post an identity to the orchestrator. Takes in optional
402    /// arguments so others can identify us on the Libp2p network.
403    /// # Errors
404    /// If we were unable to serve the request
405    fn post_identity(
406        &mut self,
407        libp2p_address: Option<Multiaddr>,
408        libp2p_public_key: Option<PeerId>,
409    ) -> Result<u16, ServerError> {
410        let node_index = self.latest_index;
411        self.latest_index += 1;
412
413        if usize::from(node_index) >= self.config.config.num_nodes_with_stake.get() {
414            return Err(ServerError {
415                status: StatusCode::BAD_REQUEST,
416                message: "Network has reached capacity".to_string(),
417            });
418        }
419
420        // If the orchestrator is set up for libp2p and we have supplied the proper
421        // Libp2p data, add our node to the list of bootstrap nodes.
422        if self.config.libp2p_config.clone().is_some()
423            && let (Some(libp2p_public_key), Some(libp2p_address)) =
424                (libp2p_public_key, libp2p_address)
425        {
426            // Push to our bootstrap nodes
427            self.config
428                .libp2p_config
429                .as_mut()
430                .unwrap()
431                .bootstrap_nodes
432                .push((libp2p_public_key, libp2p_address));
433        }
434        Ok(node_index)
435    }
436
437    // Assumes nodes will set their own index that they received from the
438    // 'identity' endpoint
439    fn post_getconfig(&mut self, _node_index: u16) -> Result<NetworkConfig<TYPES>, ServerError> {
440        Ok(self.config.clone())
441    }

443    // Assumes one node do not get twice
444    fn get_tmp_node_index(&mut self) -> Result<u16, ServerError> {
445        let tmp_node_index = self.tmp_latest_index;
446        self.tmp_latest_index += 1;
447
448        if usize::from(tmp_node_index) >= self.config.config.num_nodes_with_stake.get() {
449            return Err(ServerError {
450                status: StatusCode::BAD_REQUEST,
451                message: "Node index getter for key pair generation has reached capacity"
452                    .to_string(),
453            });
454        }
455        Ok(tmp_node_index)
456    }

458    fn register_public_key(
459        &mut self,
460        pubkey: &mut Vec<u8>,

462        libp2p_address: Option<Multiaddr>,

467        } else {
468            self.register_unknown(pubkey, da_requested, libp2p_address, libp2p_public_key)
469        }

493    fn get_start(&self) -> Result<bool, ServerError> {
494        // println!("{}", self.start);
495        if !self.start {
496            return Err(ServerError {
497                status: StatusCode::BAD_REQUEST,
498                message: "Network is not ready to start".to_string(),
499            });
500        }
501        Ok(self.start)
502    }
503
504    // Assumes nodes do not post 'ready' twice
505    fn post_ready(&mut self, peer_config: &PeerConfig<TYPES>) -> Result<(), ServerError> {
506        // If we have not disabled registration verification.
507        // Is this node allowed to connect?
508        if !self
509            .config
510            .config
511            .known_nodes_with_stake
512            .contains(peer_config)
513        {
514            return Err(ServerError {
515                status: StatusCode::FORBIDDEN,
516                message: "You are unauthorized to register with the orchestrator".to_string(),
517            });
518        }
519
520        // `HashSet::insert()` returns whether the node was newly inserted (true) or not
521        if self.nodes_connected.insert(peer_config.clone()) {
522            tracing::error!(
523                "Node {peer_config} connected. Total nodes connected: {}",
524                self.nodes_connected.len()
525            );
526        }
527
528        // i.e. nodes_connected >= num_nodes_with_stake * (start_threshold.0 / start_threshold.1)
529        if self.nodes_connected.len() as u64 * self.config.config.start_threshold.1
530            >= (self.config.config.num_nodes_with_stake.get() as u64)
531                * self.config.config.start_threshold.0
532        {
533            self.accepting_new_keys = false;
534            self.manual_start_allowed = false;
535            self.start = true;
536        }
537
538        Ok(())
539    }
540
541    /// Manually start the network
542    fn post_manual_start(&mut self, password_bytes: Vec<u8>) -> Result<(), ServerError> {
543        if !self.manual_start_allowed {
544            return Err(ServerError {
545                status: StatusCode::FORBIDDEN,
546                message: "Configs have already been distributed to nodes, and the network can no \
547                          longer be started manually."
548                    .to_string(),
549            });
550        }
551
552        let password = String::from_utf8(password_bytes)
553            .expect("Failed to decode raw password as UTF-8 string.");
554
555        // Check that the password matches
556        if self.config.manual_start_password != Some(password) {
557            return Err(ServerError {
558                status: StatusCode::FORBIDDEN,
559                message: "Incorrect password.".to_string(),
560            });
561        }
562
563        let registered_nodes_with_stake = self.config.config.known_nodes_with_stake.len();
564        let registered_da_nodes = self.config.config.known_da_nodes.len();
565
566        if registered_da_nodes > 1 {
567            self.config.config.num_nodes_with_stake =
568                std::num::NonZeroUsize::new(registered_nodes_with_stake)
569                    .expect("Failed to convert to NonZeroUsize; this should be impossible.");
570
571            self.config.config.da_staked_committee_size = registered_da_nodes;
572        } else {
573            return Err(ServerError {
574                status: StatusCode::FORBIDDEN,
575                message: format!(
576                    "We cannot manually start the network, because we only have \
577                     {registered_nodes_with_stake} nodes with stake registered, with \
578                     {registered_da_nodes} DA nodes."
579                ),
580            });
581        }
582
583        self.accepting_new_keys = false;
584        self.manual_start_allowed = false;
585        self.peer_pub_ready = true;
586        self.start = true;
587
588        Ok(())
589    }

591    // Aggregates results of the run from all nodes
592    fn post_run_results(&mut self, metrics: BenchResults) -> Result<(), ServerError> {
593        if metrics.total_transactions_committed != 0 {
594            // Deal with the bench results
595            if self.bench_results.total_transactions_committed == 0 {
596                self.bench_results = metrics;
597            } else {
598                // Deal with the bench results from different nodes
599                let cur_metrics = self.bench_results.clone();
600                self.bench_results.avg_latency_in_sec = (metrics.avg_latency_in_sec
601                    * metrics.num_latency
602                    + cur_metrics.avg_latency_in_sec * cur_metrics.num_latency)
603                    / (metrics.num_latency + cur_metrics.num_latency);
604                self.bench_results.num_latency += metrics.num_latency;
605                self.bench_results.minimum_latency_in_sec = metrics
606                    .minimum_latency_in_sec
607                    .min(cur_metrics.minimum_latency_in_sec);
608                self.bench_results.maximum_latency_in_sec = metrics
609                    .maximum_latency_in_sec
610                    .max(cur_metrics.maximum_latency_in_sec);
611                self.bench_results.throughput_bytes_per_sec = metrics
612                    .throughput_bytes_per_sec
613                    .max(cur_metrics.throughput_bytes_per_sec);
614                self.bench_results.total_transactions_committed = metrics
615                    .total_transactions_committed
616                    .max(cur_metrics.total_transactions_committed);
617                self.bench_results.total_time_elapsed_in_sec = metrics
618                    .total_time_elapsed_in_sec
619                    .max(cur_metrics.total_time_elapsed_in_sec);
620                self.bench_results.total_num_views =
621                    metrics.total_num_views.min(cur_metrics.total_num_views);
622                self.bench_results.failed_num_views =
623                    metrics.failed_num_views.max(cur_metrics.failed_num_views);
624            }
625        }
626        self.nodes_post_results += 1;
627        if self.bench_results.partial_results == "Unset" {
628            self.bench_results.partial_results = "One".to_string();
629            self.bench_results.printout();
630            self.output_to_csv();
631        }
632        if self.bench_results.partial_results == "One"
633            && self.nodes_post_results >= (self.config.config.da_staked_committee_size as u64 / 2)
634        {
635            self.bench_results.partial_results = "HalfDA".to_string();
636            self.bench_results.printout();
637            self.output_to_csv();
638        }
639        if self.bench_results.partial_results == "HalfDA"
640            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64 / 2)
641        {
642            self.bench_results.partial_results = "Half".to_string();
643            self.bench_results.printout();
644            self.output_to_csv();
645        }
646        if self.bench_results.partial_results != "Full"
647            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64)
648        {
649            self.bench_results.partial_results = "Full".to_string();
650            self.bench_results.printout();
651            self.output_to_csv();
652        }
653        Ok(())
654    }
655
656    fn post_builder(&mut self, builder: Url) -> Result<(), ServerError> {
657        self.builders.push(builder);
658        Ok(())
659    }
660
661    fn get_builders(&self) -> Result<Vec<Url>, ServerError> {
662        if !matches!(self.config.builder, BuilderType::External)
663            && self.builders.len() != self.config.config.da_staked_committee_size
664        {
665            return Err(ServerError {
666                status: StatusCode::NOT_FOUND,
667                message: "Not all builders are registered yet".to_string(),
668            });
669        }
670        Ok(self.builders.clone())
671    }

674/// Shared, lock-guarded orchestrator state, cloned into every axum handler via `State`.
675type SharedOrchestratorState<TYPES> = Arc<RwLock<OrchestratorState<TYPES>>>;

677/// Wire format of the orchestrator: [`OrchestratorVersion`] VBS framing and the `ServerError`
678/// envelope, which `OrchestratorClient` decodes.
679struct OrchestratorWireFormat;
680
681impl WireFormat for OrchestratorWireFormat {
682    type Error = ServerError;
683    type Version = OrchestratorVersion;
684
685    fn status(err: &ServerError) -> StatusCode {
686        err.status
687    }
688
689    fn serialize_failure(message: String) -> ServerError {
690        ServerError {
691            status: StatusCode::INTERNAL_SERVER_ERROR,
692            message,
693        }
694    }
695}
696
697fn respond<T: Serialize>(headers: &HeaderMap, result: Result<T, ServerError>) -> Response {
698    wire::respond::<OrchestratorWireFormat, _>(headers, result)
699}
700
701fn malformed_body() -> ServerError {
702    ServerError {
703        status: StatusCode::BAD_REQUEST,
704        message: "Malformed body".to_string(),
705    }
706}
707
708/// Decode a POST body from `OrchestratorClient`.
709///
710/// `client.rs` builds these bodies by calling `body_binary` on an already vbs-serialized
711/// `Vec<u8>`, so the wire body is `[4-byte version][8-byte bincode `Vec<u8>` length][inner
712/// vbs-serialized payload]`; the first 12 bytes are an artifact of that double encoding and must
713/// be stripped before decoding the inner payload.
714fn decode_wrapped_body<T: DeserializeOwned>(body: &[u8]) -> Result<T, ServerError> {
715    body.get(12..)
716        .and_then(|inner| Serializer::<OrchestratorVersion>::deserialize(inner).ok())
717        .ok_or_else(malformed_body)
718}
719
720/// Like [`decode_wrapped_body`], but for the `ready` route, whose inner payload is a raw
721/// [`PeerConfig::to_bytes`] blob rather than a second vbs-serialized value.
722fn decode_wrapped_peer_config<TYPES: NodeType>(
723    body: &[u8],
724) -> Result<PeerConfig<TYPES>, ServerError> {
725    body.get(12..)
726        .and_then(PeerConfig::<TYPES>::from_bytes)
727        .ok_or_else(malformed_body)
728}

840async fn post_builder<TYPES: NodeType>(
841    State(state): State<SharedOrchestratorState<TYPES>>,
842    headers: HeaderMap,
843    body: Bytes,
844) -> Response {
845    let result = match decode_wrapped_body::<Vec<Url>>(&body) {
846        Ok(urls) => {
847            let mut reachable = urls
848                .into_iter()
849                .map(|url| async {
850                    let client: http_client::Client<ClientErr, OrchestratorVersion> =
851                        http_client::Client::builder(url.clone()).build();
852                    client
853                        .connect(Some(Duration::from_secs(2)))
854                        .await
855                        .then_some(url)
856                })
857                .collect::<FuturesUnordered<_>>()
858                .filter_map(futures::future::ready);
859            match reachable.next().await {
860                Some(url) => state.write().await.post_builder(url),
861                None => Err(ServerError {
862                    status: StatusCode::BAD_REQUEST,
863                    message: "No reachable addresses".to_string(),
864                }),
865            }
866        },
867        Err(err) => Err(err),
868    };
869    respond(&headers, result)
870}

880/// Builds the `api` module's routes. Existing clients call these both directly (e.g.
881/// `api/identity`) and under a major-version prefix (`v0/api/identity`), so the `/api` tree is
882/// mounted at the root and under `/v0`.
883fn api_router<TYPES: NodeType>() -> Router<SharedOrchestratorState<TYPES>> {
884    Router::new()
885        .route("/healthcheck", get(healthcheck))
886        .route("/identity", post(post_identity::<TYPES>))
887        .route("/config/{node_index}", post(post_getconfig::<TYPES>))
888        .route("/get_tmp_node_index", post(get_tmp_node_index::<TYPES>))
889        .route("/pubkey/{is_da}", post(post_pubkey::<TYPES>))
890        .route("/peer_pub_ready", get(peer_pubconfig_ready::<TYPES>))
891        .route(
892            "/post_config_after_peer_collected",
893            post(post_config_after_peer_collected::<TYPES>),
894        )
895        .route("/ready", post(post_ready::<TYPES>))
896        .route("/start", get(get_start::<TYPES>))
897        .route("/results", post(post_results::<TYPES>))
898        .route("/manual_start", post(post_manual_start::<TYPES>))
899        .route("/builders", get(get_builders::<TYPES>))
900        .route("/builder", post(post_builder::<TYPES>))
901}
902
903/// Builds the full router: the app-level `healthcheck`, plus the `api` module served both
904/// unversioned and under `/v0`. Every response carries permissive CORS headers.
905fn app<TYPES: NodeType>(state: SharedOrchestratorState<TYPES>) -> Router {
906    let api = Router::new().nest("/api", api_router::<TYPES>());
907    Router::new()
908        .route("/healthcheck", get(healthcheck))
909        .merge(api.clone())
910        .nest("/v0", api)
911        .with_state(state)
912        .layer(cors_layer())
913}

915/// Runs the orchestrator
916/// # Errors
917/// This errors if axum runs into an issue during serving
918pub async fn run_orchestrator<TYPES: NodeType>(
919    mut network_config: NetworkConfig<TYPES>,
920    url: Url,
921) -> io::Result<()> {
922    let env_password = std::env::var("ORCHESTRATOR_MANUAL_START_PASSWORD");
923
924    if env_password.is_ok() {
925        tracing::warn!(
926            "Took orchestrator manual start password from the environment variable: \
927             ORCHESTRATOR_MANUAL_START_PASSWORD={:?}",
928            env_password
929        );
930        network_config.manual_start_password = env_password.ok();
931    }
932
933    // Try to overwrite the network_config public keys
934    // from the file the env var points to, or panic.
935    {
936        let env_public_keys = std::env::var("ORCHESTRATOR_PUBLIC_KEYS");
937
938        if let Ok(filepath) = env_public_keys {
939            #[allow(clippy::panic)]
940            let config_file_as_string: String = fs::read_to_string(filepath.clone())
941                .unwrap_or_else(|_| panic!("Could not read config file located at {filepath}"));
942
943            let file: PublicKeysFile<TYPES> =
944                toml::from_str::<PublicKeysFile<TYPES>>(&config_file_as_string)
945                    .expect("Unable to convert config file to TOML");
946
947            network_config.public_keys = file.public_keys;
948        }
949    }
950
951    network_config.config.known_nodes_with_stake = network_config
952        .public_keys
953        .iter()
954        .map(|keys| PeerConfig {
955            stake_table_entry: keys
956                .stake_table_key
957                .stake_table_entry(U256::from(keys.stake)),
958            state_ver_key: keys.state_ver_key.clone(),
959            connect_info: keys.connect_info.clone(),
960        })
961        .collect();
962
963    network_config.config.known_da_nodes = network_config
964        .public_keys
965        .iter()
966        .filter(|keys| keys.da)
967        .map(|keys| PeerConfig {
968            stake_table_entry: keys
969                .stake_table_key
970                .stake_table_entry(U256::from(keys.stake)),
971            state_ver_key: keys.state_ver_key.clone(),
972            connect_info: keys.connect_info.clone(),
973        })
974        .collect();
975
976    let state: SharedOrchestratorState<TYPES> =
977        Arc::new(RwLock::new(OrchestratorState::new(network_config)));
978    let app = app::<TYPES>(state);
979
980    let host = url
981        .host_str()
982        .ok_or_else(|| io::Error::other(format!("orchestrator url missing host: {url}")))?;
983    let port = url
984        .port_or_known_default()
985        .ok_or_else(|| io::Error::other(format!("orchestrator url missing port: {url}")))?;
986    let listener = TcpListener::bind((host, port)).await?;
987
988    tracing::error!("listening on {url:?}");
989    axum::serve(listener, app).await
990}
