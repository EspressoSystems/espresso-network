52/// The state of the orchestrator
53#[derive(Default, Clone)]
54#[allow(clippy::struct_excessive_bools)]
55struct OrchestratorState<TYPES: NodeType> {
56    /// Tracks the latest node index we have generated a configuration for
57    latest_index: u16,
58    /// Tracks the latest temporary index we have generated for init validator's key pair
59    tmp_latest_index: u16,
60    /// The network configuration
61    config: NetworkConfig<TYPES>,
62    /// Whether the network configuration has been updated with all the peer's public keys/configs
63    peer_pub_ready: bool,
64    /// A map from public keys to `(node_index, is_da)`.
65    pub_posted: HashMap<Vec<u8>, (u64, bool)>,
66    /// Whether nodes should start their HotShot instances
67    /// Will be set to true once all nodes post they are ready to start
68    start: bool,
69    /// The total nodes that have posted they are ready to start
70    nodes_connected: HashSet<PeerConfig<TYPES>>,
71    /// The results of the benchmarks
72    bench_results: BenchResults,
73    /// The number of nodes that have posted their results
74    nodes_post_results: u64,
75    /// Whether the orchestrator can be started manually
76    manual_start_allowed: bool,
77    /// Whether we are still accepting new keys for registration
78    accepting_new_keys: bool,
79    /// whether we are using a fixed stake table, disabling public key registration
80    fixed_stake_table: bool,
81}
82
83impl<TYPES: NodeType> OrchestratorState<TYPES> {
84    /// create a new [`OrchestratorState`]
85    pub fn new(network_config: NetworkConfig<TYPES>) -> Self {
86        let mut peer_pub_ready = false;
87        let mut fixed_stake_table = false;
88
89        if network_config.config.known_nodes_with_stake.is_empty() {
90            println!(
91                "No nodes were loaded from the config file. Nodes will be allowed to register \
92                 dynamically."
93            );
94        } else {
95            println!("Initializing orchestrator with fixed stake table.");
96            peer_pub_ready = true;
97            fixed_stake_table = true;
98        }
99
100        OrchestratorState {
101            latest_index: 0,
102            tmp_latest_index: 0,
103            config: network_config,
104            peer_pub_ready,
105            pub_posted: HashMap::new(),
106            nodes_connected: HashSet::new(),
107            start: false,
108            bench_results: BenchResults::default(),
109            nodes_post_results: 0,
110            manual_start_allowed: true,
111            accepting_new_keys: true,
112            fixed_stake_table,
113        }
114    }
115
116    /// Output the results to a csv file according to orchestrator state
117    pub fn output_to_csv(&self) {
118        let output_csv = BenchResultsDownloadConfig {
119            commit_sha: self.config.commit_sha.clone(),
120            total_nodes: self.config.config.num_nodes_with_stake.into(),
121            da_committee_size: self.config.config.da_staked_committee_size,
122            fixed_leader_for_gpuvid: self.config.config.fixed_leader_for_gpuvid,
123            transactions_per_round: self.config.transactions_per_round,
124            transaction_size: self.bench_results.transaction_size_in_bytes,
125            rounds: self.config.rounds,
126            partial_results: self.bench_results.partial_results.clone(),
127            avg_latency_in_sec: self.bench_results.avg_latency_in_sec,
128            minimum_latency_in_sec: self.bench_results.minimum_latency_in_sec,
129            maximum_latency_in_sec: self.bench_results.maximum_latency_in_sec,
130            throughput_bytes_per_sec: self.bench_results.throughput_bytes_per_sec,
131            total_transactions_committed: self.bench_results.total_transactions_committed,
132            total_time_elapsed_in_sec: self.bench_results.total_time_elapsed_in_sec,
133            total_num_views: self.bench_results.total_num_views,
134            failed_num_views: self.bench_results.failed_num_views,
135            committee_type: self.bench_results.committee_type.clone(),
136        };
137        // Open the CSV file in append mode
138        let results_csv_file = OpenOptions::new()
139            .create(true)
140            .append(true) // Open in append mode
141            .open("scripts/benchmarks_results/results.csv")
142            .unwrap();
143        // Open a file for writing
144        let mut wtr = Writer::from_writer(results_csv_file);
145        let _ = wtr.serialize(output_csv);
146        let _ = wtr.flush();
147        println!("Results successfully saved in scripts/benchmarks_results/results.csv");
148    }
149}

152pub trait OrchestratorApi<TYPES: NodeType> {
153    /// Post an identity to the orchestrator.
154    /// # Errors
155    /// If we were unable to serve the request
156    fn post_identity(&mut self) -> Result<u16, ServerError>;
157    /// post endpoint for each node's config
158    /// # Errors
159    /// if unable to serve
160    fn post_getconfig(&mut self, _node_index: u16) -> Result<NetworkConfig<TYPES>, ServerError>;
161    /// get endpoint for the next available temporary node index
162    /// # Errors
163    /// if unable to serve
164    fn get_tmp_node_index(&mut self) -> Result<u16, ServerError>;
165    /// post endpoint for each node's public key
166    /// # Errors
167    /// if unable to serve
168    fn register_public_key(
169        &mut self,
170        pubkey: &mut Vec<u8>,
171        is_da: bool,
172    ) -> Result<(u64, bool), ServerError>;
173    /// post endpoint for whether or not all peers public keys are ready
174    /// # Errors
175    /// if unable to serve
176    fn peer_pub_ready(&self) -> Result<bool, ServerError>;
177    /// get endpoint for the network config after all peers public keys are collected
178    /// # Errors
179    /// if unable to serve
180    fn post_config_after_peer_collected(&mut self) -> Result<NetworkConfig<TYPES>, ServerError>;
181    /// get endpoint for whether or not the run has started
182    /// # Errors
183    /// if unable to serve
184    fn get_start(&self) -> Result<bool, ServerError>;
185    /// post endpoint for the results of the run
186    /// # Errors
187    /// if unable to serve
188    fn post_run_results(&mut self, metrics: BenchResults) -> Result<(), ServerError>;
189    /// A node POSTs its public key to let the orchestrator know that it is ready
190    /// # Errors
191    /// if unable to serve
192    fn post_ready(&mut self, peer_config: &PeerConfig<TYPES>) -> Result<(), ServerError>;
193    /// post endpoint for manually starting the orchestrator
194    /// # Errors
195    /// if unable to serve
196    fn post_manual_start(&mut self, password_bytes: Vec<u8>) -> Result<(), ServerError>;
197}

199impl<TYPES: NodeType> OrchestratorState<TYPES>
200where
201    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
202{
203    /// register a node with an unknown public key.
204    /// this method should be used when we don't have a fixed stake table
205    fn register_unknown(
206        &mut self,
207        pubkey: &mut Vec<u8>,
208        da_requested: bool,
209    ) -> Result<(u64, bool), ServerError> {
210        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
211            return Ok((*node_index, *is_da));
212        }
213
214        if !self.accepting_new_keys {
215            return Err(ServerError {
216                status: StatusCode::FORBIDDEN,
217                message: "Network has been started manually, and is no longer registering new \
218                          keys."
219                    .to_string(),
220            });
221        }
222
223        let node_index = self.pub_posted.len() as u64;
224
225        // Deserialize the public key
226        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
227
228        self.config
229            .config
230            .known_nodes_with_stake
231            .push(staked_pubkey.clone());
232
233        let mut added_to_da = false;
234
235        let da_full =
236            self.config.config.known_da_nodes.len() >= self.config.config.da_staked_committee_size;
237
238        #[allow(clippy::nonminimal_bool)]
239        // We add the node to the DA committee depending on either its node index or whether it requested membership.
240        //
241        // Since we issue `node_index` incrementally, if we are deciding DA membership by node_index
242        // we only need to check that the DA committee is not yet full.
243        //
244        // Note: this logically simplifies to (self.config.indexed_da || da_requested) && !da_full,
245        // but writing it that way makes it a little less clear to me.
246        if (self.config.indexed_da || (!self.config.indexed_da && da_requested)) && !da_full {
247            self.config.config.known_da_nodes.push(staked_pubkey);
248            added_to_da = true;
249        }
250
251        self.pub_posted
252            .insert(pubkey.clone(), (node_index, added_to_da));
253
254        tracing::error!("Posted public key for node_index {node_index}");
255
256        // node_index starts at 0, so once it matches `num_nodes_with_stake`
257        // we will have registered one node too many. hence, we want `node_index + 1`.
258        if node_index + 1 >= (self.config.config.num_nodes_with_stake.get() as u64) {
259            self.peer_pub_ready = true;
260            self.accepting_new_keys = false;
261        }
262        Ok((node_index, added_to_da))
263    }
264
265    /// register a node on the fixed stake table, which was loaded at startup
266    fn register_from_list(
267        &mut self,
268        pubkey: &mut Vec<u8>,
269        da_requested: bool,
270    ) -> Result<(u64, bool), ServerError> {
271        // if we've already registered this node before, we just retrieve its info from `pub_posted`
272        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
273            return Ok((*node_index, *is_da));
274        }
275
276        // Deserialize the public key
277        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
278
279        // Check if the node is allowed to connect, returning its index and config entry if so.
280        let Some((node_index, node_config)) =
281            self.config.public_keys.iter().enumerate().find(|keys| {
282                keys.1.stake_table_key == staked_pubkey.stake_table_entry.public_key()
283            })
284        else {
285            return Err(ServerError {
286                status: StatusCode::FORBIDDEN,
287                message: "You are unauthorized to register with the orchestrator".to_string(),
288            });
289        };
290
291        // Check that our recorded DA status for the node matches what the node actually requested
292        if node_config.da != da_requested {
293            return Err(ServerError {
294                status: StatusCode::BAD_REQUEST,
295                message: format!(
296                    "Mismatch in DA status in registration for node {}. DA requested: {}, \
297                     expected: {}",
298                    node_index, da_requested, node_config.da
299                ),
300            });
301        }
302
303        let added_to_da = node_config.da;
304
305        self.pub_posted
306            .insert(pubkey.clone(), (node_index as u64, added_to_da));
307
308        tracing::error!("Node {node_index} has registered.");
309
310        Ok((node_index as u64, added_to_da))
311    }
312}

314impl<TYPES: NodeType> OrchestratorApi<TYPES> for OrchestratorState<TYPES>
315where
316    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
317{
318    fn post_identity(&mut self) -> Result<u16, ServerError> {
319        let node_index = self.latest_index;
320        self.latest_index += 1;
321
322        if usize::from(node_index) >= self.config.config.num_nodes_with_stake.get() {
323            return Err(ServerError {
324                status: StatusCode::BAD_REQUEST,
325                message: "Network has reached capacity".to_string(),
326            });
327        }
328        Ok(node_index)
329    }
330
331    // Assumes nodes will set their own index that they received from the
332    // 'identity' endpoint
333    fn post_getconfig(&mut self, _node_index: u16) -> Result<NetworkConfig<TYPES>, ServerError> {
334        Ok(self.config.clone())
335    }
336
337    // Assumes one node do not get twice
338    fn get_tmp_node_index(&mut self) -> Result<u16, ServerError> {
339        let tmp_node_index = self.tmp_latest_index;
340        self.tmp_latest_index += 1;
341
342        if usize::from(tmp_node_index) >= self.config.config.num_nodes_with_stake.get() {
343            return Err(ServerError {
344                status: StatusCode::BAD_REQUEST,
345                message: "Node index getter for key pair generation has reached capacity"
346                    .to_string(),
347            });
348        }
349        Ok(tmp_node_index)
350    }
351
352    fn register_public_key(
353        &mut self,
354        pubkey: &mut Vec<u8>,
355        da_requested: bool,
356    ) -> Result<(u64, bool), ServerError> {
357        if self.fixed_stake_table {
358            self.register_from_list(pubkey, da_requested)
359        } else {
360            self.register_unknown(pubkey, da_requested)
361        }
362    }
363
364    fn peer_pub_ready(&self) -> Result<bool, ServerError> {
365        if !self.peer_pub_ready {
366            return Err(ServerError {
367                status: StatusCode::BAD_REQUEST,
368                message: "Peer's public configs are not ready".to_string(),
369            });
370        }
371        Ok(self.peer_pub_ready)
372    }
373
374    fn post_config_after_peer_collected(&mut self) -> Result<NetworkConfig<TYPES>, ServerError> {
375        if !self.peer_pub_ready {
376            return Err(ServerError {
377                status: StatusCode::BAD_REQUEST,
378                message: "Peer's public configs are not ready".to_string(),
379            });
380        }
381
382        Ok(self.config.clone())
383    }
384
385    fn get_start(&self) -> Result<bool, ServerError> {
386        // println!("{}", self.start);
387        if !self.start {
388            return Err(ServerError {
389                status: StatusCode::BAD_REQUEST,
390                message: "Network is not ready to start".to_string(),
391            });
392        }
393        Ok(self.start)
394    }
395
396    // Assumes nodes do not post 'ready' twice
397    fn post_ready(&mut self, peer_config: &PeerConfig<TYPES>) -> Result<(), ServerError> {
398        // If we have not disabled registration verification.
399        // Is this node allowed to connect?
400        if !self
401            .config
402            .config
403            .known_nodes_with_stake
404            .contains(peer_config)
405        {
406            return Err(ServerError {
407                status: StatusCode::FORBIDDEN,
408                message: "You are unauthorized to register with the orchestrator".to_string(),
409            });
410        }
411
412        // `HashSet::insert()` returns whether the node was newly inserted (true) or not
413        if self.nodes_connected.insert(peer_config.clone()) {
414            tracing::error!(
415                "Node {peer_config} connected. Total nodes connected: {}",
416                self.nodes_connected.len()
417            );
418        }
419
420        // i.e. nodes_connected >= num_nodes_with_stake * (start_threshold.0 / start_threshold.1)
421        if self.nodes_connected.len() as u64 * self.config.config.start_threshold.1
422            >= (self.config.config.num_nodes_with_stake.get() as u64)
423                * self.config.config.start_threshold.0
424        {
425            self.accepting_new_keys = false;
426            self.manual_start_allowed = false;
427            self.start = true;
428        }
429
430        Ok(())
431    }
432
433    /// Manually start the network
434    fn post_manual_start(&mut self, password_bytes: Vec<u8>) -> Result<(), ServerError> {
435        if !self.manual_start_allowed {
436            return Err(ServerError {
437                status: StatusCode::FORBIDDEN,
438                message: "Configs have already been distributed to nodes, and the network can no \
439                          longer be started manually."
440                    .to_string(),
441            });
442        }
443
444        let password = String::from_utf8(password_bytes)
445            .expect("Failed to decode raw password as UTF-8 string.");
446
447        // Check that the password matches
448        if self.config.manual_start_password != Some(password) {
449            return Err(ServerError {
450                status: StatusCode::FORBIDDEN,
451                message: "Incorrect password.".to_string(),
452            });
453        }
454
455        let registered_nodes_with_stake = self.config.config.known_nodes_with_stake.len();
456        let registered_da_nodes = self.config.config.known_da_nodes.len();
457
458        if registered_da_nodes > 1 {
459            self.config.config.num_nodes_with_stake =
460                std::num::NonZeroUsize::new(registered_nodes_with_stake)
461                    .expect("Failed to convert to NonZeroUsize; this should be impossible.");
462
463            self.config.config.da_staked_committee_size = registered_da_nodes;
464        } else {
465            return Err(ServerError {
466                status: StatusCode::FORBIDDEN,
467                message: format!(
468                    "We cannot manually start the network, because we only have \
469                     {registered_nodes_with_stake} nodes with stake registered, with \
470                     {registered_da_nodes} DA nodes."
471                ),
472            });
473        }
474
475        self.accepting_new_keys = false;
476        self.manual_start_allowed = false;
477        self.peer_pub_ready = true;
478        self.start = true;
479
480        Ok(())
481    }
482
483    // Aggregates results of the run from all nodes
484    fn post_run_results(&mut self, metrics: BenchResults) -> Result<(), ServerError> {
485        if metrics.total_transactions_committed != 0 {
486            // Deal with the bench results
487            if self.bench_results.total_transactions_committed == 0 {
488                self.bench_results = metrics;
489            } else {
490                // Deal with the bench results from different nodes
491                let cur_metrics = self.bench_results.clone();
492                self.bench_results.avg_latency_in_sec = (metrics.avg_latency_in_sec
493                    * metrics.num_latency
494                    + cur_metrics.avg_latency_in_sec * cur_metrics.num_latency)
495                    / (metrics.num_latency + cur_metrics.num_latency);
496                self.bench_results.num_latency += metrics.num_latency;
497                self.bench_results.minimum_latency_in_sec = metrics
498                    .minimum_latency_in_sec
499                    .min(cur_metrics.minimum_latency_in_sec);
500                self.bench_results.maximum_latency_in_sec = metrics
501                    .maximum_latency_in_sec
502                    .max(cur_metrics.maximum_latency_in_sec);
503                self.bench_results.throughput_bytes_per_sec = metrics
504                    .throughput_bytes_per_sec
505                    .max(cur_metrics.throughput_bytes_per_sec);
506                self.bench_results.total_transactions_committed = metrics
507                    .total_transactions_committed
508                    .max(cur_metrics.total_transactions_committed);
509                self.bench_results.total_time_elapsed_in_sec = metrics
510                    .total_time_elapsed_in_sec
511                    .max(cur_metrics.total_time_elapsed_in_sec);
512                self.bench_results.total_num_views =
513                    metrics.total_num_views.min(cur_metrics.total_num_views);
514                self.bench_results.failed_num_views =
515                    metrics.failed_num_views.max(cur_metrics.failed_num_views);
516            }
517        }
518        self.nodes_post_results += 1;
519        if self.bench_results.partial_results == "Unset" {
520            self.bench_results.partial_results = "One".to_string();
521            self.bench_results.printout();
522            self.output_to_csv();
523        }
524        if self.bench_results.partial_results == "One"
525            && self.nodes_post_results >= (self.config.config.da_staked_committee_size as u64 / 2)
526        {
527            self.bench_results.partial_results = "HalfDA".to_string();
528            self.bench_results.printout();
529            self.output_to_csv();
530        }
531        if self.bench_results.partial_results == "HalfDA"
532            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64 / 2)
533        {
534            self.bench_results.partial_results = "Half".to_string();
535            self.bench_results.printout();
536            self.output_to_csv();
537        }
538        if self.bench_results.partial_results != "Full"
539            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64)
540        {
541            self.bench_results.partial_results = "Full".to_string();
542            self.bench_results.printout();
543            self.output_to_csv();
544        }
545        Ok(())
546    }
547}

552fn malformed_body() -> ServerError {
553    ServerError {
554        status: StatusCode::BAD_REQUEST,
555        message: "Malformed body".to_string(),
556    }
557}
558
559/// Decode a POST body from `OrchestratorClient`.
560///
561/// `client.rs` builds these bodies by calling `body_binary` on an already vbs-serialized
562/// `Vec<u8>`, so the wire body is `[4-byte version][8-byte bincode `Vec<u8>` length][inner
563/// vbs-serialized payload]`; the first 12 bytes are an artifact of that double encoding and must
564/// be stripped before decoding the inner payload.
565fn decode_wrapped_body<T: DeserializeOwned>(body: &[u8]) -> Result<T, ServerError> {
566    body.get(12..)
567        .and_then(|inner| Serializer::<OrchestratorVersion>::deserialize(inner).ok())
568        .ok_or_else(malformed_body)
569}
570
571/// Like [`decode_wrapped_body`], but for the `ready` route, whose inner payload is a raw
572/// [`PeerConfig::to_bytes`] blob rather than a second vbs-serialized value.
573fn decode_wrapped_peer_config<TYPES: NodeType>(
574    body: &[u8],
575) -> Result<PeerConfig<TYPES>, ServerError> {
576    body.get(12..)
577        .and_then(PeerConfig::<TYPES>::from_bytes)
578        .ok_or_else(malformed_body)
579}

610async fn post_pubkey<TYPES: NodeType>(
611    State(state): State<SharedOrchestratorState<TYPES>>,
612    Path(is_da): Path<bool>,
613    headers: HeaderMap,
614    body: Bytes,
615) -> Response {
616    let result = match decode_wrapped_body::<Vec<u8>>(&body) {

618        Err(err) => Err(err),
619    };
620    wire::respond(&headers, result)
621}

681/// Builds the `api` module's routes. Existing clients call these both directly (e.g.
682/// `api/identity`) and under a major-version prefix (`v0/api/identity`), so the `/api` tree is
683/// mounted at the root and under `/v0`.
684fn api_router<TYPES: NodeType>() -> Router<SharedOrchestratorState<TYPES>> {
685    Router::new()
686        .route("/healthcheck", get(healthcheck))
687        .route("/identity", post(post_identity::<TYPES>))
688        .route("/config/{node_index}", post(post_getconfig::<TYPES>))
689        .route("/get_tmp_node_index", post(get_tmp_node_index::<TYPES>))
690        .route("/pubkey/{is_da}", post(post_pubkey::<TYPES>))
691        .route("/peer_pub_ready", get(peer_pubconfig_ready::<TYPES>))
692        .route(
693            "/post_config_after_peer_collected",
694            post(post_config_after_peer_collected::<TYPES>),
695        )
696        .route("/ready", post(post_ready::<TYPES>))
697        .route("/start", get(get_start::<TYPES>))
698        .route("/results", post(post_results::<TYPES>))
699        .route("/manual_start", post(post_manual_start::<TYPES>))
700}
701
702/// Builds the full router: the app-level `healthcheck`, plus the `api` module served both
703/// unversioned and under `/v0`. Every response carries permissive CORS headers.
704fn app<TYPES: NodeType>(state: SharedOrchestratorState<TYPES>) -> Router {
705    let api = Router::new().nest("/api", api_router::<TYPES>());
706    Router::new()
707        .route("/healthcheck", get(healthcheck))
708        .merge(api.clone())
709        .nest("/v0", api)
710        .with_state(state)
711        .layer(cors_layer())
712}

714/// Runs the orchestrator
715/// # Errors
716/// This errors if axum runs into an issue during serving
717pub async fn run_orchestrator<TYPES: NodeType>(
718    mut network_config: NetworkConfig<TYPES>,
719    url: Url,
720) -> io::Result<()> {
721    let env_password = std::env::var("ORCHESTRATOR_MANUAL_START_PASSWORD");
722
723    if env_password.is_ok() {
724        tracing::warn!(
725            "Took orchestrator manual start password from the environment variable: \
726             ORCHESTRATOR_MANUAL_START_PASSWORD={:?}",
727            env_password
728        );
729        network_config.manual_start_password = env_password.ok();
730    }
731
732    // Try to overwrite the network_config public keys
733    // from the file the env var points to, or panic.
734    {
735        let env_public_keys = std::env::var("ORCHESTRATOR_PUBLIC_KEYS");
736
737        if let Ok(filepath) = env_public_keys {
738            #[allow(clippy::panic)]
739            let config_file_as_string: String = fs::read_to_string(filepath.clone())
740                .unwrap_or_else(|_| panic!("Could not read config file located at {filepath}"));
741
742            let file: PublicKeysFile<TYPES> =
743                toml::from_str::<PublicKeysFile<TYPES>>(&config_file_as_string)
744                    .expect("Unable to convert config file to TOML");
745
746            network_config.public_keys = file.public_keys;
747        }
748    }
749
750    network_config.config.known_nodes_with_stake = network_config
751        .public_keys
752        .iter()
753        .map(|keys| PeerConfig {
754            stake_table_entry: keys
755                .stake_table_key
756                .stake_table_entry(U256::from(keys.stake)),
757            state_ver_key: keys.state_ver_key.clone(),
758            connect_info: keys.connect_info.clone(),
759        })
760        .collect();
761
762    network_config.config.known_da_nodes = network_config
763        .public_keys
764        .iter()
765        .filter(|keys| keys.da)
766        .map(|keys| PeerConfig {
767            stake_table_entry: keys
768                .stake_table_key
769                .stake_table_entry(U256::from(keys.stake)),
770            state_ver_key: keys.state_ver_key.clone(),
771            connect_info: keys.connect_info.clone(),
772        })
773        .collect();
774
775    let state: SharedOrchestratorState<TYPES> =
776        Arc::new(RwLock::new(OrchestratorState::new(network_config)));
777    let app = app::<TYPES>(state);
778
779    let host = url
780        .host_str()
781        .ok_or_else(|| io::Error::other(format!("orchestrator url missing host: {url}")))?;
782    let port = url
783        .port_or_known_default()
784        .ok_or_else(|| io::Error::other(format!("orchestrator url missing port: {url}")))?;
785    let listener = TcpListener::bind((host, port)).await?;
786
787    tracing::error!("listening on {url:?}");
788    axum::serve(listener, app).await
789}
