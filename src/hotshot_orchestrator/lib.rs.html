65/// Generate an keypair based on a `seed` and an `index`
66/// # Panics
67/// This panics if libp2p is unable to generate a secret key from the seed
68#[must_use]
69pub fn libp2p_generate_indexed_identity(seed: [u8; 32], index: u64) -> Keypair {
70    let mut hasher = blake3::Hasher::new();
71    hasher.update(&seed);
72    hasher.update(&index.to_le_bytes());
73    let new_seed = *hasher.finalize().as_bytes();
74    let sk_bytes = SecretKey::try_from_bytes(new_seed).unwrap();
75    <EdKeypair as From<SecretKey>>::from(sk_bytes).into()
76}

78/// The state of the orchestrator
79#[derive(Default, Clone)]
80#[allow(clippy::struct_excessive_bools)]
81struct OrchestratorState<TYPES: NodeType> {
82    /// Tracks the latest node index we have generated a configuration for
83    latest_index: u16,
84    /// Tracks the latest temporary index we have generated for init validator's key pair
85    tmp_latest_index: u16,
86    /// The network configuration
87    config: NetworkConfig<TYPES>,
88    /// Whether the network configuration has been updated with all the peer's public keys/configs
89    peer_pub_ready: bool,
90    /// A map from public keys to `(node_index, is_da)`.
91    pub_posted: HashMap<Vec<u8>, (u64, bool)>,
92    /// Whether nodes should start their HotShot instances
93    /// Will be set to true once all nodes post they are ready to start
94    start: bool,
95    /// The total nodes that have posted they are ready to start
96    nodes_connected: HashSet<PeerConfig<TYPES>>,
97    /// The results of the benchmarks
98    bench_results: BenchResults,
99    /// The number of nodes that have posted their results
100    nodes_post_results: u64,
101    /// Whether the orchestrator can be started manually
102    manual_start_allowed: bool,
103    /// Whether we are still accepting new keys for registration
104    accepting_new_keys: bool,
105    /// Builder address pool
106    builders: Vec<Url>,
107    /// whether we are using a fixed stake table, disabling public key registration
108    fixed_stake_table: bool,
109}

111impl<TYPES: NodeType> OrchestratorState<TYPES> {
112    /// create a new [`OrchestratorState`]
113    pub fn new(network_config: NetworkConfig<TYPES>) -> Self {
114        let mut peer_pub_ready = false;
115        let mut fixed_stake_table = false;
116
117        if network_config.config.known_nodes_with_stake.is_empty() {
118            println!(
119                "No nodes were loaded from the config file. Nodes will be allowed to register \
120                 dynamically."
121            );
122        } else {
123            println!("Initializing orchestrator with fixed stake table.");
124            peer_pub_ready = true;
125            fixed_stake_table = true;
126        }
127
128        let builders = if matches!(network_config.builder, BuilderType::External) {
129            network_config.config.builder_urls.clone().into()
130        } else {
131            vec![]
132        };
133
134        OrchestratorState {
135            latest_index: 0,
136            tmp_latest_index: 0,
137            config: network_config,
138            peer_pub_ready,
139            pub_posted: HashMap::new(),
140            nodes_connected: HashSet::new(),
141            start: false,
142            bench_results: BenchResults::default(),
143            nodes_post_results: 0,
144            manual_start_allowed: true,
145            accepting_new_keys: true,
146            builders,
147            fixed_stake_table,
148        }
149    }

151    /// Output the results to a csv file according to orchestrator state
152    pub fn output_to_csv(&self) {
153        let output_csv = BenchResultsDownloadConfig {
154            commit_sha: self.config.commit_sha.clone(),
155            total_nodes: self.config.config.num_nodes_with_stake.into(),
156            da_committee_size: self.config.config.da_staked_committee_size,
157            fixed_leader_for_gpuvid: self.config.config.fixed_leader_for_gpuvid,
158            transactions_per_round: self.config.transactions_per_round,
159            transaction_size: self.bench_results.transaction_size_in_bytes,
160            rounds: self.config.rounds,
161            partial_results: self.bench_results.partial_results.clone(),
162            avg_latency_in_sec: self.bench_results.avg_latency_in_sec,
163            minimum_latency_in_sec: self.bench_results.minimum_latency_in_sec,
164            maximum_latency_in_sec: self.bench_results.maximum_latency_in_sec,
165            throughput_bytes_per_sec: self.bench_results.throughput_bytes_per_sec,
166            total_transactions_committed: self.bench_results.total_transactions_committed,
167            total_time_elapsed_in_sec: self.bench_results.total_time_elapsed_in_sec,
168            total_num_views: self.bench_results.total_num_views,
169            failed_num_views: self.bench_results.failed_num_views,
170            committee_type: self.bench_results.committee_type.clone(),
171        };
172        // Open the CSV file in append mode
173        let results_csv_file = OpenOptions::new()
174            .create(true)
175            .append(true) // Open in append mode
176            .open("scripts/benchmarks_results/results.csv")
177            .unwrap();
178        // Open a file for writing
179        let mut wtr = Writer::from_writer(results_csv_file);
180        let _ = wtr.serialize(output_csv);
181        let _ = wtr.flush();
182        println!("Results successfully saved in scripts/benchmarks_results/results.csv");
183    }

249impl<TYPES: NodeType> OrchestratorState<TYPES>
250where
251    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
252{
253    /// register a node with an unknown public key.
254    /// this method should be used when we don't have a fixed stake table
255    fn register_unknown(
256        &mut self,
257        pubkey: &mut Vec<u8>,
258        da_requested: bool,
259        libp2p_address: Option<Multiaddr>,
260        libp2p_public_key: Option<PeerId>,
261    ) -> Result<(u64, bool), ServerError> {
262        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
263            return Ok((*node_index, *is_da));
264        }
265
266        if !self.accepting_new_keys {
267            return Err(ServerError {
268                status: tide_disco::StatusCode::FORBIDDEN,
269                message: "Network has been started manually, and is no longer registering new \
270                          keys."
271                    .to_string(),
272            });
273        }
274
275        let node_index = self.pub_posted.len() as u64;
276
277        // Deserialize the public key
278        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
279
280        self.config
281            .config
282            .known_nodes_with_stake
283            .push(staked_pubkey.clone());
284
285        let mut added_to_da = false;
286
287        let da_full =
288            self.config.config.known_da_nodes.len() >= self.config.config.da_staked_committee_size;
289
290        #[allow(clippy::nonminimal_bool)]
291        // We add the node to the DA committee depending on either its node index or whether it requested membership.
292        //
293        // Since we issue `node_index` incrementally, if we are deciding DA membership by node_index
294        // we only need to check that the DA committee is not yet full.
295        //
296        // Note: this logically simplifies to (self.config.indexed_da || da_requested) && !da_full,
297        // but writing it that way makes it a little less clear to me.
298        if (self.config.indexed_da || (!self.config.indexed_da && da_requested)) && !da_full {
299            self.config.config.known_da_nodes.push(staked_pubkey);
300            added_to_da = true;
301        }
302
303        self.pub_posted
304            .insert(pubkey.clone(), (node_index, added_to_da));
305
306        // If the orchestrator is set up for libp2p and we have supplied the proper
307        // Libp2p data, add our node to the list of bootstrap nodes.
308        if self.config.libp2p_config.clone().is_some()
309            && let (Some(libp2p_public_key), Some(libp2p_address)) =
310                (libp2p_public_key, libp2p_address)
311        {
312            // Push to our bootstrap nodes
313            self.config
314                .libp2p_config
315                .as_mut()
316                .unwrap()
317                .bootstrap_nodes
318                .push((libp2p_public_key, libp2p_address));
319        }
320
321        tracing::error!("Posted public key for node_index {node_index}");
322
323        // node_index starts at 0, so once it matches `num_nodes_with_stake`
324        // we will have registered one node too many. hence, we want `node_index + 1`.
325        if node_index + 1 >= (self.config.config.num_nodes_with_stake.get() as u64) {
326            self.peer_pub_ready = true;
327            self.accepting_new_keys = false;
328        }
329        Ok((node_index, added_to_da))
330    }
331
332    /// register a node on the fixed stake table, which was loaded at startup
333    fn register_from_list(
334        &mut self,
335        pubkey: &mut Vec<u8>,
336        da_requested: bool,
337        libp2p_address: Option<Multiaddr>,
338        libp2p_public_key: Option<PeerId>,
339    ) -> Result<(u64, bool), ServerError> {
340        // if we've already registered this node before, we just retrieve its info from `pub_posted`
341        if let Some((node_index, is_da)) = self.pub_posted.get(pubkey) {
342            return Ok((*node_index, *is_da));
343        }
344
345        // Deserialize the public key
346        let staked_pubkey = PeerConfig::<TYPES>::from_bytes(pubkey).unwrap();
347
348        // Check if the node is allowed to connect, returning its index and config entry if so.
349        let Some((node_index, node_config)) =
350            self.config.public_keys.iter().enumerate().find(|keys| {
351                keys.1.stake_table_key == staked_pubkey.stake_table_entry.public_key()
352            })
353        else {
354            return Err(ServerError {
355                status: tide_disco::StatusCode::FORBIDDEN,
356                message: "You are unauthorized to register with the orchestrator".to_string(),
357            });
358        };
359
360        // Check that our recorded DA status for the node matches what the node actually requested
361        if node_config.da != da_requested {
362            return Err(ServerError {
363                status: tide_disco::StatusCode::BAD_REQUEST,
364                message: format!(
365                    "Mismatch in DA status in registration for node {}. DA requested: {}, \
366                     expected: {}",
367                    node_index, da_requested, node_config.da
368                ),
369            });
370        }
371
372        let added_to_da = node_config.da;
373
374        self.pub_posted
375            .insert(pubkey.clone(), (node_index as u64, added_to_da));
376
377        // If the orchestrator is set up for libp2p and we have supplied the proper
378        // Libp2p data, add our node to the list of bootstrap nodes.
379        if self.config.libp2p_config.clone().is_some()
380            && let (Some(libp2p_public_key), Some(libp2p_address)) =
381                (libp2p_public_key, libp2p_address)
382        {
383            // Push to our bootstrap nodes
384            self.config
385                .libp2p_config
386                .as_mut()
387                .unwrap()
388                .bootstrap_nodes
389                .push((libp2p_public_key, libp2p_address));
390        }
391
392        tracing::error!("Node {node_index} has registered.");
393
394        Ok((node_index as u64, added_to_da))
395    }
396}

398impl<TYPES: NodeType> OrchestratorApi<TYPES> for OrchestratorState<TYPES>
399where
400    TYPES::SignatureKey: serde::Serialize + Clone + SignatureKey + 'static,
401{
402    /// Post an identity to the orchestrator. Takes in optional
403    /// arguments so others can identify us on the Libp2p network.
404    /// # Errors
405    /// If we were unable to serve the request
406    fn post_identity(
407        &mut self,
408        libp2p_address: Option<Multiaddr>,
409        libp2p_public_key: Option<PeerId>,
410    ) -> Result<u16, ServerError> {
411        let node_index = self.latest_index;
412        self.latest_index += 1;
413
414        if usize::from(node_index) >= self.config.config.num_nodes_with_stake.get() {
415            return Err(ServerError {
416                status: tide_disco::StatusCode::BAD_REQUEST,
417                message: "Network has reached capacity".to_string(),
418            });
419        }
420
421        // If the orchestrator is set up for libp2p and we have supplied the proper
422        // Libp2p data, add our node to the list of bootstrap nodes.
423        if self.config.libp2p_config.clone().is_some()
424            && let (Some(libp2p_public_key), Some(libp2p_address)) =
425                (libp2p_public_key, libp2p_address)
426        {
427            // Push to our bootstrap nodes
428            self.config
429                .libp2p_config
430                .as_mut()
431                .unwrap()
432                .bootstrap_nodes
433                .push((libp2p_public_key, libp2p_address));
434        }
435        Ok(node_index)
436    }
437
438    // Assumes nodes will set their own index that they received from the
439    // 'identity' endpoint
440    fn post_getconfig(&mut self, _node_index: u16) -> Result<NetworkConfig<TYPES>, ServerError> {
441        Ok(self.config.clone())
442    }

444    // Assumes one node do not get twice
445    fn get_tmp_node_index(&mut self) -> Result<u16, ServerError> {

453                    .to_string(),
454            });
455        }
456        Ok(tmp_node_index)

459    fn register_public_key(
460        &mut self,

494    fn get_start(&self) -> Result<bool, ServerError> {
495        // println!("{}", self.start);
496        if !self.start {
497            return Err(ServerError {
498                status: tide_disco::StatusCode::BAD_REQUEST,
499                message: "Network is not ready to start".to_string(),
500            });
501        }
502        Ok(self.start)

504
505    // Assumes nodes do not post 'ready' twice
506    fn post_ready(&mut self, peer_config: &PeerConfig<TYPES>) -> Result<(), ServerError> {
507        // If we have not disabled registration verification.
508        // Is this node allowed to connect?
509        if !self
510            .config
511            .config
512            .known_nodes_with_stake
513            .contains(peer_config)
514        {
515            return Err(ServerError {
516                status: tide_disco::StatusCode::FORBIDDEN,
517                message: "You are unauthorized to register with the orchestrator".to_string(),
518            });
519        }
520
521        // `HashSet::insert()` returns whether the node was newly inserted (true) or not
522        if self.nodes_connected.insert(peer_config.clone()) {
523            tracing::error!(
524                "Node {peer_config} connected. Total nodes connected: {}",
525                self.nodes_connected.len()
526            );
527        }
528
529        // i.e. nodes_connected >= num_nodes_with_stake * (start_threshold.0 / start_threshold.1)
530        if self.nodes_connected.len() as u64 * self.config.config.start_threshold.1
531            >= (self.config.config.num_nodes_with_stake.get() as u64)
532                * self.config.config.start_threshold.0
533        {
534            self.accepting_new_keys = false;
535            self.manual_start_allowed = false;
536            self.start = true;
537        }
538
539        Ok(())
540    }
541
542    /// Manually start the network
543    fn post_manual_start(&mut self, password_bytes: Vec<u8>) -> Result<(), ServerError> {
544        if !self.manual_start_allowed {
545            return Err(ServerError {
546                status: tide_disco::StatusCode::FORBIDDEN,
547                message: "Configs have already been distributed to nodes, and the network can no \
548                          longer be started manually."
549                    .to_string(),
550            });
551        }
552
553        let password = String::from_utf8(password_bytes)
554            .expect("Failed to decode raw password as UTF-8 string.");
555
556        // Check that the password matches
557        if self.config.manual_start_password != Some(password) {
558            return Err(ServerError {
559                status: tide_disco::StatusCode::FORBIDDEN,
560                message: "Incorrect password.".to_string(),
561            });
562        }
563
564        let registered_nodes_with_stake = self.config.config.known_nodes_with_stake.len();
565        let registered_da_nodes = self.config.config.known_da_nodes.len();
566
567        if registered_da_nodes > 1 {
568            self.config.config.num_nodes_with_stake =
569                std::num::NonZeroUsize::new(registered_nodes_with_stake)
570                    .expect("Failed to convert to NonZeroUsize; this should be impossible.");
571
572            self.config.config.da_staked_committee_size = registered_da_nodes;
573        } else {
574            return Err(ServerError {
575                status: tide_disco::StatusCode::FORBIDDEN,
576                message: format!(
577                    "We cannot manually start the network, because we only have \
578                     {registered_nodes_with_stake} nodes with stake registered, with \
579                     {registered_da_nodes} DA nodes."
580                ),
581            });
582        }
583
584        self.accepting_new_keys = false;
585        self.manual_start_allowed = false;
586        self.peer_pub_ready = true;
587        self.start = true;
588
589        Ok(())
590    }

592    // Aggregates results of the run from all nodes
593    fn post_run_results(&mut self, metrics: BenchResults) -> Result<(), ServerError> {
594        if metrics.total_transactions_committed != 0 {
595            // Deal with the bench results
596            if self.bench_results.total_transactions_committed == 0 {
597                self.bench_results = metrics;
598            } else {
599                // Deal with the bench results from different nodes
600                let cur_metrics = self.bench_results.clone();
601                self.bench_results.avg_latency_in_sec = (metrics.avg_latency_in_sec
602                    * metrics.num_latency
603                    + cur_metrics.avg_latency_in_sec * cur_metrics.num_latency)
604                    / (metrics.num_latency + cur_metrics.num_latency);
605                self.bench_results.num_latency += metrics.num_latency;
606                self.bench_results.minimum_latency_in_sec = metrics
607                    .minimum_latency_in_sec
608                    .min(cur_metrics.minimum_latency_in_sec);
609                self.bench_results.maximum_latency_in_sec = metrics
610                    .maximum_latency_in_sec
611                    .max(cur_metrics.maximum_latency_in_sec);
612                self.bench_results.throughput_bytes_per_sec = metrics
613                    .throughput_bytes_per_sec
614                    .max(cur_metrics.throughput_bytes_per_sec);
615                self.bench_results.total_transactions_committed = metrics
616                    .total_transactions_committed
617                    .max(cur_metrics.total_transactions_committed);
618                self.bench_results.total_time_elapsed_in_sec = metrics
619                    .total_time_elapsed_in_sec
620                    .max(cur_metrics.total_time_elapsed_in_sec);
621                self.bench_results.total_num_views =
622                    metrics.total_num_views.min(cur_metrics.total_num_views);
623                self.bench_results.failed_num_views =
624                    metrics.failed_num_views.max(cur_metrics.failed_num_views);
625            }
626        }
627        self.nodes_post_results += 1;
628        if self.bench_results.partial_results == "Unset" {
629            self.bench_results.partial_results = "One".to_string();
630            self.bench_results.printout();
631            self.output_to_csv();
632        }
633        if self.bench_results.partial_results == "One"
634            && self.nodes_post_results >= (self.config.config.da_staked_committee_size as u64 / 2)
635        {
636            self.bench_results.partial_results = "HalfDA".to_string();
637            self.bench_results.printout();
638            self.output_to_csv();
639        }
640        if self.bench_results.partial_results == "HalfDA"
641            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64 / 2)
642        {
643            self.bench_results.partial_results = "Half".to_string();
644            self.bench_results.printout();
645            self.output_to_csv();
646        }
647        if self.bench_results.partial_results != "Full"
648            && self.nodes_post_results >= (self.config.config.num_nodes_with_stake.get() as u64)
649        {
650            self.bench_results.partial_results = "Full".to_string();
651            self.bench_results.printout();
652            self.output_to_csv();
653        }
654        Ok(())
655    }
656
657    fn post_builder(&mut self, builder: Url) -> Result<(), ServerError> {
658        self.builders.push(builder);
659        Ok(())
660    }
661
662    fn get_builders(&self) -> Result<Vec<Url>, ServerError> {
663        if !matches!(self.config.builder, BuilderType::External)
664            && self.builders.len() != self.config.config.da_staked_committee_size
665        {
666            return Err(ServerError {
667                status: tide_disco::StatusCode::NOT_FOUND,
668                message: "Not all builders are registered yet".to_string(),
669            });
670        }
671        Ok(self.builders.clone())
672    }
673}

675/// Shared, lock-guarded orchestrator state, cloned into every axum handler via `State`.
676type SharedOrchestratorState<TYPES> = Arc<RwLock<OrchestratorState<TYPES>>>;

678/// Wire format of the orchestrator: [`OrchestratorVersion`] VBS framing and tide-disco's
679/// `ServerError` envelope, which `OrchestratorClient` decodes.
680struct OrchestratorWireFormat;
681
682impl WireFormat for OrchestratorWireFormat {
683    type Error = ServerError;
684    type Version = OrchestratorVersion;
685
686    fn status(err: &ServerError) -> StatusCode {
687        StatusCode::from_u16(u16::from(err.status)).unwrap_or(StatusCode::INTERNAL_SERVER_ERROR)
688    }
689
690    fn serialize_failure(message: String) -> ServerError {
691        ServerError {
692            status: tide_disco::StatusCode::INTERNAL_SERVER_ERROR,
693            message,
694        }
695    }
696}
697
698fn respond<T: Serialize>(headers: &HeaderMap, result: Result<T, ServerError>) -> Response {
699    wire::respond::<OrchestratorWireFormat, _>(headers, result)
700}
701
702fn malformed_body() -> ServerError {
703    ServerError {
704        status: tide_disco::StatusCode::BAD_REQUEST,
705        message: "Malformed body".to_string(),
706    }
707}
708
709/// Decode a POST body from `OrchestratorClient`.
710///
711/// `client.rs` builds these bodies by calling `body_binary` on an already vbs-serialized
712/// `Vec<u8>`, so the wire body is `[4-byte version][8-byte bincode `Vec<u8>` length][inner
713/// vbs-serialized payload]`; the first 12 bytes are an artifact of that double encoding and must
714/// be stripped before decoding the inner payload.
715fn decode_wrapped_body<T: DeserializeOwned>(body: &[u8]) -> Result<T, ServerError> {
716    body.get(12..)
717        .and_then(|inner| Serializer::<OrchestratorVersion>::deserialize(inner).ok())
718        .ok_or_else(malformed_body)
719}
720
721/// Like [`decode_wrapped_body`], but for the `ready` route, whose inner payload is a raw
722/// [`PeerConfig::to_bytes`] blob rather than a second vbs-serialized value.
723fn decode_wrapped_peer_config<TYPES: NodeType>(
724    body: &[u8],
725) -> Result<PeerConfig<TYPES>, ServerError> {
726    body.get(12..)
727        .and_then(PeerConfig::<TYPES>::from_bytes)
728        .ok_or_else(malformed_body)
729}

841async fn post_builder<TYPES: NodeType>(
842    State(state): State<SharedOrchestratorState<TYPES>>,
843    headers: HeaderMap,
844    body: Bytes,
845) -> Response {
846    let result = match decode_wrapped_body::<Vec<Url>>(&body) {
847        Ok(urls) => {
848            let mut reachable = urls
849                .into_iter()
850                .map(|url| async {
851                    let client: http_client::Client<ClientErr, OrchestratorVersion> =
852                        http_client::Client::builder(url.clone()).build();
853                    client
854                        .connect(Some(Duration::from_secs(2)))
855                        .await
856                        .then_some(url)
857                })
858                .collect::<FuturesUnordered<_>>()
859                .filter_map(futures::future::ready);
860            match reachable.next().await {
861                Some(url) => state.write().await.post_builder(url),
862                None => Err(ServerError {
863                    status: tide_disco::StatusCode::BAD_REQUEST,
864                    message: "No reachable addresses".to_string(),
865                }),
866            }
867        },
868        Err(err) => Err(err),
869    };
870    respond(&headers, result)
871}

881/// Builds the `api` module's routes. tide-disco served these both directly (e.g. `api/identity`)
882/// and under a major-version prefix (`v0/api/identity`), redirecting the former to the latter; we
883/// serve both forms directly instead, by mounting the `/api` tree at the root and under `/v0`.
884fn api_router<TYPES: NodeType>() -> Router<SharedOrchestratorState<TYPES>> {
885    Router::new()
886        .route("/healthcheck", get(healthcheck))
887        .route("/identity", post(post_identity::<TYPES>))
888        .route("/config/{node_index}", post(post_getconfig::<TYPES>))
889        .route("/get_tmp_node_index", post(get_tmp_node_index::<TYPES>))
890        .route("/pubkey/{is_da}", post(post_pubkey::<TYPES>))
891        .route("/peer_pub_ready", get(peer_pubconfig_ready::<TYPES>))
892        .route(
893            "/post_config_after_peer_collected",
894            post(post_config_after_peer_collected::<TYPES>),
895        )
896        .route("/ready", post(post_ready::<TYPES>))
897        .route("/start", get(get_start::<TYPES>))
898        .route("/results", post(post_results::<TYPES>))
899        .route("/manual_start", post(post_manual_start::<TYPES>))
900        .route("/builders", get(get_builders::<TYPES>))
901        .route("/builder", post(post_builder::<TYPES>))
902}

904/// Builds the full router: the app-level `healthcheck`, plus the `api` module served both
905/// unversioned and under `/v0`. Like tide-disco, every response carries permissive CORS headers.
906fn app<TYPES: NodeType>(state: SharedOrchestratorState<TYPES>) -> Router {
907    let api = Router::new().nest("/api", api_router::<TYPES>());
908    Router::new()
909        .route("/healthcheck", get(healthcheck))
910        .merge(api.clone())
911        .nest("/v0", api)
912        .with_state(state)
913        .layer(cors_layer())
914}

916/// Runs the orchestrator
917/// # Errors
918/// This errors if axum runs into an issue during serving
919pub async fn run_orchestrator<TYPES: NodeType>(
920    mut network_config: NetworkConfig<TYPES>,
921    url: Url,
922) -> io::Result<()> {
923    let env_password = std::env::var("ORCHESTRATOR_MANUAL_START_PASSWORD");
924
925    if env_password.is_ok() {
926        tracing::warn!(
927            "Took orchestrator manual start password from the environment variable: \
928             ORCHESTRATOR_MANUAL_START_PASSWORD={:?}",
929            env_password
930        );
931        network_config.manual_start_password = env_password.ok();
932    }
933
934    // Try to overwrite the network_config public keys
935    // from the file the env var points to, or panic.
936    {
937        let env_public_keys = std::env::var("ORCHESTRATOR_PUBLIC_KEYS");
938
939        if let Ok(filepath) = env_public_keys {
940            #[allow(clippy::panic)]
941            let config_file_as_string: String = fs::read_to_string(filepath.clone())
942                .unwrap_or_else(|_| panic!("Could not read config file located at {filepath}"));
943
944            let file: PublicKeysFile<TYPES> =
945                toml::from_str::<PublicKeysFile<TYPES>>(&config_file_as_string)
946                    .expect("Unable to convert config file to TOML");
947
948            network_config.public_keys = file.public_keys;
949        }
950    }
951
952    network_config.config.known_nodes_with_stake = network_config
953        .public_keys
954        .iter()
955        .map(|keys| PeerConfig {
956            stake_table_entry: keys
957                .stake_table_key
958                .stake_table_entry(U256::from(keys.stake)),
959            state_ver_key: keys.state_ver_key.clone(),
960            connect_info: keys.connect_info.clone(),
961        })
962        .collect();
963
964    network_config.config.known_da_nodes = network_config
965        .public_keys
966        .iter()
967        .filter(|keys| keys.da)
968        .map(|keys| PeerConfig {
969            stake_table_entry: keys
970                .stake_table_key
971                .stake_table_entry(U256::from(keys.stake)),
972            state_ver_key: keys.state_ver_key.clone(),
973            connect_info: keys.connect_info.clone(),
974        })
975        .collect();
976
977    let state: SharedOrchestratorState<TYPES> =
978        Arc::new(RwLock::new(OrchestratorState::new(network_config)));
979    let app = app::<TYPES>(state);
980
981    let host = url
982        .host_str()
983        .ok_or_else(|| io::Error::other(format!("orchestrator url missing host: {url}")))?;
984    let port = url
985        .port_or_known_default()
986        .ok_or_else(|| io::Error::other(format!("orchestrator url missing port: {url}")))?;
987    let listener = TcpListener::bind((host, port)).await?;
988
989    tracing::error!("listening on {url:?}");
990    axum::serve(listener, app).await
991}
