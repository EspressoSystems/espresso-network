341    async fn get_reward_amounts(
342        &self,
343        height: u64,
344        offset: u64,
345        limit: u64,
346    ) -> anyhow::Result<Self::RewardAmounts> {
347        if limit > 10000 {
348            return Err(bad_request(format!(
349                "limit {} exceeds maximum allowed value of 10000",
350                limit
351            )));
352        }
353
354        let tree_bytes = self.data_source.load_tree(height).await.map_err(|err| {
355            not_found(format!(
356                "failed to load reward tree at height {}: {}",
357                height, err
358            ))
359        })?;
360
361        let tree_data: InternalRewardTreeData =
362            bincode::deserialize(&tree_bytes).map_err(|err| {
363                not_found(format!(
364                    "failed to deserialize RewardMerkleTreeV2Data at height {}: {}",
365                    height, err
366                ))
367            })?;
368
369        let offset_usize = offset as usize;
370        let limit_usize = limit as usize;
371
372        if offset_usize > tree_data.balances.len() {
373            return Err(not_found(format!("offset {} out of bounds", offset)));
374        }
375
376        let end = std::cmp::min(offset_usize + limit_usize, tree_data.balances.len());
377        let slice = &tree_data.balances[offset_usize..end];
378
379        let result: Vec<(alloy::primitives::Address, InternalRewardAmount)> = slice
380            .iter()
381            .rev()
382            .map(|(account, amount)| (account.0, *amount))
383            .collect();
384
385        Ok(result)
386    }

473    async fn get_namespace_proof(
474        &self,
475        block_id: v1::availability::BlockId,
476        namespace: u32,
477    ) -> anyhow::Result<Self::NamespaceProofQueryData> {
478        let ns_id = NamespaceId::from(namespace);
479
480        // Convert v1 BlockId to hotshot BlockId
481        let hs_block_id = match block_id {
482            v1::availability::BlockId::Height(h) => HsBlockId::Number(h as usize),
483            v1::availability::BlockId::Hash(h) => {
484                let hash = h
485                    .parse()
486                    .map_err(|_| bad_request(format!("invalid block hash: {}", h)))?;
487                HsBlockId::Hash(hash)
488            },
489            v1::availability::BlockId::PayloadHash(h) => {
490                let payload_hash = h
491                    .parse()
492                    .map_err(|_| bad_request(format!("invalid payload hash: {}", h)))?;
493                HsBlockId::PayloadHash(payload_hash)
494            },
495        };
496
497        // Fetch block and VID common data
498        let ds = &*self.data_source;
499        let timeout = FETCH_TIMEOUT;
500        let (block_fetch, vid_fetch) =
501            join!(ds.get_block(hs_block_id), ds.get_vid_common(hs_block_id));
502        let (block, vid_common) = join!(
503            block_fetch.with_timeout(timeout),
504            vid_fetch.with_timeout(timeout)
505        );
506
507        let block =
508            block.ok_or_else(|| not_found(format!("block {} not available", hs_block_id)))?;
509        let vid_common = vid_common.ok_or_else(|| {
510            not_found(format!(
511                "VID common for block {} not available",
512                hs_block_id
513            ))
514        })?;
515
516        // Namespace absent from the block: an empty result, not an error.
517        let ns_table = block.payload().ns_table();
518        let Some(ns_index) = ns_table.find_ns_id(&ns_id) else {
519            return Ok(espresso_types::NamespaceProofQueryData {
520                transactions: vec![],
521                proof: None,
522            });
523        };
524
525        // Generate namespace proof
526        let Some(proof) = NsProof::new(block.payload(), &ns_index, vid_common.common()) else {
527            // Failed to generate proof - namespace exists but proof generation failed
528            return Ok(espresso_types::NamespaceProofQueryData {
529                transactions: vec![],
530                proof: None,
531            });
532        };
533
534        let transactions = proof.export_all_txs(&ns_id);
535
536        Ok(espresso_types::NamespaceProofQueryData {
537            transactions,
538            proof: Some(proof),
539        })
540    }

542    async fn get_namespace_proof_range(
543        &self,
544        from: u64,
545        until: u64,
546        namespace: u32,
547    ) -> anyhow::Result<Vec<Self::NamespaceProofQueryData>> {
548        let ns_id = NamespaceId::from(namespace);
549
550        // Validate range
551        if until <= from {
552            return Err(bad_request(format!(
553                "invalid range: until ({}) must be greater than from ({})",
554                until, from
555            )));
556        }
557
558        let range_size = until - from;
559        const MAX_RANGE: u64 = 100;
560        if range_size > MAX_RANGE {
561            return Err(range_exceeded(format!(
562                "range too large: {} blocks (max {})",
563                range_size, MAX_RANGE
564            )));
565        }
566
567        // Fetch blocks and VID common data for the range
568        let (blocks_stream, vids_stream) = join!(
569            self.data_source
570                .get_block_range(from as usize..until as usize),
571            self.data_source
572                .get_vid_common_range(from as usize..until as usize)
573        );
574
575        let blocks: Vec<_> = blocks_stream
576            .then(|block| async move { block.resolve().await })
577            .collect()
578            .await;
579        let vids: Vec<_> = vids_stream
580            .then(|vid| async move { vid.resolve().await })
581            .collect()
582            .await;
583
584        if blocks.len() != vids.len() {
585            return Err(anyhow::anyhow!(
586                "mismatch between blocks and VID common data"
587            ));
588        }
589
590        // Generate proofs for each block
591        let mut proofs = Vec::new();
592
593        for (block, vid) in blocks.into_iter().zip(vids) {
594            let ns_table = block.payload().ns_table();
595
596            // Check if namespace exists in this block
597            if let Some(ns_index) = ns_table.find_ns_id(&ns_id) {
598                if let Some(proof) = NsProof::new(block.payload(), &ns_index, vid.common()) {
599                    let transactions = proof.export_all_txs(&ns_id);
600                    proofs.push(espresso_types::NamespaceProofQueryData {
601                        transactions,
602                        proof: Some(proof),
603                    });
604                } else {
605                    // Failed to generate proof - return empty result for this block
606                    proofs.push(espresso_types::NamespaceProofQueryData {
607                        transactions: vec![],
608                        proof: None,
609                    });
610                }
611            } else {
612                // Namespace not present in this block
613                proofs.push(espresso_types::NamespaceProofQueryData {
614                    transactions: vec![],
615                    proof: None,
616                });
617            }
618        }
619
620        Ok(proofs)
621    }

645                        NamespaceProofQueryData {
646                            transactions: vec![],

649                    }
650                } else {
651                    NamespaceProofQueryData {
652                        transactions: vec![],
653                        proof: None,

670            v1::availability::BlockId::Height(h) => HsBlockId::Number(h as usize),
671            v1::availability::BlockId::Hash(h) => {
672                let hash = h
673                    .parse()
674                    .map_err(|_| anyhow::anyhow!("invalid block hash: {}", h))?;
675                HsBlockId::Hash(hash)
676            },
677            v1::availability::BlockId::PayloadHash(h) => {
678                let payload_hash = h
679                    .parse()
680                    .map_err(|_| anyhow::anyhow!("invalid payload hash: {}", h))?;
681                HsBlockId::PayloadHash(payload_hash)
682            },
683        };
684
685        let ds = &*self.data_source;
686        let timeout = FETCH_TIMEOUT;
687        let (block_fetch, vid_fetch) =
688            join!(ds.get_block(hs_block_id), ds.get_vid_common(hs_block_id));
689        let (block, vid_common) = join!(
690            block_fetch.with_timeout(timeout),
691            vid_fetch.with_timeout(timeout)
692        );
693
694        let block = block.ok_or_else(|| anyhow::anyhow!("block not found"))?;
695        let vid_common = vid_common.ok_or_else(|| anyhow::anyhow!("VID common data not found"))?;
696
697        let ns_table = block.payload().ns_table();
698        let ns_index = ns_table
699            .find_ns_id(&ns_id)
700            .ok_or_else(|| anyhow::anyhow!("namespace {} not present in block", namespace))?;
701
702        if NsProof::new(block.payload(), &ns_index, vid_common.common()).is_some() {
703            return Err(anyhow::anyhow!("block was correctly encoded"));
704        }
705
706        // Block has incorrect encoding: fetch VID shares to construct the proof.
707        let vid_shares_future = ds
708            .request_vid_shares(block.height(), vid_common.clone(), Duration::from_secs(40))
709            .await;
710        let mut vid_shares = vid_shares_future
711            .await
712            .map_err(|e| anyhow::anyhow!("failed to fetch VID shares: {e:#}"))?;
713
714        if let Ok(local_share) = ds.vid_share(block.height() as usize).await {
715            vid_shares.push(local_share);
716        }
717
718        let avidm_shares: Vec<AvidMShare> = vid_shares
719            .into_iter()
720            .filter_map(|s| {
721                if let VidShare::V1(s) = s {
722                    Some(s)
723                } else {
724                    None
725                }
726            })
727            .collect();
728
729        match NsProof::v1_1_new_with_incorrect_encoding(
730            &avidm_shares,
731            ns_table,
732            &ns_index,
733            &vid_common.payload_hash(),
734            vid_common.common(),
735        ) {
736            Some(NsProof::V1IncorrectEncoding(proof)) => Ok(proof),
737            _ => Err(anyhow::anyhow!(
738                "failed to generate incorrect encoding proof"
739            )),
740        }
741    }

816fn enforce_range(from: usize, until: usize, limit: usize) -> anyhow::Result<()> {
817    if until.saturating_sub(from) > limit {
818        return Err(range_exceeded(format!(
819            "range {from}..{until} exceeds limit {limit}"
820        )));
821    }
822    Ok(())
823}

825/// Check a ranges request against the same per-request object limit the range endpoints enforce,
826/// and convert it for the data source.
827///
828/// Bounding the total heights also bounds how many ranges a request may carry, since every range
829/// covers at least one height.
830fn validate_ranges(ranges: Vec<Range<u64>>, limit: usize) -> anyhow::Result<Vec<Range<u64>>> {
831    let mut total = 0usize;
832    for range in &ranges {
833        if range.is_empty() {
834            return Err(bad_request(format!(
835                "empty or inverted range {}..{}",
836                range.start, range.end
837            )));
838        }
839        // Heights are bound into the query as i64, so anything past that range cannot be queried
840        // and would only be a way to overflow the accounting below.
841        if range.end > i64::MAX as u64 {
842            return Err(bad_request(format!("height {} out of range", range.end)));
843        }
844
845        total = total
846            .checked_add((range.end - range.start) as usize)
847            .ok_or_else(|| range_exceeded(format!("ranges cover more than {limit} heights")))?;
848        if total > limit {
849            return Err(range_exceeded(format!(
850                "ranges cover more than {limit} heights"
851            )));
852        }
853    }
854
855    // The light client pairs leaves with proofs positionally, so a height must appear once, in
856    // order.
857    if !ranges.is_sorted_by(|a, b| a.end <= b.start) {
858        return Err(bad_request("ranges must be ascending and disjoint"));
859    }
860
861    Ok(ranges)
862}
