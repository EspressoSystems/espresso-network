26/// Retries of one request after its first attempt. The seed is on the startup path, so it retries
27/// at once; the backfill leaves retries to its fill loop.
28const REQUEST_RETRIES: u32 = 4;
29
30/// Most requests of one seed in flight; peers answer 4 requests per key at a time.
31const MAX_CONCURRENT_REQUESTS: usize = 3;
32
33/// The LC-contract seed of the catchup, behind `CatchupSeedFn`; steps in
34/// `doc/agents/architecture.md`.
35///
36/// `F` and `blockCommRoot` come from the SNARK-verified LC contract. Every installed table comes
37/// from a header proven against the block Merkle root of a header at `F` that `verify_header`
38/// accepted. Proven heights are below `F`, installed epochs above `start`, and every table has a
39/// block reward.
40pub struct LcContractSeed {
41    /// Seed attempts that installed a window.
42    seeded: Arc<dyn Counter>,
43    /// Asked to fill the epochs below the window after each install.
44    backfill: StakeTableBackfill,
45}
46
47impl LcContractSeed {
48    /// `None` when `chain_config` names no stake table contract to read the LC contract from.
49    pub fn new(chain_config: &ChainConfig, metrics: &dyn Metrics) -> Option<Self> {
50        chain_config.stake_table_contract.map(|_| Self {
51            seeded: metrics
52                .subgroup("catchup".into())
53                .create_counter("lc_contract_seeds".into(), None)
54                .into(),
55            backfill: StakeTableBackfill::new(metrics),
56        })
57    }
58
59    /// Sets the catchup hook of `coordinator` to this seed. Returns the backfill that the seed
60    /// asks for a fill after each install; the caller runs it.
61    pub fn install(self, coordinator: &EpochMembershipCoordinator<SeqTypes>) -> StakeTableBackfill {
62        let backfill = self.backfill.clone();
63        let seed = Arc::new(self);
64        coordinator.set_catchup_seed(Arc::new(move |target, start, coordinator| {
65            let seed = Arc::clone(&seed);
66            async move { seed.run(target, start, coordinator).await }.boxed()
67        }));
68        backfill
69    }
70
71    /// One seed attempt for `catchup(target)` with held pair top `start`: the walk start, `start`
72    /// when not needed, the last installed epoch when seeded.
73    async fn run(
74        &self,
75        target: EpochNumber,
76        start: EpochNumber,
77        coordinator: EpochMembershipCoordinator<SeqTypes>,
78    ) -> anyhow::Result<EpochNumber> {
79        let started = tokio::time::Instant::now();
80        let fetcher = coordinator.membership().fetcher();
81        let lc_contract_state = fetcher.lc_contract_finalized_state().await?;
82        let epoch_height = *coordinator.epoch_height();
83        let seed_epoch = lc_contract_state.seed_epoch(epoch_height);
84        let log = |outcome: &str| {
85            tracing::info!(
86                target: "announce::catchup",
87                %target,
88                held_start = %start,
89                lc_contract_height = lc_contract_state.height,
90                %seed_epoch,
91                outcome,
92                duration_ms = started.elapsed().as_millis() as u64,
93                "lc contract seed"
94            );
95        };
96        let Some(window) = lc_contract_state.seed_window(
97            epoch_height,
98            coordinator.membership().first_epoch(),
99            start,
100            target,
101        ) else {
102            log("not needed");
103            return Ok(start);
104        };
105
106        let heights = window.root_heights(epoch_height).collect();
107        let (preceding, roots) = fetch_headers(
108            fetcher.peers(),
109            &lc_contract_state,
110            window.preceding_root,
111            heights,
112        )
113        .await?;
114        self.install_roots(&coordinator, preceding, roots).await?;
115
116        self.seeded.add(1);
117        log("seeded");
118        Ok(*window.epochs.end())
119    }
120
121    /// Installs `roots`, then asks for a fill whether or not the install failed: a failed batch
122    /// keeps the tables it installed.
123    async fn install_roots(
124        &self,
125        coordinator: &EpochMembershipCoordinator<SeqTypes>,
126        preceding: Option<Header>,
127        roots: Vec<Header>,
128    ) -> anyhow::Result<()> {
129        let installed = coordinator
130            .membership()
131            .install_epoch_roots(preceding, roots, coordinator)
132            .await;
133        self.backfill.request_fill();
134        Ok(installed?)
135    }
136}

138/// The preceding root (the block time source of the lowest installed reward) split off, and the
139/// headers at `proven_heights`, ascending, proven against the root of the header at `F`.
140async fn fetch_headers(
141    peers: &dyn StateCatchup,
142    lc_contract_state: &LcContractFinalizedState,
143    preceding_height: Option<u64>,
144    proven_heights: Vec<u64>,
145) -> anyhow::Result<(Option<Header>, Vec<Header>)> {
146    let anchor = retry_request(|retry| fetch_anchor_root(peers, lc_contract_state, retry)).await?;
147    let heights = preceding_height.into_iter().chain(proven_heights);
148    let mut headers: Vec<Header> = stream::iter(heights)
149        .map(|height| {
150            retry_request(move |retry| {
151                fetch_proven_header(peers, lc_contract_state, anchor, height, retry)
152            })
153        })
154        .buffered(MAX_CONCURRENT_REQUESTS)
155        .try_collect()
156        .await?;
157    let preceding = preceding_height.map(|_| headers.remove(0));
158    Ok((preceding, headers))
159}

161/// `request` with up to `REQUEST_RETRIES` retries; attempt `n` (0-based) is called with `n`.
162async fn retry_request<T, Fut>(request: impl Fn(usize) -> Fut) -> anyhow::Result<T>
163where
164    Fut: Future<Output = anyhow::Result<T>>,
165{
166    let attempts = AtomicUsize::new(0);
167    // Not `peers.backoff()`: disabled on `ParallelStateCatchup`, unreachable on request-response.
168    BackoffParams::default()
169        .retry_if(
170            REQUEST_RETRIES,
171            |_| true,
172            || request(attempts.fetch_add(1, Ordering::SeqCst)),
173        )
174        .await
175}

190    /// The preceding root is split off and the roots come back in the order asked.
191    #[tokio::test(start_paused = true)]
192    async fn fetch_headers_ok() {
193        let (peers, lc_contract_state) = mock_with_headers(&[5, 15, 25, 35, 45, 55]).await;
194
195        let (preceding, roots) = fetch_headers(
196            &peers,
197            &lc_contract_state,
198            Some(5),
199            vec![15, 25, 35, 45, 55],
200        )
201        .await
202        .unwrap();
203
204        let heights: Vec<_> = roots.iter().map(Header::height).collect();
205        assert_eq!(heights, [15, 25, 35, 45, 55]);
206        assert_eq!(preceding.map(|root| root.height()), Some(5));
207    }

209    /// A request is retried up to `REQUEST_RETRIES` times, then the error surfaces.
210    #[rstest::rstest]
211    #[case::retried_ok(2, 3, true)]
212    #[case::bounded_fails(usize::MAX, REQUEST_RETRIES as usize + 1, false)]
213    #[tokio::test(start_paused = true)]
214    async fn fetch_headers_retries_failed_request(
215        #[case] failing_calls: usize,
216        #[case] expected_calls: usize,
217        #[case] expect_ok: bool,
218    ) {
219        let (peers, lc_contract_state) = mock_with_headers(&[15]).await;
220        let peers = peers.with_failing_header_calls(failing_calls);
221
222        let result = fetch_headers(&peers, &lc_contract_state, None, vec![15]).await;
223
224        assert_eq!(result.is_ok(), expect_ok);
225        if let Err(err) = result {
226            assert!(
227                format!("{err:#}").contains("the header proof at height 15"),
228                "{err:#}"
229            );
230        }
231        assert_eq!(peers.proven_header_calls(), expected_calls);
232        let attempts: Vec<usize> = (0..expected_calls).collect();
233        assert_eq!(peers.proven_header_retries(), attempts);
234    }
