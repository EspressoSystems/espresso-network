28/// Epochs whose root headers are fetched together.
29const CHUNK_EPOCHS: usize = 20;
30
31/// Wait after the first failed attempt of a fill; doubles per failed attempt.
32const FIRST_BACKOFF: Duration = Duration::from_secs(5);
33
34/// The longest wait between two attempts of a fill.
35const MAX_BACKOFF: Duration = Duration::from_secs(600);
36
37/// The failed attempt of a fill from which failures are logged as errors: the first whose wait
38/// reaches `MAX_BACKOFF`. Counted in consecutive failures, so a request that restarts the backoff
39/// does not lower it.
40const ERROR_FROM_ATTEMPT: usize = 8;
41
42/// The highest retry index sent with a request. The peers wait `catchup_base_timeout * (index +
43/// 1)` for an answer, 10 s by default at this cap; a request-response provider ignores the index.
44const MAX_REQUEST_RETRY: usize = 4;
45
46/// Bound on listing the stored epochs and on storing one table: its storage reads and writes.
47const EPOCH_DEADLINE: Duration = Duration::from_secs(60);
48
49/// Stores the stake table of every epoch below the highest stored one that storage lacks, newest
50/// first, from root headers proven against the LC contract. A fill starts at `run` and on each `request_fill`;
51/// between fills the task is idle.
52#[derive(Clone)]
53pub struct StakeTableBackfill {
54    /// Starts a fill. A request during a fill or a backoff starts the next attempt at once and
55    /// restarts the backoff at `FIRST_BACKOFF`.
56    requested: Arc<Notify>,
57    /// Epochs without a stored table: failed ones and those with a hash mismatch.
58    missing: Arc<dyn Gauge>,
59    /// Consecutive failed attempts of the current fill; 0 when it completes.
60    failed_attempts: Arc<dyn Gauge>,
61    /// Tables stored.
62    stored: Arc<dyn Counter>,
63    /// Epochs not stored because the derived hash differs from the decided one.
64    hash_mismatches: Arc<dyn Counter>,
65}
66
67impl StakeTableBackfill {
68    /// A backfill that registers its metrics under `catchup`.
69    pub fn new(metrics: &dyn Metrics) -> Self {
70        let catchup = metrics.subgroup("catchup".into());
71        Self {
72            requested: Arc::new(Notify::new()),
73            missing: catchup
74                .create_gauge("stake_table_backfill_missing_epochs".into(), None)
75                .into(),
76            failed_attempts: catchup
77                .create_gauge("stake_table_backfill_failed_attempts".into(), None)
78                .into(),
79            stored: catchup
80                .create_counter("stake_table_backfill_stored".into(), None)
81                .into(),
82            hash_mismatches: catchup
83                .create_counter("stake_table_backfill_hash_mismatches".into(), None)
84                .into(),
85        }
86    }
87
88    /// Asks the task for a fill.
89    pub fn request_fill(&self) {
90        self.requested.notify_one();
91    }
92
93    /// Fills at once, then on each request. Start it once the catchup provider is complete.
94    pub async fn run(self, coordinator: EpochMembershipCoordinator<SeqTypes>) {
95        let membership = coordinator.membership();
96        let sources = FillSources {
97            store: membership,
98            peers: membership.fetcher().peers(),
99            layout: EpochLayout {
100                first_epoch: membership.first_epoch(),
101                epoch_height: *coordinator.epoch_height(),
102            },
103        };
104        self.serve(&sources).await
105    }
106
107    /// The loop of `run`. `mismatched` outlives the fills: those epochs are not retried.
108    async fn serve(&self, sources: &FillSources<'_>) -> ! {
109        let mut mismatched = BTreeSet::new();
110        loop {
111            self.fill(sources, &mut mismatched).await;
112            self.requested.notified().await;
113        }
114    }
115
116    /// Runs attempts until one leaves no retryable epoch. A failed attempt is followed by a
117    /// doubling backoff, cut short by a request.
118    async fn fill(&self, sources: &FillSources<'_>, mismatched: &mut BTreeSet<EpochNumber>) {
119        let mut failed_attempts = 0;
120        let mut backoff = FIRST_BACKOFF;
121        while let Err(err) = self
122            .store_missing(sources, mismatched, failed_attempts.min(MAX_REQUEST_RETRY))
123            .await
124        {
125            failed_attempts += 1;
126            self.failed_attempts.set(failed_attempts);
127            log_failed_attempt(failed_attempts, backoff, &err);
128            backoff = tokio::select! {
129                () = sleep(backoff) => (backoff * 2).min(MAX_BACKOFF),
130                () = self.requested.notified() => FIRST_BACKOFF,
131            };
132        }
133        self.failed_attempts.set(0);
134    }
135
136    /// One attempt: stores the missing tables, highest epoch first, requesting each header once
137    /// with `request_retry`. Newest first, so that peers lacking the oldest roots still leave the
138    /// epochs next to the seeded window stored.
139    ///
140    /// Every missing epoch is below a seeded window, which ends at the LC contract epoch plus one,
141    /// so its roots are below the LC contract height and provable. An epoch above it would mean an
142    /// out-of-order install and fails its header request, which reports through the failed attempts.
143    ///
144    /// Errors when a read before the epochs failed, or when a chunk of epochs stored nothing and
145    /// had a failure; a retry can fix those.
146    async fn store_missing(
147        &self,
148        sources: &FillSources<'_>,
149        mismatched: &mut BTreeSet<EpochNumber>,
150        request_retry: usize,
151    ) -> anyhow::Result<()> {
152        // Bounded: the listing holds the persistence lock the live path stores under.
153        let stored = timeout(EPOCH_DEADLINE, sources.store.stored_stake_epochs())
154            .await
155            .with_context(|| format!("listing the stored epochs exceeded {EPOCH_DEADLINE:?}"))?
156            .context("listing the epochs with a stored stake table")?;
157        let missing = missing_epochs(sources.layout, &stored);
158        self.missing.set(missing.len());
159        let retryable: Vec<_> = missing
160            .into_iter()
161            .filter(|epoch| !mismatched.contains(epoch))
162            .collect();
163        if retryable.is_empty() {
164            return Ok(());
165        }
166
167        let lc_contract_state = sources
168            .store
169            .lc_contract_finalized_state()
170            .await
171            .context("reading the LC contract state")?;
172        let anchor = fetch_anchor_root(sources.peers, &lc_contract_state, request_retry).await?;
173        // Outside `EPOCH_DEADLINE`: on a deployment without a known or cached initial supply this
174        // is a long L1 scan, which a per-epoch deadline would cancel on every epoch.
175        sources
176            .store
177            .ensure_initial_supply()
178            .await
179            .context("reading the initial token supply")?;
180
181        let mut failures = Vec::new();
182        for chunk in retryable.chunks(CHUNK_EPOCHS) {
183            let failed_before = failures.len();
184            let mut chunk_stored = false;
185            let headers = sources
186                .fetch_chunk_headers(&lc_contract_state, anchor, request_retry, chunk)
187                .await;
188            for &epoch in chunk {
189                let outcome = sources.store_epoch(epoch, &headers).await;
190                chunk_stored |= matches!(outcome, Ok(StakeTableStoreOutcome::Stored));
191                if let Err(err) = self.record_outcome(epoch, outcome, mismatched) {
192                    failures.push((epoch, err));
193                }
194            }
195            // Deliberate: a chunk that stores nothing also blocks the epochs below it. Epochs that
196            // cannot be stored need investigation, which the failed-attempts gauge and error log call for.
197            if !chunk_stored && failures.len() > failed_before {
198                break;
199            }
200        }
201        if let Some((epoch, err)) = failures.first() {
202            bail!(
203                "{} epochs failed, the highest is epoch {epoch}: {err:#}",
204                failures.len()
205            );
206        }
207        Ok(())
208    }
209
210    /// Counts the outcome of storing `epoch`; the error of a failed store is passed on.
211    fn record_outcome(
212        &self,
213        epoch: EpochNumber,
214        outcome: anyhow::Result<StakeTableStoreOutcome>,
215        mismatched: &mut BTreeSet<EpochNumber>,
216    ) -> anyhow::Result<()> {
217        match outcome? {
218            StakeTableStoreOutcome::Stored => {
219                tracing::info!(target: "announce::catchup", %epoch, "stored stake table");
220                self.stored.add(1);
221                self.missing.update(-1);
222            },
223            StakeTableStoreOutcome::HashMismatch { decided, derived } => {
224                tracing::error!(
225                    %epoch,
226                    ?decided,
227                    ?derived,
228                    "stake table hash differs from the decided hash, epoch skipped"
229                );
230                mismatched.insert(epoch);
231                self.hash_mismatches.add(1);
232            },
233        }
234        Ok(())
235    }
236}

238/// Logs failed attempt number `attempt`: a warning before `ERROR_FROM_ATTEMPT`, an error from it.
239fn log_failed_attempt(attempt: usize, retry_in: Duration, err: &anyhow::Error) {
240    if attempt < ERROR_FROM_ATTEMPT {
241        tracing::warn!(attempt, ?retry_in, "backfill attempt failed: {err:#}");
242    } else {
243        tracing::error!(attempt, ?retry_in, "backfill is stuck: {err:#}");
244    }
245}

247/// What a fill reads from and writes to.
248struct FillSources<'a> {
249    /// Where the tables are read and written; the membership in production.
250    store: &'a dyn StakeTableStore,
251    /// Where the proven root headers come from.
252    peers: &'a dyn StateCatchup,
253    /// The epoch numbering of the chain.
254    layout: EpochLayout,
255}
256
257impl FillSources<'_> {
258    /// The root headers of `chunk` by height, each requested once and one at a time, so that the
259    /// requests of a backfill and a concurrent seed stay within the peers' limit per key. A
260    /// header that cannot be fetched fails only the epochs that need it.
261    ///
262    /// Each header is requested once per attempt, so with peers that answer only part of the
263    /// requests a chunk that needed all its headers would rarely complete; failing only the
264    /// dependent epochs keeps progress.
265    async fn fetch_chunk_headers(
266        &self,
267        lc_contract_state: &LcContractFinalizedState,
268        anchor: BlockMerkleCommitment,
269        request_retry: usize,
270        chunk: &[EpochNumber],
271    ) -> HashMap<u64, anyhow::Result<Header>> {
272        let heights: BTreeSet<u64> = chunk
273            .iter()
274            .flat_map(|epoch| {
275                let heights = self.layout.root_heights(*epoch);
276                [Some(heights.root), Some(heights.next), heights.preceding]
277            })
278            .flatten()
279            .collect();
280        let mut headers = HashMap::new();
281        for height in heights {
282            let header =
283                fetch_proven_header(self.peers, lc_contract_state, anchor, height, request_retry)
284                    .await;
285            headers.insert(height, header);
286        }
287        headers
288    }
289
290    /// Stores the table of `epoch` from its roots in `headers`, within `EPOCH_DEADLINE`.
291    async fn store_epoch(
292        &self,
293        epoch: EpochNumber,
294        headers: &HashMap<u64, anyhow::Result<Header>>,
295    ) -> anyhow::Result<StakeTableStoreOutcome> {
296        let header = |height: u64| match &headers[&height] {
297            Ok(header) => Ok(header),
298            Err(err) => bail!("no root header at height {height}: {err:#}"),
299        };
300        let heights = self.layout.root_heights(epoch);
301        let root = header(heights.root)?;
302        let next_root = header(heights.next)?;
303        let preceding = heights.preceding.map(header).transpose()?;
304        timeout(
305            EPOCH_DEADLINE,
306            self.store
307                .store_derived_stake_table(root, preceding, next_root),
308        )
309        .await
310        .with_context(|| format!("storing the table exceeded {EPOCH_DEADLINE:?}"))?
311    }
312}

364/// The epoch numbering of the chain.
365#[derive(Clone, Copy, Debug)]
366struct EpochLayout {
367    /// The first epoch of the membership.
368    first_epoch: EpochNumber,
369    /// Blocks per epoch.
370    epoch_height: u64,
371}
372
373impl EpochLayout {
374    /// The heights of the epoch roots that the table of `epoch` reads.
375    fn root_heights(&self, epoch: EpochNumber) -> EpochRootHeights {
376        EpochRootHeights {
377            root: root_block_in_epoch(*epoch - 2, self.epoch_height),
378            next: root_block_in_epoch(*epoch - 1, self.epoch_height),
379            preceding: preceding_root_height(epoch, self.first_epoch, self.epoch_height),
380        }
381    }
382}
383
384/// The epoch roots behind the table of one epoch.
385struct EpochRootHeights {
386    /// The root that defines the table.
387    root: u64,
388    /// The root one epoch above, which decides the table's hash.
389    next: u64,
390    /// The root one epoch below `root`, when the block reward reads it.
391    preceding: Option<u64>,
392}
393
394/// The epochs from the first one an epoch root defines up to the highest in `stored` that are
395/// not in `stored`, descending. `first_epoch` is at least 1, so `first_epoch + 2` is that first
396/// epoch. Pure.
397fn missing_epochs(layout: EpochLayout, stored: &BTreeSet<EpochNumber>) -> Vec<EpochNumber> {
398    let Some(highest) = stored.last() else {
399        return Vec::new();
400    };
401    let lowest = *layout.first_epoch + 2;
402    (lowest..=**highest)
403        .rev()
404        .map(EpochNumber::new)
405        .filter(|epoch| !stored.contains(epoch))
406        .collect()
407}

547        async fn store_derived_stake_table(
548            &self,
549            root: &Header,
550            preceding_root: Option<&Header>,
551            next_root: &Header,
552        ) -> anyhow::Result<StakeTableStoreOutcome> {
553            let epoch = epoch_from_block_number(root.height(), EPOCH_HEIGHT) + 2;
554            self.calls.lock().push((
555                epoch,
556                preceding_root.map(Header::height),
557                next_root.height(),
558            ));
559            if self.hanging.contains(&epoch) {
560                std::future::pending::<()>().await;
561            }
562            anyhow::ensure!(!self.failing.contains(&epoch), "injected failure");
563            if self.mismatched.contains(&epoch) {
564                return Ok(StakeTableStoreOutcome::HashMismatch {
565                    decided: Commitment::from_raw([1; 32]),
566                    derived: Commitment::from_raw([2; 32]),
567                });
568            }
569            self.stored.lock().insert(EpochNumber::new(epoch));
570            Ok(StakeTableStoreOutcome::Stored)
571        }
572    }
