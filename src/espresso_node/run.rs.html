24pub enum NodeContext<P: SequencerPersistence> {
25    Validator(Box<SequencerContext<network::Production, P>>),
26    Follower(Box<FollowerContext<P>>),
27}

29impl<P: SequencerPersistence> NodeContext<P> {
30    pub async fn start(&mut self) -> anyhow::Result<()> {
31        match self {
32            Self::Validator(ctx) => ctx.start_consensus().await,
33            // Following began in `init_follower_node`.
34            Self::Follower(_) => {},
35        }
36        Ok(())
37    }
38
39    pub async fn join(&mut self) {
40        match self {
41            Self::Validator(ctx) => ctx.join().await,
42            Self::Follower(ctx) => ctx.join().await,
43        }
44    }
45
46    pub async fn shut_down(&mut self) {
47        match self {
48            Self::Validator(ctx) => ctx.shut_down().await,
49            Self::Follower(ctx) => ctx.shut_down().await,
50        }
51    }
52}

54pub async fn main(migrated_envs: Vec<(&str, &str)>) -> anyhow::Result<()> {
55    espresso_types::assert_node_feature();
56    let opt = Options::parse();
57    let mut modules = opt.modules();
58
59    // Genesis carries the chain ID, which selects the default telemetry
60    // endpoint. Load it before telemetry init; the genesis log line is emitted
61    // later, once the subscriber is installed.
62    let genesis = Genesis::load(&opt.genesis_file).await?;
63    let telemetry_endpoint: Option<Url> = opt.telemetry.endpoint.clone().or_else(|| {
64        default_telemetry_endpoint(genesis.chain_config.chain_id).map(|s| {
65            s.parse()
66                .expect("default telemetry endpoint is a valid URL")
67        })
68    });
69    let telemetry_enabled = opt.telemetry.logs_enable || opt.telemetry.metrics_enable;
70
71    // Build telemetry before logging so its layer attaches to the global
72    // subscriber. The registry is still `None` here (the API setup runs later);
73    // the metrics push is attached then via `attach_metrics_push`.
74    let (mut telemetry_handle, deferred_warnings, telemetry_init_error) =
75        match (opt.key_set.clone().try_into(), telemetry_endpoint.as_ref()) {
76            (Ok(KeySet { staking, .. }), Some(endpoint)) => match telemetry::init(
77                &opt.telemetry,
78                &staking,
79                opt.identity.node_name.as_deref(),
80                opt.identity.company_name.as_deref(),
81                endpoint,
82                telemetry::registry(),
83            ) {
84                Ok((h, warns)) => (h, warns, None),
85                Err(e) => (None, Vec::new(), Some(e.context("telemetry init failed"))),
86            },
87            // Telemetry requested but no endpoint resolved (unknown chain, no
88            // override). Stay off and warn so the misconfig is visible.
89            (Ok(_), None) if telemetry_enabled => (
90                None,
91                vec![format!(
92                    "telemetry enabled but no endpoint resolved for chain {}; set \
93                     ESPRESSO_NODE_TELEMETRY_ENDPOINT",
94                    genesis.chain_config.chain_id
95                )],
96                None,
97            ),
98            (Err(_), _) if telemetry_enabled && modules.follower.is_some() => (
99                None,
100                vec![
101                    "telemetry enabled but a follower has no staking key to identify itself with; \
102                     continuing without telemetry"
103                        .into(),
104                ],
105                None,
106            ),
107            // Keyset error (surfaced later) or telemetry not requested.
108            _ => (None, Vec::new(), None),
109        };
110    let otel_layer = telemetry_handle.as_ref().and_then(|h| h.tracing_layer());
111    opt.logging.init_with_otel(otel_layer);
112    for w in deferred_warnings {
113        tracing::warn!("{w}");
114    }
115    if let Some(e) = telemetry_init_error {
116        tracing::error!("{e:#}; continuing without telemetry");
117    }
118    espresso_utils::env_compat::log_migrated_env_vars(&migrated_envs);
119    log_cpu_probe(genesis.drb_difficulty.max(genesis.drb_upgrade_difficulty)).await;
120
121    tracing::warn!(?modules, "sequencer starting up");
122
123    let public_node_config = PublicNodeConfig::new(&opt, &modules, &genesis);
124
125    tracing::warn!(?genesis, "genesis");
126
127    let result = if let Some(storage) = modules.storage_fs.take() {
128        run_with_storage(
129            genesis,
130            modules,
131            opt,
132            storage,
133            public_node_config,
134            telemetry_handle.as_mut(),
135        )
136        .await
137    } else if let Some(storage) = modules.storage_sql.take() {
138        run_with_storage(
139            genesis,
140            modules,
141            opt,
142            storage,
143            public_node_config,
144            telemetry_handle.as_mut(),
145        )
146        .await
147    } else {
148        // Persistence is required. If none is provided, just use the local file system.
149        run_with_storage(
150            genesis,
151            modules,
152            opt,
153            persistence::fs::Options::default(),
154            public_node_config,
155            telemetry_handle.as_mut(),
156        )
157        .await
158    };
159
160    if let Some(h) = telemetry_handle {
161        h.shutdown();
162    }
163    result
164}

166async fn run_with_storage<S>(
167    genesis: Genesis,
168    modules: Modules,
169    opt: Options,
170    storage_opt: S,
171    public_node_config: PublicNodeConfig,
172    telemetry_handle: Option<&mut telemetry::TelemetryHandle>,
173) -> anyhow::Result<()>
174where
175    S: DataSourceOptions,
176{
177    let mut ctx = init_with_storage(genesis, modules, opt, storage_opt, public_node_config).await?;
178
179    // The API setup deposited the prometheus Registry into `telemetry::REGISTRY`
180    // (if the HTTP module was configured). Attach the metrics push task now,
181    // before consensus starts churning.
182    if let (Some(handle), Some(registry)) = (telemetry_handle, telemetry::registry()) {
183        handle.attach_metrics_push(registry);
184    }
185
186    ctx.start().await?;
187
188    tokio::select! {
189        () = ctx.join() => tracing::warn!("consensus stopped; exiting"),
190        _ = espresso_utils::shutdown::wait_for_shutdown_signal() => ctx.shut_down().await,
191    }
192
193    Ok(())
194}

196pub async fn init_with_storage<S>(
197    genesis: Genesis,
198    mut modules: Modules,
199    opt: Options,
200    mut storage_opt: S,
201    public_node_config: PublicNodeConfig,
202) -> anyhow::Result<NodeContext<S::Persistence>>
203where
204    S: DataSourceOptions,
205{
206    let follower_params = match (&modules.follower, &modules.query) {
207        (Some(follower), Some(query)) => Some(opt.follower_params(follower, query)?),
208        _ => None,
209    };
210    let l1_params = L1Params {
211        urls: opt.l1_provider_url,
212        options: opt.l1_options,
213    };
214
215    if let Some(params) = follower_params {
216        let ctx = init_follower_with_storage(
217            genesis,
218            modules,
219            params,
220            l1_params,
221            storage_opt,
222            public_node_config,
223        )
224        .await?;
225        return Ok(NodeContext::Follower(Box::new(ctx)));
226    }
227
228    let KeySet {
229        staking,
230        state,
231        x25519,
232    } = opt.key_set.try_into()?;
233
234    let network_params = NetworkParams {
235        cdn_endpoint: opt.cdn_endpoint,
236        cliquenet_bind_addr: opt.cliquenet_bind_address,
237        cliquenet_advertise_addr: opt.cliquenet_advertise_address,
238        x25519_secret_key: x25519,
239        libp2p_advertise_address: opt.libp2p_advertise_address,
240        libp2p_bind_address: opt.libp2p_bind_address,
241        libp2p_bootstrap_nodes: opt.libp2p_bootstrap_nodes,
242        orchestrator_url: opt.orchestrator_url,
243        builder_urls: opt.builder_urls,
244        state_relay_server_url: opt.state_relay_server_url,
245        public_api_url: opt.public_api_url,
246        private_staking_key: staking,
247        private_state_key: state,
248        config_peers: opt.config_peers,
249        catchup: CatchupParams {
250            state_peers: opt.state_peers,
251            backoff: opt.catchup_backoff,
252            base_timeout: opt.catchup_base_timeout,
253            local_timeout: opt.local_catchup_timeout,
254        },
255        bootstrap_epoch_catchup_timeout: opt.bootstrap_epoch_catchup_timeout,
256        libp2p_history_gossip: opt.libp2p_history_gossip,
257        libp2p_history_length: opt.libp2p_history_length,
258        libp2p_max_ihave_length: opt.libp2p_max_ihave_length,
259        libp2p_max_ihave_messages: opt.libp2p_max_ihave_messages,
260        libp2p_max_gossip_transmit_size: opt.libp2p_max_gossip_transmit_size,
261        libp2p_max_direct_transmit_size: opt.libp2p_max_direct_transmit_size,
262        libp2p_mesh_outbound_min: opt.libp2p_mesh_outbound_min,
263        libp2p_mesh_n: opt.libp2p_mesh_n,
264        libp2p_mesh_n_high: opt.libp2p_mesh_n_high,
265        libp2p_heartbeat_interval: opt.libp2p_heartbeat_interval,
266        libp2p_mesh_n_low: opt.libp2p_mesh_n_low,
267        libp2p_published_message_ids_cache_time: opt.libp2p_published_message_ids_cache_time,
268        libp2p_iwant_followup_time: opt.libp2p_iwant_followup_time,
269        libp2p_max_messages_per_rpc: opt.libp2p_max_messages_per_rpc,
270        libp2p_gossip_retransmission: opt.libp2p_gossip_retransmission,
271        libp2p_flood_publish: opt.libp2p_flood_publish,
272        libp2p_duplicate_cache_time: opt.libp2p_duplicate_cache_time,
273        libp2p_fanout_ttl: opt.libp2p_fanout_ttl,
274        libp2p_heartbeat_initial_delay: opt.libp2p_heartbeat_initial_delay,
275        libp2p_gossip_factor: opt.libp2p_gossip_factor,
276        libp2p_gossip_lazy: opt.libp2p_gossip_lazy,
277        libp2p_dht_put_quorum: opt.libp2p_dht_put_quorum,
278    };
279
280    let proposal_fetcher_config = opt.proposal_fetcher_config;
281    let empty_block_delay = opt.empty_block_delay;
282
283    let persistence = storage_opt.create().await?;
284
285    // Initialize HotShot. If the user requested the HTTP module, we must initialize the handle in
286    // a special way, in order to populate the API with consensus metrics. Otherwise, we initialize
287    // the handle directly, with no metrics.
288    let ctx = match modules.http.take() {
289        Some(http_opt) => {
290            let http_opt = api_options(http_opt, modules, &storage_opt, public_node_config);
291            http_opt
292                .serve(move |metrics, consumer, storage| {
293                    async move {
294                        init_node(
295                            genesis,
296                            network_params,
297                            metrics,
298                            persistence,
299                            l1_params,
300                            storage,
301                            consumer,
302                            opt.is_da,
303                            opt.identity,
304                            proposal_fetcher_config,
305                            empty_block_delay,
306                        )
307                        .await
308                    }
309                    .boxed()
310                })
311                .await?
312        },
313        None => {
314            init_node(
315                genesis,
316                network_params,
317                Box::new(NoMetrics),
318                persistence,
319                l1_params,
320                None,
321                NullEventConsumer,
322                opt.is_da,
323                opt.identity,
324                proposal_fetcher_config,
325                empty_block_delay,
326            )
327            .await?
328        },
329    };
330
331    Ok(NodeContext::Validator(Box::new(ctx)))
332}

334/// Module parsing has checked that the http, query and storage-sql modules are present.
335async fn init_follower_with_storage<S>(
336    genesis: Genesis,
337    mut modules: Modules,
338    params: FollowerParams,
339    l1_params: L1Params,
340    mut storage_opt: S,
341    public_node_config: PublicNodeConfig,
342) -> anyhow::Result<FollowerContext<S::Persistence>>
343where
344    S: DataSourceOptions,
345{
346    let http_opt = modules
347        .http
348        .take()
349        .context("a follower needs the http module")?;
350    let http_opt = api_options(http_opt, modules, &storage_opt, public_node_config);
351    let persistence = storage_opt.create().await?;
352    http_opt
353        .serve(move |metrics, sink, _storage| {
354            async move {
355                init_follower_node(genesis, params, metrics, persistence, l1_params, sink).await
356            }
357            .boxed()
358        })
359        .await
360        .map(FollowerContext::new)
361}

363fn api_options<S: DataSourceOptions>(
364    http_opt: api::options::Http,
365    modules: Modules,
366    storage_opt: &S,
367    public_node_config: PublicNodeConfig,
368) -> api::Options {
369    let mut http_opt = api::Options::from(http_opt);
370    if let Some(query) = modules.query {
371        http_opt = storage_opt.enable_query_module(http_opt, query);
372    }
373    if let Some(submit) = modules.submit {
374        http_opt = http_opt.submit(submit);
375    }
376    if let Some(status) = modules.status {
377        http_opt = http_opt.status(status);
378    }
379    if let Some(catchup) = modules.catchup {
380        http_opt = http_opt.catchup(catchup);
381    }
382    if modules.hotshot_events.is_some() {
383        tracing::warn!(
384            "the hotshot-events module is deprecated and ignored: the node no longer serves \
385             /hotshot-events. Remove it from the command line."
386        );
387    }
388    if let Some(explorer) = modules.explorer {
389        http_opt = http_opt.explorer(explorer);
390    }
391    if let Some(light_client) = modules.light_client {
392        http_opt = http_opt.light_client(light_client);
393    }
394    if let Some(config) = modules.config {
395        http_opt = http_opt
396            .config(config)
397            .public_node_config(public_node_config);
398    }
399    http_opt
400}
