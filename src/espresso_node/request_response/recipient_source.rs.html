21#[derive(Clone)]
22pub struct RecipientSource<P: SequencerPersistence> {
23    pub consensus_handle: Arc<ConsensusHandle<SeqTypes, Arc<P>>>,
24    /// A copy of the membership coordinator
25    pub memberships: EpochMembershipCoordinator<SeqTypes>,
26    /// The public key of the node
27    pub public_key: PubKey,
28    /// Keys of the validators active at the L1 finalized block at startup, held or not; responders
29    /// are these and the keys of the consensus-epoch table that the network holds at request time,
30    /// deduplicated, without self.
31    pub l1_finalized_peers: Vec<PubKey>,
32    /// The network, for the keys it holds at request time.
33    pub network: Sender<SeqTypes>,
34}
35
36/// Implement the RecipientSourceTrait, which allows the request-response protocol to derive the
37/// intended recipients for a given request
38#[async_trait]
39impl<P: SequencerPersistence> RecipientSourceTrait<Request, PubKey> for RecipientSource<P> {
40    async fn get_expected_responders(&self, _request: &Request) -> Result<Vec<PubKey>> {
41        let current_epoch = self
42            .consensus_handle
43            .current_epoch()
44            .await
45            .unwrap_or(EpochNumber::genesis());
46        let membership = responder_membership(&self.memberships, current_epoch)?;
47
48        let table_keys = membership
49            .stake_table()
50            .map(|entry| entry.stake_table_entry.stake_key);
51        Ok(responders(
52            table_keys,
53            &self.l1_finalized_peers,
54            &self.network.peer_keys(),
55            &self.public_key,
56        ))
57    }
58}
59
60/// The `table_keys` and `l1_finalized_peers` that are in `held`, deduplicated, without `own_key`.
61fn responders(
62    table_keys: impl Iterator<Item = PubKey>,
63    l1_finalized_peers: &[PubKey],
64    held: &HashSet<PubKey>,
65    own_key: &PubKey,
66) -> Vec<PubKey> {
67    let keys: BTreeSet<PubKey> = table_keys
68        .chain(l1_finalized_peers.iter().copied())
69        .filter(|key| held.contains(key) && key != own_key)
70        .collect();
71    keys.into_iter().collect()
72}

74/// The stake table to draw responders from: the current epoch's, else the previous epoch's, else
75/// the newest one this node holds.
76///
77/// The newest covers a node whose consensus is below the epochs it holds tables for, such as one
78/// started on an empty database at the genesis epoch. It always resolves, since the genesis pair is
79/// never pruned. Without it every request fails before it is sent, including the ones catchup
80/// makes to bring the node to the network's epoch.
81fn responder_membership(
82    memberships: &EpochMembershipCoordinator<SeqTypes>,
83    current_epoch: EpochNumber,
84) -> Result<EpochMembership<SeqTypes>> {
85    let previous_epoch = EpochNumber::new(current_epoch.saturating_sub(1));
86    let newest_epoch = memberships.membership().highest_known_epoch();
87    [Some(current_epoch), Some(previous_epoch), newest_epoch]
88        .into_iter()
89        .flatten()
90        .find_map(|epoch| {
91            memberships
92                .stake_table_for_epoch(epoch)
93                .inspect_err(|err| debug!(%epoch, %err, "no stake table to draw responders from"))
94                .ok()
95        })
96        .with_context(|| {
97            format!(
98                "no stake table for epoch {current_epoch}, epoch {previous_epoch} or the newest \
99                 epoch held ({newest_epoch:?})"
100            )
101        })
102}

115    /// Decaf's first epoch, far above the genesis epoch.
116    const FIRST_EPOCH: u64 = 1054;
117
118    const EPOCH_HEIGHT: u64 = 100;
119
120    const STAKERS: u64 = 4;
121
122    fn key(i: u64) -> PubKey {
123        BLSPubKey::generated_from_seed_indexed([0; 32], i).0
124    }
125
126    #[test]
127    fn responders_union_of_held_without_self() {
128        let own = key(0);
129        let table = [own, key(1), key(2)];
130        let l1_finalized = [key(2), key(3), key(4), own];
131        let held = HashSet::from([key(1), key(3), own]);
132
133        let out = responders(table.into_iter(), &l1_finalized, &held, &own);
134
135        let mut expected = vec![key(1), key(3)];
136        expected.sort();
137        assert_eq!(out, expected);
138    }

140    /// A membership holding the stake tables for `FIRST_EPOCH` and the epoch after it.
141    fn memberships() -> EpochMembershipCoordinator<SeqTypes> {
142        let peers: Vec<PeerConfig<SeqTypes>> = (0..STAKERS)
143            .map(|i| {
144                ValidatorConfig::<SeqTypes>::generated_from_seed_indexed(
145                    [7u8; 32],
146                    i,
147                    U256::from(100),
148                    true,
149                )
150                .public_config()
151            })
152            .collect();
153        let committees = EpochCommittees::new(
154            peers.clone(),
155            peers,
156            None,
157            Fetcher::mock(),
158            EPOCH_HEIGHT,
159            EpochNumber::new(FIRST_EPOCH),
160        )
161        .expect("genesis without duplicate keys");
162        EpochMembershipCoordinator::new(
163            committees,
164            std::num::NonZeroU64::new(EPOCH_HEIGHT).unwrap(),
165            &TestStorage::default(),
166        )
167    }

169    /// A node started on an empty database has consensus at the genesis epoch, below the first
170    /// epoch, so neither that epoch nor the one before it has a stake table. Its responders must
171    /// still come from the stake tables it holds, or every request fails before it is sent.
172    #[tokio::test]
173    async fn responders_before_the_first_epoch_come_from_the_newest_stake_table() {
174        let membership = responder_membership(&memberships(), EpochNumber::genesis()).unwrap();
175
176        assert_eq!(membership.epoch(), EpochNumber::new(FIRST_EPOCH + 1));
177        assert_eq!(membership.stake_table().len(), STAKERS as usize);
178    }
179
180    #[tokio::test]
181    async fn responders_come_from_the_current_epoch_when_its_stake_table_is_held() {
182        let membership =
183            responder_membership(&memberships(), EpochNumber::new(FIRST_EPOCH)).unwrap();
184
185        assert_eq!(membership.epoch(), EpochNumber::new(FIRST_EPOCH));
186    }
