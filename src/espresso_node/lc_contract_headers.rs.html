1//! Headers proven against the LC contract state, for the catchup seed and the stake table
2//! backfill.

4use anyhow::Context;
5use espresso_types::{
6    BlockMerkleCommitment, Header, LcContractFinalizedState, traits::StateCatchup,
7};
8
9/// The block Merkle root of the header at `lc_contract_state.height`, which `verify_header`
10/// accepted. One request; `retry` scales the peers' timeout.
11pub(crate) async fn fetch_anchor_root(
12    peers: &dyn StateCatchup,
13    lc_contract_state: &LcContractFinalizedState,
14    retry: usize,
15) -> anyhow::Result<BlockMerkleCommitment> {
16    peers
17        .try_fetch_lc_contract_block_merkle_root(retry, lc_contract_state)
18        .await
19        .with_context(|| {
20            format!(
21                "fetching the LC contract header at height {}",
22                lc_contract_state.height
23            )
24        })
25}
26
27/// The header at `height`, proven against `anchor` from `fetch_anchor_root`. One request.
28pub(crate) async fn fetch_proven_header(
29    peers: &dyn StateCatchup,
30    lc_contract_state: &LcContractFinalizedState,
31    anchor: BlockMerkleCommitment,
32    height: u64,
33    retry: usize,
34) -> anyhow::Result<Header> {
35    peers
36        .try_fetch_proven_header(retry, lc_contract_state.height, anchor, height)
37        .await
38        .with_context(|| format!("fetching the header proof at height {height}"))
39}
