78/// Inbound request-response messages are dropped (not queued) when this channel is full. A node
79/// catching up has many batched requests in flight, each answered by up to `request_batch_size`
80/// peers per batch interval, so this must comfortably exceed that fan-in.
81const REQUEST_RESPONSE_CHANNEL_CAPACITY: usize = 256;

126    #[tracing::instrument(skip_all, fields(node_id = initializer.instance_state().node_id))]
127    #[allow(clippy::too_many_arguments)]
128    pub async fn init<F>(
129        network_config: NetworkConfig<SeqTypes>,
130        upgrade: versions::Upgrade,
131        validator_config: ValidatorConfig<SeqTypes>,
132        membership_coordinator: EpochMembershipCoordinator<SeqTypes>,
133        initializer: HotShotInitializer<SeqTypes>,
134        anchor_view: Option<ViewNumber>,
135        storage: Option<RequestResponseStorage>,
136        state_catchup: ParallelStateCatchup,
137        persistence: Arc<P>,
138        coordinator_network: F,
139        state_relay_server: Option<Url>,
140        metrics: &dyn Metrics,
141        stake_table_capacity: usize,
142        event_consumer: impl PersistenceEventConsumer + 'static,
143        proposal_fetcher_cfg: ProposalFetcherConfig,
144        empty_block_delay: Duration,
145        block_sizes: BTreeMap<Version, u64>,
146        leader_trace_dir: Option<PathBuf>,
147    ) -> anyhow::Result<Self>
148    where
149        F: AsyncFnOnce(UpgradeLock<SeqTypes>) -> Result<Cliquenet<SeqTypes>, NetworkError>,
150    {
151        let config = &network_config.config;
152        let pub_key = validator_config.public_key;
153        tracing::info!(%pub_key, "initializing consensus");
154        let upgrade_lock = restored_upgrade_lock(upgrade, &initializer);
155        let version = running_version(&upgrade_lock);
156        ensure!(
157            version >= NEW_PROTOCOL_VERSION,
158            "the chain runs version {version}, which predates the new protocol \
159             ({NEW_PROTOCOL_VERSION}), the only consensus this node runs"
160        );
161
162        let instance_state = initializer.instance_state().clone();
163
164        // Stick our node ID in `metrics` so it is easily accessible via the status API.
165        metrics
166            .create_gauge("node_index".into(), None)
167            .set(instance_state.node_id as usize);
168
169        // Start L1 client if it isn't already.
170        instance_state.l1_client.spawn_tasks().await;
171
172        info!(target: "announce", ?initializer, "starting up sequencer context with initializer");
173
174        let stake_table = config.hotshot_stake_table();
175        let stake_table_commit = stake_table.commitment(stake_table_capacity)?;
176        let stake_table_epoch = None;
177        let should_vote =
178            state_signature::should_vote(&stake_table, &validator_config.state_public_key);
179
180        seed_membership(&membership_coordinator, config, &upgrade_lock, &initializer).await;
181
182        // Before the anchor `apply_epoch` in `connect_startup_peers`, which can start a catchup.
183        let mut stake_table_backfill = None;
184        if let Some(seed) = LcContractSeed::new(&instance_state.chain_config, metrics) {
185            stake_table_backfill = Some(seed.install(&membership_coordinator));
186        }
187
188        let mut coordinator_network = coordinator_network(upgrade_lock.clone()).await?;
189
190        let l1_finalized_peers = connect_startup_peers(
191            &mut coordinator_network,
192            &membership_coordinator,
193            &initializer,
194            version,
195            pub_key,
196        )
197        .await?;
198        let network_sender = coordinator_network.sender().clone();
199
200        // Restore the persisted lock so the new protocol resumes with the lock
201        // it actually held, not the older decided-anchor QC.
202        let locked_qc = persistence
203            .load_high_qc2()
204            .await
205            .context("loading persisted locked QC")?;
206        // And the epoch boundary certificate a re-vote and the next epoch's
207        // first block name.
208        let boundary_qc = persistence
209            .load_boundary_qc2()
210            .await
211            .context("loading persisted boundary QC")?;
212        let anchor_cert2 = persistence
213            .load_cert2(initializer.anchor_leaf().view_number())
214            .await
215            .context("loading the anchor's cert2")?;
216
217        let leader_tracer = leader_trace_dir
218            .map(|dir| {
219                let node_id = instance_state.node_id;
220                let path = dir.join(format!("leader_trace_node{node_id}.csv"));
221                CsvLeaderTracer::new(node_id, &path)
222                    .map(|tracer| Arc::new(tracer) as LeaderTracerHandle)
223                    .with_context(|| format!("creating leader trace {}", path.display()))
224            })
225            .transpose()?;
226
227        let (external_tx, external_rx) = unbounded_channel();
228        let coordinator = Coordinator::maker()
229            .membership_coordinator(membership_coordinator.clone())
230            .network(coordinator_network)
231            .initializer(&initializer)
232            .upgrade_lock(upgrade_lock.clone())
233            .public_key(validator_config.public_key)
234            .private_key(validator_config.private_key.clone())
235            .state_private_key(validator_config.state_private_key.clone())
236            .stake_table_capacity(stake_table_capacity)
237            .timeout_duration(Duration::from_secs(10))
238            .empty_block_delay(empty_block_delay)
239            .block_sizes(block_sizes)
240            .storage(Arc::clone(&persistence))
241            .metrics(metrics)
242            .consensus_metrics(ConsensusMetricsValue::new(metrics))
243            .maybe_locked_qc(locked_qc)
244            .maybe_boundary_qc(boundary_qc)
245            .maybe_anchor_cert2(anchor_cert2)
246            .maybe_leader_tracer(leader_tracer)
247            .upgrade_config(UpgradeConfig {
248                start_proposing_view: config.start_proposing_view,
249                stop_proposing_view: config.stop_proposing_view,
250                start_voting_view: config.start_voting_view,
251                stop_voting_view: config.stop_voting_view,
252                start_proposing_time: config.start_proposing_time,
253                stop_proposing_time: config.stop_proposing_time,
254                start_voting_time: config.start_voting_time,
255                stop_voting_time: config.stop_voting_time,
256            })
257            .external_tx(external_tx)
258            .make();
259
260        let consensus_handle = Arc::new(ConsensusHandle::new(
261            coordinator,
262            external_rx,
263            upgrade_lock,
264            membership_coordinator.clone(),
265            &initializer,
266            EXTERNAL_EVENT_CHANNEL_SIZE,
267            metrics,
268        ));
269
270        let mut state_signer = StateSigner::new(
271            validator_config.state_private_key.clone(),
272            validator_config.state_public_key.clone(),
273            stake_table_commit,
274            stake_table_epoch,
275            stake_table_capacity,
276            should_vote,
277        );
278        if let Some(url) = state_relay_server {
279            state_signer = state_signer.with_relay_server(url);
280        }
281
282        // Create the channel for sending outbound messages from the external event handler
283        let (outbound_message_sender, outbound_message_receiver) =
284            channel(OUTBOUND_MESSAGE_CHANNEL_CAPACITY);
285        let (request_response_sender, request_response_receiver) =
286            channel(REQUEST_RESPONSE_CHANNEL_CAPACITY);
287
288        // Configure the request-response protocol
289        let request_response_config = RequestResponseConfig {
290            incoming_request_ttl: Duration::from_secs(40),
291            incoming_request_timeout: Duration::from_secs(5),
292            request_batch_size: 5,
293            request_batch_interval: Duration::from_secs(2),
294            // Permits are held while a response is derived (possibly from SQL), and a peer
295            // catching up legitimately issues several distinct requests concurrently
296            max_incoming_requests: 32,
297            max_incoming_requests_per_key: 4,
298        };
299
300        // Create the request-response protocol
301        let request_response_protocol = RequestResponseProtocol::new(
302            request_response_config,
303            RequestResponseSender::new(outbound_message_sender),
304            request_response_receiver,
305            RecipientSource {
306                memberships: membership_coordinator.clone(),
307                consensus_handle: consensus_handle.clone(),
308                public_key: validator_config.public_key,
309                l1_finalized_peers,
310                network: network_sender,
311            },
312            DataSource {
313                node_state: instance_state.clone(),
314                storage,
315                persistence: persistence.clone(),
316                consensus_handle: consensus_handle.clone(),
317            },
318            validator_config.public_key,
319            validator_config.private_key.clone(),
320        );
321
322        // Add the request-response protocol to the list of providers for state catchup. Since the interior is mutable,
323        // the request-response protocol will now retroactively be used anywhere we passed in the original struct (e.g. in consensus
324        // itself)
325        state_catchup.add_provider(Arc::new(request_response_protocol.clone()));
326
327        // Create the external event handler
328        let mut tasks = TaskList::default();
329        // After `add_provider`: the backfill fetches through the complete catchup provider.
330        if let Some(backfill) = stake_table_backfill {
331            tasks.spawn(
332                "stake table backfill",
333                backfill.run(membership_coordinator.clone()),
334            );
335        }
336        let external_event_handler = ExternalEventHandler::new(
337            &mut tasks,
338            request_response_sender,
339            outbound_message_receiver,
340            consensus_handle.clone(),
341            pub_key,
342            metrics
343                .subgroup("request_response".into())
344                .create_counter("inbound_dropped".into(), None)
345                .into(),
346        )
347        .await
348        .with_context(|| "Failed to create external event handler")?;
349
350        Ok(Self::new(
351            consensus_handle,
352            persistence,
353            state_signer,
354            external_event_handler,
355            request_response_protocol,
356            instance_state,
357            network_config,
358            validator_config,
359            event_consumer,
360            anchor_view,
361            proposal_fetcher_cfg,
362            metrics,
363        )
364        .with_task_list(tasks))
365    }

367    /// Constructor
368    #[allow(clippy::too_many_arguments)]
369    fn new(
370        consensus_handle: Arc<ConsensusHandle<SeqTypes, Arc<P>>>,
371        persistence: Arc<P>,
372        state_signer: StateSigner<SequencerApiVersion>,
373        external_event_handler: ExternalEventHandler,
374        request_response_protocol: RequestResponseProtocol<P>,
375        node_state: NodeState,
376        network_config: NetworkConfig<SeqTypes>,
377        validator_config: ValidatorConfig<SeqTypes>,
378        event_consumer: impl PersistenceEventConsumer + 'static,
379        anchor_view: Option<ViewNumber>,
380        proposal_fetcher_cfg: ProposalFetcherConfig,
381        metrics: &dyn Metrics,
382    ) -> Self {
383        let events = consensus_handle.event_stream();
384
385        let node_id = node_state.node_id;
386        let mut ctx = Self {
387            consensus_handle,
388            persistence: persistence.clone(),
389            state_signatures: Arc::clone(&state_signer.signatures),
390            request_response_protocol,
391            tasks: Default::default(),
392            detached: false,
393            wait_for_orchestrator: None,
394            node_state,
395            network_config,
396            validator_config,
397        };
398
399        // Spawn proposal fetching tasks.
400        proposal_fetcher_cfg.spawn(
401            &mut ctx.tasks,
402            ctx.consensus_handle.clone(),
403            persistence.clone(),
404            metrics,
405        );
406
407        // Shared between the event loop and the background decide processor.
408        let event_consumer = Arc::new(event_consumer);
409
410        // Wakes the background decide processor. `watch` coalesces: the processor is cursor-driven,
411        // so it only needs the latest decided view.
412        let (decide_tx, decide_rx) = watch::channel::<DecideSignal>(None);
413
414        // Background decide processor: query-service ingestion + GC, decoupled from the event loop.
415        ctx.spawn(
416            "decide processor",
417            process_decided_events_task(
418                persistence.clone(),
419                event_consumer.clone(),
420                decide_rx,
421                anchor_view,
422                DecideProcessorMetrics::new(metrics),
423            ),
424        );
425
426        let (signer_tx, signer_rx) = broadcast::channel(SIGNER_QUEUE_CAPACITY);
427        ctx.spawn(
428            "state signer",
429            sign_decided_leaves(ctx.consensus_handle.clone(), state_signer, signer_rx),
430        );
431
432        // Event loop. On a decide this only does the leaf write, then signals `decide_tx`.
433        ctx.spawn(
434            "event handler",
435            handle_events(
436                node_id,
437                events,
438                persistence,
439                signer_tx,
440                external_event_handler,
441                event_consumer,
442                decide_tx,
443            ),
444        );
445
446        ctx
447    }

523    pub async fn start_consensus(&self) {
524        if let Some((orchestrator_client, peer_config)) = &self.wait_for_orchestrator {

529        } else {

562    /// Stop participating in consensus.
563    pub async fn shut_down(&mut self) {
564        tracing::info!("shutting down SequencerContext");
565        self.consensus_handle.shut_down().await;
566        self.tasks.shut_down();
567        self.node_state.l1_client.shut_down_tasks().await;
568
569        // Since we've already shut down, we can set `detached` so the drop
570        // handler doesn't call `shut_down` again.
571        self.detached = true;
572    }

574    /// Wait for consensus to complete.
575    ///
576    /// Under normal conditions, this function will block forever, which is a convenient way of
577    /// keeping the main thread from exiting as long as there are still active background tasks.
578    /// It returns once the coordinator has stopped, so a node whose coordinator died exits
579    /// instead of serving its last state.
580    pub async fn join(&mut self) {
581        select! {
582            () = self.tasks.join() => {},
583            () = self.consensus_handle.stopped() => tracing::error!("coordinator stopped"),
584        }
585    }

599/// A decided certificate at or below the configured base is from a completed upgrade; kept, it
600/// would make `UpgradeLock::version` fail once the target moves on.
601fn restored_upgrade_lock(
602    upgrade: versions::Upgrade,
603    initializer: &HotShotInitializer<SeqTypes>,
604) -> UpgradeLock<SeqTypes> {
605    let decided_upgrade_certificate = initializer
606        .decided_upgrade_certificate()
607        .filter(|cert| cert.data.new_version > upgrade.base)
608        .cloned();
609    tracing::info!(?upgrade, ?decided_upgrade_certificate, "resuming consensus");
610    UpgradeLock::from_certificate(upgrade, &decided_upgrade_certificate)
611}

613/// The version the chain runs now: the decided upgrade's, or the base when none was decided.
614fn running_version(upgrade_lock: &UpgradeLock<SeqTypes>) -> Version {
615    upgrade_lock
616        .decided_upgrade_cert()
617        .map(|cert| cert.data.new_version)
618        .unwrap_or(upgrade_lock.upgrade().base)
619}

621/// Applies the stored anchor's epoch to `network` and dials the active validators at the L1
622/// finalized block that it does not hold. Returns the keys of all of them with connect info,
623/// without `own_key`.
624///
625/// Held keys keep the connect info of the held tables, which may be older than L1's. A dial error
626/// fails startup. The first `apply_epoch` for a higher epoch with a table drops the dialed peers
627/// outside that epoch's window.
628async fn connect_startup_peers(
629    network: &mut Cliquenet<SeqTypes>,
630    membership_coordinator: &EpochMembershipCoordinator<SeqTypes>,
631    initializer: &HotShotInitializer<SeqTypes>,
632    version: Version,
633    own_key: PubKey,
634) -> anyhow::Result<Vec<PubKey>> {
635    // Dials the N-1/N/N+1 window of the stored anchor; a node behind it is seeded after start.
636    let anchor_epoch = initializer
637        .anchor_leaf()
638        .epoch(initializer.epoch_height())
639        .unwrap_or_else(EpochNumber::genesis);
640    if let Err(err) = network.apply_epoch(anchor_epoch, membership_coordinator) {
641        tracing::warn!(%anchor_epoch, %err, "coordinator apply_epoch failed at startup");
642    }
643
644    let instance_state = initializer.instance_state();
645    if instance_state.chain_config.stake_table_contract.is_none() {
646        return Ok(Vec::new());
647    }
648
649    let started = Instant::now();
650    let held = network.sender().peer_keys();
651    let l1_finalized_peers = {
652        let l1_block = instance_state
653            .l1_client
654            .finalized_number_within_wait()
655            .await
656            .context("dialing the validators active on L1")?;
657        let validators = match membership_coordinator
658            .membership()
659            .fetcher()
660            .validator_set_at(l1_block, version)
661            .await
662        {
663            Ok(set) => set
664                .active_validators()
665                .values()
666                .map(|validator| {
667                    (
668                        *validator.stake_table_key(),
669                        validator.x25519_key,
670                        validator.p2p_addr.clone(),
671                    )
672                })
673                .collect(),
674            Err(err) if is_no_valid_validators(&err) => Vec::new(),
675            Err(err) => return Err(err.context("dialing the validators active on L1")),
676        };
677        let (l1_active_keys, targets) =
678            startup_dial_targets(&own_key, validators.into_iter(), &held);
679        let targets = targets
680            .into_iter()
681            .map(|(key, (x25519_key, addr))| (key, (x25519_key.into(), addr)))
682            .collect();
683        network
684            .add_peers(Role::Active, targets)
685            .context("dialing the validators active on L1")?;
686        l1_active_keys
687    };
688    info!(
689        target: "announce::catchup",
690        %anchor_epoch,
691        l1_finalized_peers = l1_finalized_peers.len(),
692        duration_ms = started.elapsed().as_millis() as u64,
693        "startup peers"
694    );
695    Ok(l1_finalized_peers)
696}

706/// A validator key with its x25519 key and address.
707type DialTarget = (PubKey, (x25519::PublicKey, NetAddr));
708
709/// The keys of the `validators` other than `own_key` that have an x25519 key and an address, and
710/// the `(key, (x25519 key, address))` of those that `held` lacks.
711fn startup_dial_targets(
712    own_key: &PubKey,
713    validators: impl Iterator<Item = (PubKey, Option<x25519::PublicKey>, Option<NetAddr>)>,
714    held: &HashSet<PubKey>,
715) -> (Vec<PubKey>, Vec<DialTarget>) {
716    let connectable: Vec<_> = validators
717        .filter(|(key, ..)| key != own_key)
718        .filter_map(|(key, x25519_key, p2p_addr)| Some((key, (x25519_key?, p2p_addr?))))
719        .collect();
720    let keys = connectable.iter().map(|(key, _)| *key).collect();
721    let to_dial = connectable
722        .into_iter()
723        .filter(|(key, _)| !held.contains(key))
724        .collect();
725    (keys, to_dial)
726}

750impl<P: SequencerPersistence> Drop for SequencerContext<P> {
751    fn drop(&mut self) {
752        if !self.detached {
753            // Spawn a task to shut down the context
754            let consensus_handle = self.consensus_handle.clone();
755            let tasks_clone = self.tasks.clone();
756            let node_state_clone = self.node_state.clone();
757
758            spawn(async move {
759                tracing::info!("shutting down SequencerContext");
760                consensus_handle.shut_down().await;
761                tasks_clone.shut_down();
762                node_state_clone.l1_client.shut_down_tasks().await;
763            });
764
765            // Set `detached` so the drop handler doesn't call `shut_down` again.
766            self.detached = true;
767        }
768    }
769}

771/// Latest decided view, sent from the event loop to the background decide processor. `None` is
772/// the initial/no-op value of the `watch` channel.
773type DecideSignal = Option<ViewNumber>;

775/// Metrics for the background decide processor. `backlog` (decided - processed) is the key signal:
776/// sustained growth means staging tables accumulate (no data lost, but disk grows).
777struct DecideProcessorMetrics {

788        Self {

790                .create_gauge("last_decided".into(), Some("view".into()))
791                .into(),
792            last_processed: metrics
793                .create_gauge("last_processed".into(), Some("view".into()))
794                .into(),
795            backlog: metrics
796                .create_gauge("backlog".into(), Some("view".into()))
797                .into(),
798            duration: metrics
799                .create_histogram("process_duration".into(), Some("seconds".into()))

801            failures: metrics.create_counter("failures".into(), None).into(),
802        }

806#[tracing::instrument(skip_all, fields(node_id))]
807async fn handle_events<P, C>(
808    node_id: u64,
809    mut events: impl Stream<Item = CoordinatorEvent<SeqTypes>> + Unpin,
810    persistence: Arc<P>,
811    signer_tx: broadcast::Sender<DecidedLeaf>,
812    external_event_handler: ExternalEventHandler,
813    event_consumer: Arc<C>,
814    decide_tx: watch::Sender<DecideSignal>,
815) where
816    P: SequencerPersistence,
817    C: PersistenceEventConsumer + 'static,
818{
819    while let Some(event) = events.next().await {
820        tracing::debug!(node_id, ?event, "consensus event");
821
822        match &event {
823            CoordinatorEvent::ExternalMessageReceived { data, .. } => {
824                if let Err(err) = external_event_handler.handle_event(data).await {
825                    tracing::warn!("Failed to handle external message: {:?}", err);
826                }
827            },
828            CoordinatorEvent::BlockPayloadReconstructed { .. } => {
829                // Forward straight to the consumer: reconstructed payloads might not yet
830                // have been stored by consensus storage,
831                // Query service verifies the block against a decided leaf before storing it.
832                if let Err(err) = event_consumer.handle_event(&event).await {
833                    tracing::warn!("failed to handle reconstructed payload: {err:#}");
834                }
835            },
836            _ => {},
837        }
838
839        if let Some(leaf) = DecidedLeaf::from_event(&event) {
840            let _ = signer_tx.send(leaf);
841        }
842
843        // Critical path: only persist the decided leaves, then signal the background processor.
844        // Signalling after the persist means it never reads ahead of committed state.
845        if let Some(signal) = persistence
846            .persist_event(&event, event_consumer.as_ref())
847            .await
848        {
849            // Keep the max view: a gap-fill decide signals an *older* view
850            // and must not hide a newer, unconsumed tip signal.
851            decide_tx.send_modify(|current| match current {
852                Some(view) if *view > signal => {},
853                _ => *current = Some(signal),
854            });
855        }
856    }
857}
858
859/// Decides the signer may fall behind by before it drops the oldest ones.
860const SIGNER_QUEUE_CAPACITY: usize = 128;
861
862/// Signs and relays the newest leaf of every decide, in order, off the event loop, because a
863/// relay post can take as long as its HTTP timeout. The relay needs a threshold of signatures on
864/// one height, so every node must sign the same heights. Signing every decide keeps that while
865/// posts fall behind, and dropping only the oldest once the queue is full keeps it as well.
866async fn sign_decided_leaves<P>(
867    consensus_handle: Arc<ConsensusHandle<SeqTypes, Arc<P>>>,
868    mut state_signer: StateSigner<SequencerApiVersion>,
869    mut leaves: broadcast::Receiver<DecidedLeaf>,
870) where
871    P: SequencerPersistence,
872{
873    loop {
874        match leaves.recv().await {
875            Ok(leaf) => {
876                state_signer
877                    .handle_decide(&leaf, consensus_handle.as_ref())
878                    .await;
879            },
880            Err(broadcast::error::RecvError::Lagged(skipped)) => {
881                tracing::warn!(skipped, "state signer fell behind, dropped oldest decides");
882            },
883            Err(broadcast::error::RecvError::Closed) => break,
884        }
885    }
886}
887
888const PROCESS_RETRY_INTERVAL: Duration = Duration::from_secs(30);
889
890/// Turns persisted decided leaves into query-service decide events and GCs processed data.
891/// Decoupled from [`handle_events`] so slow ingestion/GC can't stall (or drop) consensus events;
892/// cursor-driven, so it can lag without losing data.
893#[tracing::instrument(skip_all)]
894async fn process_decided_events_task<P, C>(
895    persistence: Arc<P>,
896    consumer: Arc<C>,
897    mut decide_rx: watch::Receiver<DecideSignal>,
898    anchor_view: Option<ViewNumber>,
899    metrics: DecideProcessorMetrics,
900) where
901    P: SequencerPersistence,
902    C: PersistenceEventConsumer + 'static,
903{
904    // Highest view confirmed processed, for the backlog gauge. Floored at the anchor view; the
905    // cursor reported below raises it.
906    let mut last_processed = anchor_view.map(|v| v.u64()).unwrap_or(0);
907
908    // Process leaves persisted before a previous shutdown but not yet handled.
909    if let Some(view) = anchor_view {
910        match persistence
911            .process_decided_events(view, consumer.as_ref())
912            .await
913        {
914            Ok(processed) => {
915                if let Some(v) = processed {
916                    last_processed = last_processed.max(v.u64());
917                }
918            },
919            Err(err) => tracing::warn!(
920                "failed to process decided leaves on startup, chain may not be up to date: {err:#}"
921            ),
922        }
923    }
924
925    // Reused on a timeout to re-attempt the most recent decide when no new one has arrived.
926    let mut latest: DecideSignal = None;
927
928    loop {
929        // Wait for the next decide, retrying the most recent one if none arrives within the timeout.
930        match tokio::time::timeout(PROCESS_RETRY_INTERVAL, decide_rx.changed()).await {
931            Ok(Ok(())) => latest = *decide_rx.borrow_and_update(),
932            Ok(Err(_)) => {
933                tracing::info!("decide signal channel closed, stopping decide processor");
934                return;
935            },
936            Err(_) => {}, // Timed out; fall through to retry `latest`.
937        }
938
939        let Some(view) = latest else {
940            continue;
941        };
942        let decided = view.u64();
943        metrics.last_decided.set(decided as usize);
944        metrics
945            .backlog
946            .set(decided.saturating_sub(last_processed) as usize);
947
948        let start = Instant::now();
949        let result = persistence
950            .process_decided_events(view, consumer.as_ref())
951            .await;
952        metrics.duration.add_point(start.elapsed().as_secs_f64());
953
954        match result {
955            Ok(processed) => {
956                // Advance from the real cursor, not `decided`: if ingestion/GC lagged, `processed`
957                // stays behind and the backlog gauge reflects it.
958                if let Some(v) = processed {
959                    last_processed = last_processed.max(v.u64());
960                }
961                // reset latest if we have processed all the decided leaves
962                if let Some(view) = latest
963                    && last_processed >= view.u64()
964                {
965                    latest = None;
966                }
967                metrics.last_processed.set(last_processed as usize);
968                metrics
969                    .backlog
970                    .set(decided.saturating_sub(last_processed) as usize);
971            },
972            Err(err) => {
973                // Cursor not advanced, so this range is retried next iteration; no data is lost.
974                metrics.failures.add(1);
975                tracing::warn!(?view, "deferred decide processing failed: {err:#}");
976            },
977        }
978    }
979}

981#[derive(Debug, Default, Clone)]
982#[allow(clippy::type_complexity)]
983pub struct TaskList(Arc<Mutex<Vec<(String, JoinHandle<()>)>>>);
984
985macro_rules! spawn_with_log_level {
986    ($this:expr, $lvl:expr, $name:expr, $task: expr) => {
987        let name = $name.to_string();
988        let task = {
989            let name = name.clone();
990            let span = tracing::span!($lvl, "background task", name);
991            spawn(
992                async move {
993                    tracing::event!($lvl, "spawning background task");
994                    let res = $task.await;
995                    tracing::event!($lvl, ?res, "background task exited");
996                }
997                .instrument(span),
998            )
999        };
1000        $this.0.lock().push((name, task));
1001    };
1002}
1003
1004impl TaskList {
1005    /// Spawn a background task attached to this [`TaskList`].
1006    ///
1007    /// When this [`TaskList`] is dropped or [`shut_down`](Self::shut_down), background tasks will
1008    /// be cancelled in the reverse order that they were spawned.
1009    pub fn spawn(&mut self, name: impl Display, task: impl Future<Output: Debug> + Send + 'static) {
1010        spawn_with_log_level!(self, Level::INFO, name, task);
1011    }
1012
1013    /// Spawn a short-lived background task attached to this [`TaskList`].
1014    ///
1015    /// When this [`TaskList`] is dropped or [`shut_down`](Self::shut_down), background tasks will
1016    /// be cancelled in the reverse order that they were spawned.
1017    ///
1018    /// The only difference between a short-lived background task and a [long-lived](Self::spawn)
1019    /// one is how urgently logging related to the task is treated.
1020    pub fn spawn_short_lived(
1021        &mut self,
1022        name: impl Display,
1023        task: impl Future<Output: Debug> + Send + 'static,
1024    ) {
1025        spawn_with_log_level!(self, Level::DEBUG, name, task);
1026    }
1027
1028    /// Stop all background tasks.
1029    pub fn shut_down(&self) {
1030        let tasks: Vec<(String, JoinHandle<()>)> = self.0.lock().drain(..).collect();
1031        for (name, task) in tasks.into_iter().rev() {
1032            tracing::info!(name, "cancelling background task");
1033            task.abort();
1034        }
1035    }
1036
1037    /// Wait for all background tasks to complete.
1038    pub async fn join(&mut self) {
1039        let tasks: Vec<(String, JoinHandle<()>)> = self.0.lock().drain(..).collect();
1040        join_all(tasks.into_iter().map(|(_, task)| task)).await;
1041    }
1042
1043    pub fn extend(&mut self, tasks: TaskList) {
1044        self.0.lock().extend(
1045            tasks
1046                .0
1047                .lock()
1048                .drain(..)
1049                .collect::<Vec<(String, JoinHandle<()>)>>(),
1050        );
1051    }
1052}
1053
1054impl Drop for TaskList {
1055    fn drop(&mut self) {
1056        self.shut_down()
1057    }
1058}
