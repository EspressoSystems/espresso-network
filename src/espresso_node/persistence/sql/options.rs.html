80/// The path of the SQLite database under the base directory `path`.
81///
82/// The `sqlite` directory is created when the database is opened, not here: the same directory
83/// is also read by Postgres and file system nodes, which never open it.
84pub fn build_sqlite_path(path: &str) -> anyhow::Result<PathBuf> {
85    Ok(PathBuf::from_str(path)?.join("sqlite").join("database"))

407/// Pruning parameters.
408#[derive(Parser, Clone, Copy, Debug)]
409pub struct PruningOptions {
410    /// Threshold for pruning, specified in bytes.
411    /// If the disk usage surpasses this threshold, pruning is initiated for data older than the specified minimum retention period.
412    /// Pruning continues until the disk usage drops below the MAX USAGE.
413    #[clap(long, env = "ESPRESSO_NODE_PRUNER_PRUNING_THRESHOLD", value_parser = parse_size)]
414    pub(crate) pruning_threshold: Option<u64>,
415
416    /// Minimum retention period.
417    /// Data is retained for at least this duration, even if there's no free disk space.
418    #[clap(
419        long,
420        env = "ESPRESSO_NODE_PRUNER_MINIMUM_RETENTION",
421        value_parser = parse_duration,
422    )]
423    pub(crate) minimum_retention: Option<Duration>,
424
425    /// Minimum retention period for Merklized state.
426    /// State is retained for at least this duration, even if there's no free disk space.
427    #[clap(
428        long,
429        env = "ESPRESSO_NODE_PRUNER_STATE_MINIMUM_RETENTION",
430        value_parser = parse_duration,
431    )]
432    state_minimum_retention: Option<Duration>,
433
434    /// Target retention period.
435    /// Data older than this is pruned to free up space.
436    #[clap(
437        long,
438        env = "ESPRESSO_NODE_PRUNER_TARGET_RETENTION",
439        value_parser = parse_duration,
440    )]
441    pub(crate) target_retention: Option<Duration>,
442
443    /// Target retention period for Merklized state.
444    /// State older than this is pruned to free up space, but never at or past the newest
445    /// merklized state, which the state writer resumes from.
446    #[clap(
447        long,
448        env = "ESPRESSO_NODE_PRUNER_STATE_TARGET_RETENTION",
449        value_parser = parse_duration,
450    )]
451    state_target_retention: Option<Duration>,
452
453    /// Batch size for pruning.
454    /// This is the number of blocks worth of data to delete in a single transaction. Heights that
455    /// hold no data are skipped without counting, so a batch, and the `Pruned to height` log line,
456    /// can advance by more than this many heights at once.
457    #[clap(long, env = "ESPRESSO_NODE_PRUNER_BATCH_SIZE")]
458    pub(crate) batch_size: Option<u64>,
459
460    /// Maximum disk usage (in basis points).
461    ///
462    /// Pruning stops once the disk usage falls below this value, even if
463    /// some data older than the `MINIMUM_RETENTION` remains. Values range
464    /// from 0 (0%) to 10000 (100%).
465    #[clap(long, env = "ESPRESSO_NODE_PRUNER_MAX_USAGE")]
466    pub(crate) max_usage: Option<u16>,
467
468    /// Interval for running the pruner.
469    #[clap(
470        long,
471        env = "ESPRESSO_NODE_PRUNER_INTERVAL",
472        value_parser = parse_duration,
473    )]
474    pub(crate) interval: Option<Duration>,
475
476    /// Number of SQLite pages to vacuum from the freelist
477    /// during each pruner cycle.
478    /// This value corresponds to `N` in the SQLite PRAGMA `incremental_vacuum(N)`,
479    #[clap(long, env = "ESPRESSO_NODE_PRUNER_INCREMENTAL_VACUUM_PAGES")]
480    pub(crate) pages: Option<u64>,
481}

489impl From<PruningOptions> for PrunerCfg {
490    fn from(opt: PruningOptions) -> Self {
491        let mut cfg = PrunerCfg::new();
492        if let Some(threshold) = opt.pruning_threshold {
493            cfg = cfg.with_pruning_threshold(threshold);
494        }
495        if let Some(min) = opt.minimum_retention {
496            cfg = cfg.with_minimum_retention(min);
497        }
498        if let Some(min) = opt.state_minimum_retention {
499            cfg = cfg.with_state_minimum_retention(min);
500        }
501        if let Some(target) = opt.target_retention {
502            cfg = cfg.with_target_retention(target);
503        }
504        if let Some(target) = opt.state_target_retention {
505            cfg = cfg.with_state_target_retention(target);
506        }
507        if let Some(batch) = opt.batch_size {
508            cfg = cfg.with_batch_size(batch);
509        }
510        if let Some(max) = opt.max_usage {
511            cfg = cfg.with_max_usage(max);
512        }
513        if let Some(interval) = opt.interval {
514            cfg = cfg.with_interval(interval);
515        }
516
517        if let Some(pages) = opt.pages {
518            cfg = cfg.with_incremental_vacuum_pages(pages)
519        }
520
521        cfg = cfg.with_state_tables(vec![
522            BlockMerkleTree::state_type().to_string(),
523            FeeMerkleTree::state_type().to_string(),
524        ]);
525
526        cfg
527    }
528}

530/// Pruning parameters for ephemeral consensus storage.
531#[derive(Parser, Clone, Copy, Debug)]
532pub struct ConsensusPruningOptions {
533    /// Number of views to try to retain in consensus storage before data that hasn't been archived
534    /// is garbage collected.
535    ///
536    /// The longer this is, the more certain that all data will eventually be archived, even if
537    /// there are temporary problems with archive storage or partially missing data. This can be set
538    /// very large, as most data is garbage collected as soon as it is finalized by consensus. This
539    /// setting only applies to views which never get decided (ie forks in consensus) and views for
540    /// which this node is partially offline. These should be exceptionally rare.
541    ///
542    /// Note that in extreme scenarios, data may be garbage collected even before TARGET_RETENTION
543    /// views, if consensus storage exceeds TARGET_USAGE. For a hard lower bound on how long
544    /// consensus data will be retained, see MINIMUM_RETENTION.
545    ///
546    /// The default of 302000 views equates to approximately 1 week (604800 seconds) at an average
547    /// view time of 2s.
548    #[clap(
549        name = "TARGET_RETENTION",
550        long = "consensus-storage-target-retention",
551        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_TARGET_RETENTION",
552        default_value = "302000"
553    )]
554    pub(crate) target_retention: u64,
555
556    /// Minimum number of views to try to retain in consensus storage before data that hasn't been
557    /// archived is garbage collected.
558    ///
559    /// This bound allows data to be retained even if consensus storage occupies more than
560    /// TARGET_USAGE. This can be used to ensure sufficient time to move consensus data to archival
561    /// storage as necessary, even under extreme circumstances where otherwise garbage collection
562    /// would kick in based on TARGET_RETENTION.
563    ///
564    /// The default of 130000 views equates to approximately 3 days (259200 seconds) at an average
565    /// view time of 2s.
566    #[clap(
567        name = "MINIMUM_RETENTION",
568        long = "consensus-storage-minimum-retention",
569        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_MINIMUM_RETENTION",
570        default_value = "130000"
571    )]
572    pub(crate) minimum_retention: u64,
573
574    /// Amount (in bytes) of data to retain in consensus storage before garbage collecting more
575    /// aggressively.
576    ///
577    /// See also TARGET_RETENTION and MINIMUM_RETENTION.
578    #[clap(
579        name = "TARGET_USAGE",
580        long = "consensus-storage-target-usage",
581        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_TARGET_USAGE",
582        default_value = "1000000000"
583    )]
584    pub(crate) target_usage: u64,

668/// How far behind the newest persisted decide an out-of-order ("gap-fill") decide can still
669/// arrive. Mirrors `DECIDE_BUFFER` in `hotshot-new-protocol`; a leaf missing further behind the
670/// watermark than this will never be filled in by consensus.
671pub(crate) const DECIDE_GAP_FILL_HORIZON: u64 = 20;
672
673/// Whether the height gap directly below the leaf at `view` can still be filled by a late
674/// decide: the missing view (at most `view - 1`) must be within [`DECIDE_GAP_FILL_HORIZON`] of
675/// `watermark`, the newest persisted decide.
676pub(crate) fn within_gap_fill_horizon(view: u64, watermark: u64) -> bool {
677    view.saturating_sub(1) + DECIDE_GAP_FILL_HORIZON > watermark
678}

704        );

707        // mutating the process environment that concurrent tests read.

709        for value in ["true", "TRUE", "t", "1", "yes", "Y", "on"] {
