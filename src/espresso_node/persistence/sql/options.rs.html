80/// The path of the SQLite database under the base directory `path`.
81///
82/// The `sqlite` directory is created when the database is opened, not here: the same directory
83/// is also read by Postgres and file system nodes, which never open it.

85    Ok(PathBuf::from_str(path)?.join("sqlite").join("database"))

405/// Pruning parameters.
406#[derive(Parser, Clone, Copy, Debug)]
407pub struct PruningOptions {
408    /// Threshold for pruning, specified in bytes.
409    /// If the disk usage surpasses this threshold, pruning is initiated for data older than the specified minimum retention period.
410    /// Pruning continues until the disk usage drops below the MAX USAGE.
411    #[clap(long, env = "ESPRESSO_NODE_PRUNER_PRUNING_THRESHOLD", value_parser = parse_size)]
412    pub(crate) pruning_threshold: Option<u64>,
413
414    /// Minimum retention period.
415    /// Data is retained for at least this duration, even if there's no free disk space.
416    #[clap(
417        long,
418        env = "ESPRESSO_NODE_PRUNER_MINIMUM_RETENTION",
419        value_parser = parse_duration,
420    )]
421    pub(crate) minimum_retention: Option<Duration>,
422
423    /// Minimum retention period for Merklized state.
424    /// State is retained for at least this duration, even if there's no free disk space.
425    #[clap(
426        long,
427        env = "ESPRESSO_NODE_PRUNER_STATE_MINIMUM_RETENTION",
428        value_parser = parse_duration,
429    )]
430    state_minimum_retention: Option<Duration>,
431
432    /// Target retention period.
433    /// Data older than this is pruned to free up space.
434    #[clap(
435        long,
436        env = "ESPRESSO_NODE_PRUNER_TARGET_RETENTION",
437        value_parser = parse_duration,
438    )]
439    pub(crate) target_retention: Option<Duration>,
440
441    /// Target retention period for Merklized state.
442    /// State older than this is pruned to free up space, but never at or past the newest
443    /// merklized state, which the state writer resumes from.
444    #[clap(
445        long,
446        env = "ESPRESSO_NODE_PRUNER_STATE_TARGET_RETENTION",
447        value_parser = parse_duration,
448    )]
449    state_target_retention: Option<Duration>,
450
451    /// Batch size for pruning.
452    /// This is the number of blocks worth of data to delete in a single transaction. Heights that
453    /// hold no data are skipped without counting, so a batch, and the `Pruned to height` log line,
454    /// can advance by more than this many heights at once.
455    #[clap(long, env = "ESPRESSO_NODE_PRUNER_BATCH_SIZE")]
456    pub(crate) batch_size: Option<u64>,
457
458    /// Maximum disk usage (in basis points).
459    ///
460    /// Pruning stops once the disk usage falls below this value, even if
461    /// some data older than the `MINIMUM_RETENTION` remains. Values range
462    /// from 0 (0%) to 10000 (100%).
463    #[clap(long, env = "ESPRESSO_NODE_PRUNER_MAX_USAGE")]
464    pub(crate) max_usage: Option<u16>,
465
466    /// Interval for running the pruner.
467    #[clap(
468        long,
469        env = "ESPRESSO_NODE_PRUNER_INTERVAL",
470        value_parser = parse_duration,
471    )]
472    pub(crate) interval: Option<Duration>,
473
474    /// Number of SQLite pages to vacuum from the freelist
475    /// during each pruner cycle.
476    /// This value corresponds to `N` in the SQLite PRAGMA `incremental_vacuum(N)`,
477    #[clap(long, env = "ESPRESSO_NODE_PRUNER_INCREMENTAL_VACUUM_PAGES")]
478    pub(crate) pages: Option<u64>,
479}

487impl From<PruningOptions> for PrunerCfg {
488    fn from(opt: PruningOptions) -> Self {
489        let mut cfg = PrunerCfg::new();
490        if let Some(threshold) = opt.pruning_threshold {
491            cfg = cfg.with_pruning_threshold(threshold);
492        }
493        if let Some(min) = opt.minimum_retention {
494            cfg = cfg.with_minimum_retention(min);
495        }
496        if let Some(min) = opt.state_minimum_retention {
497            cfg = cfg.with_state_minimum_retention(min);
498        }
499        if let Some(target) = opt.target_retention {
500            cfg = cfg.with_target_retention(target);
501        }
502        if let Some(target) = opt.state_target_retention {
503            cfg = cfg.with_state_target_retention(target);
504        }
505        if let Some(batch) = opt.batch_size {
506            cfg = cfg.with_batch_size(batch);
507        }
508        if let Some(max) = opt.max_usage {
509            cfg = cfg.with_max_usage(max);
510        }
511        if let Some(interval) = opt.interval {
512            cfg = cfg.with_interval(interval);
513        }
514
515        if let Some(pages) = opt.pages {
516            cfg = cfg.with_incremental_vacuum_pages(pages)
517        }
518
519        cfg = cfg.with_state_tables(vec![
520            BlockMerkleTree::state_type().to_string(),
521            FeeMerkleTree::state_type().to_string(),
522        ]);
523
524        cfg
525    }
526}

528/// Pruning parameters for ephemeral consensus storage.
529#[derive(Parser, Clone, Copy, Debug)]
530pub struct ConsensusPruningOptions {
531    /// Number of views to try to retain in consensus storage before data that hasn't been archived
532    /// is garbage collected.
533    ///
534    /// The longer this is, the more certain that all data will eventually be archived, even if
535    /// there are temporary problems with archive storage or partially missing data. This can be set
536    /// very large, as most data is garbage collected as soon as it is finalized by consensus. This
537    /// setting only applies to views which never get decided (ie forks in consensus) and views for
538    /// which this node is partially offline. These should be exceptionally rare.
539    ///
540    /// Note that in extreme scenarios, data may be garbage collected even before TARGET_RETENTION
541    /// views, if consensus storage exceeds TARGET_USAGE. For a hard lower bound on how long
542    /// consensus data will be retained, see MINIMUM_RETENTION.
543    ///
544    /// The default of 302000 views equates to approximately 1 week (604800 seconds) at an average
545    /// view time of 2s.
546    #[clap(
547        name = "TARGET_RETENTION",
548        long = "consensus-storage-target-retention",
549        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_TARGET_RETENTION",
550        default_value = "302000"
551    )]
552    pub(crate) target_retention: u64,
553
554    /// Minimum number of views to try to retain in consensus storage before data that hasn't been
555    /// archived is garbage collected.
556    ///
557    /// This bound allows data to be retained even if consensus storage occupies more than
558    /// TARGET_USAGE. This can be used to ensure sufficient time to move consensus data to archival
559    /// storage as necessary, even under extreme circumstances where otherwise garbage collection
560    /// would kick in based on TARGET_RETENTION.
561    ///
562    /// The default of 130000 views equates to approximately 3 days (259200 seconds) at an average
563    /// view time of 2s.
564    #[clap(
565        name = "MINIMUM_RETENTION",
566        long = "consensus-storage-minimum-retention",
567        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_MINIMUM_RETENTION",
568        default_value = "130000"
569    )]
570    pub(crate) minimum_retention: u64,
571
572    /// Amount (in bytes) of data to retain in consensus storage before garbage collecting more
573    /// aggressively.
574    ///
575    /// See also TARGET_RETENTION and MINIMUM_RETENTION.
576    #[clap(
577        name = "TARGET_USAGE",
578        long = "consensus-storage-target-usage",
579        env = "ESPRESSO_NODE_CONSENSUS_STORAGE_TARGET_USAGE",
580        default_value = "1000000000"
581    )]
582    pub(crate) target_usage: u64,
583
584    /// How often to measure consensus storage against TARGET_USAGE.
585    ///
586    /// On SQLite a measurement scans every page of the consensus tables. Between measurements,
587    /// each decide prunes only to TARGET_RETENTION.
588    #[clap(

595    pub(crate) usage_check_interval: Duration,

605/// 40001).  Under SERIALIZABLE isolation these aborts are expected and safe to retry from scratch.

620        default_value = "10ms",
621        value_parser = parse_duration
622    )]
623    pub(super) backoff_base: Duration,
624
625    /// Maximum delay between retries after a PostgreSQL serialization conflict.

666/// How far behind the newest persisted decide an out-of-order ("gap-fill") decide can still
667/// arrive. Mirrors `DECIDE_BUFFER` in `hotshot-new-protocol`; a leaf missing further behind the
668/// watermark than this will never be filled in by consensus.
669pub(crate) const DECIDE_GAP_FILL_HORIZON: u64 = 20;
670
671/// Whether the height gap directly below the leaf at `view` can still be filled by a late
672/// decide: the missing view (at most `view - 1`) must be within [`DECIDE_GAP_FILL_HORIZON`] of
673/// `watermark`, the newest persisted decide.
674pub(crate) fn within_gap_fill_horizon(view: u64, watermark: u64) -> bool {
675    view.saturating_sub(1) + DECIDE_GAP_FILL_HORIZON > watermark
676}
