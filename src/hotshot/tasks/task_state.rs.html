34/// Trait for creating task states.
35#[async_trait]
36pub trait CreateTaskState<TYPES, I>
37where
38    TYPES: NodeType,
39    I: NodeImplementation<TYPES>,
40{
41    /// Function to create the task state from a given `SystemContextHandle`.
42    async fn create_from(handle: &SystemContextHandle<TYPES, I>) -> Self;
43}
44
45#[async_trait]
46impl<TYPES: NodeType, I: NodeImplementation<TYPES>> CreateTaskState<TYPES, I>
47    for NetworkRequestState<TYPES, I>
48{
49    async fn create_from(handle: &SystemContextHandle<TYPES, I>) -> Self {
50        Self {
51            network: Arc::clone(&handle.hotshot.network),
52            consensus: OuterConsensus::new(handle.hotshot.consensus()),
53            view: handle.cur_view().await,
54            delay: handle.hotshot.config.data_request_delay,
55            membership_coordinator: handle.hotshot.membership_coordinator.clone(),
56            public_key: handle.public_key().clone(),
57            private_key: handle.private_key().clone(),
58            id: handle.hotshot.id,
59            shutdown_flag: Arc::new(AtomicBool::new(false)),
60            spawned_tasks: BTreeMap::new(),
61            epoch_height: handle.epoch_height,
62        }
63    }
64}

215            latest_voted_view: handle.cur_view().await,
216            vote_dependencies: BTreeMap::new(),
217            network: Arc::clone(&handle.hotshot.network),
218            membership: handle.hotshot.membership_coordinator.clone(),
219            output_event_stream: handle.hotshot.external_event_stream.0.clone(),
220            id: handle.hotshot.id,
221            storage: handle.storage.clone(),
222            storage_metrics: handle.storage_metrics(),
223            upgrade_lock: handle.hotshot.upgrade_lock.clone(),
224            epoch_height: handle.hotshot.config.epoch_height,
225            consensus_metrics,
226            first_epoch: None,
227            stake_table_capacity: handle.hotshot.config.stake_table_capacity,
228            da_committees: handle.hotshot.config.da_committees.clone(),
229        }
230    }
