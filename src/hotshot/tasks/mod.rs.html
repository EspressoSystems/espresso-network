53/// Add tasks for network requests and responses
54pub async fn add_request_network_task<TYPES: NodeType, I: NodeImplementation<TYPES>>(
55    handle: &mut SystemContextHandle<TYPES, I>,
56) {
57    let state = NetworkRequestState::<TYPES, I>::create_from(handle).await;
58
59    let task = Task::new(
60        state,
61        handle.internal_event_stream.0.clone(),
62        handle.internal_event_stream.1.activate_cloned(),
63    );
64    handle.consensus_registry.run_task(task);
65}

67/// Add a task which responds to requests on the network.
68pub fn add_response_task<TYPES: NodeType, I: NodeImplementation<TYPES>>(
69    handle: &mut SystemContextHandle<TYPES, I>,
70) {
71    let state = NetworkResponseState::<TYPES>::new(
72        handle.hotshot.consensus(),
73        handle.membership_coordinator.clone(),
74        handle.public_key().clone(),
75        handle.private_key().clone(),
76        handle.hotshot.id,
77        handle.hotshot.upgrade_lock.clone(),
78    );
79    handle.network_registry.register(run_response_task::<TYPES>(
80        state,
81        handle.internal_event_stream.1.activate_cloned(),
82        handle.internal_event_stream.0.clone(),
83    ));
84}

86/// Add a task which updates our queue length metric at a set interval
87pub fn add_queue_len_task<TYPES: NodeType, I: NodeImplementation<TYPES>>(
88    handle: &mut SystemContextHandle<TYPES, I>,
89) {
90    let consensus = handle.hotshot.consensus();
91    let rx = handle.internal_event_stream.1.clone();
92    let shutdown_signal = create_shutdown_event_monitor(handle).fuse();
93    let task_handle = spawn(async move {
94        futures::pin_mut!(shutdown_signal);
95        loop {
96            futures::select! {
97                () = shutdown_signal => {
98                    return;
99                },
100                () = sleep(Duration::from_millis(500)).fuse() => {
101                    consensus.read().await.metrics.internal_event_queue_len.set(rx.len());
102                }
103            }
104        }
105    });
106    handle.network_registry.register(task_handle);
107}

109/// Add the network task to handle messages and publish events.
110#[allow(clippy::missing_panics_doc)]
111pub fn add_network_message_task<
112    TYPES: NodeType,
113    I: NodeImplementation<TYPES>,
114    NET: ConnectedNetwork<TYPES::SignatureKey>,
115>(
116    handle: &mut SystemContextHandle<TYPES, I>,
117    channel: &Arc<NET>,
118) {
119    let upgrade_lock = handle.hotshot.upgrade_lock.clone();
120
121    let network_state: NetworkMessageTaskState<TYPES> = NetworkMessageTaskState {
122        internal_event_stream: handle.internal_event_stream.0.clone(),
123        external_event_stream: handle.output_event_stream.0.clone(),
124        public_key: handle.public_key().clone(),
125        upgrade_lock: upgrade_lock.clone(),
126        id: handle.hotshot.id,
127    };
128
129    let network = Arc::clone(channel);
130    let mut state = network_state.clone();
131    let shutdown_signal = create_shutdown_event_monitor(handle).fuse();
132    let task_handle = spawn(async move {
133        futures::pin_mut!(shutdown_signal);
134
135        loop {
136            // Wait for one of the following to resolve:
137            futures::select! {
138                // Wait for a shutdown signal
139                () = shutdown_signal => {
140                    tracing::error!("Shutting down network message task");
141                    return;
142                }
143
144                // Wait for a message from the network
145                message = network.recv_message().fuse() => {
146                    // Make sure the message did not fail
147                    let Ok(message) = message else {
148                        continue;
149                    };
150
151                    // Deserialize the message and get the version
152                    let (deserialized_message, version): (Message<TYPES>, Version) = match upgrade_lock.deserialize(&message) {
153                        Ok(message) => message,
154                        Err(e) => {
155                            tracing::error!("Failed to deserialize message: {:?}", e);
156                            continue;
157                        }
158                    };
159
160                    // Special case: external messages (version 0.0). We want to make sure it is an external message
161                    // and warn and continue otherwise.
162                    if version == EXTERNAL_MESSAGE_VERSION
163                        && !matches!(deserialized_message.kind, MessageKind::<TYPES>::External(_))
164                    {
165                        tracing::warn!("Received a non-external message with version 0.0");
166                        continue;
167                    }
168
169                    // Handle the message
170                    state.handle_message(deserialized_message).await;
171                }
172            }
173        }
174    });
175    handle.network_registry.register(task_handle);
176}

178/// Add the network task to handle events and send messages.
179pub fn add_network_event_task<
180    TYPES: NodeType,
181    I: NodeImplementation<TYPES>,
182    NET: ConnectedNetwork<TYPES::SignatureKey>,
183>(
184    handle: &mut SystemContextHandle<TYPES, I>,
185    network: Arc<NET>,
186) {
187    let network_state: NetworkEventTaskState<_, _, _> = NetworkEventTaskState {
188        network,
189        view: ViewNumber::genesis(),
190        epoch: genesis_epoch_from_version(handle.hotshot.upgrade_lock.upgrade().base),
191        membership_coordinator: handle.membership_coordinator.clone(),
192        storage: handle.storage(),
193        storage_metrics: handle.storage_metrics(),
194        consensus: OuterConsensus::new(handle.consensus()),
195        upgrade_lock: handle.hotshot.upgrade_lock.clone(),
196        transmit_tasks: BTreeMap::new(),
197        epoch_height: handle.epoch_height,
198        id: handle.hotshot.id,
199    };
200    let task = Task::new(
201        network_state,
202        handle.internal_event_stream.0.clone(),
203        handle.internal_event_stream.1.activate_cloned(),
204    );
205    handle.consensus_registry.run_task(task);
206}

208/// Adds consensus-related tasks to a `SystemContextHandle`.
209pub async fn add_consensus_tasks<TYPES: NodeType, I: NodeImplementation<TYPES>>(
210    handle: &mut SystemContextHandle<TYPES, I>,
211) {
212    handle.add_task(ViewSyncTaskState::<TYPES>::create_from(handle).await);
213    handle.add_task(VidTaskState::<TYPES, I>::create_from(handle).await);
214    handle.add_task(DaTaskState::<TYPES, I>::create_from(handle).await);
215    handle.add_task(TransactionTaskState::<TYPES>::create_from(handle).await);
216
217    let upgrade = handle.hotshot.upgrade_lock.upgrade();
218
219    // clear the loaded certificate if it's now outdated
220    handle.hotshot.upgrade_lock.apply(|cert| {
221        if cert
222            .as_ref()
223            .is_some_and(|c| upgrade.base >= c.data.new_version)
224        {
225            tracing::warn!("Discarding loaded upgrade certificate due to version configuration.");
226            *cert = None
227        }
228    });
229
230    // only spawn the upgrade task if we are actually configured to perform an upgrade.
231    if upgrade.base < upgrade.target {
232        tracing::warn!("Consensus was started with an upgrade configured. Spawning upgrade task.");
233        handle.add_task(UpgradeTaskState::<TYPES>::create_from(handle).await);
234    }
235
236    {
237        use hotshot_task_impls::{
238            consensus::ConsensusTaskState, quorum_proposal::QuorumProposalTaskState,
239            quorum_proposal_recv::QuorumProposalRecvTaskState, quorum_vote::QuorumVoteTaskState,
240        };
241
242        handle.add_task(QuorumProposalTaskState::<TYPES, I>::create_from(handle).await);
243        handle.add_task(QuorumVoteTaskState::<TYPES, I>::create_from(handle).await);
244        handle.add_task(QuorumProposalRecvTaskState::<TYPES, I>::create_from(handle).await);
245        handle.add_task(ConsensusTaskState::<TYPES, I>::create_from(handle).await);
246    }
247    add_queue_len_task(handle);
248}

250/// Creates a monitor for shutdown events.
251///
252/// # Returns
253/// A `BoxFuture<'static, ()>` that resolves when a `HotShotEvent::Shutdown` is detected.
254///
255/// # Usage
256/// Use in `select!` macros or similar constructs for graceful shutdowns:
257#[must_use]
258pub fn create_shutdown_event_monitor<TYPES: NodeType, I: NodeImplementation<TYPES>>(
259    handle: &SystemContextHandle<TYPES, I>,
260) -> BoxFuture<'static, ()> {
261    // Activate the cloned internal event stream
262    let mut event_stream = handle.internal_event_stream.1.activate_cloned();
263
264    // Create a future that completes when the `HotShotEvent::Shutdown` is received
265    async move {
266        loop {
267            match event_stream.recv_direct().await {
268                Ok(event) => {
269                    if matches!(event.as_ref(), HotShotEvent::Shutdown) {
270                        return;
271                    }
272                },
273                Err(RecvError::Closed) => {
274                    return;
275                },
276                Err(e) => {
277                    tracing::error!("Shutdown event monitor channel recv error: {}", e);
278                },
279            }
280        }
281    }
282    .boxed()
283}

285/// adds tasks for sending/receiving messages to/from the network.
286pub async fn add_network_tasks<TYPES: NodeType, I: NodeImplementation<TYPES>>(
287    handle: &mut SystemContextHandle<TYPES, I>,
288) {
289    add_network_message_and_request_receiver_tasks(handle).await;
290
291    add_network_event_tasks(handle);
292}

294/// Adds the `NetworkMessageTaskState` tasks and the request / receiver tasks.
295pub async fn add_network_message_and_request_receiver_tasks<
296    TYPES: NodeType,
297    I: NodeImplementation<TYPES>,
298>(
299    handle: &mut SystemContextHandle<TYPES, I>,
300) {
301    let network = Arc::clone(&handle.network);
302
303    add_network_message_task(handle, &network);
304
305    add_request_network_task(handle).await;
306    add_response_task(handle);
307}
