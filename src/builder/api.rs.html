50/// Wire format of the builder API: [`WireVersion`] VBS framing and the [`BuilderApiError`]
51/// envelope.
52struct BuilderWireFormat;
53
54impl WireFormat for BuilderWireFormat {
55    type Error = BuilderApiError;
56    type Version = WireVersion;
57
58    fn status(err: &BuilderApiError) -> AxumStatusCode {
59        // Maps tide-disco's `StatusCode` (what `BuilderApiError::status()` returns) onto axum's.
60        AxumStatusCode::from_u16(u16::from(err.status()))
61            .unwrap_or(AxumStatusCode::INTERNAL_SERVER_ERROR)
62    }
63
64    fn serialize_failure(message: String) -> BuilderApiError {
65        BuilderApiError::Custom {
66            message,
67            status: StatusCode::INTERNAL_SERVER_ERROR,
68        }
69    }
70}

72fn respond<T: Serialize>(headers: &HeaderMap, result: Result<T, BuilderApiError>) -> Response {
73    wire::respond::<BuilderWireFormat, _>(headers, result)
74}
75
76fn decode_body<T: DeserializeOwned>(headers: &HeaderMap, body: &[u8]) -> Result<T, RequestError> {
77    wire::decode_body::<WireVersion, T>(headers, body).map_err(|failure| match failure {
78        DecodeFailure::Json(_) => RequestError::Json,
79        DecodeFailure::Binary(_) => RequestError::Binary,
80        DecodeFailure::UnsupportedContentType => RequestError::UnsupportedContentType,
81    })
82}

84fn tb64_request_error(field: &str) -> BuilderApiError {
85    BuilderApiError::Request(RequestError::TaggedBase64 {
86        reason: format!("invalid tagged base64 for {field}"),
87    })
88}

90/// Parses a hash-type path parameter (`parent_hash`, `block_hash`, `transaction_hash`). Any
91/// failure is an `Error::Request`, which `BuilderApiError::status()` maps to 400, like tide's
92/// `blob_param` path did for these params.
93fn parse_hash_param<T>(value: &str, field: &str) -> Result<T, BuilderApiError>
94where
95    T: for<'a> TryFrom<&'a TaggedBase64>,
96{
97    let tb64: TaggedBase64 = value.parse().map_err(|_| tb64_request_error(field))?;
98    T::try_from(&tb64).map_err(|_| tb64_request_error(field))
99}
100
101/// Parses a key/signature path parameter, mirroring `try_extract_param`: a wrong-type value is a
102/// `Custom` error carrying 422. Note `BuilderApiError::status()` returns 500 for every `Custom`,
103/// so the wire status is 500, as it was under tide.
104fn parse_key_param<T>(value: &str, field: &str) -> Result<T, BuilderApiError>
105where
106    T: for<'a> TryFrom<&'a TaggedBase64>,
107{
108    let tb64: TaggedBase64 = value.parse().map_err(|_| tb64_request_error(field))?;
109    T::try_from(&tb64).map_err(|_| BuilderApiError::Custom {
110        message: format!("Invalid {field}"),
111        status: StatusCode::UNPROCESSABLE_ENTITY,
112    })
113}

115type Sender = <SeqTypes as NodeType>::SignatureKey;
116type Signature = <Sender as SignatureKey>::PureAssembledSignatureType;
117
118fn parse_sender_signature(
119    sender: &str,
120    signature: &str,
121) -> Result<(Sender, Signature), BuilderApiError> {
122    let sender = parse_key_param::<Sender>(sender, "sender")?;
123    let signature = parse_key_param::<Signature>(signature, "signature")?;
124    Ok((sender, signature))
125}

131async fn available_blocks(
132    State(state): State<SharedState>,
133    headers: HeaderMap,
134    Path((parent_hash, view_number, sender, signature)): Path<(String, u64, String, String)>,
135) -> Response {
136    let result: Result<Vec<AvailableBlockInfo<_>>, BuilderApiError> = async {
137        let hash = parse_hash_param::<VidCommitment>(&parent_hash, "parent_hash")?;
138        let (sender, signature) = parse_sender_signature(&sender, &signature)?;
139        state
140            .available_blocks(&hash, view_number, sender, &signature)
141            .await
142            .map_err(|source| BuilderApiError::BlockAvailable {
143                source,
144                resource: hash.to_string(),
145            })
146    }
147    .await;
148    respond(&headers, result)
149}

246async fn submit_txn(State(state): State<SharedState>, headers: HeaderMap, body: Bytes) -> Response {
247    let result = async {
248        let tx: <SeqTypes as NodeType>::Transaction =
249            decode_body(&headers, &body).map_err(BuilderApiError::TxnUnpack)?;
250        let hash = tx.commit();
251        state
252            .submit_txns(vec![tx])
253            .await
254            .map_err(BuilderApiError::TxnSubmit)?;
255        Ok(hash)
256    }
257    .await;
258    respond(&headers, result)
259}

261async fn submit_batch(
262    State(state): State<SharedState>,
263    headers: HeaderMap,
264    body: Bytes,
265) -> Response {
266    let result = async {
267        let txns: Vec<<SeqTypes as NodeType>::Transaction> =
268            decode_body(&headers, &body).map_err(BuilderApiError::TxnUnpack)?;
269        let hashes = txns.iter().map(|tx| tx.commit()).collect::<Vec<_>>();
270        state
271            .submit_txns(txns)
272            .await
273            .map_err(BuilderApiError::TxnSubmit)?;
274        Ok(hashes)
275    }
276    .await;
277    respond(&headers, result)
278}

296fn block_info_router(state: SharedState) -> Router {
297    Router::new()
298        .route(
299            "/availableblocks/{parent_hash}/{view_number}/{sender}/{signature}",
300            get(available_blocks),
301        )
302        .route(
303            "/claimblock/{block_hash}/{view_number}/{sender}/{signature}",
304            get(claim_block),
305        )
306        .route(
307            "/claimblockwithnumnodes/{block_hash}/{view_number}/{sender}/{signature}/{num_nodes}",
308            get(claim_block_with_num_nodes),
309        )
310        .route(
311            "/claimheaderinput/{block_hash}/{view_number}/{sender}/{signature}",
312            get(claim_header_input),
313        )
314        .route(
315            "/claimheaderinput/v2/{block_hash}/{view_number}/{sender}/{signature}",
316            get(claim_header_input_v2),
317        )
318        .route("/builderaddress", get(builder_address))
319        .with_state(state)
320}

322fn txn_submit_router(state: SharedState) -> Router {

330/// Builds the full router: `healthcheck`, plus `block_info` and `txn_submit`, served both
331/// unversioned and under `/v0` (both modules were registered with API version major `0`, tide's
332/// convention for the module's only registered major version). Like tide-disco, every response
333/// carries permissive CORS headers.
334pub fn router(state: ProxyGlobalState<SeqTypes>) -> Router {
335    let state: SharedState = Arc::new(state);
336    let api = Router::new()
337        .nest("/block_info", block_info_router(state.clone()))
338        .nest("/txn_submit", txn_submit_router(state));
339    Router::new()
340        .route("/healthcheck", get(healthcheck))
341        .merge(api.clone())
342        .nest("/v0", api)
343        .layer(body_limit_layer())
344        .layer(cors_layer())
345}

392    /// tide-disco served both modules at `/v0/<module>/...` and redirected the unversioned paths
393    /// there, so both forms must route. Nesting the version after the module path instead
394    /// (`/block_info/v0/...`) silently 404s the canonical versioned URLs.
395    #[tokio::test]
396    async fn versioned_and_unversioned_module_paths_route() {
397        for uri in [
398            "/block_info/builderaddress",
399            "/v0/block_info/builderaddress",
400            "/txn_submit/status/abc",
401            "/v0/txn_submit/status/abc",
402        ] {
403            assert_ne!(
404                request(Method::GET, uri).await.status(),
405                AxumStatusCode::NOT_FOUND,
406                "{uri} did not route"
407            );
408        }
409        assert_eq!(
410            request(Method::GET, "/block_info/v0/builderaddress")
411                .await
412                .status(),
413            AxumStatusCode::NOT_FOUND,
414            "the version belongs before the module path, not after"
415        );
416    }

418    /// Like tide-disco, every response carries permissive CORS headers; the node's builder client
419    /// is not a browser, but the demo tooling and dashboards are.
420    #[tokio::test]
421    async fn responses_carry_cors_headers() {
422        for (method, uri) in [
423            (Method::GET, "/healthcheck"),
424            (Method::GET, "/v0/block_info/builderaddress"),
425            (Method::GET, "/no/such/route"),
426        ] {
427            let resp = request(method.clone(), uri).await;

430                    .get(header::ACCESS_CONTROL_ALLOW_ORIGIN)
431                    .unwrap_or_else(|| panic!("no CORS header on {uri}")),
432                "*",
433                "{uri}"
434            );
