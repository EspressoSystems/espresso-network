60#[derive(Clone, Copy, Debug, Default, Deserialize, Serialize, Hash, PartialEq, Eq)]
61pub struct L1Snapshot {
62    /// The relevant snapshot of the L1 includes a reference to the current head of the L1 chain.
63    ///
64    /// Note that the L1 head is subject to changing due to a reorg. However, no reorg will change
65    /// the _number_ of this block in the chain: L1 block numbers will always be sequentially
66    /// increasing. Therefore, the sequencer does not have to worry about reorgs invalidating this
67    /// snapshot.
68    pub head: u64,
69
70    /// The snapshot also includes information about the latest finalized L1 block.
71    ///
72    /// Since this block is finalized (ie cannot be reorged) we can include specific information
73    /// about the particular block, such as its hash and timestamp.
74    ///
75    /// This block may be `None` in the rare case where Espresso has started shortly after the
76    /// genesis of the L1, and the L1 has yet to finalize a block. In all other cases it will be
77    /// `Some`.
78    pub finalized: Option<L1BlockInfo>,
79}

223        default_value = "60m",

229    ///
230    /// This prevents infinite retries by panicking if the total number of retries exceed the maximum duration.
231    /// This is helpful in cases where the RPC block range limit or the event return limit is hit,
232    /// or if there is an outage. In such cases, panicking ensures that the node operator can take
233    /// action instead of the node getting stuck indefinitely. This is necessary because the stake table is constructed
234    /// from the fetched events, and is required for node to participate in consensus.
235    #[clap(
236        long,
237        env = "ESPRESSO_L1_EVENTS_MAX_RETRY_DURATION",
238        default_value = "20m",
239        value_parser = parse_duration,
240    )]
241    pub l1_events_max_retry_duration: Duration,
242
243    /// A block range which is expected to contain the finalized heads of all L1 provider chains.
244    ///
245    /// It is assumed that if a block `n` is known to be finalized according to a certain
246    /// provider, then any block less than `n - L1_FINALIZED_SAFETY_MARGIN` is finalized
247    /// _according to any provider_. In other words, if we fail over from one provider to another,
248    /// the second provider will never be lagging the first by more than this margin.
249    ///
250    /// This allows us to quickly query for very old finalized blocks by number. Without this
251    /// assumption, we always need to verify that a block is finalized by fetching all blocks in a
252    /// hash chain between the known finalized block and the desired block, recomputing and checking
253    /// the hashes. This is fine and good for blocks very near the finalized head, but for
254    /// extremely old blocks it is prohibitively expensive, and these old blocks are extremely
255    /// unlikely to be unfinalized anyways.
256    ///
257    /// Set to `unlimited` to hash-chain verify every block instead.
258    #[clap(
259        long,
260        env = "ESPRESSO_L1_FINALIZED_SAFETY_MARGIN",
261        default_value = "100"
262    )]
263    pub l1_finalized_safety_margin: L1SafetyMargin,
264
265    #[clap(skip = Arc::<Box<dyn Metrics>>::new(Box::new(NoMetrics)))]
266    pub metrics: Arc<Box<dyn Metrics>>,
267}

278/// An Ethereum provider and configuration to interact with the L1.
279///
280/// This client runs asynchronously, updating an in-memory snapshot of the relevant L1 information
281/// each time a new L1 block is published. The main advantage of this is that we can update the L1
282/// state at the pace of the L1, instead of the much faster pace of HotShot consensus.This makes it
283/// easy to use a subscription instead of polling for new blocks, vastly reducing the number of L1
284/// RPC calls we make.
285pub struct L1Client {
286    /// The alloy provider used for L1 communication with wallet and default fillers
287    #[deref]
288    pub provider: L1Provider,
289    /// Actual transport used in `self.provider`
290    /// i.e. the `t` variable in `ProviderBuilder::new().on_client(RpcClient::new(t, is_local))`
291    pub transport: SwitchingTransport,
292    /// Shared state updated by an asynchronous task which polls the L1.
293    pub(crate) state: Arc<Mutex<L1State>>,
294    /// Channel used by the async update task to send events to clients.
295    pub(crate) sender: Sender<L1Event>,
296    /// Receiver for events from the async update task.
297    pub(crate) receiver: InactiveReceiver<L1Event>,
298    /// Async task which updates the shared state.
299    pub(crate) update_task: Arc<L1UpdateTask>,
300}
301
302#[cfg(feature = "node")]
303impl Provider for L1Client {
304    fn root(&self) -> &RootProvider {
305        self.provider.root()
306    }
307}

309/// In-memory view of the L1 state, updated asynchronously.
310#[cfg(feature = "node")]
311#[derive(Debug)]
312pub(crate) struct L1State {
313    pub(crate) snapshot: L1Snapshot,
314    pub(crate) finalized: LruCache<u64, L1BlockInfoWithParent>,
315    pub(crate) last_finalized: Option<u64>,
316    /// When a waiter last refreshed the head from the RPC, throttling `refresh_head` to one call
317    /// per `WAIT_REFRESH_INTERVAL` across every concurrent waiter.
318    pub(crate) last_head_refresh: Option<Instant>,
319    /// When a waiter last refreshed the finalized block from the RPC, throttling
320    /// `refresh_finalized` to one call per `WAIT_REFRESH_INTERVAL` across every concurrent
321    /// waiter. Independent of `last_head_refresh`, so a head wait can't starve a finalized wait
322    /// or vice versa.
323    pub(crate) last_finalized_refresh: Option<Instant>,
324}

347/// An RPC client with multiple remote (HTTP) providers.
348///
349/// This client utilizes one RPC provider at a time, but if it detects that the provider is in a
350/// failing state, it will automatically switch to the next provider in its list.
351#[cfg(feature = "node")]
352#[derive(Clone, derive_more::Debug)]
353pub struct SwitchingTransport {

355    pub(crate) current_transport: Arc<RwLock<SingleTransport>>,
356    /// The list of configured HTTP URLs to use for RPC requests
357    #[debug("{:?}", redact_urls(urls.iter()))]
358    pub(crate) urls: Arc<Vec<Url>>,
359    pub(crate) opt: Arc<L1ClientOptions>,
360    pub(crate) metrics: L1ClientMetrics,
361    pub(crate) switch_notify: Arc<Notify>,
362}
