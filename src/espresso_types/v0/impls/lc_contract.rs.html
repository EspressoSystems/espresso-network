15/// The LC contract's `finalizedState` at one L1 finalized block: `F` and the block commitment
16/// root of the header at `F`. Only headers below `F` are provable against that header.
17///
18/// An `epoch_height` of 5 or less is unsupported: the chain cannot run that way.
19#[derive(Clone, Debug, PartialEq)]
20pub struct LcContractFinalizedState {
21    /// `finalizedState.blockHeight`: `F`.
22    pub height: u64,
23    /// `finalizedState.blockCommRoot`: hash of the block Merkle root of the header at `F`
24    /// (`hotshot_types::light_client::CircuitField`).
25    pub block_comm_root: CircuitField,
26}
27
28/// What a seed installs for a catchup to a target epoch.
29#[derive(Clone, Debug, PartialEq)]
30pub struct SeedWindow {
31    /// Epochs whose stake tables are installed, ascending.
32    pub epochs: RangeInclusive<EpochNumber>,
33    /// Height of the epoch root below the lowest one, when its block reward reads it.
34    pub preceding_root: Option<u64>,
35}
36
37impl SeedWindow {
38    /// Heights of the epoch roots that define the window's tables, ascending.
39    pub fn root_heights(&self, epoch_height: u64) -> impl Iterator<Item = u64> {
40        (self.epochs.start().u64()..=self.epochs.end().u64())
41            .map(move |epoch| root_block_in_epoch(epoch - 2, epoch_height))
42    }
43}
44
45/// Height of the epoch root below the one that defines the table of `epoch`, when the block reward
46/// of that table reads its timestamp; `None` near genesis.
47pub fn preceding_root_height(
48    epoch: EpochNumber,
49    first_epoch: EpochNumber,
50    epoch_height: u64,
51) -> Option<u64> {
52    (epoch > first_epoch + 4).then(|| root_block_in_epoch(*epoch - 3, epoch_height))
53}
54
55impl LcContractFinalizedState {
56    /// The epoch of `F`.
57    pub fn seed_epoch(&self, epoch_height: u64) -> EpochNumber {
58        EpochNumber::new(epoch_from_block_number(self.height, epoch_height))
59    }
60
61    /// The window a seed installs, or `None` when the walk needs no seed.
62    pub fn seed_window(
63        &self,
64        epoch_height: u64,
65        first_epoch: EpochNumber,
66        // The highest epoch whose stake table and the one below it are held; the walk starts here.
67        start: EpochNumber,
68        target: EpochNumber,
69    ) -> Option<SeedWindow> {
70        let contract_epoch = self.seed_epoch(epoch_height);
71        // The walk reaches the target, or the contract epoch, on its own from `start`.
72        if target <= start + 1 || contract_epoch <= start + 1 {
73            return None;
74        }
75        // Only epoch roots below the finalized height can be proven.
76        let last = target.min(contract_epoch + 1);
77        // Five tables cover the DRB and block rewards of the target; held tables are not
78        // installed again; epoch 3 is the first table an epoch root defines.
79        let first = EpochNumber::new(last.saturating_sub(4).max(*start + 1).max(3));
80        let preceding_root = preceding_root_height(first, first_epoch, epoch_height);
81        Some(SeedWindow {
82            epochs: first..=last,
83            preceding_root,
84        })
85    }
86
87    /// Accepts `header` iff it is at `F` and `Header::get_light_client_state(..).block_comm_root`
88    /// equals the LC contract's, and returns its block Merkle root, the only part to trust. Pure.
89    pub fn verify_header(&self, header: &Header) -> anyhow::Result<BlockMerkleCommitment> {
90        ensure!(
91            header.height() == self.height,
92            "header is at height {}, expected {}",
93            header.height(),
94            self.height
95        );
96        let block_comm_root = header
97            .get_light_client_state(ViewNumber::new(0))?
98            .block_comm_root;
99        ensure!(
100            block_comm_root == self.block_comm_root,
101            "header at height {} does not match the LC contract block commitment root",
102            self.height
103        );
104        Ok(header.block_merkle_tree_root())
105    }
106}

273    proptest! {
274        #[test]
275        fn seed_window_ok(
276            height in 1u64..100_000,
277            epoch_height in prop::sample::select(vec![10u64, 100]),
278            first_epoch in 1u64..300,
279            start_above_first in 1u64..120,
280            target in 0u64..500,
281        ) {
282            let lc = finalized_state(height);
283            let contract_epoch = *lc.seed_epoch(epoch_height);
284            let start = first_epoch + start_above_first;
285            let window = lc.seed_window(
286                epoch_height,
287                EpochNumber::new(first_epoch),
288                EpochNumber::new(start),
289                EpochNumber::new(target),
290            );
291
292            prop_assert_eq!(
293                window.is_some(),
294                target > start + 1 && contract_epoch > start + 1
295            );
296            let Some(window) = window else { return Ok(()) };
297            let (first, last) = (window.epochs.start().u64(), window.epochs.end().u64());
298            let heights: Vec<u64> = window.root_heights(epoch_height).collect();
299
300            prop_assert!(first >= first_epoch + 2 && first > start);
301            prop_assert!(last <= target && last <= contract_epoch + 1);
302            prop_assert_eq!(heights.len() as u64, last + 1 - first);
303            for (epoch, root) in (first..=last).zip(&heights) {
304                prop_assert_eq!(*root, root_block_in_epoch(epoch - 2, epoch_height));
305                prop_assert!(*root > 0 && *root < height);
306            }
307            // The DRB of the target needs the two tables below it, installed or held.
308            for below in [last - 2, last - 1] {
309                prop_assert!((first..=last).contains(&below) || below == start || below + 1 == start);
310            }
311            // The reward of the lowest table reads the root one epoch below its own root unless
312            // the epoch before that root's is within the first two.
313            let previous_epoch = epoch_from_block_number(heights[0], epoch_height) - 1;
314            prop_assert_eq!(
315                window.preceding_root,
316                (previous_epoch > first_epoch + 1).then(|| heights[0] - epoch_height)
317            );
318        }
319    }
