1//! Types related to a namespace payload and its transaction table.
2//!
3//! All code that needs to know the binary format of a namespace payload and its
4//! transaction table is restricted to this file.
5//!
6//! There are many newtypes in this file to facilitate transaction proofs.
7//!
8//! # Binary format of a namespace payload
9//!
10//! Any sequence of bytes is a valid [`NsPayload`].
11//!
12//! A namespace payload consists of two concatenated byte sequences:
13//! - `tx_table`: transaction table
14//! - `tx_payloads`: transaction payloads
15//!
16//! # Transaction table
17//!
18//! Byte lengths for the different items that could appear in a `tx_table` are
19//! specified in local private constants [`NUM_TXS_BYTE_LEN`],
20//! [`TX_OFFSET_BYTE_LEN`].
21//!
22//! ## Number of entries in the transaction table
23//!
24//! The first [`NUM_TXS_BYTE_LEN`] bytes of the `tx_table` indicate the number
25//! `n` of entries in the table as a little-endian unsigned integer. If the
26//! entire namespace payload byte length is smaller than [`NUM_TXS_BYTE_LEN`]
27//! then the missing bytes are zero-padded.
28//!
29//! The bytes in the namespace payload beyond the first [`NUM_TXS_BYTE_LEN`]
30//! bytes encode entries in the `tx_table`. Each entry consumes exactly
31//! [`TX_OFFSET_BYTE_LEN`] bytes.
32//!
33//! The number `n` could be anything, including a number much larger than the
34//! number of entries that could fit in the namespace payload. As such, the
35//! actual number of entries in the `tx_table` is defined as the minimum of `n`
36//! and the maximum number of whole `tx_table` entries that could fit in the
37//! namespace payload.
38//!
39//! The `tx_payloads` consist of any bytes in the namespace payload beyond the
40//! `tx_table`.
41//!
42//! ## Transaction table entry
43//!
44//! Each entry in the `tx_table` is exactly [`TX_OFFSET_BYTE_LEN`] bytes. These
45//! bytes indicate the end-index of a transaction in the namespace payload
46//! bytes. This end-index is a little-endian unsigned integer.
47//!
48//! This offset is relative to the end of the `tx_table` within the current
49//! namespace.
50//!
51//! ### Example
52//!
53//! Suppose a block payload has 3000 bytes and 3 namespaces of 1000 bytes each.
54//! Suppose the `tx_table` for final namespace in the block has byte length 100,
55//! and suppose an entry in that `tx_table` indicates an end-index of `10`. The
56//! actual end-index of that transaction relative to the current namespace is
57//! `110`: `10` bytes for the offset plus `100` bytes for the `tx_table`.
58//! Relative to the entire block payload, the end-index of that transaction is
59//! `2110`: `10` bytes for the offset plus `100` bytes for the `tx_table` plus
60//! `2000` bytes for this namespace.
61//!
62//! # How to deduce a transaction's byte range
63//!
64//! In order to extract the payload bytes of a single transaction `T` from the
65//! namespace payload one needs both the start- and end-indices for `T`.
66//!
67//! See [`TxPayloadRange::new`] for clarification. What follows is a description
68//! of what's implemented in [`TxPayloadRange::new`].
69//!
70//! If `T` occupies the `i`th entry in the `tx_table` for `i>0` then the
71//! start-index for `T` is defined as the end-index of the `(i-1)`th entry in
72//! the table.
73//!
74//! Thus, both start- and end-indices for any transaction `T` can be read from a
75//! contiguous, constant-size byte range in the `tx_table`. This property
76//! facilitates transaction proofs.
77//!
78//! The start-index of the 0th entry in the table is implicitly defined to be
79//! `0`.
80//!
81//! The start- and end-indices `(declared_start, declared_end)` declared in the
82//! `tx_table` could be anything. As such, the actual start- and end-indices
83//! `(start, end)` are defined so as to ensure that the byte range is
84//! well-defined and in-bounds for the namespace payload:
85//! ```ignore
86//! end = min(declared_end, namespace_payload_byte_length)
87//! start = min(declared_start, end)
88//! ```
89//!
90//! To get the byte range for `T` relative to the current namespace, the above
91//! range is translated by the byte length of the `tx_table` *as declared in the
92//! `tx_table` itself*, suitably truncated to fit within the current namespace.
93//!
94//! In particular, if the `tx_table` declares a huge number `n` of entries that
95//! cannot fit into the namespace payload then all transactions in this
96//! namespace have a zero-length byte range whose start- and end-indices are
97//! both `namespace_payload_byte_length`.
98//!
99//! In a "honestly-prepared" `tx_table` the end-index of the final transaction
100//! equals the byte length of the namespace payload minus the byte length of the
101//! `tx_table`. (Otherwise the namespace payload might have bytes that are not
102//! included in any transaction.)
103//!
104//! It is possible that a `tx_table` table could indicate two distinct
105//! transactions whose byte ranges overlap, though no "honestly-prepared"
106//! `tx_table` would do this.

123impl NumTxs {
124    /// Returns the minimum of:
125    /// - `num_txs`
126    /// - The maximum number of tx table entries that could fit in a namespace
127    ///   whose byte length is `byte_len`.
128    pub fn new(num_txs: &NumTxsUnchecked, byte_len: &NsPayloadByteLen) -> Self {
129        Self(std::cmp::min(
130            // Number of txs declared in the tx table
131            num_txs.0,
132            // Max number of tx table entries that could fit in the namespace payload
133            byte_len.0.saturating_sub(NUM_TXS_BYTE_LEN) / TX_OFFSET_BYTE_LEN,
134        ))
135    }
136
137    pub fn in_bounds(&self, index: &TxIndex) -> bool {
138        index.0 < self.0
139    }
140}

188    {

190    }

193impl<'de> Deserialize<'de> for TxTableEntries {
194    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
195    where
196        D: Deserializer<'de>,
197    {
198        let bytes = Vec::<u8>::deserialize(deserializer)?;
199        Self::try_from_payload_bytes(&bytes).ok_or_else(|| {
200            serde::de::Error::invalid_length(bytes.len(), &"one or two 4-byte tx table entries")
201        })
202    }
203}

205impl TxTableEntries {
206    const TWO_ENTRIES_BYTE_LEN: usize = 2 * TX_OFFSET_BYTE_LEN;
207
208    pub fn to_payload_bytes(&self) -> Vec<u8> {
209        let mut bytes = Vec::with_capacity(Self::TWO_ENTRIES_BYTE_LEN);
210        if let Some(prev) = self.prev {
211            bytes.extend(usize_to_bytes::<TX_OFFSET_BYTE_LEN>(prev));
212        }
213        bytes.extend(usize_to_bytes::<TX_OFFSET_BYTE_LEN>(self.cur));
214        bytes
215    }
216
217    fn try_from_payload_bytes(bytes: &[u8]) -> Option<Self> {
218        match bytes.len() {
219            TX_OFFSET_BYTE_LEN => Some(Self {
220                cur: usize_from_bytes::<TX_OFFSET_BYTE_LEN>(bytes),
221                prev: None,
222            }),
223            Self::TWO_ENTRIES_BYTE_LEN => Some(Self {
224                cur: usize_from_bytes::<TX_OFFSET_BYTE_LEN>(&bytes[TX_OFFSET_BYTE_LEN..]),
225                prev: Some(usize_from_bytes::<TX_OFFSET_BYTE_LEN>(
226                    &bytes[..TX_OFFSET_BYTE_LEN],
227                )),
228            }),
229            _ => None,
230        }
231    }
232}

241impl TxTableEntriesRange {
242    pub fn new(index: &TxIndex) -> Self {
243        let start = if index.0 == 0 {
244            // Special case: the desired range includes only one entry from
245            // the tx table: the first entry. This entry starts immediately
246            // following the bytes that encode the tx table length.
247            NUM_TXS_BYTE_LEN
248        } else {
249            // The desired range starts at the beginning of the previous tx
250            // table entry.
251            (index.0 - 1)
252                .saturating_mul(TX_OFFSET_BYTE_LEN)
253                .saturating_add(NUM_TXS_BYTE_LEN)
254        };
255        // The desired range ends at the end of this transaction's tx table entry
256        let end = index
257            .0
258            .saturating_add(1)
259            .saturating_mul(TX_OFFSET_BYTE_LEN)
260            .saturating_add(NUM_TXS_BYTE_LEN);
261        Self(start..end)
262    }
263}

285impl TxPayloadRange {
286    pub fn new(
287        num_txs: &NumTxsUnchecked,
288        tx_table_entries: &TxTableEntries,
289        byte_len: &NsPayloadByteLen,
290    ) -> Self {
291        let tx_table_byte_len = num_txs
292            .0
293            .saturating_mul(TX_OFFSET_BYTE_LEN)
294            .saturating_add(NUM_TXS_BYTE_LEN);
295        let end = tx_table_entries
296            .cur
297            .saturating_add(tx_table_byte_len)
298            .min(byte_len.0);
299        let start = tx_table_entries
300            .prev
301            .unwrap_or(0)
302            .saturating_add(tx_table_byte_len)
303            .min(end);
304        Self(start..end)
305    }
306}

342impl<'a> NsPayloadBuilder<'a> {
343    /// Add a transaction's payload to this namespace
344    pub fn append_tx(&mut self, tx: &'a Transaction) {
345        self.bodies_len += tx.payload().len();
346        self.txs.push(tx);
347    }
348
349    /// Byte length of the serialized namespace.
350    pub(crate) fn byte_len(&self) -> usize {
351        Self::tx_table_header_byte_len() + self.txs.len() * TX_OFFSET_BYTE_LEN + self.bodies_len
352    }
353
354    /// Write the serialized namespace to the start of `out` and return the byte count.
355    pub(crate) fn write_into(&self, out: &mut [u8]) -> Result<usize, BlockBuildingError> {
356        let len = self.byte_len();
357        let actual = out.len();
358        let out = out
359            .get_mut(..len)
360            .ok_or(BlockBuildingError::NsPayloadBufferTooShort {
361                expected: len,
362                actual,
363            })?;
364        let num_txs = NumTxsUnchecked(self.txs.len());
365        let (header, rest) = out.split_at_mut(NUM_TXS_BYTE_LEN);
366        header.copy_from_slice(&num_txs.to_payload_bytes());
367        let (entries, bodies) = rest.split_at_mut(self.txs.len() * TX_OFFSET_BYTE_LEN);
368        let mut end = 0;
369        for (entry, tx) in entries
370            .as_chunks_mut::<TX_OFFSET_BYTE_LEN>()
371            .0
372            .iter_mut()
373            .zip(&self.txs)
374        {
375            let body = tx.payload();
376            bodies[end..end + body.len()].copy_from_slice(body);
377            end += body.len();
378            *entry = usize_to_bytes::<TX_OFFSET_BYTE_LEN>(end);
379        }
380        debug_assert_eq!(end, self.bodies_len);
381        Ok(len)
382    }
383
384    /// Byte length of a tx table header.
385    pub const fn tx_table_header_byte_len() -> usize {
386        NUM_TXS_BYTE_LEN
387    }
388
389    /// Byte length of a single tx table entry.
390    pub const fn tx_table_entry_byte_len() -> usize {
391        TX_OFFSET_BYTE_LEN
392    }
393}

399    #[test]
400    fn tx_table_entries_round_trip() {
401        for entries in [
402            TxTableEntries { cur: 7, prev: None },
403            TxTableEntries {
404                cur: 7,
405                prev: Some(0),
406            },
407        ] {
408            let bytes = bincode::serialize(&entries).unwrap();
409            assert_eq!(
410                bincode::deserialize::<TxTableEntries>(&bytes).unwrap(),
411                entries
412            );
413            let json = serde_json::to_string(&entries).unwrap();
414            assert_eq!(
415                serde_json::from_str::<TxTableEntries>(&json).unwrap(),
416                entries
417            );
418        }
419    }

421    #[test]
422    fn tx_table_entries_reject_unexpected_byte_len() {
423        for len in [0, 3, 5, 12] {
424            let bytes = vec![0u8; len];

426                bincode::deserialize::<TxTableEntries>(&bincode::serialize(&bytes).unwrap())

431                serde_json::from_str::<TxTableEntries>(&serde_json::to_string(&bytes).unwrap())
432                    .is_err(),
433                "json accepted {len} bytes"
