17impl ADVZNsProof {
18    /// Returns the payload bytes for the `index`th namespace, along with a
19    /// proof of correctness for those bytes. Returns `None` on error.
20    ///
21    /// The namespace payload [`NsPayloadOwned`] is included as a hidden field
22    /// in the returned [`NsProof`]. A conventional API would instead return
23    /// `(NsPayload, NsProof)` and [`NsProof`] would not contain the namespace
24    /// payload.
25    /// ([`TxProof::new`](crate::block::namespace_payload::TxProof::new)
26    /// conforms to this convention.) In the future we should change this API to
27    /// conform to convention. But that would require a change to our RPC
28    /// endpoint API, which is a hassle.
29    pub fn new(payload: &Payload, index: &NsIndex, common: &ADVZCommon) -> Option<ADVZNsProof> {
30        let payload_byte_len = payload.byte_len();
31        if !payload_byte_len.is_consistent(common) {
32            tracing::warn!(
33                "payload byte len {} inconsistent with common {}",
34                payload_byte_len,
35                ADVZScheme::get_payload_byte_len(common)
36            );
37            return None; // error: payload byte len inconsistent with common
38        }
39        if !payload.ns_table().in_bounds(index) {
40            tracing::warn!("ns_index {:?} out of bounds", index);
41            return None; // error: index out of bounds
42        }
43        let ns_payload_range = payload.ns_table().ns_range(index, &payload_byte_len);
44
45        // TODO vid_scheme() arg should be u32 to match get_num_storage_nodes
46        // https://github.com/EspressoSystems/HotShot/issues/3298
47        let vid = advz_scheme(
48            ADVZScheme::get_num_storage_nodes(common).try_into().ok()?, // error: failure to convert u32 to usize
49        );
50
51        let ns_proof = if ns_payload_range.as_block_range().is_empty() {
52            None
53        } else {
54            Some(
55                vid.payload_proof(payload.encode(), ns_payload_range.as_block_range())
56                    .ok()?, // error: internal to payload_proof()
57            )
58        };
59
60        Some(ADVZNsProof {
61            ns_index: index.clone(),
62            ns_payload: payload.read_ns_payload(&ns_payload_range).to_owned(),
63            ns_proof,
64        })
65    }

67    /// Verify a [`NsProof`] against a payload commitment. Returns `None` on
68    /// error or if verification fails.
69    ///
70    /// There is no [`NsPayload`](crate::block::namespace_payload::NsPayload)
71    /// arg because this data is already included in the [`NsProof`]. See
72    /// [`NsProof::new`] for discussion.
73    ///
74    /// If verification is successful then return `(Vec<Transaction>,
75    /// NamespaceId)` obtained by post-processing the underlying
76    /// [`NsPayload`](crate::block::namespace_payload::NsPayload). Why? This
77    /// method might be run by a client in a WASM environment who might be
78    /// running non-Rust code, in which case the client is unable to perform
79    /// this post-processing himself.
80    pub fn verify(
81        &self,
82        ns_table: &NsTable,
83        commit: &VidCommitment,
84        common: &ADVZCommon,
85    ) -> Option<(Vec<Transaction>, NamespaceId)> {
86        match commit {
87            VidCommitment::V0(commit) => {
88                ADVZScheme::is_consistent(commit, common).ok()?;
89                if !ns_table.in_bounds(&self.ns_index) {
90                    return None; // error: index out of bounds
91                }
92
93                let range = ns_table
94                    .ns_range(&self.ns_index, &PayloadByteLen::from_vid_common(common))
95                    .as_block_range();
96
97                match (&self.ns_proof, range.is_empty()) {
98                    (Some(proof), false) => {
99                        // TODO advz_scheme() arg should be u32 to match get_num_storage_nodes
100                        // https://github.com/EspressoSystems/HotShot/issues/3298
101                        let vid = advz_scheme(
102                            ADVZScheme::get_num_storage_nodes(common).try_into().ok()?, // error: failure to convert u32 to usize
103                        );
104
105                        vid.payload_verify(
106                            Statement {
107                                payload_subslice: self.ns_payload.as_bytes_slice(),
108                                range,
109                                commit,
110                                common,
111                            },
112                            proof,
113                        )
114                        .ok()? // error: internal to payload_verify()
115                        .ok()?; // verification failure
116                    },
117                    (None, true) => {
118                        if !self.ns_payload.as_bytes_slice().is_empty() {
119                            tracing::info!(
120                                "ns verify: nonempty ns payload for an empty ns payload range"
121                            );
122                            return None;
123                        }
124                    },
125                    (None, false) => {
126                        tracing::error!(
127                            "ns verify: missing proof for nonempty ns payload range {:?}",
128                            range
129                        );
130                        return None;
131                    },
132                    (Some(_), true) => {
133                        tracing::error!("ns verify: unexpected proof for empty ns payload range");
134                        return None;
135                    },
136                }
137
138                // verification succeeded, return some data
139                let ns_id = ns_table.read_ns_id_unchecked(&self.ns_index);
140                Some((self.ns_payload.export_all_txs(&ns_id), ns_id))
141            },
142            _ => None,
143        }
144    }
145
146    /// Return all transactions in the namespace whose payload is proven by
147    /// `self`. The namespace ID for each returned [`Transaction`] is set to
148    /// `ns_id`.
149    ///
150    /// # Design warning
151    ///
152    /// This method relies on a promise that a [`NsProof`] stores the entire
153    /// namespace payload. If in the future we wish to remove the payload from a
154    /// [`NsProof`] then this method can no longer be supported.
155    ///
156    /// In that case, use the following a workaround:
157    /// - Given a [`NamespaceId`], get a [`NsIndex`] `i` via
158    ///   [`NsTable::find_ns_id`].
159    /// - Use `i` to get a
160    ///   [`NsPayload`](crate::block::namespace_payload::NsPayload) `p` via
161    ///   [`Payload::ns_payload`].
162    /// - Use `p` to get the desired [`Vec<Transaction>`] via
163    ///   [`NsPayload::export_all_txs`](crate::block::namespace_payload::NsPayload::export_all_txs).
164    ///
165    /// This workaround duplicates the work done in [`NsProof::new`]. If you
166    /// don't like that then you could instead hack [`NsProof::new`] to return a
167    /// pair `(NsProof, Vec<Transaction>)`.
168    pub fn export_all_txs(&self, ns_id: &NamespaceId) -> Vec<Transaction> {
169        self.ns_payload.export_all_txs(ns_id)
170    }
171}

188    #[test_log::test(tokio::test(flavor = "multi_thread"))]
189    async fn ns_proof() {
190        let test_cases = vec![
191            vec![
192                vec![5, 8, 8],
193                vec![7, 9, 11],
194                vec![10, 5, 8],
195                vec![7, 8, 9],
196                vec![],
197            ],
198            vec![vec![1, 2, 3], vec![4, 5, 6]],
199            vec![],
200        ];
201
202        let mut rng = jf_utils::test_rng();
203        let mut tests = ValidTest::many_from_tx_lengths(test_cases, &mut rng);
204
205        struct BlockInfo {
206            block: Payload,
207            vid: VidDisperse<ADVZScheme>,
208            ns_proofs: Vec<ADVZNsProof>,
209        }
210
211        let blocks: Vec<BlockInfo> = {
212            // compute blocks separately to avoid async error `captured variable
213            // cannot escape `FnMut` closure body` caused by mutable variable `vid`
214            // below.
215            let blocks_only = future::join_all(tests.iter().map(|t| async {
216                Payload::from_transactions(&t.all_txs(), &Default::default(), &Default::default())
217                    .await
218                    .unwrap()
219                    .0
220            }))
221            .await;
222
223            let mut vid = advz_scheme(10);
224            blocks_only
225                .into_iter()
226                .map(|block| {
227                    let vid = vid.disperse(block.encode()).unwrap();
228                    let ns_proofs: Vec<ADVZNsProof> = block
229                        .ns_table()
230                        .iter()
231                        .map(|ns_index| ADVZNsProof::new(&block, &ns_index, &vid.common).unwrap())
232                        .collect();
233                    BlockInfo {
234                        block,
235                        vid,
236                        ns_proofs,
237                    }
238                })
239                .collect()
240        };
241
242        // sanity: verify all valid namespace proofs
243        for (
244            BlockInfo {
245                block,
246                vid,
247                ns_proofs,
248            },
249            test,
250        ) in blocks.iter().zip(tests.iter_mut())
251        {
252            for ns_proof in ns_proofs.iter() {
253                let ns_id = block.ns_table().read_ns_id(&ns_proof.ns_index).unwrap();
254                let txs = test
255                    .nss
256                    .remove(&ns_id)
257                    .unwrap_or_else(|| panic!("namespace {ns_id} missing from test"));
258
259                // verify ns_proof
260                let (ns_proof_txs, ns_proof_ns_id) = ns_proof
261                    .verify(
262                        block.ns_table(),
263                        &VidCommitment::V0(vid.commit),
264                        &vid.common,
265                    )
266                    .unwrap_or_else(|| panic!("namespace {ns_id} proof verification failure"));
267
268                assert_eq!(ns_proof_ns_id, ns_id);
269                assert_eq!(ns_proof_txs, txs);
270            }
271        }
272
273        assert!(blocks.len() >= 2, "need at least 2 test_cases");
274
275        let ns_proof_0_0 = &blocks[0].ns_proofs[0];
276        let ns_table_0 = blocks[0].block.ns_table();
277        let ns_table_1 = blocks[1].block.ns_table();
278        let vid_commit_0 = &VidCommitment::V0(blocks[0].vid.commit);
279        let vid_commit_1 = &VidCommitment::V0(blocks[1].vid.commit);
280        let vid_common_0 = &blocks[0].vid.common;
281        let vid_common_1 = &blocks[1].vid.common;
282
283        // mix and match ns_table, vid_commit, vid_common
284        {
285            // wrong ns_table
286            assert!(
287                ns_proof_0_0
288                    .verify(ns_table_1, vid_commit_0, vid_common_0)
289                    .is_none()
290            );
291
292            // wrong vid commitment
293            assert!(
294                ns_proof_0_0
295                    .verify(ns_table_0, vid_commit_1, vid_common_0)
296                    .is_none()
297            );
298
299            // wrong vid common
300            assert!(
301                ns_proof_0_0
302                    .verify(ns_table_0, vid_commit_0, vid_common_1)
303                    .is_none()
304            );
305
306            // wrong ns_proof
307            assert!(
308                ns_proof_0_0
309                    .verify(ns_table_1, vid_commit_1, vid_common_1)
310                    .is_none()
311            );
312        }
313
314        // hack the proof
315        {
316            ns_proof_0_0
317                .verify(ns_table_0, vid_commit_0, vid_common_0)
318                .expect("sanity: correct proof should succeed");
319
320            let wrong_ns_index_ns_proof_0_0 = ADVZNsProof {
321                ns_index: blocks[0].ns_proofs[1].ns_index.clone(),
322                ..ns_proof_0_0.clone()
323            };
324            assert!(
325                wrong_ns_index_ns_proof_0_0
326                    .verify(ns_table_0, vid_commit_0, vid_common_0)
327                    .is_none()
328            );
329
330            let wrong_ns_payload_ns_proof_0_0 = ADVZNsProof {
331                ns_payload: blocks[0].ns_proofs[1].ns_payload.clone(),
332                ..ns_proof_0_0.clone()
333            };
334            assert!(
335                wrong_ns_payload_ns_proof_0_0
336                    .verify(ns_table_0, vid_commit_0, vid_common_0)
337                    .is_none()
338            );
339
340            let wrong_proof_ns_proof_0_0 = ADVZNsProof {
341                ns_proof: blocks[0].ns_proofs[1].ns_proof.clone(),
342                ..ns_proof_0_0.clone()
343            };
344            assert!(
345                wrong_proof_ns_proof_0_0
346                    .verify(ns_table_0, vid_commit_0, vid_common_0)
347                    .is_none()
348            );
349        }

352    /// Consensus rejects a namespace table with an empty namespace, so this
353    /// guards callers that pass `verify` a table they did not validate.
354    #[test_log::test(tokio::test(flavor = "multi_thread"))]
355    async fn empty_ns_proof_rejects_nonempty_payload() {
356        let mut rng = jf_utils::test_rng();
357        let test = ValidTest::from_tx_lengths(vec![vec![5, 8]], &mut rng);
358        let mut block =
359            Payload::from_transactions(&test.all_txs(), &Default::default(), &Default::default())
360                .await
361                .unwrap()
362                .0;
363        let byte_len = block.byte_len().as_usize();
364        let mut ns_table = NsTableBuilder::new();
365        ns_table.append_entry(block.ns_table().read_ns_id(&NsIndex(0)).unwrap(), byte_len);
366        ns_table.append_entry(NamespaceId::from(u32::MAX), byte_len);
367        *block.ns_table_mut() = ns_table.into_ns_table();
368
369        let disperse = advz_scheme(10).disperse(block.encode()).unwrap();
370        let commit = VidCommitment::V0(disperse.commit);
371        let empty_ns_proof = ADVZNsProof::new(&block, &NsIndex(1), &disperse.common).unwrap();
372        let (txs, _) = empty_ns_proof
373            .verify(block.ns_table(), &commit, &disperse.common)
374            .unwrap();
375        assert!(txs.is_empty());
376
377        let forged = ADVZNsProof {
378            ns_payload: ADVZNsProof::new(&block, &NsIndex(0), &disperse.common)
379                .unwrap()
380                .ns_payload,
381            ..empty_ns_proof
382        };
383        assert!(
384            forged
385                .verify(block.ns_table(), &commit, &disperse.common)
386                .is_none()
387        );
388    }
