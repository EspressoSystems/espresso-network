29impl From<ValidatorConfig<SeqTypes>> for PublicValidatorConfig {

33            private_key: _,
34            stake_value,
35            state_public_key,

51        }
52    }

276impl PublicNetworkConfig {
277    pub fn into_network_config(
278        self,
279        my_own_validator_config: ValidatorConfig<SeqTypes>,
280    ) -> anyhow::Result<NetworkConfig<SeqTypes>> {
281        let node_index = self
282            .config
283            .known_nodes_with_stake
284            .iter()
285            .position(|peer| peer.stake_table_entry.stake_key == my_own_validator_config.public_key)
286            .unwrap_or(0) as u64;
287
288        Ok(NetworkConfig {
289            rounds: self.rounds,
290            indexed_da: self.indexed_da,
291            transactions_per_round: self.transactions_per_round,
292            manual_start_password: self.manual_start_password,
293            num_bootrap: self.num_bootrap,
294            next_view_timeout: self.next_view_timeout,
295            view_sync_timeout: self.view_sync_timeout,
296            builder_timeout: self.builder_timeout,
297            data_request_delay: self.data_request_delay,
298            node_index,
299            seed: self.seed,
300            transaction_size: self.transaction_size,
301            key_type_name: self.key_type_name,
302            config: self.config.into_hotshot_config(),
303            commit_sha: self.commit_sha,
304            builder: self.builder,
305            random_builder: self.random_builder,
306            public_keys: Vec::new(),
307        })
308    }

366    /// A node from a release before this one fetches its config from peers and fails to start
367    /// unless the JSON carries a libp2p section, so the key is served with no bootstrap nodes.
368    #[test]
369    fn serves_an_empty_libp2p_section_for_older_peers() {
370        let config = PublicNetworkConfig::from(NetworkConfig::<SeqTypes>::default());
371        let json = serde_json::to_value(&config).unwrap();
372        assert_eq!(
373            json["libp2p_config"],
374            serde_json::json!({ "bootstrap_nodes": [] })
375        );
376        // Incoming values, including the non-empty lists older peers serve, are ignored.
377        let mut with_peers = json.clone();
378        with_peers["libp2p_config"] =
379            serde_json::json!({ "bootstrap_nodes": [["12D3KooW", "/ip4/10.0.0.1/tcp/1769"]] });
380        let parsed: PublicNetworkConfig = serde_json::from_value(with_peers).unwrap();
381        assert!(parsed.libp2p_config.bootstrap_nodes.is_empty());
382    }
