20/// The smallest range holding both.
21fn hull_of<T: Ord + Copy>(a: &Range<T>, b: &Range<T>) -> Range<T> {
22    a.start.min(b.start)..a.end.max(b.end)
23}

25/// Rows of `map` overlapping `heights`, whole.
26fn overlapping_rows(

28    heights: &Range<Height>,
29) -> Vec<(Range<Height>, UnitState)> {
30    map.overlapping(heights)

35/// Applies `observation` at `now` to the unit `key` of `rows`; whether the row changed.
36fn observe_unit<K: Ord + Copy, U: Ord + Copy>(
37    rows: &mut BTreeMap<(K, U), UnitState>,
38    key: (K, U),
39    observation: Observation,
40    now: DateTime<Utc>,
41) -> bool {
42    let old = rows.get(&key).copied();
43    let new = transition(old, observation, now);
44    match new {
45        Some(state) => rows.insert(key, state),
46        None => rows.remove(&key),
47    };
48    new != old
49}

51/// Applies `observation` at `now` to every unit of `piece`: the part of each row inside `piece`
52/// and the heights no row covers (read as `None`) go through `transition` and are written back.
53fn observe_piece(
54    map: &mut RangeMap<Height, UnitState>,
55    piece: Range<Height>,
56    observation: Observation,
57    now: DateTime<Utc>,
58) {
59    let mut parts: Vec<(Range<Height>, Option<UnitState>)> = map
60        .overlapping(&piece)
61        .map(|(row, state)| {
62            let inside = row.start.max(piece.start)..row.end.min(piece.end);
63            (inside, Some(*state))
64        })
65        .chain(map.gaps(&piece).map(|gap| (gap, None)))
66        .collect();
67    parts.sort_by_key(|(range, _)| range.start);
68    map.remove(piece);
69    for (range, old) in parts {
70        if let Some(new) = transition(old, observation, now) {
71            map.insert(range, new);
72        }
73    }
74}

76/// Removes the rows of `kind` whose unit satisfies `dropped`; the units they covered, `None` when
77/// there were none.
78fn drop_units<K: Ord + Copy, U: Ord + Copy>(
79    rows: &mut BTreeMap<(K, U), UnitState>,
80    kind: K,
81    dropped: impl Fn(U) -> bool,
82    next: impl Fn(U) -> U,
83) -> Option<Range<U>> {
84    let units: Vec<U> = rows
85        .keys()
86        .filter(|&&(row_kind, unit)| row_kind == kind && dropped(unit))
87        .map(|&(_, unit)| unit)
88        .collect();
89    let (first, last) = (*units.first()?, *units.last()?);
90    for unit in units {
91        rows.remove(&(kind, unit));
92    }
93    Some(first..next(last))
94}

113impl RowKey {
114    /// The kind label.
115    pub fn kind(&self) -> DataKind {
116        match self {
117            Self::Range { kind, .. } => (*kind).into(),
118            Self::Point { kind, .. } => (*kind).into(),
119            Self::Epoch { kind, .. } => (*kind).into(),
120        }
121    }
122
123    /// Heights or epochs covered: range length, or 1.
124    pub fn unit_count(&self) -> u64 {
125        match self {
126            Self::Range { heights, .. } => height_count(heights),
127            Self::Point { .. } | Self::Epoch { .. } => 1,
128        }
129    }
130
131    /// Probe of the row's first `CHUNK_LEN` heights (`kind::next_chunk` from the start), or of the
132    /// point or epoch; the page link.
133    pub fn first_probe(&self) -> Probe {
134        match self {
135            Self::Range { kind, heights } => Probe::Range {
136                kind: *kind,
137                heights: next_chunk(heights.start, heights.end)
138                    .expect("a row covers at least one height"),
139            },
140            Self::Point { kind, height } => Probe::Point {
141                kind: *kind,
142                height: *height,
143            },
144            Self::Epoch { kind, epoch } => Probe::Epoch {
145                kind: *kind,
146                epoch: *epoch,
147            },
148        }
149    }
150
151    /// First unit, for ordering: range start, height or epoch number.
152    pub fn first_unit(&self) -> u64 {
153        match self {
154            Self::Range { heights, .. } => heights.start.0,
155            Self::Point { height, .. } => height.0,
156            Self::Epoch { epoch, .. } => epoch.0,
157        }
158    }
159
160    /// Block of the first unit: range start, point height, or the epoch's root block (`None` when
161    /// `Off` or on overflow).
162    pub fn first_block(&self, epochs: Epochs) -> Option<Height> {
163        match self {
164            Self::Range { heights, .. } => Some(heights.start),
165            Self::Point { height, .. } => Some(*height),
166            Self::Epoch { kind, epoch } => epochs.root_block(*kind, *epoch),
167        }
168    }
169
170    /// The units covered: heights or epoch numbers.
171    pub fn units(&self) -> Range<u64> {
172        let first = self.first_unit();
173        first..first + self.unit_count()
174    }
175
176    /// The probe of the first `CHUNK_LEN` heights and the rest of a range key; point and epoch
177    /// keys have no rest.
178    pub fn split_first_chunk(self) -> (Probe, Option<RowKey>) {
179        let probe = self.first_probe();
180        match (self, &probe) {
181            (Self::Range { kind, heights }, Probe::Range { heights: chunk, .. })
182                if chunk.end < heights.end =>
183            {
184                (
185                    probe.clone(),
186                    Some(Self::Range {
187                        kind,
188                        heights: chunk.end..heights.end,
189                    }),
190                )
191            },
192            _ => (probe, None),
193        }
194    }

215/// The only kinds of change to a missing set.

217pub enum MissingUpdate {
218    /// A probe result.
219    Observed(ProbeResult),
220    /// Drops heights below the window start: range rows are trimmed, point rows below go, epoch

229/// Units of one kind whose image rows an update rewrote; one variant per unit family. The region
230/// is the hull of every row it overlaps before and after, and rows of a kind are disjoint, so

232/// on the start.

234pub enum ImageRegion {
235    Range {
236        kind: RangeKind,
237        heights: Range<Height>,
238    },
239    Points {
240        kind: PointKind,
241        heights: Range<Height>,
242    },
243    Epochs {
244        kind: EpochKind,
245        epochs: Range<Epoch>,
246    },
247}

289    /// From image rows; errors on range rows of a kind that overlap (`overlapping` non-empty
290    /// before `insert`), or touch with equal state (row count after loading below the input
291    /// count).
292    pub fn from_rows(
293        epochs: Epochs,
294        rows: impl IntoIterator<Item = MissingRow>,
295    ) -> anyhow::Result<Self> {
296        let mut set = Self::new(epochs);
297        let mut input_rows = 0;
298        for MissingRow { key, state } in rows {
299            input_rows += 1;
300            match key {
301                RowKey::Range { kind, heights } => {
302                    ensure!(
303                        heights.start < heights.end,
304                        "empty {kind:?} row {heights:?}"
305                    );
306                    let map = set.ranges.entry(kind).or_default();
307                    ensure!(
308                        map.overlapping(&heights).next().is_none(),
309                        "overlapping {kind:?} rows at {heights:?}"
310                    );
311                    map.insert(heights, state);
312                },
313                RowKey::Point { kind, height } => {
314                    ensure!(
315                        set.points.insert((kind, height), state).is_none(),
316                        "duplicate {kind:?} row at {height}"
317                    );
318                },
319                RowKey::Epoch { kind, epoch } => {
320                    ensure!(
321                        set.epoch_rows.insert((kind, epoch), state).is_none(),
322                        "duplicate {kind:?} row at epoch {}",
323                        epoch.0
324                    );
325                },
326            }
327        }
328        ensure!(
329            set.row_count() == input_rows,
330            "touching rows with equal state in the image"
331        );
332        Ok(set)
333    }

335    /// Applies one update at `now`: a probe result applies `transition` to each piece it covers
336    /// (range rows split at piece bounds, units without a row read `None`); returns the regions it
337    /// rewrote. The only mutator.
338    pub fn apply(&mut self, update: MissingUpdate, now: DateTime<Utc>) -> ImageDiff {
339        let rows_before = self.row_count();
340        let regions = match update {
341            MissingUpdate::Observed(result) => self.observe(result, now).into_iter().collect(),
342            MissingUpdate::ForgetBelow(window_start) => self.forget_below(window_start),
343            MissingUpdate::ForgetAbove(settled_end) => self.forget_above(settled_end),
344        };
345        ImageDiff {
346            regions,
347            rows_removed: rows_before.saturating_sub(self.row_count()),
348        }
349    }

357    /// Applies one probe result; the region it rewrote, `None` when nothing changed.
358    fn observe(&mut self, result: ProbeResult, now: DateTime<Utc>) -> Option<ImageRegion> {
359        match result {
360            ProbeResult::Range { kind, findings } => self.observe_range(kind, &findings, now),
361            ProbeResult::Point {
362                kind,
363                height,
364                observation,
365            } => observe_unit(&mut self.points, (kind, height), observation, now).then_some(
366                ImageRegion::Points {
367                    kind,
368                    heights: height..height.next(),
369                },
370            ),
371            ProbeResult::Epoch {
372                kind,
373                epoch,
374                observation,
375            } => observe_unit(&mut self.epoch_rows, (kind, epoch), observation, now).then_some(
376                ImageRegion::Epochs {
377                    kind,
378                    epochs: epoch..epoch.next(),
379                },
380            ),
381        }
382    }

384    /// Applies the pieces of `findings`; the region (the hull of the probed range and the rows
385    /// overlapping it before and after) when a row changed, `None` otherwise.
386    fn observe_range(
387        &mut self,
388        kind: RangeKind,
389        findings: &RangeFindings,
390        now: DateTime<Utc>,
391    ) -> Option<ImageRegion> {
392        let map = self.ranges.entry(kind).or_default();
393        let probed = findings.heights();
394        let before = overlapping_rows(map, probed);
395        let region = before
396            .iter()
397            .fold(probed.clone(), |hull, (range, _)| hull_of(&hull, range));
398        for (piece, observation) in findings.pieces() {
399            observe_piece(map, piece, observation, now);
400        }
401        let after = overlapping_rows(map, &region);
402        if before == after {
403            return None;
404        }
405        let heights = after
406            .iter()
407            .fold(region, |hull, (range, _)| hull_of(&hull, range));
408        Some(ImageRegion::Range { kind, heights })
409    }

411    /// Drops what lies below `window_start`; the rewritten regions.
412    fn forget_below(&mut self, window_start: Height) -> Vec<ImageRegion> {
413        if window_start == Height(0) {
414            return Vec::new();
415        }
416        let mut regions = Vec::new();
417        let below = Height(0)..window_start;
418        for (&kind, map) in &mut self.ranges {
419            let rows = overlapping_rows(map, &below);
420            if let Some(heights) = rows
421                .iter()
422                .map(|(range, _)| range.clone())
423                .reduce(|hull, range| hull_of(&hull, &range))
424            {
425                map.remove(below.clone());
426                regions.push(ImageRegion::Range { kind, heights });
427            }
428        }
429        for &kind in PointKind::VARIANTS {
430            let dropped = |height: Height| height < window_start;
431            if let Some(heights) = drop_units(&mut self.points, kind, dropped, Height::next) {
432                regions.push(ImageRegion::Points { kind, heights });
433            }
434        }
435        let epochs = self.epochs;
436        for &kind in EpochKind::VARIANTS {
437            let dropped = |epoch: Epoch| {
438                epochs
439                    .root_block(kind, epoch)
440                    .is_some_and(|block| block < window_start)
441            };
442            if let Some(epochs) = drop_units(&mut self.epoch_rows, kind, dropped, Epoch::next) {
443                regions.push(ImageRegion::Epochs { kind, epochs });
444            }
445        }
446        regions
447    }

449    /// Drops what lies at or above `settled_end`; the rewritten regions.
450    fn forget_above(&mut self, settled_end: Height) -> Vec<ImageRegion> {
451        let mut regions = Vec::new();
452        let above = settled_end..Height(u64::MAX);
453        for (&kind, map) in &mut self.ranges {
454            let hull = overlapping_rows(map, &above)
455                .iter()
456                .map(|(range, _)| range.clone())
457                .reduce(|hull, range| hull_of(&hull, &range));
458            if let Some(heights) = hull {
459                map.remove(above.clone());
460                regions.push(ImageRegion::Range { kind, heights });
461            }
462        }
463        for &kind in PointKind::VARIANTS {
464            let dropped = |height: Height| height >= settled_end;
465            if let Some(heights) = drop_units(&mut self.points, kind, dropped, Height::next) {
466                regions.push(ImageRegion::Points { kind, heights });
467            }
468        }
469        let epochs = self.epochs;
470        for &kind in EpochKind::VARIANTS {
471            let dropped = |epoch: Epoch| {
472                epochs
473                    .root_block(kind, epoch)
474                    .is_some_and(|block| block > settled_end)
475            };
476            if let Some(epochs) = drop_units(&mut self.epoch_rows, kind, dropped, Epoch::next) {
477                regions.push(ImageRegion::Epochs { kind, epochs });
478            }
479        }
480        regions
481    }

552    /// `Known::Gap` when `heights` lies inside one `NotFound` row of `kind`, else `Known::Missing`.
553    pub fn known(&self, kind: RangeKind, heights: &Range<Height>) -> Known {
554        let inside_not_found = self
555            .ranges
556            .get(&kind)
557            .and_then(|map| map.get_key_value(&heights.start))
558            .is_some_and(|(row, state)| {
559                row.end >= heights.end && matches!(state, UnitState::NotFound { .. })
560            });
561        if inside_not_found {
562            Known::Gap
563        } else {
564            Known::Missing
565        }
566    }

577    /// One per `DataKind`: units missing and the oldest row by (`first_missing_at`, first unit).
578    pub fn tallies(&self) -> Vec<KindTally> {
579        let mut tallies: Vec<KindTally> = DataKind::VARIANTS
580            .iter()
581            .map(|&kind| KindTally {
582                kind,
583                missing_count: 0,
584                oldest: None,

586            .collect();
587        for row in self.rows() {
588            let tally = tallies
589                .iter_mut()
590                .find(|tally| tally.kind == row.key.kind())
591                .expect("a tally exists per kind");
592            tally.missing_count += row.key.unit_count();
593            let older = tally
594                .oldest
595                .as_ref()
596                .is_none_or(|old| row.age_order() < old.age_order());
597            if older {
598                tally.oldest = Some(row);
599            }
600        }
601        tallies
602    }

1624                kind,

1626                len,
1627                shape,

1630            prop::sample::select(PointKind::VARIANTS),

1650            4 => range,

1655            4 => prop_oneof![0u64..12, 0u64..700].prop_map(Op::Advance),

1661    /// Findings of `shape` over `heights`.
1662    fn findings_of(heights: Range<Height>, shape: &Shape) -> RangeFindings {
1663        let (mask, tail) = match shape {
1664            Shape::Uniform(observation) => return RangeFindings::uniform(heights, *observation),
1665            Shape::Mask { mask, tail } => (mask, tail),
1666        };
1667        let start = heights.start.0;
1668        let end = tail.map_or(mask.len(), |(offset, _)| offset as usize);
1669        let mut findings = RangeFindings::new(heights.clone());
1670        let mut run: Option<usize> = None;
1671        for (i, &on) in mask[..end].iter().chain([&false]).enumerate() {
1672            match (on, run) {
1673                (true, None) => run = Some(i),
1674                (false, Some(from)) => {
1675                    let gap = Height(start + from as u64)..Height(start + i as u64);
1676                    findings.mark(gap, Observation::NotFound);
1677                    run = None;
1678                },
1679                _ => {},
1680            }
1681        }
1682        if let Some((offset, failure)) = tail {
1683            findings.mark(
1684                Height(start + offset)..heights.end,
1685                Observation::Failed(*failure),
1686            );
1687        }
1688        findings
1689    }

1691    /// The result a retry of `probe` finds.
1692    fn found_result(probe: &Probe, found: Answer) -> ProbeResult {
1693        let unit = |observation| match probe {
1694            Probe::Point { kind, height } => ProbeResult::Point {
1695                kind: *kind,
1696                height: *height,
1697                observation,
1698            },
1699            Probe::Epoch { kind, epoch } => ProbeResult::Epoch {
1700                kind: *kind,
1701                epoch: *epoch,
1702                observation,
1703            },
1704            Probe::Range { .. } => unreachable!("units only"),
1705        };
1706        let uniform = |observation| match probe {
1707            Probe::Range { kind, heights } => ProbeResult::Range {
1708                kind: *kind,
1709                findings: RangeFindings::uniform(heights.clone(), observation),
1710            },
1711            _ => unit(observation),
1712        };
1713        let split = |first, second| match probe {
1714            Probe::Range { kind, heights } if height_count(heights) > 1 => {
1715                let mid = Height(heights.start.0 + height_count(heights) / 2);
1716                let mut findings = RangeFindings::new(heights.clone());
1717                if first != Observation::Present {
1718                    findings.mark(heights.start..mid, first);
1719                }
1720                findings.mark(mid..heights.end, second);
1721                ProbeResult::Range {
1722                    kind: *kind,
1723                    findings,
1724                }
1725            },
1726            _ => uniform(second),
1727        };
1728        match found {
1729            Answer::Present => uniform(Observation::Present),
1730            Answer::NotFound => uniform(Observation::NotFound),
1731            Answer::Failed => uniform(Observation::Failed(server_error())),
1732            Answer::Transport => uniform(Observation::Failed(RequestFailure::Transport)),
1733            Answer::GapThenFailed => {
1734                split(Observation::NotFound, Observation::Failed(server_error()))
1735            },
1736            Answer::PresentThenGap => split(Observation::Present, Observation::NotFound),
1737        }
1738    }

1755    /// A maximal run of consecutive units of one range kind with one state; one run per other
1756    /// unit.
1757    struct Run {
1758        kind: DataKind,
1759        units: Range<u64>,
1760        state: UnitState,
1761    }

1763    impl Model {
1764        fn observe_unit(&mut self, unit: Unit, observation: Observation, now: DateTime<Utc>) {
1765            match crate::state::transition(self.units.get(&unit).copied(), observation, now) {
1766                Some(state) => self.units.insert(unit, state),
1767                None => self.units.remove(&unit),
1768            };
1769        }
1770
1771        fn apply(&mut self, result: &ProbeResult, now: DateTime<Utc>) {
1772            match result {
1773                ProbeResult::Range { kind, findings } => {
1774                    for (piece, observation) in findings.pieces() {
1775                        for height in piece.start.0..piece.end.0 {
1776                            self.observe_unit(((*kind).into(), height), observation, now);
1777                        }
1778                    }
1779                },
1780                ProbeResult::Point {
1781                    kind,
1782                    height,
1783                    observation,
1784                } => self.observe_unit(((*kind).into(), height.0), *observation, now),
1785                ProbeResult::Epoch {
1786                    kind,
1787                    epoch,
1788                    observation,
1789                } => self.observe_unit(((*kind).into(), epoch.0), *observation, now),
1790            }
1791        }
1792
1793        fn forget_above(&mut self, settled_end: u64) {
1794            self.units.retain(|&(kind, unit), _| match kind.class() {
1795                KindClass::Range(_) | KindClass::Point(_) => unit < settled_end,
1796                KindClass::Epoch(kind) => EPOCHS
1797                    .root_block(kind, Epoch(unit))
1798                    .is_none_or(|block| block.0 <= settled_end),
1799            });
1800        }

1802        fn forget_below(&mut self, window_start: u64) {
1803            self.units.retain(|&(kind, unit), _| match kind.class() {
1804                KindClass::Range(_) | KindClass::Point(_) => unit >= window_start,
1805                KindClass::Epoch(kind) => EPOCHS
1806                    .root_block(kind, Epoch(unit))
1807                    .is_none_or(|block| block.0 >= window_start),
1808            });
1809        }
1810
1811        fn runs(&self) -> Vec<Run> {
1812            let mut runs: Vec<Run> = Vec::new();
1813            for (&(kind, unit), &state) in &self.units {
1814                let extends = matches!(kind.class(), KindClass::Range(_))
1815                    && runs.last().is_some_and(|run| {
1816                        run.kind == kind && run.units.end == unit && run.state == state
1817                    });
1818                if extends {
1819                    runs.last_mut().unwrap().units.end += 1;
1820                } else {
1821                    runs.push(Run {
1822                        kind,
1823                        units: unit..unit + 1,
1824                        state,
1825                    });
1826                }
1827            }
1828            runs
1829        }

1834                KindClass::Range(_) => Some(run.units.end),
1835                KindClass::Point(_) => Some(run.units.end),
1836                KindClass::Epoch(kind) => {

1911    /// The probes of `run`: its units cut at `CHUNK_LEN` from the start.
1912    fn run_probes(run: &Run) -> Vec<Probe> {
1913        match run.kind.class() {
1914            KindClass::Range(kind) => {
1915                let mut probes = Vec::new();
1916                let mut start = run.units.start;
1917                while start < run.units.end {
1918                    let end = run.units.end.min(start + CHUNK_LEN);
1919                    probes.push(Probe::Range {
1920                        kind,
1921                        heights: h(start..end),
1922                    });
1923                    start = end;
1924                }
1925                probes
1926            },
1927            KindClass::Point(kind) => vec![Probe::Point {
1928                kind,
1929                height: Height(run.units.start),
1930            }],
1931            KindClass::Epoch(kind) => vec![Probe::Epoch {
1932                kind,
1933                epoch: Epoch(run.units.start),
1934            }],
1935        }
1936    }

1938    fn check_retry_queue(
1939        set: &MissingSet,
1940        model: &Model,
1941        now: DateTime<Utc>,
1942    ) -> Result<(), TestCaseError> {
1943        let runs = model.runs();
1944        let queue = RetryQueue::new(set, now);
1945        let representative = runs
1946            .iter()
1947            .filter(|run| run.state.is_confirmed() && run.state.retry().next_at <= now)
1948            .min_by_key(|run| {
1949                (
1950                    run.state.retry().next_at,
1951                    run.state.first_missing_at(),
1952                    run.units.start,
1953                    run.kind,
1954                )
1955            })
1956            .map(|run| run_probes(run).remove(0));
1957        prop_assert_eq!(&queue.representative, &representative);
1958
1959        let due = |failed: bool| -> Vec<Probe> {
1960            let mut due: Vec<_> = runs
1961                .iter()
1962                .filter(|run| {
1963                    let is_failed = matches!(run.state, UnitState::Failed { .. });
1964                    !run.state.is_confirmed()
1965                        && is_failed == failed
1966                        && run.state.retry().next_at <= now
1967                })
1968                .collect();
1969            due.sort_by_key(|run| (run.state.retry().next_at, run.kind, run.units.start));
1970            due.into_iter().flat_map(run_probes).collect()
1971        };
1972        prop_assert_eq!(drain(queue.not_found), due(false));
1973        prop_assert_eq!(drain(queue.failed), due(true));
1974
1975        let confirmed: Vec<Probe> = runs
1976            .iter()
1977            .filter(|run| run.state.is_confirmed())
1978            .flat_map(run_probes)
1979            .collect();
1980        prop_assert_eq!(drain(ChunkQueue::confirmed(set)), confirmed);
1981        Ok(())
1982    }

2018        set: &MissingSet,
2019        diff: &ImageDiff,

2021    ) -> Result<(), TestCaseError> {
2022        for region in &diff.regions {

2029                    "{row:?} outside {region:?}"

2031                for unit in row.key.units() {
2032                    image.insert((kind, unit), row.state);
2033                }
2034            }
2035        }

2056    /// What an op does at `now`.
2057    enum Step {
2058        Advance(u64),
2059        Skip,
2060        Update(MissingUpdate, Option<Probe>),
2061    }
2062
2063    fn step_of(op: &Op, set: &MissingSet, now: DateTime<Utc>) -> Step {
2064        let observed = |result| Step::Update(MissingUpdate::Observed(result), None);
2065        match op {
2066            Op::Advance(secs) => Step::Advance(*secs),
2067            Op::Forget(height) => Step::Update(MissingUpdate::ForgetBelow(Height(*height)), None),
2068            Op::ForgetAbove(height) => {
2069                Step::Update(MissingUpdate::ForgetAbove(Height(*height)), None)
2070            },
2071            Op::Range {
2072                kind,
2073                start,
2074                len,
2075                shape,
2076            } => observed(ProbeResult::Range {
2077                kind: *kind,
2078                findings: findings_of(h(*start..*start + *len), shape),
2079            }),
2080            Op::Point {
2081                kind,
2082                height,
2083                observation,
2084            } => observed(point_result(*kind, *height, *observation)),
2085            Op::Epoch {
2086                kind,
2087                epoch,
2088                observation,
2089            } => observed(epoch_result(*kind, *epoch, *observation)),
2090            Op::Retry(found) => {
2091                let mut queue = RetryQueue::new(set, now);
2092                let probe = queue
2093                    .not_found
2094                    .next_probe()
2095                    .or_else(|| queue.failed.next_probe());
2096                probe.map_or(Step::Skip, |probe| {
2097                    let result = found_result(&probe, *found);
2098                    Step::Update(MissingUpdate::Observed(result), Some(probe))
2099                })
2100            },
2101            Op::Representative(found) => {
2102                let probe = RetryQueue::new(set, now).representative;
2103                probe.map_or(Step::Skip, |probe| {
2104                    let result = found_result(&probe, *found);
2105                    Step::Update(MissingUpdate::Observed(result), Some(probe))
2106                })
2107            },
2108        }
2109    }

2111    /// Runs `ops` against the set and the model, checking every invariant after each step.
2112    fn run_ops(ops: Vec<Op>) -> Result<(), TestCaseError> {
2113        let mut set = MissingSet::new(EPOCHS);
2114        let mut model = Model::default();
2115        let mut image = BTreeMap::new();
2116        let mut now = at(T0);
2117        for op in ops {
2118            let (update, probe) = match step_of(&op, &set, now) {
2119                Step::Advance(secs) => {
2120                    now += seconds(secs);
2121                    continue;
2122                },
2123                Step::Skip => continue,
2124                Step::Update(update, probe) => (update, probe),
2125            };
2126            let runs_before = model.runs().len();
2127            match &update {
2128                MissingUpdate::Observed(result) => model.apply(result, now),
2129                MissingUpdate::ForgetBelow(window_start) => model.forget_below(window_start.0),
2130                MissingUpdate::ForgetAbove(settled_end) => model.forget_above(settled_end.0),
2131            }
2132            let diff = set.apply(update, now);
2133            let forgotten = matches!(op, Op::Forget(_));
2134            if forgotten {
2135                prop_assert_eq!(
2136                    diff.rows_removed,
2137                    runs_before.saturating_sub(model.runs().len())
2138                );
2139            }
2140            apply_diff(&set, &diff, &mut image)?;
2141            check_units(&set, &model)?;
2142            check_row_order(&set)?;
2143            check_tallies(&set, &model)?;
2144            check_retry_queue(&set, &model, now)?;
2145            check_known_and_listed(&set, &model)?;
2146            if let Some(probe) = probe {
2147                check_liveness(&set, &probe, now)?;
2148            }
2149        }
2150        Ok(())
2151    }
