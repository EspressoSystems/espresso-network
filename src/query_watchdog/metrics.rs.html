10/// `query_watchdog_*` families on the process `Registry`.
11pub struct WatchdogMetrics {
12    /// Unlabeled; the alert total, sum of `NodeSummary.not_found_units` over nodes. A label-less
13    /// vec has no series until the first `set_not_found_units`.
14    not_found_units: IntGaugeVec,
15    /// `{node,kind}`, heights or epochs per kind.
16    missing: IntGaugeVec,
17    /// `{node,kind}`.
18    oldest_missing_age_seconds: IntGaugeVec,
19    /// `{node}`.
20    follow_lag_blocks: IntGaugeVec,
21    /// `{node}`, highest node height minus the node's own height; no alert.
22    node_lag_blocks: IntGaugeVec,
23    /// `{node}`.
24    merklized_state_lag_blocks: IntGaugeVec,
25    /// `{node,kind,status}`.
26    fetch_errors_total: IntCounterVec,
27    /// `{node}`, unix seconds.
28    last_sweep_completed_seconds: IntGaugeVec,
29    /// Max block height of all watched nodes.
30    tip_height: IntGauge,
31    /// `{node}`.
32    follow_next_height: IntGaugeVec,
33    /// `{node}`, 1 while a sweep runs, 0 while it waits.
34    sweep_running: IntGaugeVec,
35    /// `{node}`, next height of the running sweep; 0 while waiting.
36    sweep_next_height: IntGaugeVec,
37    /// `{node}`, end height of the running sweep; 0 while waiting.
38    sweep_until_height: IntGaugeVec,
39    /// `{node}`, estimated time to finish the running sweep; absent while waiting or unknown.
40    sweep_eta_seconds: IntGaugeVec,
41    /// `{node,kind}`, every probe sent.
42    probes_total: IntCounterVec,
43    /// `{node,kind}`, objects the node lacked locally and served after a fetch from peers; no
44    /// alert.
45    local_misses_total: IntCounterVec,
46}

142    /// Every per-node gauge from one report: `missing` and
143    /// `oldest_missing_age_seconds` per kind, follow and sweep gauges, ETA (series removed when
144    /// the report has none), last completed sweep.
145    pub fn export_node(&self, report: &NodeReport) -> anyhow::Result<()> {
146        let label = report.node.label.as_str();
147        let summary = &report.summary;
148        self.follow_lag_blocks
149            .with_label_values(&[label])
150            .set(i64::try_from(summary.follow_lag_blocks)?);
151        self.node_lag_blocks
152            .with_label_values(&[label])
153            .set(i64::try_from(report.node_lag_blocks)?);
154        if let Some(lag) = summary.merklized_state_lag_blocks {
155            self.merklized_state_lag_blocks
156                .with_label_values(&[label])
157                .set(i64::try_from(lag)?);
158        }
159        for kind in &summary.kinds {
160            let labels = [label, kind.kind.into()];
161            self.missing
162                .with_label_values(&labels)
163                .set(i64::try_from(kind.missing_count)?);
164            let age = kind
165                .oldest_age
166                .map_or(Ok(0), |age| i64::try_from(age.as_secs()))?;
167            self.oldest_missing_age_seconds
168                .with_label_values(&labels)
169                .set(age);
170        }
171        self.export_progress(report)
172    }
173
174    /// Follow, sweep and last-sweep gauges; sweep heights read 0 unless `Running`.
175    fn export_progress(&self, report: &NodeReport) -> anyhow::Result<()> {
176        let labels = [report.node.label.as_str()];
177        let progress = &report.progress;
178        let (running, next, until) = match &progress.sweep {
179            SweepState::Running { sweep, .. } => (1, sweep.next_height, sweep.until_height),
180            _ => (0, Height(0), Height(0)),
181        };
182        self.follow_next_height
183            .with_label_values(&labels)
184            .set(i64::try_from(progress.follow_next_height.0)?);
185        self.sweep_running.with_label_values(&labels).set(running);
186        match report.sweep_eta {
187            Some(eta) => self
188                .sweep_eta_seconds
189                .with_label_values(&labels)
190                .set(i64::try_from(eta.remaining.as_secs())?),
191            None => {
192                // Fails only when the series does not exist yet.
193                self.sweep_eta_seconds.remove_label_values(&labels).ok();
194            },
195        }
196        self.sweep_next_height
197            .with_label_values(&labels)
198            .set(i64::try_from(next.0)?);
199        self.sweep_until_height
200            .with_label_values(&labels)
201            .set(i64::try_from(until.0)?);
202        if let Some(completed_at) = progress.sweep.last_completed() {
203            self.last_sweep_completed_seconds
204                .with_label_values(&labels)
205                .set(completed_at.timestamp());
206        }
207        Ok(())
208    }

210    /// Sets the alert total.
211    pub fn set_not_found_units(&self, total: u64) -> anyhow::Result<()> {
212        self.not_found_units
213            .with_label_values(&[])
214            .set(i64::try_from(total)?);
215        Ok(())
216    }
217
218    /// Sets the tip height.
219    pub fn set_tip_height(&self, tip: Height) -> anyhow::Result<()> {
220        self.tip_height.set(i64::try_from(tip.0)?);
221        Ok(())
222    }
223
224    /// Counts one probe of `kind` under `{node_label, kind}`.
225    pub fn count_probe(&self, node_label: &str, kind: DataKind) {
226        self.probes_total
227            .with_label_values(&[node_label, kind.into()])
228            .inc();
229    }
230
231    /// Counts one object of `kind` served only after a fetch from peers.
232    pub fn count_local_miss(&self, node_label: &str, kind: DataKind) {
233        self.local_misses_total
234            .with_label_values(&[node_label, kind.into()])
235            .inc();
236    }
237
238    /// Counts one failure under `{node_label, target.label(), error.not_served().label()}`.
239    pub fn count_fetch_error(&self, node_label: &str, target: FetchTarget, error: &FetchError) {
240        self.fetch_errors_total
241            .with_label_values(&[node_label, target.label(), error.not_served().label()])
242            .inc();
243    }

246/// Creates a `query_watchdog_{name}` gauge family and registers it.
247fn gauge(
248    registry: &Registry,
249    name: &str,
250    help: &str,
251    labels: &[&str],
252) -> prometheus::Result<IntGaugeVec> {
253    let gauge = IntGaugeVec::new(Opts::new(format!("query_watchdog_{name}"), help), labels)?;
254    registry.register(Box::new(gauge.clone()))?;
255    Ok(gauge)
256}
257
258/// Creates a `query_watchdog_{name}` counter family and registers it.
259fn counter(
260    registry: &Registry,
261    name: &str,

263    labels: &[&str],
264) -> prometheus::Result<IntCounterVec> {
265    let counter = IntCounterVec::new(Opts::new(format!("query_watchdog_{name}"), help), labels)?;
266    registry.register(Box::new(counter.clone()))?;
267    Ok(counter)
268}

274    /// Names of the families `registry` currently exports.
275    fn family_names(registry: &Registry) -> Vec<String> {
276        registry
277            .gather()
278            .iter()
279            .map(|family| family.get_name().to_string())
280            .collect()
281    }
282
283    #[test]
284    fn the_alert_total_is_absent_until_first_set() {
285        let registry = Registry::new();
286        let metrics = WatchdogMetrics::new(&registry).unwrap();
287        let name = "query_watchdog_not_found_units".to_string();
288        assert!(!family_names(&registry).contains(&name));
289
290        metrics.set_not_found_units(0).unwrap();
291        assert!(family_names(&registry).contains(&name));
