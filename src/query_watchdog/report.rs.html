1//! The reporter's view of a node: gauges every tick, the `summary` line every few ticks.
2
3use std::time::Duration;
4
5use anyhow::Context;
6use chrono::{DateTime, Utc};
7
8use crate::{
9    node::WatchedNode,
10    schedule::NodeProgress,
11    view::{NodeSummary, NodeViewSlot, SweepEta, Tips, WatchCounters},
12};
13
14/// Time between gauge writes; the Datadog OpenMetrics default scrape interval.
15pub const GAUGE_INTERVAL: Duration = Duration::from_secs(15);
16/// Gauge ticks per `summary` line: one line per 60 s.
17pub const TICKS_PER_SUMMARY: u64 = 4;
18
19/// One node as the reporter sees it: its view and the counters of its last summary.
20pub struct ReportedNode {
21    slot: NodeViewSlot,
22    last_logged: WatchCounters,
23}
24
25impl ReportedNode {
26    /// Nothing logged yet.
27    pub fn new(slot: NodeViewSlot) -> Self {
28        Self {
29            slot,
30            last_logged: WatchCounters::default(),
31        }
32    }
33
34    /// The latest view at `now` against the node's live height in `tips`. Errors when the node has published no
35    /// view. The view's `borrow()` ends before it returns, so logging never holds the channel lock
36    /// a publishing node watch waits on.
37    pub fn report(&self, tips: &Tips, now: DateTime<Utc>) -> anyhow::Result<NodeReport> {
38        let guard = self.slot.view.borrow();
39        let view = guard
40            .as_ref()
41            .with_context(|| format!("{} has published no view", self.slot.node.label))?;
42        let label = &self.slot.node.label;
43        Ok(NodeReport {
44            node: self.slot.node.clone(),
45            summary: view.summary(tips.node(label).unwrap_or_default(), now),
46            node_lag_blocks: tips.node_lag_blocks(label),
47            counters: view.counters.clone(),
48            progress: view.progress,
49            sweep_eta: view.sweep_eta,
50        })
51    }
52
53    /// The `summary` INFO line (`node`, `errors`, `forgotten`, `sweep_eta`, `summary`) and `node unhealthy` when
54    /// `errors.failures()` since the previous call is not empty; `errors` and `forgotten` since the
55    /// previous call.
56    pub fn log(&mut self, report: &NodeReport) {
57        let node = &report.node.label;
58        let since = report.counters.since(&self.last_logged);
59        tracing::info!(
60            %node,
61            errors = ?since.errors,
62            forgotten = since.forgotten_rows,
63            sweep_eta = report
64                .sweep_eta
65                .map(|eta| humantime::format_duration(eta.remaining).to_string()),
66            summary = %report.summary,
67            "summary"
68        );
69        let failures = since.errors.failures();
70        if !failures.is_empty() {
71            tracing::warn!(%node, errors = ?failures, "node unhealthy");
72        }
73        self.last_logged = report.counters.clone();
74    }
75}
76
77/// One node's state at one tick.
78#[derive(Clone, Debug)]
79pub struct NodeReport {
80    pub node: WatchedNode,
81    pub summary: NodeSummary,
82    /// Blocks behind the highest node; never pages.
83    pub node_lag_blocks: u64,
84    /// Cumulative, from the view.
85    pub counters: WatchCounters,
86    pub progress: NodeProgress,
87    pub sweep_eta: Option<SweepEta>,
88}
