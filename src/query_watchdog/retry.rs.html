13/// Time per retry step in which probes may start; follow and sweep keep the rest (HEAD's value).
14pub const RETRY_BUDGET: Duration = Duration::from_secs(5);
15
16/// Time between representative probes of one node: the 60 s retry cap of a confirmed row.
17pub const REPRESENTATIVE_INTERVAL: Duration = Duration::from_secs(60);

19/// Rows yielding probes of at most `CHUNK_LEN` heights, front first.
20#[derive(Clone, Debug, Default)]
21pub struct ChunkQueue(VecDeque<RowKey>);
22
23impl ChunkQueue {
24    /// Every confirmed row of `set` in `rows()` order: the recovery queue.
25    pub fn confirmed(set: &MissingSet) -> Self {
26        Self(
27            set.rows()
28                .filter(|row| row.state.is_confirmed())
29                .map(|row| row.key)
30                .collect(),
31        )
32    }
33
34    /// `split_first_chunk` of the front row; the rest stays at the front. `None` when empty.
35    pub fn next_probe(&mut self) -> Option<Probe> {
36        let (probe, rest) = self.0.pop_front()?.split_first_chunk();
37        if let Some(rest) = rest {
38            self.0.push_front(rest);
39        }
40        Some(probe)
41    }
42
43    /// Drops what lies below `window_start`: ranges are clipped, points go by height, epoch rows
44    /// by root block (rows without one stay), as `MissingUpdate::ForgetBelow`.
45    pub fn forget_below(&mut self, window_start: Height, epochs: Epochs) {
46        self.0.retain_mut(|key| match key {
47            RowKey::Range { heights, .. } => {
48                heights.start = heights.start.max(window_start);
49                heights.start < heights.end
50            },
51            RowKey::Point { height, .. } => *height >= window_start,
52            RowKey::Epoch { kind, epoch } => epochs
53                .root_block(*kind, *epoch)
54                .is_none_or(|block| block >= window_start),
55        });
56    }
57
58    /// No row left.
59    pub fn is_empty(&self) -> bool {
60        self.0.is_empty()
61    }
62}
63
64/// What one retry step may probe at one `now`.
65#[derive(Clone, Debug)]
66pub struct RetryQueue {
67    /// First chunk of the due confirmed row (`next_at <= now`) with the earliest `next_at`, ties by
68    /// (`first_missing_at`, first unit, kind): the longest overdue row, so confirmed rows are
69    /// re-probed in rotation.
70    pub representative: Option<Probe>,
71    /// Not-found rows below confirmation with `next_at <= now`, in (`next_at`, kind, first unit)
72    /// order.
73    pub not_found: ChunkQueue,
74    /// Failed rows with `next_at <= now`, in the same order; they never count, so they go last.
75    pub failed: ChunkQueue,
76}
77
78impl RetryQueue {
79    /// The representative and the due unconfirmed rows of `set` at `now`.
80    pub fn new(set: &MissingSet, now: DateTime<Utc>) -> Self {
81        let (confirmed, unconfirmed): (Vec<MissingRow>, Vec<MissingRow>) =
82            set.rows().partition(|row| row.state.is_confirmed());
83        let representative = confirmed
84            .iter()
85            .filter(|row| row.state.retry().next_at <= now)
86            .min_by_key(|row| (row.state.retry().next_at, row.age_order()))
87            .map(|row| row.key.first_probe());
88        let (failed, not_found) = unconfirmed
89            .into_iter()
90            .filter(|row| row.state.retry().next_at <= now)
91            .partition(|row| matches!(row.state, UnitState::Failed { .. }));
92        Self {
93            representative,
94            not_found: Self::queue(not_found),
95            failed: Self::queue(failed),
96        }
97    }
98
99    /// `rows` in (`next_at`, kind, first unit) order.
100    fn queue(mut rows: Vec<MissingRow>) -> ChunkQueue {
101        rows.sort_by_key(|row| {
102            (
103                row.state.retry().next_at,
104                row.key.kind(),
105                row.key.first_unit(),
106            )
107        });
108        ChunkQueue(rows.into_iter().map(|row| row.key).collect())
109    }
110}
