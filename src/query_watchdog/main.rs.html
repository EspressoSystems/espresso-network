51/// Watches archive query nodes for heights they fail to serve.
52#[derive(Debug, Parser)]
53struct Options {
54    /// Metrics and healthcheck port.
55    #[arg(long, env = "ESPRESSO_QUERY_WATCHDOG_PORT", default_value_t = 8080)]
56    port: u16,
57    /// Base URLs without `/v1`, comma-separated.
58    #[arg(
59        long,
60        env = "ESPRESSO_QUERY_WATCHDOG_NODES",
61        value_delimiter = ',',
62        required = true
63    )]
64    nodes: Vec<Url>,
65    /// SQLite state file; default `state.sqlite` in the platform data directory of `query-watchdog`.
66    #[arg(long, env = "ESPRESSO_QUERY_WATCHDOG_DB_PATH")]
67    db_path: Option<PathBuf>,
68    /// Delete the stored state at startup and log a WARN (pass it once, not in the standing
69    /// config).
70    #[arg(long, env = "ESPRESSO_QUERY_WATCHDOG_RESET")]
71    reset: bool,
72    /// Pacing.
73    #[command(flatten)]
74    watch: WatchOptions,
75}
76
77impl Options {
78    /// `--db-path`, or `state.sqlite` in the platform data directory; errors when there is none.
79    fn resolve_db_path(&self) -> anyhow::Result<PathBuf> {
80        if let Some(path) = &self.db_path {
81            return Ok(path.clone());
82        }
83        let dirs = ProjectDirs::from("", "espresso", "query-watchdog")
84            .context("no home directory found for the default state file, pass --db-path")?;
85        Ok(dirs.data_dir().join("state.sqlite"))
86    }
87}

89/// Parses options and runs the watchdog; a fatal error is also logged, so it reaches the log
90/// stream (JSON included) before the process exits.
91#[tokio::main]
92async fn main() -> anyhow::Result<()> {
93    let options = Options::parse();
94    init_logging();
95    let result = run(options).await;
96    if let Err(error) = &result {
97        tracing::error!(error = format!("{error:#}"), "fatal");
98    }
99    result
100}
101
102/// As HEAD, plus `NodeHttp::new(ProbeTimeouts::DEFAULT)`, `fetch_epochs`, `start_reporter` on a
103/// second receiver of each view channel, and `Reporter::run` in the `select!`.
104async fn run(options: Options) -> anyhow::Result<()> {
105    let nodes = options
106        .nodes
107        .iter()
108        .cloned()
109        .map(WatchedNode::new)
110        .collect::<anyhow::Result<Vec<_>>>()?;
111    let labels: Vec<_> = nodes.iter().map(|node| node.label.clone()).collect();
112    ensure!(
113        labels.iter().collect::<BTreeSet<_>>().len() == labels.len(),
114        "duplicate node labels in {labels:?}"
115    );
116    let db_path = options.resolve_db_path()?;
117    log_config(&options, &db_path, &labels);
118
119    let store = Store::open(&db_path).await?;
120    let forgotten = store.forget_other_nodes(&labels).await?;
121    tracing::info!(count = forgotten, "forgot nodes");
122
123    let registry = Registry::new();
124    let metrics = Arc::new(WatchdogMetrics::new(&registry)?);
125    let listener = TcpListener::bind(("0.0.0.0", options.port))
126        .await
127        .with_context(|| format!("binding port {}", options.port))?;
128    let (senders, slots): (Vec<_>, Vec<_>) = nodes
129        .iter()
130        .map(|node| {
131            let (sender, view) = view::channel();
132            let slot = NodeViewSlot {
133                node: node.clone(),
134                view,
135            };
136            (sender, slot)
137        })
138        .unzip();
139    let (block_times_sender, block_times) = block_time::channel();
140    let server = tokio::spawn(http::serve(
141        listener,
142        AppState::new(registry, slots.clone(), block_times),
143    ));
144    log_serving(options.port);
145
146    let system = System {
147        clock: WallClock,
148        http: NodeHttp::new(ProbeTimeouts::DEFAULT)?,
149    };
150    let watchdog = Watchdog::new(system, options.watch, metrics, store);
151    let epochs = watchdog.fetch_epochs(&nodes).await?;
152    match epochs {
153        Epochs::On(config) => tracing::info!(
154            epoch_height = config.epoch_height.get(),
155            epoch_start_block = config.epoch_start_block.0,
156            "epochs"
157        ),
158        Epochs::Off => tracing::info!(epochs = "off", "epochs"),
159    }
160
161    let (tip_poller, tips) = watchdog.start_tip_poller(nodes.clone()).await?;
162    if options.reset {
163        watchdog.reset_state().await?;
164    }
165    let watches = try_join_all(
166        nodes
167            .into_iter()
168            .zip(senders)
169            .map(|(node, view)| watchdog.start_node(node, tips.clone(), epochs, view)),
170    )
171    .await?;
172    let reporter = watchdog.start_reporter(tips, slots.clone());
173    let block_time_poller = watchdog.start_block_time_poller(slots, block_times_sender);
174
175    tokio::select! {
176        result = try_join_all(watches.into_iter().map(NodeWatch::run)) => result.map(drop),
177        result = reporter.run() => result.map(drop),
178        result = tip_poller.run() => result.map(drop),
179        never = block_time_poller.run() => match never {},
180        result = server => result?,
181        signal = shutdown_signal() => {
182            tracing::debug!(signal = signal?, "shutting down");
183            Ok(())
184        },
185    }
186}

188/// The `config` INFO line.
189fn log_config(options: &Options, db_path: &Path, labels: &[String]) {
190    let watch = &options.watch;
191    tracing::info!(
192        nodes = ?labels,
193        db_path = %db_path.display(),
194        sweep_interval = %humantime::format_duration(watch.sweep_interval),
195        sweep_once = watch.sweep_once,
196        request_interval = %humantime::format_duration(watch.request_interval),
197        retention_blocks = watch.retention_blocks,
198        retention = watch.retention_blocks.is_none().then_some("archive"),
199        "config"
200    );
201}

213/// fmt subscriber on stdout, `EnvFilter::from_default_env()`, JSON when `RUST_LOG_FORMAT=json`.
214fn init_logging() {
215    // `RUST_LOG` unset means INFO, not the `from_default_env` default of ERROR.
216    let filter = EnvFilter::builder()
217        .with_default_directive(LevelFilter::INFO.into())
218        .from_env_lossy();
219    let builder = tracing_subscriber::fmt()
220        .with_env_filter(filter)
221        .with_writer(std::io::stdout);
222    if std::env::var("RUST_LOG_FORMAT").is_ok_and(|format| format == "json") {
223        builder.json().init();
224    } else {
225        builder.init();
226    }
227}

229/// Resolves on SIGINT or SIGTERM and returns its name.
230async fn shutdown_signal() -> anyhow::Result<&'static str> {
231    let mut sigterm = signal(SignalKind::terminate())?;
232    tokio::select! {
233        result = tokio::signal::ctrl_c() => result.map(|()| "SIGINT").map_err(Into::into),
234        _ = sigterm.recv() => Ok("SIGTERM"),
235    }
236}
