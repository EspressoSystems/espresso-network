47/// `axum::serve` of a `Router` with `/healthcheck` (`{"status":"available"}`, 200),
48/// `/status/metrics` (`TextEncoder::encode_to_string(&registry.gather())`, 500 on encode error)
49/// and `/`, the page of missing ranges per node.
50pub async fn serve(listener: TcpListener, app: AppState) -> anyhow::Result<()> {
51    let router = Router::new()
52        .route("/", get(missing_page))
53        .route("/healthcheck", get(healthcheck))
54        .route("/status/metrics", get(encode_metrics))
55        .with_state(app);
56    axum::serve(listener, router).await?;
57    Ok(())
58}

60/// Body of `/healthcheck`.
61#[derive(Serialize)]
62struct Health {
63    status: &'static str,
64}
65
66/// Always available while the process serves.
67async fn healthcheck() -> Json<Health> {
68    Json(Health {
69        status: "available",
70    })
71}

73/// Text exposition of the registry.
74async fn encode_metrics(State(app): State<AppState>) -> Result<String, StatusCode> {
75    TextEncoder::new()
76        .encode_to_string(&app.registry.gather())
77        .map_err(|error| {
78            tracing::warn!(%error, "encoding metrics");
79            StatusCode::INTERNAL_SERVER_ERROR
80        })
81}

83/// HTML page from the latest view of every node and the block times; never reads the store.
84async fn missing_page(State(app): State<AppState>) -> Html<String> {
85    let views: Vec<_> = app.nodes.iter().map(|slot| slot.view.borrow()).collect();
86    let nodes: Vec<_> = app
87        .nodes
88        .iter()
89        .zip(&views)
90        .map(|(slot, view)| (&slot.node, view.as_ref()))
91        .collect();
92    let block_times = app.block_times.borrow();
93    let page = Page {
94        nodes: &nodes,
95        block_times: &block_times,
96        generated_at: app.clock.now(),
97    };
98    Html(page.render())
99}

101/// One render of `/`: the views, the block times and the render time; every age is at
102/// `generated_at`.
103pub(crate) struct Page<'a> {
104    pub nodes: &'a [(&'a WatchedNode, Option<&'a NodeView>)],
105    pub block_times: &'a BlockTimes,
106    pub generated_at: DateTime<Utc>,
107}
108
109impl Page<'_> {
110    /// Status line, `generated at`, endpoint links, the overview table, then one section per node.
111    pub(crate) fn render(&self) -> String {
112        let total = self.not_found_units();
113        let (color, background, text) = if total > 0 {
114            ("red", "#f8d7da", "#721c24")
115        } else {
116            ("green", "#d4edda", "#155724")
117        };
118        let mut page = String::from(
119            "<!DOCTYPE html><html><head><title>query-watchdog</title></\
120             head><body><h1>query-watchdog</h1>",
121        );
122        write!(
123            page,
124            "<p style=\"background:{background};color:{text};padding:0.5em\">{color}: {total} \
125             not-found units</p><p>generated at {} UTC</p><ul><li><a \
126             href=\"/status/metrics\">/status/metrics</a></li><li><a \
127             href=\"/healthcheck\">/healthcheck</a></li></ul>",
128            time_text(self.generated_at)
129        )
130        .expect("writing to a String");
131        self.write_overview(&mut page).expect("writing to a String");
132        for (node, view) in self.nodes {
133            self.write_node(&mut page, node, *view)
134                .expect("writing to a String");
135        }
136        page.push_str("</body></html>");
137        page
138    }
139
140    /// Sum of `not_found_units` over the published views.
141    fn not_found_units(&self) -> u64 {
142        self.nodes
143            .iter()
144            .filter_map(|(_, view)| *view)
145            .map(|view| view.not_found_units)
146            .sum()
147    }
148
149    /// One table with a row per node; full times are in `title` attributes.
150    fn write_overview(&self, out: &mut String) -> fmt::Result {
151        out.write_str(
152            "<table border=\"1\"><tr><th>node</th><th>not found</th><th>published</th><th>newest \
153             block</th><th>follow lag</th><th>sweep</th><th>last sweep</th></tr>",
154        )?;
155        for (node, view) in self.nodes {
156            let link = node_link(node);
157            match view {
158                None => write!(
159                    out,
160                    "<tr><td>{link}</td><td>-</td><td>-</td><td>-</td><td>-</td><td>not \
161                     started</td><td>-</td></tr>"
162                )?,
163                Some(view) => {
164                    let summary = view.summary(view.frontier.tip(), self.generated_at);
165                    let newest = view
166                        .newest_block()
167                        .map_or_else(|| "-".to_owned(), |height| height.to_string());
168                    let last_sweep = match view.progress.sweep.last_completed() {
169                        Some(done) => self.age_cell(done),
170                        None => "<td>never</td>".to_owned(),
171                    };
172                    write!(
173                        out,
174                        "<tr><td>{link}</td><td>{}</td>{}<td>{newest}</td><td>{}</\
175                         td>{}{last_sweep}</tr>",
176                        summary.not_found_units,
177                        self.age_cell(view.published_at),
178                        summary.follow_lag_blocks,
179                        self.sweep_cell(view)
180                    )?;
181                },
182            }
183        }
184        out.write_str("</table>")
185    }
186
187    /// `<td title="<time>"><age></td>`.
188    fn age_cell(&self, at: DateTime<Utc>) -> String {
189        format!(
190            "<td title=\"{}\">{}</td>",
191            time_text(at),
192            view::short_duration(view::age_at(at, self.generated_at))
193        )
194    }
195
196    /// `running 45.0% (eta 12m)`, `next in 6d 23h`, `due now` or `done`.
197    fn sweep_cell(&self, view: &NodeView) -> String {
198        let (text, title) = match &view.progress.sweep {
199            SweepState::Running { sweep, .. } => {
200                let eta = view.sweep_eta.as_ref();
201                let text = match eta {
202                    Some(eta) => format!(
203                        "running {:.1}% (eta {})",
204                        sweep.percent_done(),
205                        view::short_duration(eta.remaining)
206                    ),
207                    None => format!("running {:.1}%", sweep.percent_done()),
208                };
209                (text, eta.map(|eta| eta.finish))
210            },
211            SweepState::NeverSwept | SweepState::Waiting { .. } => {
212                match view.next_sweep_start.filter(|at| *at > self.generated_at) {
213                    Some(at) => (
214                        format!(
215                            "next in {}",
216                            view::short_duration(view::age_at(self.generated_at, at))
217                        ),
218                        Some(at),
219                    ),
220                    None => ("due now".to_owned(), None),
221                }
222            },
223            SweepState::Done { .. } => ("done".to_owned(), None),
224        };
225        match title {
226            Some(at) => format!("<td title=\"{}\">{}</td>", time_text(at), escape(&text)),
227            None => format!("<td>{}</td>", escape(&text)),
228        }
229    }

231    /// Heading, missing counts and missing table of one node.
232    fn write_node(
233        &self,
234        out: &mut String,
235        node: &WatchedNode,
236        view: Option<&NodeView>,
237    ) -> fmt::Result {
238        write!(out, "<h2>{}</h2>", node_link(node))?;
239        let Some(view) = view else {
240            return out.write_str("<p>nothing missing</p>");
241        };
242        let summary = view.summary(view.frontier.tip(), self.generated_at);
243        if let Some(line) = missing_line(&summary.kinds) {
244            write!(out, "<p>{}</p>", escape(&line))?;
245        }
246        self.write_missing_table(out, node, view)
247    }
248
249    /// The listed rows with first block time, missing for, last tried, next try and link.
250    fn write_missing_table(
251        &self,
252        out: &mut String,
253        node: &WatchedNode,
254        view: &NodeView,
255    ) -> fmt::Result {
256        let listed = &view.listed;
257        if listed.shown.is_empty() {
258            return out.write_str("<p>nothing missing</p>");
259        }
260        out.write_str(
261            "<table border=\"1\"><tr><th>kind</th><th>range_start</th><th>range_end</th><th>first \
262             block time</th><th>missing for</th><th>last tried</th><th>next \
263             try</th><th>link</th></tr>",
264        )?;
265        for row in &listed.shown {
266            let key = &row.key;
267            let url = escape(node.route_url(&key.first_probe().route()).as_str());
268            let kind: &str = key.kind().into();
269            let first_block = match key.first_block(view.epochs) {
270                None => "-".to_owned(),
271                Some(height) => self.block_times.get(height).map_or_else(
272                    || "time unknown".to_owned(),
273                    |time| {
274                        let age = view::age_at(time, self.generated_at);
275                        format!("{} ({} ago)", time_text(time), view::short_duration(age))
276                    },
277                ),
278            };
279            let age = view::short_duration(view::age_at(
280                row.state.first_missing_at(),
281                self.generated_at,
282            ));
283            let tries = TryTimes::at(&row.state.retry(), self.generated_at);
284            let units = key.units();
285            write!(
286                out,
287                "<tr><td>{kind}</td><td>{}</td><td>{}</td><td>{first_block}</td><td>{age}</\
288                 td><td>{}</td><td>{}</td><td><a href=\"{url}\">{url}</a></td></tr>",
289                units.start,
290                units.end,
291                tries.last_tried_text(),
292                tries.next_try_text()
293            )?;
294        }
295        out.write_str("</table>")?;
296        let hidden = listed.total - listed.shown.len() as u64;
297        if hidden > 0 {
298            write!(out, "<p>{hidden} more not shown</p>")?;
299        }
300        Ok(())
301    }
302}

304/// Link to the node's base URL labelled with its label.
305fn node_link(node: &WatchedNode) -> String {
306    format!(
307        "<a href=\"{}\">{}</a>",
308        escape(node.base_url.as_str()),
309        escape(&node.label)
310    )

313/// `2026-11-06 14:56:22`, UTC.
314fn time_text(at: DateTime<Utc>) -> String {
315    at.format("%Y-%m-%d %H:%M:%S").to_string()
316}

318/// `missing: leaf 300 heights, block 2 heights`, or `None` when nothing is missing.
319fn missing_line(kinds: &[KindSummary]) -> Option<String> {
320    let parts: Vec<_> = kinds
321        .iter()
322        .filter(|kind| kind.missing_count > 0)
323        .map(|kind| {
324            format!(
325                "{} {} {}",
326                <&str>::from(kind.kind),
327                kind.missing_count,
328                kind.kind.unit_name()
329            )
330        })
331        .collect();
332    (!parts.is_empty()).then(|| format!("missing: {}", parts.join(", ")))
333}
334
335/// Escapes text for HTML content and double-quoted attributes.
336fn escape(text: &str) -> String {
337    text.replace('&', "&amp;")
338        .replace('<', "&lt;")
339        .replace('>', "&gt;")
340        .replace('"', "&quot;")
341}

404    /// Times of the first blocks of `views`' rows, one second per height after 1000 s.
405    fn times_of(views: &[&NodeView]) -> BlockTimes {
406        let mut times = BlockTimes::default();
407        for height in views.iter().flat_map(|view| view.missing_row_blocks()) {
408            times.insert(height, at(1_000 + height.0 as i64));
409        }
410        times
