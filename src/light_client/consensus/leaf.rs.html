21/// Data sufficient to convince a client that a certain leaf is finalized.
22///
23/// There are different types of proofs for different scenarios and protocol versions. New proof
24/// types can be added while remaining compatible with old serialized proofs.
25#[derive(Clone, Debug, Default, PartialEq, Eq, Deserialize, Serialize)]
26pub enum FinalityProof {
27    /// The client has stated that they already trust the finality of this particular leaf.
28    #[default]
29    Assumption,
30
31    /// The finality follows from a 2-chain of QCs using the HotStuff2 commit rule.
32    ///
33    /// The requirements for checking finality of a leaf given a 2-chain of QCs are:
34    /// * The leaf has a protocol version indicating it was created via HotStuff2
35    /// * `committing_qc.leaf_commit() == leaf.commit()`
36    /// * `committing_qc.view_number() == leaf.view_number()`
37    /// * `deciding_qc.view_number() == committing_qc.view_number() + 1`
38    /// * Both QCs have a valid threshold signature given a stake table. When the leaf is the last
39    ///   leaf of an epoch, the deciding QC is produced in the subsequent epoch and is checked
40    ///   against the next epoch's stake table.
41    HotStuff2 {
42        committing_qc: Arc<Certificate>,
43        deciding_qc: Arc<Certificate>,
44    },
45
46    /// The finality follows from a Certificate2 in the new protocol.
47    ///
48    /// Certificate2 proves finality. The QC is needed to reconstruct the returned
49    /// [`LeafQueryData`] when the proof contains only the leaf committed by cert2.
50    NewProtocol {
51        cert2: Arc<Certificate2<SeqTypes>>,
52        leaf_qc: Box<QuorumCertificate2<SeqTypes>>,
53    },
54
55    /// The finality follows from a 3-chain of QCs using the original HotStuff commit rule.
56    ///
57    /// The requirements for checking finality of a leaf via the 3-chain rule are similar to the
58    /// `HotStuff2` finality rule, but an extra QC is required with a consecutive view number.
59    HotStuff {
60        precommit_qc: Arc<Certificate>,
61        committing_qc: Arc<Certificate>,
62        deciding_qc: Arc<Certificate>,
63    },
64}
65/// A hint that allows a verifier to verify a proof.
66///
67/// The hint should be supplied by the verifier (e.g. the light client). It represents the root of
68/// trust for this proof.
69#[derive(Clone, Copy, Debug)]
70pub enum LeafProofHint<'a, Q> {
71    /// The root of trust is a quorum for a particular epoch.
72    ///
73    /// This quorum can be used to verify the [`Certificate`]s that make up a
74    /// [`HotStuff`](FinalityProof::HotStuff) or [`HotStuff2`](FinalityProof::HotStuff2)
75    /// [`FinalityProof`].
76    Quorum(&'a Q),
77
78    /// The root of trust is an existing leaf that is already known by the verifier to be finalized.
79    ///
80    /// This can be used to check that the leaf chain in a [`LeafProof`] leads up to a leaf that is
81    /// finalized, and thus the whole chain is finalized.
82    Assumption(&'a Leaf2),
83}
84
85impl<'a> LeafProofHint<'a, StakeTableQuorum<EpochMembership<SeqTypes>>> {
86    /// Construct a [`LeafProofHint`] from a known-finalized leaf, where the quorum type doesn't
87    /// matter.
88    ///
89    /// If the quorum type matters (even though it will not be used in verification whenever the
90    /// known-finalized leaf is used), use `LeafProofHint::<Q>::Assumption(leaf)` to specify the
91    /// quorum type `Q` explicitly.
92    pub fn assumption(leaf: &'a Leaf2) -> Self {
93        Self::Assumption(leaf)
94    }
95}

97/// A proof that a leaf is finalized.
98#[derive(Clone, Debug, Default, PartialEq, Eq, Deserialize, Serialize)]
99pub struct LeafProof {
100    /// A chain of leaves from the requested leaf to a provably finalized leaf.
101    ///
102    /// The chain is in chronological order, so `leaves[0]` is the requested leaf and
103    /// `leaves.last()` is a leaf which is known or can be proven to be finalized. The chain is
104    /// joined by `parent_commitment`, so it can be validated by recomputing the commitment of each
105    /// leaf and comparing to the parent commitment of the next.
106    leaves: Vec<Leaf2>,
107
108    /// Some extra data proving finality for the last leaf in `leaves`.
109    proof: FinalityProof,
110}
111
112impl LeafProof {
113    /// Verify the proof.
114    ///
115    /// If successful, returns the leaf which is proven finalized.
116    pub async fn verify(
117        &self,
118        hint: LeafProofHint<'_, impl Quorum>,
119    ) -> Result<LeafQueryData<SeqTypes>> {
120        let mut leaves = self.leaves.iter();
121        let leaf = leaves.next().context("empty leaf chain")?;
122
123        // The QC signing `leaf`. This either comes from the next leaf in the chain, or from the
124        // final QC chain in case the leaf chain contains only this single leaf.
125        let mut opt_qc = None;
126
127        // Verify chaining by recomputing hashes.
128        let mut curr = leaf;
129        for next in leaves {
130            ensure!(curr.commit() == next.parent_commitment());
131            curr = next;
132
133            if opt_qc.is_none() {
134                // Get the QC signing `leaf` from the justify QC of the subsequent leaf.
135                opt_qc = Some(next.justify_qc().clone());
136            }
137        }
138
139        // Check that the final leaf is actually finalized and get the QC which signs it.
140        let final_qc = match (&self.proof, hint) {
141            (FinalityProof::Assumption, LeafProofHint::Assumption(finalized)) => {
142                // The prover claims that we already have a finalized leaf whose parent is the
143                // current leaf.
144                ensure!(finalized.parent_commitment() == curr.commit());
145                finalized.justify_qc().clone()
146            },
147            (
148                FinalityProof::HotStuff2 {
149                    committing_qc,
150                    deciding_qc,
151                },
152                LeafProofHint::Quorum(quorum),
153            ) => {
154                // Check that the given QCs form a 2-chain, which proves `curr` finalized under
155                // HotStuff2.
156                let ChainVersions {
157                    leaf: version,
158                    max_cert,
159                } = quorum
160                    .verify_qc_chain_and_get_version(curr, [&**committing_qc, &**deciding_qc])
161                    .await?;
162
163                // Check that HotStuff2 is the appropriate commit rule to use. HotStuff2 commit
164                // rule was introduced with the epochs version of HotShot and retired by the new
165                // protocol.
166                ensure!(version >= EPOCH_VERSION);
167                ensure!(
168                    version < NEW_PROTOCOL_VERSION,
169                    "HotStuff2 finality proof used for new-protocol leaf"
170                );
171                // The 2-chain rule is only sound if every QC in the chain predates the cutover;
172                // post-cutover QCs do not imply finality.
173                ensure!(
174                    max_cert < NEW_PROTOCOL_VERSION,
175                    "HotStuff2 finality proof spans the new-protocol cutover"
176                );
177
178                committing_qc.qc().clone()
179            },
180            (
181                FinalityProof::HotStuff {
182                    precommit_qc,
183                    committing_qc,
184                    deciding_qc,
185                },
186                LeafProofHint::Quorum(quorum),
187            ) => {
188                // Check that the given QCs form a 3-chain, which proves `curr` finalized under
189                // HotStuff.
190                let ChainVersions { leaf: version, .. } = quorum
191                    .verify_qc_chain_and_get_version(
192                        curr,
193                        [&**precommit_qc, &**committing_qc, &**deciding_qc],
194                    )
195                    .await?;
196
197                // Check that HotStuff is the appropriate commit rule to use. HotStuff commit rule
198                // was deprecated with the epochs version of HotShot.
199                ensure!(version < EPOCH_VERSION);
200
201                precommit_qc.qc().clone()
202            },
203            (FinalityProof::NewProtocol { cert2, leaf_qc }, LeafProofHint::Quorum(quorum)) => {
204                // Certificate2 proves finality for `curr`.
205                let version = curr.block_header().version();
206                ensure!(
207                    version >= NEW_PROTOCOL_VERSION,
208                    "new protocol finality proof used for pre-new-protocol leaf"
209                );
210                ensure!(
211                    cert2.data.leaf_commit == curr.commit(),
212                    "cert2 leaf commitment does not match leaf"
213                );
214                // A re-vote commits the last block of an epoch again, at a view of
215                // its own, so the commit may be later than the leaf. No other block is
216                // voted on again. These are the node's checks in
217                // `verify_new_protocol_leaf_chain`.
218                ensure!(
219                    cert2.view_number() >= curr.view_number(),
220                    "cert2 view number is earlier than the leaf"
221                );
222                ensure!(
223                    cert2.data.block_number == curr.block_header().block_number(),
224                    "cert2 block number does not match leaf"
225                );
226                if cert2.view_number() > curr.view_number() {
227                    ensure!(
228                        is_last_block(curr.height(), quorum.epoch_height()),
229                        "cert2 view is after the leaf's, which is not the last block of its epoch"
230                    );
231                }
232                let epoch = EpochNumber::new(epoch_from_block_number(
233                    curr.height(),
234                    quorum.epoch_height(),
235                ));
236                ensure!(
237                    cert2.data.epoch == epoch,
238                    "cert2 epoch {} does not match the leaf's epoch {epoch}",
239                    cert2.data.epoch
240                );
241
242                quorum
243                    .verify_cert2(cert2, version)
244                    .await
245                    .context("verifying cert2 signature")?;
246
247                leaf_qc.as_ref().clone()
248            },
249            (proof, hint) => {
250                let required = match proof {
251                    FinalityProof::Assumption => "finalized leaf",
252                    FinalityProof::HotStuff { .. }
253                    | FinalityProof::HotStuff2 { .. }
254                    | FinalityProof::NewProtocol { .. } => "quorum",
255                };
256                let supplied = match hint {
257                    LeafProofHint::Assumption(..) => "finalized leaf",
258                    LeafProofHint::Quorum(..) => "quorum",
259                };
260                bail!(
261                    "verifier supplied wrong hint for proof: proof requires a {required} but \
262                     supplied hint is {supplied}"
263                );
264            },
265        };
266
267        // Take the QC which signs the requested leaf: either the one we saved earlier, or the one
268        // signing the latest leaf in case the latest leaf is also the requested leaf.
269        let qc = opt_qc.unwrap_or(final_qc);
270
271        let info = LeafQueryData::new(leaf.clone(), qc)?;
272        Ok(info)
273    }

275    /// Append a new leaf to the proof's chain.
276    ///
277    /// Returns `true` if and only if we have enough data to prove at least the first leaf in the
278    /// chain finalized.
279    pub fn push(&mut self, new_leaf: LeafQueryData<SeqTypes>) -> bool {
280        let len = self.leaves.len();
281
282        // Check if the new leaf plus the last saved leaf contain justifying QCs that form a
283        // HotStuff2 QC chain for the leaf before. The deciding QC signs `leaves[len - 1]`, so
284        // bounding that leaf's version keeps the whole chain pre-cutover.
285        if len >= 2
286            && self.leaves[len - 2].block_header().version() >= EPOCH_VERSION
287            && self.leaves[len - 1].block_header().version() < NEW_PROTOCOL_VERSION
288        {
289            let committing_qc = Certificate::for_parent(&self.leaves[len - 1]);
290            let deciding_qc = Certificate::for_parent(new_leaf.leaf());
291            if committing_qc.view_number() == self.leaves[len - 2].view_number()
292                && deciding_qc.view_number() == committing_qc.view_number() + 1
293            {
294                // Sanity check: if we have a chain of QCs from consecutive views, they must refer
295                // to consecutive leaves.
296                debug_assert!(committing_qc.leaf_commit() == self.leaves[len - 2].commit());
297                debug_assert!(deciding_qc.leaf_commit() == self.leaves[len - 1].commit());
298
299                self.proof = FinalityProof::HotStuff2 {
300                    committing_qc: Arc::new(committing_qc),
301                    deciding_qc: Arc::new(deciding_qc),
302                };
303
304                // We don't actually need the last leaf in the chain, we just needed it for its
305                // extra justifying QC.
306                self.leaves.pop();
307
308                return true;
309            }
310        }
311
312        // Check if the new leaf plus the last saved leaf contain QCs that form a legacy HotStuff
313        // QC chain for the leaf before.
314        if len >= 3 && self.leaves[len - 3].block_header().version() < EPOCH_VERSION {
315            let precommit_qc = Certificate::for_parent(&self.leaves[len - 2]);
316            let committing_qc = Certificate::for_parent(&self.leaves[len - 1]);
317            let deciding_qc = Certificate::for_parent(new_leaf.leaf());
318            if precommit_qc.view_number() == self.leaves[len - 3].view_number()
319                && committing_qc.view_number() == precommit_qc.view_number() + 1
320                && deciding_qc.view_number() == committing_qc.view_number() + 1
321            {
322                // Sanity check: if we have a chain of QCs from consecutive views, they must refer
323                // to consecutive leaves.
324                debug_assert!(precommit_qc.leaf_commit() == self.leaves[len - 3].commit());
325                debug_assert!(committing_qc.leaf_commit() == self.leaves[len - 2].commit());
326                debug_assert!(deciding_qc.leaf_commit() == self.leaves[len - 1].commit());
327
328                self.proof = FinalityProof::HotStuff {
329                    precommit_qc: Arc::new(precommit_qc),
330                    committing_qc: Arc::new(committing_qc),
331                    deciding_qc: Arc::new(deciding_qc),
332                };
333
334                // We don't actually need the last two leaves in the chain, we just needed them for
335                // their extra justifying QCs,.
336                self.leaves.pop();
337                self.leaves.pop();
338
339                return true;
340            }
341        }
342
343        // Nothing is finalized yet, just save the new leaf.
344        self.leaves.push(new_leaf.leaf().clone());
345        false
346    }

348    /// Complete a finality proof using the new protocol Certificate2.
349    pub fn add_certificate(
350        &mut self,
351        cert2: Arc<Certificate2<SeqTypes>>,
352        leaf_qc: QuorumCertificate2<SeqTypes>,
353    ) {
354        debug_assert!(cert2.data.leaf_commit == self.leaves[self.leaves.len() - 1].commit());
355
356        self.proof = FinalityProof::NewProtocol {
357            cert2,
358            leaf_qc: Box::new(leaf_qc),
359        };
360    }

362    /// Inspect the raw finality proof within the larger proof.
363    pub fn proof(&self) -> &FinalityProof {
364        &self.proof
365    }
366
367    /// Inspect the leaf chain, requested leaf first, without verifying it.
368    pub fn leaves(&self) -> &[Leaf2] {
369        &self.leaves
370    }

372    /// The epoch number whose quorum is needed to verify this proof.
373    ///
374    /// This determines the kind of [`LeafProofHint`] needed to verify the proof. If [`Some`], then
375    /// a [`LeafProofHint::Quorum`] is needed with a quorum from this epoch. If [`None`], then a
376    /// [`LeafProofHint::Assumption`] is needed.
377    pub fn epoch(&self, epoch_height: u64) -> Result<Option<EpochNumber>> {
378        if matches!(self.proof, FinalityProof::Assumption) {
379            Ok(None)
380        } else {
381            let height = self.leaves.last().context("empty leaf chain")?.height();
382            let epoch = epoch_from_block_number(height, epoch_height);
383            Ok(Some(epoch.into()))
384        }
385    }

402    /// A new-protocol proof for `leaf` alone, by a `Certificate2` at `view` naming `epoch`.
403    fn cert2_proof(leaf: &LeafQueryData<SeqTypes>, view: u64, epoch: u64) -> LeafProof {
404        let data = Vote2Data {
405            leaf_commit: leaf.leaf().commit(),
406            epoch: EpochNumber::new(epoch),
407            block_number: leaf.leaf().height(),
408        };
409        // Unsigned: the quorums here accept any signature.
410        let commitment = data.commit();
411        let view = ViewNumber::new(view);
412        LeafProof {
413            leaves: vec![leaf.leaf().clone()],
414            proof: FinalityProof::NewProtocol {
415                cert2: Arc::new(Certificate2::new(data, commitment, view, None, PhantomData)),
416                leaf_qc: Box::new(leaf.qc().clone()),
417            },
418        }
419    }

662    async fn test_new_protocol_later_cert2_only_for_last_block_of_epoch() {
663        // Epochs of 3 blocks: height 3 is the last block of epoch 1.
664        let leaves = leaf_chain(1..=3, NEW_PROTOCOL_VERSION).await;
665        let quorum = AlwaysTrueQuorum::new(3);
666
667        // A re-vote commits the epoch's last block at a later view.
668        let last = &leaves[2];
669        let revote = cert2_proof(last, *last.leaf().view_number() + 2, 1);
670        let verified = revote.verify(LeafProofHint::Quorum(&quorum)).await.unwrap();
671        assert_eq!(verified.leaf(), last.leaf());
672
673        // No other block is committed at a later view.
674        let middle = &leaves[1];
675        let proof = cert2_proof(middle, *middle.leaf().view_number() + 2, 1);
676        let err = proof
677            .verify(LeafProofHint::Quorum(&quorum))
678            .await
679            .unwrap_err();
680        assert!(
681            err.to_string().contains("not the last block of its epoch"),
682            "{err:#}"
683        );
