150/// Message sent at the end of an epoch by the current committee
151/// to the next committee.  Both certificates are on the last block of the epoch.
152/// The protocol spec only requires the second certificate, but for consistency
153/// in the code and with the existing Proposal and Leaf structures
154/// We include the Certificate1.  This allows us to use the Certificate1 as the
155/// Justify QC on the first proposal.  The Certificate2 also required on that proposal
156/// but as next_epoch_justify_qc on the Leaf.
157///
158/// We include the proposal because the new leader in the next epoch
159/// will need it to build a header for the first block of the next epoch.
160#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, Hash, Eq)]
161#[serde(bound(deserialize = "S: Deserialize<'de>"))]
162pub struct EpochChangeMessage<T: NodeType, S> {
163    pub cert1: Certificate1<T>,
164    pub cert2: Certificate2<T>,
165    pub proposal: Proposal<T>,
166    #[serde(skip)]
167    _marker: PhantomData<fn() -> S>,
168}

198impl<T: NodeType, S> EpochChangeMessage<T, S> {
199    /// Structural validity of the message, independent of signatures.
200    ///
201    /// Every part must name the same view and block. A certificate's own view
202    /// and block number are covered by the signatures over it, and the
203    /// proposal's are covered by the leaf commitment, so agreement between them
204    /// otherwise rests on an honest signer being in the quorum. Comparing them
205    /// makes the message self-checking. `Proposal::epoch` is not covered by the
206    /// commitment at all and has no other check.
207    ///
208    /// What the embedded proposal claims about itself and its parent is checked
209    /// by the same functions the proposal path uses, except for the boundary
210    /// Cert2 only the first proposal of an epoch carries.
211    pub fn well_formed(&self, epoch_height: u64) -> Result<(), EpochChangeError> {
212        let block_number = self.cert2.data.block_number;
213        if self.cert1.view_number() != self.cert2.view_number()
214            || self.cert1.epoch() != self.cert2.epoch()
215            || self.cert1.data.leaf_commit != self.cert2.data.leaf_commit
216            || self.cert1.data.block_number != Some(block_number)
217        {
218            return Err(EpochChangeError::CertificateMismatch);
219        }
220        if !is_last_block(block_number, epoch_height) {
221            return Err(EpochChangeError::NotLastBlock);
222        }
223        if self.cert2.data.epoch != epoch_from_block_number(block_number, epoch_height).into() {
224            return Err(EpochChangeError::WrongEpoch);
225        }
226        if proposal_commitment(&self.proposal) != self.cert1.data.leaf_commit {
227            return Err(EpochChangeError::ProposalMismatch);
228        }
229        if self.proposal.view_number() != self.cert1.view_number()
230            || self.proposal.block_header.block_number() != block_number
231        {
232            return Err(EpochChangeError::ProposalCertificateMismatch);
233        }
234        epoch_matches_height(&self.proposal, epoch_height)?;
235        justify_qc_matches_parent(&self.proposal, epoch_height)?;
236        view_change_evidence_matches_parent(&self.proposal)?;
237        Ok(())
238    }
239
240    #[cfg(any(test, feature = "testing"))]
241    pub fn into_unchecked(self) -> EpochChangeMessage<T, Unchecked> {
242        EpochChangeMessage {
243            cert1: self.cert1,
244            cert2: self.cert2,
245            proposal: self.proposal,
246            _marker: PhantomData,
247        }
248    }
249}

251/// Reason an [`EpochChangeMessage`] is not [well-formed](EpochChangeMessage::well_formed).
252#[derive(Copy, Clone, Debug, thiserror::Error)]
253pub enum EpochChangeError {
254    #[error("certificates differ in view, epoch, block number or leaf commitment")]
255    CertificateMismatch,
256    #[error("certificate2 is not for the last block of an epoch")]
257    NotLastBlock,
258    #[error("certificate2's block number does not match its epoch")]
259    WrongEpoch,
260    #[error("proposal commitment does not match certificate1's leaf commitment")]
261    ProposalMismatch,
262    #[error("the embedded proposal names a different view or block than the certificates")]
263    ProposalCertificateMismatch,
264    #[error("the embedded proposal is malformed: {0}")]
265    Proposal(#[from] MalformedProposal),
266}

284    pub signature: <T::SignatureKey as SignatureKey>::PureAssembledSignatureType,
285}

287impl<T: NodeType> ProposalFetchRequest<T> {
288    pub fn new(
289        view_number: ViewNumber,
290        key: T::SignatureKey,
291        private_key: &<T::SignatureKey as SignatureKey>::PrivateKey,
292    ) -> Result<Self, <T::SignatureKey as SignatureKey>::SignError> {
293        let payload = ProposalRequestPayload { view_number, key };
294        let signature = T::SignatureKey::sign(private_key, payload.commit().as_ref())?;
295        Ok(Self { payload, signature })
296    }
297
298    pub fn validate_sender(&self, sender: &T::SignatureKey) -> bool {
299        &self.payload.key == sender
300            && self
301                .payload
302                .key
303                .validate(&self.signature, self.payload.commit().as_ref())
304    }
305}
