1//! Verifiable information dispersal (VID) for the new protocol.
2//!
3//! A block's payload is erasure-coded per namespace and spread across the
4//! committee so the block can be recovered from any subset of storage nodes
5//! whose shards cover the recovery threshold. This module owns the three stages
6//! of that lifecycle, one per submodule:
7//!
8//! - [`disperse`] -- the leader side. [`VidDisperser`] erasure-codes each
9//!   namespace, coalesces namespaces into size-balanced buckets, and unicasts
10//!   to every node a stream of [`AvidmGf2DisperseShareFragment`] messages
11//!   (one per bucket), each carrying that node's shares for the bucket's
12//!   namespaces.
13//!
14//! - [`fragments`] -- the receive side of dispersal, the mirror of
15//!   [`VidDisperser`]. [`VidFragmentAccumulator`] buffers the fragments a node
16//!   receives for its *own* share and, once every namespace has arrived,
17//!   reassembles them into a complete [`VidDisperseShare2`]. That share is then
18//!   verified, attached to this node's vote, and fed to the reconstructor.
19//!
20//! - [`reconstruct`] -- block recovery. [`VidReconstructor`] collects the
21//!   verified shares contributed by *many* voters (each node's own share,
22//!   carried on its vote) and decodes the payload once their shards cover the
23//!   recovery threshold.
24//!
25//! [`AvidmGf2DisperseShareFragment`]: hotshot_types::data::vid_disperse::AvidmGf2DisperseShareFragment
26//! [`VidDisperseShare2`]: hotshot_types::data::VidDisperseShare2
27

49/// The VID erasure parameters the committee for `epoch` fixes, matching what
50/// an honest disperser derives. Used to reject shares whose `common.param` is
51/// forged (the commitment binds `ns_commits`, not `param`) and to verify
52/// payloads fetched whole. `None` if the committee cannot be resolved.
53pub fn expected_vid_param<T: NodeType>(
54    membership: &EpochMembershipCoordinator<T>,
55    epoch: EpochNumber,
56) -> Option<AvidmGf2Param> {
57    let membership = membership.stake_table_for_epoch(Some(epoch)).ok()?;
58    let total_weight = vid_total_weight::<T, _>(membership.stake_table(), Some(epoch));
59    init_avidm_gf2_param(total_weight).ok()
60}

62/// Whether `common.ns_lens` are the namespace lengths an honest disperser
63/// derives from the header's namespace table `ns_table` at the payload length
64/// the common claims, which the table must be sized for if it is one.
65///
66/// The commitment binds `ns_commits` but not `ns_lens`, so a leader could
67/// otherwise pair honest namespace commitments with lengths shuffled between
68/// namespaces. Such a block reconstructs and decides, but its stored common
69/// then slices the payload at the wrong namespace boundaries and no namespace
70/// proof for it can be served.
71///
72/// This pins `ns_lens` to the header. Whether the header's table is sized for
73/// the payload is settled by `matches_commitment` at reconstruction, which
74/// sees the recovered bytes.
75pub(crate) fn ns_lens_match_metadata(common: &AvidmGf2Common, ns_table: &[u8]) -> bool {
76    let total = common.payload_byte_len();
77    if ns_table_payload_byte_len(ns_table).is_some_and(|claimed| claimed != total) {
78        return false;
79    }
80    let expected = parse_ns_table(total, ns_table);
81    common.ns_lens.len() == expected.len()
82        && common
83            .ns_lens
84            .iter()
85            .zip(&expected)
86            .all(|(len, range)| *len == range.len())
87}
