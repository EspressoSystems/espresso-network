1//! Per-view vote collection and certificate formation.
2//!
3//! Votes for views arrive concurrently, and each view is tallied on its own
4//! until it crosses a threshold and forms a certificate. [`VoteCollector`]
5//! owns the machinery common to every kind of vote, inspecting each vote only
6//! through the [`Ballot`] trait. It creates a task per view and voted epoch
7//! and routes votes to the appropriate task, drops duplicate and stale votes,
8//! buffers votes whose epoch is not yet resolved, and GCs decided views. How
9//! a task actually combines votes into an output is delegated to a pluggable
10//! [`Tally`] strategy.

53/// Information about a vote.
54///
55/// Used by [`VoteCollector`] to handle incoming votes, i.e. reject
56/// duplicate votes by the same signer, or stale votes for GCed views,
57/// before even tallying the vote.
58pub trait Ballot {
59    type Signer;
60
61    fn view(&self) -> ViewNumber;
62    fn epoch(&self) -> Option<EpochNumber>;
63    fn signer(&self) -> Self::Signer;
64
65    /// The epochs whose committees counting the vote reads, its own included.
66    ///
67    /// The vote waits until every one of them is known.
68    fn epochs(&self) -> impl Iterator<Item = EpochNumber> {
69        self.epoch().into_iter()
70    }
71}

182/// Accumulates epoch-root [`Vote1`]s into the (quorum, state) certificate pair.
183pub struct EpochRootTally<T: NodeType> {
184    membership: EpochMembership<T>,
185    quorum: CheckedAccumulator<T, QuorumVote2<T>, SuccessThreshold>,
186    state: LightClientStateUpdateVoteAccumulator<T>,
187    quorum_cert: Option<QuorumCertificate2<T>>,
188    state_cert: Option<LightClientStateUpdateCertificateV2<T>>,

191impl<T: NodeType> Tally<T> for EpochRootTally<T> {
192    type Vote = Vote1<T>;
193    type Output = EpochRootCerts<T>;
194
195    fn new(m: EpochMembership<T>, l: UpgradeLock<T>) -> Self {
196        Self {
197            quorum: CheckedAccumulator::new(m.clone(), l.clone()),
198            state: LightClientStateUpdateVoteAccumulator {
199                vote_outcomes: HashMap::new(),
200                upgrade_lock: l,
201            },
202            membership: m,
203            quorum_cert: None,
204            state_cert: None,
205        }
206    }
207
208    fn min_votes(&self) -> usize {
209        self.quorum.min_votes()
210    }
211
212    fn has_signer(m: &EpochMembership<T>, signer: &T::SignatureKey) -> bool {
213        QuorumCertificate2::<T>::stake_table_entry(m, signer).is_some()
214    }
215
216    fn add(&mut self, vote1: Vote1<T>) -> Option<Self::Output> {
217        let Some(state_vote) = vote1.state_vote else {
218            error!(view = %vote1.vote.view_number(), "epoch-root vote1 without state vote");
219            return None;
220        };
221        let bls_key = vote1.vote.signing_key();
222
223        if self.quorum_cert.is_none() {
224            self.quorum_cert = self.quorum.add(vote1.vote);
225        }
226
227        // Unlike quorum votes, state votes are fully checked, including
228        // their signatures, by the accumulator, so the certificate does
229        // not need to be validated again.
230        if self.state_cert.is_none() {
231            self.state_cert = self
232                .state
233                .accumulate(&bls_key, &state_vote, &self.membership);
234        }
235
236        let (Some(q), Some(s)) = (&self.quorum_cert, &self.state_cert) else {
237            return None;
238        };
239        info!(view = %q.view_number(), epoch = %s.epoch, "epoch-root certificates formed");
240        let Some(e) = q.epoch() else {
241            warn!(
242                cert = type_name_of_val(q),
243                "certificate has no epoch number"
244            );
245            return None;
246        };
247        Some((ValidCert::new(q.clone(), e), s.clone()))
248    }
249}

273/// Accumulates [`TimeoutBallot`]s into a [`TimeoutCertificate3`] once their
274/// signers' combined stake reaches the threshold `Th`.
275///
276/// Signers with different locks sign different data, so unlike a
277/// [`SimpleTally`] the stake is counted across locks. As in a
278/// [`CheckedAccumulator`], votes are taken unchecked until a certificate
279/// fails to verify, and are checked one by one from then on. A ballot's lock
280/// certificate is checked along with its signature: a vote naming a lock no
281/// quorum certified is not counted, so it cannot raise the certificate's lock.
282pub struct TimeoutTally<T: NodeType, Th> {
283    membership: EpochMembership<T>,
284    upgrade_lock: UpgradeLock<T>,
285    ballots: Vec<TimeoutBallot<T>>,
286    weight: U256,
287    verify_votes: bool,
288    /// Whether each lock certificate checked so far is signed by a quorum.
289    ///
290    /// Most voters name the same lock, so once votes are checked one by one,
291    /// each certificate's signatures are checked once rather than per vote.
292    checked_locks: HashMap<Certificate1<T>, bool>,
293    _threshold: PhantomData<fn() -> Th>,
294}
295
296impl<T: NodeType, Th: Threshold<T> + Send + 'static> Tally<T> for TimeoutTally<T, Th> {
297    type Vote = TimeoutBallot<T>;
298    type Output = ValidCert<TimeoutCertificate3<T>>;
299
300    fn new(m: EpochMembership<T>, l: UpgradeLock<T>) -> Self {
301        Self {
302            membership: m,
303            upgrade_lock: l,
304            ballots: Vec::new(),
305            weight: U256::ZERO,
306            verify_votes: false,
307            checked_locks: HashMap::new(),
308            _threshold: PhantomData,
309        }
310    }
311
312    fn min_votes(&self) -> usize {
313        let table: Vec<_> = self.membership.stake_table().collect();
314        let largest = table
315            .iter()
316            .map(|peer| peer.stake_table_entry.stake())
317            .max()
318            .unwrap_or_default();
319        if largest.is_zero() {
320            return 1;
321        }
322        let nodes = table.len().max(1);
323        let votes = Th::threshold(&self.membership).div_ceil(largest);
324        usize::try_from(votes).unwrap_or(nodes).clamp(1, nodes)
325    }
326
327    fn has_signer(m: &EpochMembership<T>, signer: &T::SignatureKey) -> bool {
328        m.stake(signer).is_some()
329    }
330
331    fn add(&mut self, ballot: TimeoutBallot<T>) -> Option<Self::Output> {
332        if self.verify_votes && !self.is_valid(&ballot) {
333            return None;
334        }
335        self.weight += self.stake(&ballot);
336        self.ballots.push(ballot);
337        if self.weight < Th::threshold(&self.membership) {
338            return None;
339        }
340        match self.assemble() {
341            Some(Ok(cert)) => Some(cert),
342            Some(Err(err)) if !self.verify_votes => {
343                warn!(%err, "invalid timeout certificate formed");
344                self.recover()
345            },
346            Some(Err(err)) => {
347                error!(%err, "timeout certificate of checked votes is invalid");
348                None
349            },
350            None => None,
351        }
352    }
353}

355impl<T: NodeType, Th: Threshold<T>> TimeoutTally<T, Th> {
356    /// The stake of the ballot's signer.
357    ///
358    /// The collector admits only signers in the stake table (`has_signer`),
359    /// so one without stake here is a bug, and counts for nothing.
360    fn stake(&self, ballot: &TimeoutBallot<T>) -> U256 {
361        let signer = ballot.vote().signing_key();
362        let Some(peer) = self.membership.stake(&signer) else {
363            warn!(
364                %signer,
365                view = %ballot.vote().view_number(),
366                "timeout vote signer is not in the stake table"
367            );
368            return U256::ZERO;
369        };
370        peer.stake_table_entry.stake()
371    }
372
373    /// The certificate of the ballots so far, checked.
374    fn assemble(&self) -> Option<anytrace::Result<ValidCert<TimeoutCertificate3<T>>>> {
375        let first = self.ballots.first()?;
376        let view = first.vote().view_number();
377        let epoch = first.vote().data.epoch;
378        let latest = self
379            .ballots
380            .iter()
381            .max_by_key(|ballot| ballot.vote().data.lock)
382            .expect("there is a first ballot");
383        let entries = StakeTableEntries::from_iter(self.membership.stake_table()).0;
384        let threshold = Th::threshold(&self.membership);
385        let votes = self.ballots.iter().map(|ballot| {
386            (
387                ballot.vote().data.lock,
388                ballot.vote().signing_key(),
389                ballot.vote().signature(),
390            )
391        });
392        let result =
393            TimeoutCertificate3::assemble(view, epoch, &entries, votes, latest.lock().cloned())
394                .and_then(|cert| {
395                    verify_timeout3(
396                        &cert,
397                        &entries,
398                        threshold,
399                        *self.membership.coordinator.epoch_height(),
400                        &self.upgrade_lock,
401                        &self.membership.coordinator,
402                    )?;
403                    Ok(cert)
404                })
405                .map(|cert| {
406                    info!(%view, %epoch, lock = ?cert.lock(), "timeout certificate formed");
407                    ValidCert::new(cert, epoch)
408                });
409        Some(result)
410    }
411
412    /// Drop the ballots that fail their checks and assemble the rest again.
413    fn recover(&mut self) -> Option<ValidCert<TimeoutCertificate3<T>>> {
414        self.verify_votes = true;
415        let ballots = mem::take(&mut self.ballots);
416        self.weight = U256::ZERO;
417        for ballot in ballots {
418            if self.is_valid(&ballot) {
419                self.weight += self.stake(&ballot);
420                self.ballots.push(ballot);
421            }
422        }
423        if self.weight < Th::threshold(&self.membership) {
424            return None;
425        }
426        match self.assemble()? {
427            Ok(cert) => Some(cert),
428            Err(err) => {
429                error!(%err, "timeout certificate of checked votes is invalid");
430                None
431            },
432        }
433    }
434
435    /// Check a ballot's signature and its lock certificate.
436    fn is_valid(&mut self, ballot: &TimeoutBallot<T>) -> bool {
437        let vote = ballot.vote();
438        let Ok(data) =
439            VersionedVoteData::new(vote.data.clone(), vote.view_number(), &self.upgrade_lock)
440        else {
441            return false;
442        };
443        if !vote
444            .signing_key()
445            .validate(&vote.signature(), data.commit().as_ref())
446        {
447            warn!(view = %vote.view_number(), signer = %vote.signing_key(), "invalid timeout vote");
448            return false;
449        }
450        let TimeoutLock::Recent(cert) = ballot.timeout_lock() else {
451            return true;
452        };
453        if !cert
454            .data
455            .is_well_formed(*self.membership.coordinator.epoch_height())
456        {
457            return false;
458        }
459        let valid = match self.checked_locks.get(cert) {
460            Some(valid) => *valid,
461            None => {
462                let Ok(membership) = self
463                    .membership
464                    .coordinator
465                    .membership_for_epoch(cert.data.epoch)
466                else {
467                    return false;
468                };
469                let entries = StakeTableEntries::from_iter(membership.stake_table()).0;
470                let valid = verify_signatures(
471                    cert,
472                    &entries,
473                    membership.success_threshold(),
474                    &self.upgrade_lock,
475                )
476                .is_ok();
477                self.checked_locks.insert(cert.clone(), valid);
478                valid
479            },
480        };
481        if !valid {
482            warn!(view = %vote.view_number(), signer = %vote.signing_key(), "timeout vote with an invalid lock certificate");
483        }
484        valid
485    }
486}

488async fn run_tally<T, S>(
489    mut rx: mpsc::UnboundedReceiver<S::Vote>,
490    mut tally: S,
491) -> Option<S::Output>
492where
493    T: NodeType,
494    S: Tally<T>,
495{
496    let cap = tally.min_votes().max(1);
497    let mut buf = Vec::with_capacity(cap);
498    let mut need = cap;
499
500    loop {
501        while buf.len() < need {
502            if rx.recv_many(&mut buf, cap).await == 0 {
503                return None;
504            }
505        }
506        need = 1;
507
508        let hop = spawn_blocking(move || {
509            let out = buf.drain(..).find_map(|v| tally.add(v));
510            (tally, buf, out)
511        });
512
513        match hop.await {
514            Ok((_, _, Some(o))) => return Some(o),
515            Ok((t, b, None)) => (tally, buf) = (t, b),
516            Err(e) if e.is_panic() => resume_unwind(e.into_panic()),
517            Err(_) => return None,
518        }
519    }
520}

525/// Collects votes per view and epoch and forms certificate(s) using the strategy `S`.
526pub struct VoteCollector<T: NodeType, S: Tally<T>> {
527    /// Tasks collecting votes and forming certificates.
528    accumulators: JoinMap<Key, Option<S::Output>>,
529
530    /// Where callers submit their votes.
531    ballot_boxes: BTreeMap<Key, mpsc::UnboundedSender<S::Vote>>,
532
533    /// Votes for epochs we have yet to resolve, deduplicated by signer.
534    pending: BTreeMap<Key, HashMap<T::SignatureKey, S::Vote>>,
535
536    /// The view and epoch pairs that formed a valid certificate already.
537    completed: BTreeSet<Key>,
538
539    /// The signers per view and epoch.
540    signers: BTreeMap<Key, HashSet<T::SignatureKey>>,
541
542    /// The GC threshold.
543    lower_bound: ViewNumber,
544
545    membership: EpochMembershipCoordinator<T>,
546
547    upgrade_lock: UpgradeLock<T>,
548}
549
550impl<T: NodeType, S: Tally<T>> VoteCollector<T, S>
551where
552    <S as Tally<T>>::Vote: Ballot<Signer = T::SignatureKey>,
553{
554    pub fn new(mc: EpochMembershipCoordinator<T>, lock: UpgradeLock<T>) -> Self {
555        Self {
556            accumulators: JoinMap::new(),
557            ballot_boxes: BTreeMap::new(),
558            pending: BTreeMap::new(),
559            signers: BTreeMap::new(),
560            completed: BTreeSet::new(),
561            membership: mc,
562            upgrade_lock: lock,
563            lower_bound: ViewNumber::genesis(),
564        }
565    }
566
567    pub async fn next(&mut self) -> Option<S::Output> {
568        loop {
569            let (key, result) = self.accumulators.join_next().await?;
570            let (view, epoch) = key;
571            // However the task ended, it is gone from the `JoinMap` and will
572            // not be joined again. A sender left behind in `ballot_boxes`
573            // would keep taking votes for `key` that nothing ever reads.
574            self.ballot_boxes.remove(&key);
575            let cert = match result {
576                Ok(cert) => cert,
577                Err(err) => {
578                    if err.is_panic() {
579                        error!(%view, %epoch, %err, "vote collection task panic");
580                    }
581                    None
582                },
583            };
584            match cert {
585                Some(cert) if view >= self.lower_bound && self.completed.insert(key) => {
586                    return Some(cert);
587                },
588                // `key` was retired by `mark_completed` before its tally was
589                // joined. Nothing reopens it, so its signers stay for the
590                // timeout diagnostics.
591                _ if self.completed.contains(&key) => {},
592                // No certificate came out of `key`, so a later vote may open a
593                // new ballot box for it. Its accumulator starts empty, hence the
594                // signers this one saw have to go too, or every one of them is
595                // deduplicated away from a tally that never counted them.
596                _ => {
597                    self.signers.remove(&key);
598                },
599            }
600        }
601    }
602
603    /// Count `vote`, or hold it until the committees of its epochs are known.
604    ///
605    /// Returns the epoch whose committee a held vote waits for, so that the
606    /// caller can drive its catchup.
607    pub fn accumulate_vote(&mut self, vote: S::Vote) -> Option<EpochNumber> {
608        let view = vote.view();
609
610        let Some(epoch) = vote.epoch() else {
611            // A vote without an epoch names no committee, so it can never be tallied.
612            return None;
613        };
614
615        let key = (view, epoch);
616
617        if view < self.lower_bound || self.completed.contains(&key) {
618            return None;
619        }
620
621        // A stake table may still arrive by catchup, so keep the vote for
622        // `retry_pending_votes`.
623        let Ok(membership) = self.membership.membership_for_epoch(Some(epoch)) else {
624            self.pending
625                .entry(key)
626                .or_default()
627                .insert(vote.signer(), vote);
628            return Some(epoch);
629        };
630        let missing = vote
631            .epochs()
632            .find(|e| self.membership.membership_for_epoch(Some(*e)).is_err());
633        if let Some(missing) = missing {
634            self.pending
635                .entry(key)
636                .or_default()
637                .insert(vote.signer(), vote);
638            return Some(missing);
639        }
640
641        if !S::has_signer(&membership, &vote.signer()) {
642            return None;
643        }
644
645        // Check that we have not received a vote from this signer already.
646        if !self.signers.entry(key).or_default().insert(vote.signer()) {
647            return None;
648        }
649
650        if let Some(tx) = self.ballot_boxes.get(&key) {
651            let _ = tx.send(vote);
652            return None;
653        }
654
655        let (tx, rx) = mpsc::unbounded_channel();
656
657        let _ = tx.send(vote);
658        self.ballot_boxes.insert(key, tx);
659
660        let tally = S::new(membership, self.upgrade_lock.clone());
661        self.accumulators.spawn(key, run_tally::<T, S>(rx, tally));
662        None
663    }
664
665    /// Record that `view`'s certificate in `epoch` was obtained by other means,
666    /// e.g. from the network, so the tally for it stops.
667    ///
668    /// Drops the ballot box and buffered votes and aborts the tally. A tally
669    /// mid-way through a blocking hop finishes that hop and has its
670    /// certificate discarded by [`Self::next`]. `signers` is left in place so
671    /// the timeout diagnostics keep what the tally saw.
672    pub fn mark_completed(&mut self, view: ViewNumber, epoch: EpochNumber) {
673        if view < self.lower_bound {
674            return;
675        }
676        let key = (view, epoch);
677        self.completed.insert(key);
678        self.ballot_boxes.remove(&key);
679        self.pending.remove(&key);
680        self.accumulators.abort(&key);
681    }
682
683    pub fn retry_pending_votes(&mut self) {
684        for vote in mem::take(&mut self.pending)
685            .into_values()
686            .flat_map(|votes| votes.into_values())
687        {
688            self.accumulate_vote(vote);
689        }
690    }

692    pub fn gc(&mut self, view: ViewNumber) {
693        let floor = (view, EpochNumber::new(0));
694        self.ballot_boxes = self.ballot_boxes.split_off(&floor);
695        self.completed = self.completed.split_off(&floor);
696        self.pending = self.pending.split_off(&floor);
697        self.signers = self.signers.split_off(&floor);
698        self.lower_bound = view;
699    }

702impl<T, V, Th> VoteCollector<T, SimpleTally<T, V, Th>>
703where
704    T: NodeType,
705    V: Vote<T> + Send + 'static,
706    V::Commitment: 'static,
707    Th: Threshold<T> + Send + 'static,
708    Cert<T, V, Th>:
709        Certificate<T, V::Commitment, Voteable = V::Commitment> + HasEpoch + Send + 'static,
710{
711    /// What every epoch that voted in `view` has collected.
712    ///
713    /// Each entry counts the unique signers routed to `view`'s tally under
714    /// that epoch and, where the epoch still resolves, sums their stake
715    /// against its cert threshold. Votes name their own epoch, so a view can
716    /// hold more than one entry; that means its votes are split across
717    /// committees and neither tally need reach its threshold. Looks up each
718    /// signer's stake on demand, only intended for rare paths like timeout
719    /// diagnostics.
720    ///
721    /// An epoch that has signers always yields an entry, so an empty result
722    /// means no vote was received for `view` and nothing weaker.
723    pub fn stats(&self, view: ViewNumber) -> Vec<(EpochNumber, VoteStats)> {
724        let epochs = (view, EpochNumber::new(0))..(view + 1, EpochNumber::new(0));
725        self.signers
726            .range(epochs)
727            .filter(|(_, signers)| !signers.is_empty())
728            .map(|(&(_, epoch), signers)| {
729                let stats = VoteStats {
730                    signers: signers.len(),
731                    stake: self.membership.membership_for_epoch(Some(epoch)).ok().map(
732                        |membership| {
733                            let stake = signers
734                                .iter()
735                                .filter_map(|signer| {
736                                    Cert::<T, V, Th>::stake_table_entry(&membership, signer)
737                                })
738                                .map(|peer| peer.stake_table_entry.stake())
739                                .sum();
740                            (stake, Cert::<T, V, Th>::threshold(&membership))
741                        },
742                    ),
743                };
744                (epoch, stats)
745            })
746            .collect()
747    }
748}
