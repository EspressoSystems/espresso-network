21/// A block's payload in hand, with what the node needs to file it.
22///
23/// Produced by decoding a view's shares and, on the recovery path, by a
24/// payload a peer sent whole. Either way the bytes are bound to
25/// `payload_commitment` before this is built.
26pub struct ObtainedPayload<T: NodeType> {
27    pub view: ViewNumber,
28    pub epoch: EpochNumber,
29    pub payload_commitment: VidCommitment2,
30    pub payload: T::BlockPayload,
31    pub metadata: <T::BlockPayload as BlockPayload<T>>::Metadata,
32    pub tx_commitments: Vec<Commitment<T::Transaction>>,
33}

35/// Why a reconstruction attempt failed.
36#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
37pub enum VidReconstructErrorKind {
38    /// Unverifiable shares were weeded out; reconstruction retries once the
39    /// remaining shares cover the recovery threshold again.
40    #[error("awaiting more shares after weeding out unverifiable ones")]
41    AwaitingShares,
42    /// Every share verified yet the payload still does not re-commit: the
43    /// disperser committed to a non-codeword, so no subset can ever recover it.
44    #[error("unrecoverable: verified shares cannot decode to a payload matching the commitment")]
45    Unrecoverable,
46}

48/// A failed reconstruction attempt for one view and claimed commitment.
49#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
50#[error("VID reconstruction failed for view {view}: {kind}")]
51pub struct VidReconstructError<K> {
52    pub view: ViewNumber,
53    pub payload_commitment: VidCommitment2,
54    pub kind: VidReconstructErrorKind,
55    /// Voters whose shares failed verification against the commitment.
56    /// Provably bad and attributable: each share arrived in a message
57    /// signed by its voter.
58    pub bad_share_keys: Vec<K>,
59}

61#[derive(Default)]
62pub struct VidReconstructor<T: NodeType> {
63    /// Shares that arrived before their view's proposal, one per voter:
64    /// admitted (or dropped) once the proposal pins the view's commitment.
65    pending: BTreeMap<ViewNumber, BTreeMap<T::SignatureKey, VidDisperseShare2<T>>>,
66    /// One accumulator per view, created when its validated proposal
67    /// arrives: reconstruction needs the proposal's metadata, so only the
68    /// proposal's commitment is worth accumulating.
69    accumulators: BTreeMap<ViewNumber, VidShareAccumulator<T>>,
70    reconstructed: BTreeSet<ViewNumber>,
71    tasks: JoinSet<ReconstructResult<T>>,
72    calculations: BTreeMap<ViewNumber, AbortHandle>,
73}

75pub(crate) struct VidShareAccumulator<T: NodeType> {
76    /// The payload commitment claimed by the view's validated proposal.
77    payload_commitment: VidCommitment2,
78    metadata: Metadata<T>,
79    epoch: EpochNumber,
80    /// The VID erasure parameters the committee fixes for this view, used to
81    /// reject shares carrying a forged `common.param` (see [`Self::accept`]).
82    /// `None` if the committee could not be resolved; the param check is then
83    /// skipped, matching the previously unchecked path.
84    expected_param: Option<AvidmGf2Param>,
85    /// Common data pinned by the first admitted share; hash-bound to
86    /// `payload_commitment` (see [`Self::accept`]).
87    common: Option<AvidmGf2Common>,
88    /// Admitted shares by voter; their shard ranges are pairwise disjoint
89    /// (see [`Self::accept`]).
90    shares: BTreeMap<T::SignatureKey, AvidmGf2Share>,
91    /// Every voter whose share was admitted, including weeded ones: a voter
92    /// whose share failed verification doesn't get a second submission.
93    seen_keys: HashSet<T::SignatureKey>,
94    /// Set when a fully verified share set failed to decode to a payload
95    /// matching the commitment: the disperser is provably faulty and further
96    /// attempts are pointless.
97    exhausted: bool,
98}

100impl<T: NodeType> VidReconstructor<T> {
101    pub fn new() -> Self {
102        Self {
103            pending: BTreeMap::new(),
104            accumulators: BTreeMap::new(),
105            reconstructed: BTreeSet::new(),
106            tasks: JoinSet::new(),
107            calculations: BTreeMap::new(),
108        }
109    }
110
111    /// Pin `view` to its validated proposal's payload commitment and
112    /// metadata, and admit any shares that arrived before the proposal.
113    pub(crate) fn handle_proposal(
114        &mut self,
115        view: ViewNumber,
116        payload_commitment: VidCommitment2,
117        metadata: Metadata<T>,
118        epoch: EpochNumber,
119        expected_param: Option<AvidmGf2Param>,
120    ) {
121        if self.reconstructed.contains(&view) {
122            return;
123        }
124        let accumulator = match self.accumulators.entry(view) {
125            Entry::Occupied(existing) => {
126                // The first proposal wins: an equivocating leader cannot re-pin
127                // the view to another commitment.
128                if existing.get().payload_commitment != payload_commitment {
129                    warn!(%view, "conflicting proposal for a view pinned to another commitment");
130                }
131                return;
132            },
133            Entry::Vacant(slot) => slot.insert(VidShareAccumulator::new(
134                payload_commitment,
135                metadata,
136                epoch,
137                expected_param,
138            )),
139        };
140        for (sender, share) in self.pending.remove(&view).into_iter().flatten() {
141            accumulator.accept(view, sender, share);
142        }
143        self.try_reconstruct(view);
144    }
145
146    pub(crate) fn handle_vid_share(
147        &mut self,
148        sender: T::SignatureKey,
149        share: VidDisperseShare2<T>,
150    ) {
151        let view = share.view_number;
152        // A share carries the voter it belongs to; only the authenticated
153        // sender may contribute its own. This cheap check bounds each node to
154        // one slot and guards the pre-proposal `pending` window, where a share
155        // cannot yet be verified against a commitment.
156        if share.recipient_key != sender {
157            warn!(%view, ?sender, "VID share recipient key does not match its sender");
158            return;
159        }
160        if self.reconstructed.contains(&view) {
161            return;
162        }
163        let Some(accumulator) = self.accumulators.get_mut(&view) else {
164            // No validated proposal yet: hold the voter's share until
165            // `handle_proposal` pins the view's commitment.
166            self.pending
167                .entry(view)
168                .or_default()
169                .entry(sender)
170                .or_insert(share);
171            return;
172        };
173        accumulator.accept(view, sender, share);
174        self.try_reconstruct(view);
175    }
176
177    pub async fn next(&mut self) -> Option<ReconstructResult<T>> {
178        loop {
179            match self.tasks.join_next().await? {
180                Ok(Ok(out)) => {
181                    self.calculations.remove(&out.view);
182                    self.accumulators.remove(&out.view);
183                    self.reconstructed.insert(out.view);
184                    return Some(Ok(out));
185                },
186                Ok(Err(err)) => {
187                    self.calculations.remove(&err.view);
188                    self.handle_failed_attempt(&err);
189                    return Some(Err(err));
190                },
191                Err(err) => {
192                    if err.is_panic() {
193                        error!(%err, "VID reconstruction task panicked");
194                    }
195                    self.calculations.retain(|_, task| task.id() != err.id());
196                },
197            }
198        }
199    }
200
201    /// Apply the outcome of a failed attempt: weed the bad shares out of the
202    /// accumulator, then either mark the payload as unrecoverable or retry
203    /// (`try_reconstruct` re-checks coverage, which shares that arrived while
204    /// the attempt ran may already restore).
205    fn handle_failed_attempt(&mut self, err: &VidReconstructError<T::SignatureKey>) {
206        let Some(accumulator) = self.accumulators.get_mut(&err.view) else {
207            return;
208        };
209        // Views are pinned to one commitment for their lifetime, so a
210        // finished attempt always matches; guard anyway so a future re-pin
211        // policy can't weed the wrong accumulator.
212        if accumulator.payload_commitment != err.payload_commitment {
213            return;
214        }
215        for key in &err.bad_share_keys {
216            accumulator.shares.remove(key);
217        }
218        match err.kind {
219            VidReconstructErrorKind::Unrecoverable => accumulator.exhausted = true,
220            VidReconstructErrorKind::AwaitingShares => self.try_reconstruct(err.view),
221        }
222    }
223
224    fn try_reconstruct(&mut self, view: ViewNumber) {
225        if self.calculations.contains_key(&view) {
226            return;
227        }
228        let Some(accumulator) = self.accumulators.get(&view) else {
229            return;
230        };
231        if accumulator.exhausted || !accumulator.has_enough_shares() {
232            return;
233        }
234        // Enough shares implies an admitted share, which pinned the common.
235        let Some(common) = accumulator.common.clone() else {
236            return;
237        };
238        let payload_commitment = accumulator.payload_commitment;
239        let metadata = accumulator.metadata.clone();
240        let shares: Vec<(T::SignatureKey, AvidmGf2Share)> = accumulator
241            .shares
242            .iter()
243            .map(|(key, share)| (key.clone(), share.clone()))
244            .collect();
245        let epoch = accumulator.epoch;
246        let task = self.tasks.spawn_blocking(move || {
247            reconstruct::<T>(view, epoch, payload_commitment, common, shares, metadata)
248        });
249        self.calculations.insert(view, task);
250    }
251
252    pub fn gc(&mut self, view_number: ViewNumber) {
253        let keep = self.calculations.split_off(&view_number);
254        for handle in self.calculations.values_mut() {
255            handle.abort();
256        }
257        self.calculations = keep;
258        self.pending = self.pending.split_off(&view_number);
259        self.accumulators = self.accumulators.split_off(&view_number);
260        self.reconstructed = self.reconstructed.split_off(&view_number);
261    }
262
263    /// Stop tracking `view`.
264    ///
265    /// Either because its payload was reconstructed (or obtained elsewhere)
266    /// or because it timed out and will never be decided: record it so
267    /// `handle_vid_share` ignores later shares, drop its accumulator and
268    /// pending shares, and abort any in-flight reconstruction task.
269    pub fn retire_view(&mut self, view: ViewNumber) {
270        self.reconstructed.insert(view);
271        self.pending.remove(&view);
272        self.accumulators.remove(&view);
273        if let Some(handle) = self.calculations.remove(&view) {
274            handle.abort();
275        }
276    }
277}

279impl<T: NodeType> VidShareAccumulator<T> {
280    fn new(
281        payload_commitment: VidCommitment2,
282        metadata: Metadata<T>,
283        epoch: EpochNumber,
284        expected_param: Option<AvidmGf2Param>,
285    ) -> Self {
286        Self {
287            payload_commitment,
288            metadata,
289            epoch,
290            expected_param,
291            common: None,
292            shares: BTreeMap::new(),
293            seen_keys: HashSet::new(),
294            exhausted: false,
295        }
296    }
297
298    /// Admit `share` from the authenticated `sender`, dropping it if it fails
299    /// any intake check. The non-overlapping path stays crypto-free; a
300    /// shard-range overlap is the only case that triggers per-share
301    /// verification, via [`Self::resolve_conflict`].
302    fn accept(&mut self, view: ViewNumber, sender: T::SignatureKey, share: VidDisperseShare2<T>) {
303        if self.exhausted {
304            return;
305        }
306        if share.payload_commitment != self.payload_commitment {
307            warn!(%view, ?sender, "VID share commitment differs from the proposal's");
308            return;
309        }
310        // The commitment binds a share's `ns_commits` but not its `param`, so a
311        // Byzantine voter can pair real `ns_commits` with a forged `param` (e.g.
312        // an inflated `recovery_threshold`). Pinning that common as the
313        // verification oracle would reject every honest share, so reject it now.
314        if let Some(expected) = &self.expected_param
315            && share.common.param != *expected
316        {
317            warn!(%view, ?sender, "VID share common param differs from the committee's");
318            return;
319        }
320        // The commitment hash-binds the common, so trust it as the verification
321        // oracle only after that check; later shares must carry the same common.
322        if let Some(common) = &self.common {
323            if share.common != *common {
324                warn!(%view, ?sender, "VID share common differs from the accumulator's");
325                return;
326            }
327        } else if AvidmGf2Scheme::is_consistent(&self.payload_commitment, &share.common) {
328            self.common = Some(share.common.clone());
329        } else {
330            warn!(%view, ?sender, "VID share common is inconsistent with its commitment");
331            return;
332        }
333        // A share whose namespaces disagree on the shard range is malformed.
334        let Some(range) = share.share.range() else {
335            warn!(%view, ?sender, "VID share has an inconsistent shard range");
336            return;
337        };
338        // An empty range contributes nothing; positions past the end of the
339        // encoded payload would inflate coverage without aiding decoding.
340        if range.is_empty() || range.end > share.common.param.total_weights {
341            warn!(%view, ?sender, ?range, "VID share has an empty or out-of-bounds shard range");
342            return;
343        }
344        if self.seen_keys.contains(&sender) {
345            return;
346        }
347        // Honest dispersal assigns disjoint ranges, so an overlap with an
348        // admitted share proves a squat; resolve it (needs verification) below.
349        let conflicts: Vec<T::SignatureKey> = self
350            .shares
351            .iter()
352            .filter(|(_, admitted)| {
353                admitted
354                    .range()
355                    .is_some_and(|covered| covered.start < range.end && range.start < covered.end)
356            })
357            .map(|(key, _)| key.clone())
358            .collect();
359        if conflicts.is_empty() {
360            self.seen_keys.insert(sender.clone());
361            self.shares.insert(sender, share.share);
362            return;
363        }
364        self.resolve_conflict(view, sender, share, conflicts);
365    }
366
367    /// Resolve a shard-range collision between the incoming `share` and the
368    /// already-admitted `conflicts`: verify each against the commitment-bound
369    /// common, evict those that fail, and admit the newcomer only if it
370    /// verifies and no surviving share still covers its range.
371    fn resolve_conflict(
372        &mut self,
373        view: ViewNumber,
374        sender: T::SignatureKey,
375        share: VidDisperseShare2<T>,
376        conflicts: Vec<T::SignatureKey>,
377    ) {
378        // A conflict implies a prior admission, which pinned the common.
379        let Some(common) = self.common.clone() else {
380            return;
381        };
382        // The sender has used its one slot regardless of the outcome.
383        self.seen_keys.insert(sender.clone());
384        let mut survivor = false;
385        for key in conflicts {
386            let verified = self
387                .shares
388                .get(&key)
389                .is_some_and(|admitted| share_verifies(&common, admitted));
390            if verified {
391                survivor = true;
392            } else {
393                warn!(%view, ?key, "evicting unverifiable VID share squatting a shard range");
394                self.shares.remove(&key);
395            }
396        }
397        // A verified share still covers the contested range: the newcomer would
398        // double-cover it, so drop it.
399        if survivor {
400            return;
401        }
402        if share_verifies(&common, &share.share) {
403            self.shares.insert(sender, share.share);
404        } else {
405            warn!(%view, ?sender, "dropping unverifiable VID share at intake conflict");
406        }
407    }
408
409    fn ranges(&self) -> impl Iterator<Item = &Range<usize>> {
410        // Admitted shares always have a consistent range (checked in `accept`).
411        self.shares.values().filter_map(AvidmGf2Share::range)
412    }
413
414    /// Number of shard positions covered by the admitted shares; exact
415    /// because their ranges are disjoint.
416    fn coverage(&self) -> usize {
417        self.ranges().map(ExactSizeIterator::len).sum()
418    }
419
420    fn has_enough_shares(&self) -> bool {
421        self.common
422            .as_ref()
423            .is_some_and(|common| self.coverage() >= common.param.recovery_threshold)
424    }
425}

427/// Decode the shares and accept the result only if it re-commits to
428/// `payload_commitment`. On failure, report the shares that fail
429/// verification against the commitment (each share is self-authenticating
430/// via its merkle proofs) so they can be weeded out. If every share
431/// verifies, the payload is unrecoverable: the shares cover the recovery
432/// threshold with disjoint ranges, so the disperser committed to a
433/// non-codeword and no share subset can ever succeed.
434fn reconstruct<T: NodeType>(
435    view: ViewNumber,
436    epoch: EpochNumber,
437    payload_commitment: VidCommitment2,
438    common: AvidmGf2Common,
439    shares: Vec<(T::SignatureKey, AvidmGf2Share)>,
440    metadata: Metadata<T>,
441) -> ReconstructResult<T> {
442    let (keys, shares): (Vec<_>, Vec<_>) = shares.into_iter().unzip();
443    if let Some(bytes) =
444        decode_and_recommit::<T>(view, &common, &shares, &payload_commitment, &metadata)
445    {
446        let payload = T::BlockPayload::from_bytes(&bytes, &metadata);
447        let tx_commitments = payload.transaction_commitments(&metadata);
448        let output = ObtainedPayload {
449            view,
450            epoch,
451            payload_commitment,
452            payload,
453            metadata,
454            tx_commitments,
455        };
456        return Ok(output);
457    }
458
459    let bad_share_keys: Vec<_> = keys
460        .into_iter()
461        .zip(&shares)
462        .filter(|(_, share)| !share_verifies(&common, share))
463        .map(|(key, _)| key)
464        .collect();
465
466    let kind = if bad_share_keys.is_empty() {
467        warn!(
468            %view,
469            %payload_commitment,
470            "verified shares cannot decode to a payload matching the commitment"
471        );
472        VidReconstructErrorKind::Unrecoverable
473    } else {
474        warn!(
475            %view,
476            %payload_commitment,
477            ?bad_share_keys,
478            "weeded out VID shares that failed verification"
479        );
480        VidReconstructErrorKind::AwaitingShares
481    };
482    Err(VidReconstructError {
483        view,
484        payload_commitment,
485        kind,
486        bad_share_keys,
487    })
488}

490/// Decode `shares` and return the payload bytes only if they re-commit to
491/// `payload_commitment`. Recovery alone does not bind the decoded bytes
492/// to the commitment: a Byzantine disperser can commit to a non-codeword,
493/// and a bad share poisons the erasure decoding.
494fn decode_and_recommit<T: NodeType>(
495    view: ViewNumber,
496    common: &AvidmGf2Common,
497    shares: &[AvidmGf2Share],
498    payload_commitment: &VidCommitment2,
499    metadata: &Metadata<T>,
500) -> Option<Vec<u8>> {
501    let bytes = match AvidmGf2Scheme::recover(common, shares) {
502        Ok(bytes) => bytes,
503        Err(err) => {
504            warn!(%view, %err, "VID recovery failed");
505            return None;
506        },
507    };
508    matches_commitment::<T>(view, &common.param, metadata, &bytes, payload_commitment)
509        .then_some(bytes)
510}
511
512/// Whether `bytes` are the payload `payload_commitment` commits to.
513pub(crate) fn matches_commitment<T: NodeType>(
514    view: ViewNumber,
515    param: &AvidmGf2Param,
516    metadata: &Metadata<T>,
517    payload: &[u8],
518    payload_commitment: &VidCommitment2,
519) -> bool {
520    let ns_table = parse_ns_table(payload.len(), &metadata.encode());
521    match AvidmGf2Scheme::commit(param, payload, ns_table) {
522        Ok((recomputed, _)) if recomputed == *payload_commitment => true,
523        Ok((recomputed, _)) => {
524            warn!(
525                %view,
526                expected = %payload_commitment,
527                %recomputed,
528                "payload does not match the payload commitment"
529            );
530            false
531        },
532        Err(err) => {
533            warn!(%view, %err, "failed to recommit VID payload");
534            false
535        },
536    }
537}

539/// Whether `share` verifies against a `common` already known to be hash-bound
540/// to the commitment (a `verify_with_verified_common` success).
541fn share_verifies(common: &AvidmGf2Common, share: &AvidmGf2Share) -> bool {
542    matches!(
543        AvidmGf2Scheme::verify_share_with_verified_common(common, share),
544        Ok(Ok(()))
545    )
