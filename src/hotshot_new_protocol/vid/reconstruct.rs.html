26/// A block's payload in hand, with what the node needs to file it.
27///
28/// Produced by decoding a view's shares and, on the recovery path, by a
29/// payload a peer sent whole. Either way the bytes are bound to
30/// `payload_commitment` before this is built.
31pub struct ObtainedPayload<T: NodeType> {
32    pub view: ViewNumber,
33    pub epoch: EpochNumber,
34    pub payload_commitment: VidCommitment2,
35    pub payload: T::BlockPayload,
36    pub metadata: <T::BlockPayload as BlockPayload<T>>::Metadata,
37    pub tx_commitments: Vec<Commitment<T::Transaction>>,
38}

40/// Why a reconstruction attempt failed.
41#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
42pub enum VidReconstructErrorKind {
43    /// Unverifiable shares were weeded out; reconstruction retries once the
44    /// remaining shares cover the recovery threshold again.
45    #[error("awaiting more shares after weeding out unverifiable ones")]
46    AwaitingShares,
47    /// Every share verified yet the payload still does not re-commit: the
48    /// disperser committed to a non-codeword, so no subset can ever recover it.
49    #[error("unrecoverable: verified shares cannot decode to a payload matching the commitment")]
50    Unrecoverable,
51}

53/// A failed reconstruction attempt for one view and claimed commitment.
54#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
55#[error("VID reconstruction failed for view {view}: {kind}")]
56pub struct VidReconstructError<K> {
57    pub view: ViewNumber,
58    pub payload_commitment: VidCommitment2,
59    pub kind: VidReconstructErrorKind,
60    /// Voters whose shares failed verification against the commitment.
61    /// Provably bad and attributable: each share arrived in a message
62    /// signed by its voter.
63    pub bad_share_keys: Vec<K>,
64}

67pub struct VidReconstructor<T: NodeType> {
68    /// Shares that arrived before their view's proposal, one per voter:
69    /// admitted (or dropped) once the proposal pins the view's commitment.
70    pending: BTreeMap<ViewNumber, BTreeMap<T::SignatureKey, VidDisperseShare2<T>>>,
71    /// One accumulator per view, created when its validated proposal
72    /// arrives: reconstruction needs the proposal's metadata, so only the
73    /// proposal's commitment is worth accumulating.
74    accumulators: BTreeMap<ViewNumber, VidShareAccumulator<T>>,
75    reconstructed: BTreeSet<ViewNumber>,
76    tasks: JoinSet<ReconstructResult<T>>,
77    calculations: BTreeMap<ViewNumber, AbortHandle>,
78}

80pub(crate) struct VidShareAccumulator<T: NodeType> {
81    /// The payload commitment claimed by the view's validated proposal.
82    payload_commitment: VidCommitment2,
83    metadata: Metadata<T>,
84    epoch: EpochNumber,
85    /// The VID erasure parameters the committee fixes for this view, used to
86    /// reject shares carrying a forged `common.param` (see [`Self::accept`]).
87    /// `None` if the committee could not be resolved; the param check is then
88    /// skipped, matching the previously unchecked path.
89    expected_param: Option<AvidmGf2Param>,
90    /// Common data pinned by the first admitted share; hash-bound to
91    /// `payload_commitment` (see [`Self::accept`]).
92    common: Option<AvidmGf2Common>,
93    /// Admitted shares by voter; their shard ranges are pairwise disjoint
94    /// (see [`Self::accept`]).
95    shares: BTreeMap<T::SignatureKey, AvidmGf2Share>,
96    /// Every voter whose share was admitted, including weeded ones: a voter
97    /// whose share failed verification doesn't get a second submission.
98    seen_keys: HashSet<T::SignatureKey>,
99    /// Set when a fully verified share set failed to decode to a payload
100    /// matching the commitment: the disperser is provably faulty and further
101    /// attempts are pointless.
102    exhausted: bool,
103}

105impl<T: NodeType> VidReconstructor<T> {
106    pub fn new() -> Self {
107        Self {
108            pending: BTreeMap::new(),
109            accumulators: BTreeMap::new(),
110            reconstructed: BTreeSet::new(),
111            tasks: JoinSet::new(),
112            calculations: BTreeMap::new(),
113        }
114    }
115
116    /// Pin `view` to its validated proposal's payload commitment and
117    /// metadata, and admit any shares that arrived before the proposal.
118    pub(crate) fn handle_proposal(
119        &mut self,
120        view: ViewNumber,
121        payload_commitment: VidCommitment2,
122        metadata: Metadata<T>,
123        epoch: EpochNumber,
124        expected_param: Option<AvidmGf2Param>,
125    ) {
126        if self.reconstructed.contains(&view) {
127            return;
128        }
129        let accumulator = match self.accumulators.entry(view) {
130            Entry::Occupied(existing) => {
131                // The first proposal wins: an equivocating leader cannot re-pin
132                // the view to another commitment.
133                if existing.get().payload_commitment != payload_commitment {
134                    warn!(%view, "conflicting proposal for a view pinned to another commitment");
135                }
136                return;
137            },
138            Entry::Vacant(slot) => slot.insert(VidShareAccumulator::new(
139                payload_commitment,
140                metadata,
141                epoch,
142                expected_param,
143            )),
144        };
145        for (sender, share) in self.pending.remove(&view).into_iter().flatten() {
146            accumulator.accept(view, sender, share);
147        }
148        self.try_reconstruct(view);
149    }
150
151    pub(crate) fn handle_vid_share(
152        &mut self,
153        sender: T::SignatureKey,
154        share: VidDisperseShare2<T>,
155    ) {
156        let view = share.view_number;
157        // A share carries the voter it belongs to; only the authenticated
158        // sender may contribute its own. This cheap check bounds each node to
159        // one slot and guards the pre-proposal `pending` window, where a share
160        // cannot yet be verified against a commitment.
161        if share.recipient_key != sender {
162            warn!(%view, ?sender, "VID share recipient key does not match its sender");
163            return;
164        }
165        if self.reconstructed.contains(&view) {
166            return;
167        }
168        let Some(accumulator) = self.accumulators.get_mut(&view) else {
169            // No validated proposal yet: hold the voter's share until
170            // `handle_proposal` pins the view's commitment.
171            self.pending
172                .entry(view)
173                .or_default()
174                .entry(sender)
175                .or_insert(share);
176            return;
177        };
178        accumulator.accept(view, sender, share);
179        self.try_reconstruct(view);
180    }
181
182    pub async fn next(&mut self) -> Option<ReconstructResult<T>> {
183        loop {
184            match self.tasks.join_next().await? {
185                Ok(Ok(out)) => {
186                    self.calculations.remove(&out.view);
187                    self.accumulators.remove(&out.view);
188                    self.reconstructed.insert(out.view);
189                    return Some(Ok(out));
190                },
191                Ok(Err(err)) => {
192                    self.calculations.remove(&err.view);
193                    self.handle_failed_attempt(&err);
194                    return Some(Err(err));
195                },
196                Err(err) => {
197                    if err.is_panic() {
198                        error!(%err, "VID reconstruction task panicked");
199                    }
200                    self.calculations.retain(|_, task| task.id() != err.id());
201                },
202            }
203        }
204    }
205
206    /// Apply the outcome of a failed attempt: weed the bad shares out of the
207    /// accumulator, then either mark the payload as unrecoverable or retry
208    /// (`try_reconstruct` re-checks coverage, which shares that arrived while
209    /// the attempt ran may already restore).
210    fn handle_failed_attempt(&mut self, err: &VidReconstructError<T::SignatureKey>) {
211        let Some(accumulator) = self.accumulators.get_mut(&err.view) else {
212            return;
213        };
214        // Views are pinned to one commitment for their lifetime, so a
215        // finished attempt always matches; guard anyway so a future re-pin
216        // policy can't weed the wrong accumulator.
217        if accumulator.payload_commitment != err.payload_commitment {
218            return;
219        }
220        for key in &err.bad_share_keys {
221            accumulator.shares.remove(key);
222        }
223        match err.kind {
224            VidReconstructErrorKind::Unrecoverable => accumulator.exhausted = true,
225            VidReconstructErrorKind::AwaitingShares => self.try_reconstruct(err.view),
226        }
227    }
228
229    fn try_reconstruct(&mut self, view: ViewNumber) {
230        if self.calculations.contains_key(&view) {
231            return;
232        }
233        let Some(accumulator) = self.accumulators.get(&view) else {
234            return;
235        };
236        if accumulator.exhausted || !accumulator.has_enough_shares() {
237            return;
238        }
239        // Enough shares implies an admitted share, which pinned the common.
240        let Some(common) = accumulator.common.clone() else {
241            return;
242        };
243        let payload_commitment = accumulator.payload_commitment;
244        let metadata = accumulator.metadata.clone();
245        let shares: Vec<(T::SignatureKey, AvidmGf2Share)> = accumulator
246            .shares
247            .iter()
248            .map(|(key, share)| (key.clone(), share.clone()))
249            .collect();
250        let epoch = accumulator.epoch;
251        let task = self.tasks.spawn_blocking(move || {
252            reconstruct::<T>(view, epoch, payload_commitment, common, shares, metadata)
253        });
254        self.calculations.insert(view, task);
255    }
256
257    pub fn gc(&mut self, view_number: ViewNumber) {
258        let keep = self.calculations.split_off(&view_number);
259        for handle in self.calculations.values_mut() {
260            handle.abort();
261        }
262        self.calculations = keep;
263        self.pending = self.pending.split_off(&view_number);
264        self.accumulators = self.accumulators.split_off(&view_number);
265        self.reconstructed = self.reconstructed.split_off(&view_number);
266    }
267
268    /// Stop tracking `view`.
269    ///
270    /// Either because its payload was reconstructed (or obtained elsewhere)
271    /// or because it timed out and will never be decided: record it so
272    /// `handle_vid_share` ignores later shares, drop its accumulator and
273    /// pending shares, and abort any in-flight reconstruction task.
274    pub fn retire_view(&mut self, view: ViewNumber) {
275        self.reconstructed.insert(view);
276        self.pending.remove(&view);
277        self.accumulators.remove(&view);
278        if let Some(handle) = self.calculations.remove(&view) {
279            handle.abort();
280        }
281    }
282}

284impl<T: NodeType> VidShareAccumulator<T> {
285    fn new(
286        payload_commitment: VidCommitment2,
287        metadata: Metadata<T>,
288        epoch: EpochNumber,
289        expected_param: Option<AvidmGf2Param>,
290    ) -> Self {
291        Self {
292            payload_commitment,
293            metadata,
294            epoch,
295            expected_param,
296            common: None,
297            shares: BTreeMap::new(),
298            seen_keys: HashSet::new(),
299            exhausted: false,
300        }
301    }
302
303    /// Admit `share` from the authenticated `sender`, dropping it if it fails
304    /// any intake check. The non-overlapping path stays crypto-free; a
305    /// shard-range overlap is the only case that triggers per-share
306    /// verification, via [`Self::resolve_conflict`].
307    fn accept(&mut self, view: ViewNumber, sender: T::SignatureKey, share: VidDisperseShare2<T>) {
308        if self.exhausted {
309            return;
310        }
311        if share.payload_commitment != self.payload_commitment {
312            warn!(%view, ?sender, "VID share commitment differs from the proposal's");
313            return;
314        }
315        // The commitment binds a share's `ns_commits` but not its `param`, so a
316        // Byzantine voter can pair real `ns_commits` with a forged `param` (e.g.
317        // an inflated `recovery_threshold`). Pinning that common as the
318        // verification oracle would reject every honest share, so reject it now.
319        if let Some(expected) = &self.expected_param
320            && share.common.param != *expected
321        {
322            warn!(%view, ?sender, "VID share common param differs from the committee's");
323            return;
324        }
325        // The commitment hash-binds the common's `ns_commits` but not its
326        // `ns_lens`, which a forwarded share could shuffle against the
327        // proposal's table. Trust a common as the verification oracle only
328        // after both checks; later shares must carry the same common.
329        if let Some(common) = &self.common {
330            if share.common != *common {
331                warn!(%view, ?sender, "VID share common differs from the accumulator's");
332                return;
333            }
334        } else if !AvidmGf2Scheme::is_consistent(&self.payload_commitment, &share.common) {
335            warn!(%view, ?sender, "VID share common is inconsistent with its commitment");
336            return;
337        } else if !ns_lens_match_metadata(&share.common, &self.metadata.encode()) {
338            warn!(
339                %view, ?sender,
340                "VID share namespace lengths disagree with the proposal's namespace table"
341            );
342            return;
343        } else {
344            self.common = Some(share.common.clone());
345        }
346        // A share whose namespaces disagree on the shard range is malformed.
347        let Some(range) = share.share.range() else {
348            warn!(%view, ?sender, "VID share has an inconsistent shard range");
349            return;
350        };
351        // An empty range contributes nothing; positions past the end of the
352        // encoded payload would inflate coverage without aiding decoding.
353        if range.is_empty() || range.end > share.common.param.total_weights {
354            warn!(%view, ?sender, ?range, "VID share has an empty or out-of-bounds shard range");
355            return;
356        }
357        if self.seen_keys.contains(&sender) {
358            return;
359        }
360        // Honest dispersal assigns disjoint ranges, so an overlap with an
361        // admitted share proves a squat; resolve it (needs verification) below.
362        let conflicts: Vec<T::SignatureKey> = self
363            .shares
364            .iter()
365            .filter(|(_, admitted)| {
366                admitted
367                    .range()
368                    .is_some_and(|covered| covered.start < range.end && range.start < covered.end)
369            })
370            .map(|(key, _)| key.clone())
371            .collect();
372        if conflicts.is_empty() {
373            self.seen_keys.insert(sender.clone());
374            self.shares.insert(sender, share.share);
375            return;
376        }
377        self.resolve_conflict(view, sender, share, conflicts);
378    }
379
380    /// Resolve a shard-range collision between the incoming `share` and the
381    /// already-admitted `conflicts`: verify each against the commitment-bound
382    /// common, evict those that fail, and admit the newcomer only if it
383    /// verifies and no surviving share still covers its range.
384    fn resolve_conflict(
385        &mut self,
386        view: ViewNumber,
387        sender: T::SignatureKey,
388        share: VidDisperseShare2<T>,
389        conflicts: Vec<T::SignatureKey>,
390    ) {
391        // A conflict implies a prior admission, which pinned the common.
392        let Some(common) = self.common.clone() else {
393            return;
394        };
395        // The sender has used its one slot regardless of the outcome.
396        self.seen_keys.insert(sender.clone());
397        let mut survivor = false;
398        for key in conflicts {
399            let verified = self
400                .shares
401                .get(&key)
402                .is_some_and(|admitted| share_verifies(&common, admitted));
403            if verified {
404                survivor = true;
405            } else {
406                warn!(%view, ?key, "evicting unverifiable VID share squatting a shard range");
407                self.shares.remove(&key);
408            }
409        }
410        // A verified share still covers the contested range: the newcomer would
411        // double-cover it, so drop it.
412        if survivor {
413            return;
414        }
415        if share_verifies(&common, &share.share) {
416            self.shares.insert(sender, share.share);
417        } else {
418            warn!(%view, ?sender, "dropping unverifiable VID share at intake conflict");
419        }
420    }
421
422    fn ranges(&self) -> impl Iterator<Item = &Range<usize>> {
423        // Admitted shares always have a consistent range (checked in `accept`).
424        self.shares.values().filter_map(AvidmGf2Share::range)
425    }
426
427    /// Number of shard positions covered by the admitted shares; exact
428    /// because their ranges are disjoint.
429    fn coverage(&self) -> usize {
430        self.ranges().map(ExactSizeIterator::len).sum()
431    }
432
433    fn has_enough_shares(&self) -> bool {
434        self.common
435            .as_ref()
436            .is_some_and(|common| self.coverage() >= common.param.recovery_threshold)
437    }
438}

440/// Decode the shares and accept the result only if it re-commits to
441/// `payload_commitment`. On failure, report the shares that fail
442/// verification against the commitment (each share is self-authenticating
443/// via its merkle proofs) so they can be weeded out. If every share
444/// verifies, the payload is unrecoverable: the shares cover the recovery
445/// threshold with disjoint ranges, so the disperser committed to a
446/// non-codeword and no share subset can ever succeed.
447fn reconstruct<T: NodeType>(
448    view: ViewNumber,
449    epoch: EpochNumber,
450    payload_commitment: VidCommitment2,
451    common: AvidmGf2Common,
452    shares: Vec<(T::SignatureKey, AvidmGf2Share)>,
453    metadata: Metadata<T>,
454) -> ReconstructResult<T> {
455    let (keys, shares): (Vec<_>, Vec<_>) = shares.into_iter().unzip();
456    if let Some(bytes) =
457        decode_and_recommit::<T>(view, &common, &shares, &payload_commitment, &metadata)
458    {
459        let payload = T::BlockPayload::from_bytes(&bytes, &metadata);
460        let tx_commitments = payload.transaction_commitments(&metadata);
461        let output = ObtainedPayload {
462            view,
463            epoch,
464            payload_commitment,
465            payload,
466            metadata,
467            tx_commitments,
468        };
469        return Ok(output);
470    }
471
472    let bad_share_keys: Vec<_> = keys
473        .into_iter()
474        .zip(&shares)
475        .filter(|(_, share)| !share_verifies(&common, share))
476        .map(|(key, _)| key)
477        .collect();
478
479    let kind = if bad_share_keys.is_empty() {
480        warn!(
481            %view,
482            %payload_commitment,
483            "verified shares cannot decode to a payload matching the commitment"
484        );
485        VidReconstructErrorKind::Unrecoverable
486    } else {
487        warn!(
488            %view,
489            %payload_commitment,
490            ?bad_share_keys,
491            "weeded out VID shares that failed verification"
492        );
493        VidReconstructErrorKind::AwaitingShares
494    };
495    Err(VidReconstructError {
496        view,
497        payload_commitment,
498        kind,
499        bad_share_keys,
500    })
501}

503/// Decode `shares` and return the payload bytes only if they re-commit to
504/// `payload_commitment`. Recovery alone does not bind the decoded bytes
505/// to the commitment: a Byzantine disperser can commit to a non-codeword,
506/// and a bad share poisons the erasure decoding.
507fn decode_and_recommit<T: NodeType>(
508    view: ViewNumber,
509    common: &AvidmGf2Common,
510    shares: &[AvidmGf2Share],
511    payload_commitment: &VidCommitment2,
512    metadata: &Metadata<T>,
513) -> Option<Vec<u8>> {
514    let bytes = match AvidmGf2Scheme::recover(common, shares) {
515        Ok(bytes) => bytes,
516        Err(err) => {
517            warn!(%view, %err, "VID recovery failed");
518            return None;
519        },
520    };
521    matches_commitment(
522        view,
523        &common.param,
524        &metadata.encode(),
525        &bytes,
526        payload_commitment,
527    )
528    .then_some(bytes)
529}

531/// Whether `payload` is what `payload_commitment` commits to, sliced by the
532/// header's namespace table `ns_table`.
533///
534/// The commitment covers each namespace's bytes but not the table, and
535/// `parse_ns_table` substitutes one namespace spanning the payload for a table
536/// sized for another payload. For a one-namespace block that substitute is the
537/// honest table, so a header claiming a shorter or longer payload would
538/// recommit to the same bytes and decide with a table that misdescribes them.
539/// Refuse such a table instead: it must be sized for `payload`, or be no
540/// table at all, which the disperser also reads as one namespace.
541pub(crate) fn matches_commitment(
542    view: ViewNumber,
543    param: &AvidmGf2Param,
544    ns_table: &[u8],
545    payload: &[u8],
546    payload_commitment: &VidCommitment2,
547) -> bool {
548    if let Some(claimed) = ns_table_payload_byte_len(ns_table)
549        && claimed != payload.len()
550    {
551        warn!(
552            %view,
553            claimed,
554            actual = payload.len(),
555            "namespace table is sized for another payload"
556        );
557        return false;
558    }
559    let ns_table = parse_ns_table(payload.len(), ns_table);
560    match AvidmGf2Scheme::commit(param, payload, ns_table) {
561        Ok((recomputed, _)) if recomputed == *payload_commitment => true,
562        Ok((recomputed, _)) => {
563            warn!(
564                %view,
565                expected = %payload_commitment,
566                %recomputed,
567                "payload does not match the payload commitment"
568            );
569            false
570        },
571        Err(err) => {
572            warn!(%view, %err, "failed to recommit VID payload");
573            false
574        },
575    }
576}

578/// Whether `share` verifies against a `common` already known to be hash-bound
579/// to the commitment (a `verify_with_verified_common` success).
580fn share_verifies(common: &AvidmGf2Common, share: &AvidmGf2Share) -> bool {
581    matches!(
582        AvidmGf2Scheme::verify_share_with_verified_common(common, share),
583        Ok(Ok(()))
584    )
