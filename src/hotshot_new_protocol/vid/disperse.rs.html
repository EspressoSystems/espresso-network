46/// A block's dispersal, computed by the block builder together with its commitment, so the
47/// disperser only signs and sends it.
48pub struct BuiltDispersal<T: NodeType> {
49    params: AvidmGf2DisperseParams<T>,
50    /// One per namespace, in namespace-table order.
51    dispersals: Vec<AvidmGf2NsDispersal>,
52}
53
54impl<T: NodeType> BuiltDispersal<T> {
55    /// Erasure-codes every namespace of an encoded payload for the nodes of `stake_table`.
56    pub fn new(
57        stake_table: &[PeerConfig<T>],
58        payload: Arc<[u8]>,
59        metadata: &[u8],
60    ) -> Result<Self, VidDisperseError> {
61        let params = AvidmGf2DisperseParams::new(stake_table, payload, metadata)
62            .map_err(VidDisperseError::Vid)?;
63        let dispersals = (0..params.ns_table.len())
64            .into_par_iter()
65            .map(|ns_index| disperse_namespace(&params, ns_index))
66            .collect::<Result<_, _>>()?;
67        Ok(Self { params, dispersals })
68    }
69
70    /// The payload commitment, folded from the namespaces' commits as
71    /// [`AvidmGf2Scheme::commit`] folds them.
72    pub fn payload_commitment(&self) -> Result<VidCommitment2, VidDisperseError> {
73        let commits: Vec<_> = self
74            .dispersals
75            .iter()
76            .map(|dispersal| dispersal.commit)
77            .collect();
78        AvidmGf2Scheme::aggregate_commit(&commits)
79            .wrap()
80            .map_err(VidDisperseError::Vid)
81    }
82}

107impl<T: NodeType> VidDisperser<T> {
108    pub fn new(
109        epoch_membership_coordinator: EpochMembershipCoordinator<T>,
110        network: Sender<T>,
111        public_key: T::SignatureKey,
112        private_key: <T::SignatureKey as SignatureKey>::PrivateKey,
113    ) -> Self {
114        Self {
115            calculations: BTreeMap::new(),
116            epoch_membership_coordinator,
117            network,
118            public_key,
119            private_key,
120            tasks: JoinSet::new(),
121            duration_metric: None,
122        }
123    }
124
125    pub fn with_metrics(mut self, hist: Option<Arc<dyn Histogram>>) -> Self {
126        self.duration_metric = hist;
127        self
128    }
129
130    /// Disperses a block: sends `built` if the block builder computed it, else encodes the
131    /// payload first. The coordinator always hands over its block builder's dispersal; only the
132    /// bench node, which skips the block builder, and tests that feed `BlockBuilt` directly pass
133    /// `None`. A repeated request is dropped, `built` with it.
134    pub fn request_vid_disperse(
135        &mut self,
136        vid_disperse_request: VidDisperseRequest<T>,
137        built: Option<BuiltDispersal<T>>,
138    ) {
139        let key = (
140            vid_disperse_request.view,
141            vid_disperse_request.epoch,
142            vid_disperse_request.payload_commitment,
143        );
144        if self.calculations.contains_key(&key) {
145            return;
146        }
147        let membership = self.epoch_membership_coordinator.clone();
148        let network = self.network.clone();
149        let public_key = self.public_key.clone();
150        let private_key = self.private_key.clone();
151        let duration_metric = self.duration_metric.clone();
152        let view = vid_disperse_request.view;
153        let handle = self.tasks.spawn_blocking(move || {
154            let measurement = duration_metric.map(Measurement::start);
155            let result = handle_vid_disperse_request(
156                membership,
157                move |recipient, message| network.unicast(view, recipient, message),
158                public_key,
159                private_key,
160                vid_disperse_request,
161                built,
162            );
163            finish_measurement(measurement);
164            result
165        });
166        self.calculations.insert(key, handle);
167    }
168
169    pub async fn next(&mut self) -> Option<Result<VidDisperseOutput, VidDisperseError>> {
170        loop {
171            match self.tasks.join_next().await {
172                Some(Ok(result)) => return Some(result),
173                Some(Err(err)) => {
174                    if err.is_panic() {
175                        error!(%err, "vid disperse task panic");
176                    }
177                    continue;
178                },
179                None => return None,
180            }
181        }
182    }
183
184    pub fn gc(&mut self, view_number: ViewNumber) {
185        let keep = self.calculations.split_off(&(
186            view_number,
187            EpochNumber::new(0),
188            VidCommitment2::default(),
189        ));
190        for handle in self.calculations.values_mut() {
191            handle.abort();
192        }
193        self.calculations = keep;
194    }
195}

197fn handle_vid_disperse_request<T, S>(
198    epoch_membership_coordinator: EpochMembershipCoordinator<T>,
199    send: S,
200    public_key: T::SignatureKey,
201    private_key: <T::SignatureKey as SignatureKey>::PrivateKey,
202    vid_disperse_request: VidDisperseRequest<T>,
203    built: Option<BuiltDispersal<T>>,
204) -> Result<VidDisperseOutput, VidDisperseError>
205where
206    T: NodeType,
207    S: Fn(&T::SignatureKey, &Message<T, Validated>) -> Result<(), NetworkError> + Clone + Send,
208{
209    let view = vid_disperse_request.view;
210    let epoch = vid_disperse_request.epoch;
211    let payload_commitment = vid_disperse_request.payload_commitment;
212
213    let dispersal = match built {
214        Some(built) => Dispersal::Built(built),
215        None => Dispersal::Encode(
216            VidDisperse2::<T>::disperse_params(
217                &vid_disperse_request.block,
218                &epoch_membership_coordinator,
219                Some(epoch),
220                &vid_disperse_request.metadata,
221            )
222            .map_err(VidDisperseError::Vid)?,
223        ),
224    };
225
226    disperse_payload(
227        dispersal,
228        view,
229        epoch,
230        payload_commitment,
231        &public_key,
232        &private_key,
233        send,
234    )?;
235
236    Ok(VidDisperseOutput {
237        view,
238        payload_commitment,
239    })
240}

242/// Signs a block's dispersal and hands every recipient's fragment of each bucket of namespaces to
243/// `send`, which the disperser points at the network. A [`Dispersal::Encode`] is erasure-coded
244/// bucket by bucket first. The `block_build` bench in espresso-types passes a `send` that only
245/// serializes; the spans time each step.
246pub fn disperse_payload<T, S>(
247    dispersal: Dispersal<T>,
248    view: ViewNumber,
249    epoch: EpochNumber,
250    payload_commitment: VidCommitment2,
251    public_key: &T::SignatureKey,
252    private_key: &<T::SignatureKey as SignatureKey>::PrivateKey,
253    send: S,
254) -> Result<(), VidDisperseError>
255where
256    T: NodeType,
257    S: Fn(&T::SignatureKey, &Message<T, Validated>) -> Result<(), NetworkError> + Clone + Send,
258{
259    let (params, dispersals) = match dispersal {
260        Dispersal::Built(BuiltDispersal { params, dispersals }) => (params, Some(dispersals)),
261        Dispersal::Encode(params) => (params, None),
262    };
263
264    let signature = debug_span!("sign")
265        .in_scope(|| T::SignatureKey::sign(private_key, payload_commitment.as_ref()))
266        .map_err(|err| VidDisperseError::Sign(err.into()))?;
267
268    // Coalesce namespaces into size-balanced buckets, roughly one per core,
269    // but never below the minimum size, so a block of many tiny namespaces
270    // goes out as a few balanced messages per recipient instead of one tiny
271    // message each. Each bucket is one parallel unit and one message per
272    // recipient.
273    let ns_buckets: Vec<Vec<usize>> = {
274        // We want buckets to be at least 256 KiB, otherwise the per-message
275        // overhead is too large. Larger payloads are divided by the number
276        // of available threads to fully utilise rayon.
277        let threshold = params.payload.len().div_ceil(*NUM_THREADS).max(256 * 1024);
278        bucketize(&params.ns_table, threshold)
279    };
280    let buckets: Vec<Bucket> = match dispersals {
281        Some(dispersals) => into_buckets(dispersals, &ns_buckets)
282            .into_iter()
283            .map(Bucket::Built)
284            .collect(),
285        None => ns_buckets.into_iter().map(Bucket::Encode).collect(),
286    };
287
288    buckets.into_par_iter().try_for_each_with(
289        send,
290        |send, bucket| -> Result<(), VidDisperseError> {
291            let dispersals = bucket.dispersals(&params)?;
292            let bucket_fragments = debug_span!("fragments")
293                .in_scope(|| fragments(view, epoch, payload_commitment, &params, dispersals));
294            for fragment in bucket_fragments {
295                let recipient = fragment.recipient_key.clone();
296                let message = Message {
297                    sender: public_key.clone(),
298                    message_type: MessageType::Consensus(ConsensusMessage::VidShareFragment(
299                        SignedProposal::new(fragment, signature.clone()),
300                    )),
301                };
302                if let Err(err) = send(&recipient, &message) {
303                    if err.is_critical() {
304                        return Err(err.into());
305                    }
306                    warn!(%err, "network error while sending vid share fragment");
307                }
308            }
309            Ok(())
310        },
311    )
312}

314/// The namespace dispersals of one bucket, which go out as one fragment per recipient.
315enum Bucket {
316    /// Computed by the block builder.
317    Built(Vec<AvidmGf2NsDispersal>),
318    /// Still to encode, for these namespace indices.
319    Encode(Vec<usize>),
320}
321
322impl Bucket {
323    fn dispersals<T: NodeType>(
324        self,
325        params: &AvidmGf2DisperseParams<T>,
326    ) -> Result<Vec<AvidmGf2NsDispersal>, VidDisperseError> {
327        match self {
328            Self::Built(dispersals) => Ok(dispersals),
329            Self::Encode(ns_indices) => debug_span!("ns_encode").in_scope(|| {
330                ns_indices
331                    .into_iter()
332                    .map(|ns_index| disperse_namespace(params, ns_index))
333                    .collect()
334            }),
335        }
336    }
337}

339/// Moves each namespace's dispersal into its bucket. `dispersals` holds one per namespace, in
340/// namespace-table order, and `ns_buckets` partitions the namespace indices.
341fn into_buckets(
342    dispersals: Vec<AvidmGf2NsDispersal>,
343    ns_buckets: &[Vec<usize>],
344) -> Vec<Vec<AvidmGf2NsDispersal>> {
345    let mut dispersals: Vec<_> = dispersals.into_iter().map(Some).collect();
346    ns_buckets
347        .iter()
348        .map(|bucket| {
349            bucket
350                .iter()
351                .map(|&ns_index| {
352                    dispersals[ns_index]
353                        .take()
354                        .expect("buckets partition the namespaces")
355                })
356                .collect()
357        })
358        .collect()
359}

361/// Erasure-codes the namespace at `ns_index` of the payload in `params`.
362fn disperse_namespace<T: NodeType>(
363    params: &AvidmGf2DisperseParams<T>,
364    ns_index: usize,
365) -> Result<AvidmGf2NsDispersal, VidDisperseError> {
366    AvidmGf2Scheme::ns_disperse_one(
367        &params.param,
368        &params.weights,
369        &params.payload[params.ns_table[ns_index].clone()],
370        ns_index,
371    )
372    .wrap()
373    .map_err(VidDisperseError::Vid)
374}

376/// One fragment per recipient in `params`, carrying its shares of `dispersals`.
377fn fragments<T: NodeType>(
378    view: ViewNumber,
379    epoch: EpochNumber,
380    payload_commitment: VidCommitment2,
381    params: &AvidmGf2DisperseParams<T>,
382    dispersals: Vec<AvidmGf2NsDispersal>,
383) -> Vec<AvidmGf2DisperseShareFragment<T>> {
384    let mut pieces = vec![Vec::new(); params.recipients.len()];
385    for dispersal in dispersals {
386        for (pieces, ns_share) in pieces.iter_mut().zip(dispersal.shares) {
387            pieces.push(AvidmGf2NamespacePiece {
388                ns_index: dispersal.ns_index,
389                ns_payload_byte_len: dispersal.payload_byte_len,
390                ns_commit: dispersal.commit,
391                ns_share,
392            });
393        }
394    }
395    params
396        .recipients
397        .iter()
398        .zip(pieces)
399        .map(|(recipient, pieces)| AvidmGf2DisperseShareFragment {
400            view_number: view,
401            epoch: Some(epoch),
402            target_epoch: Some(epoch),
403            payload_commitment,
404            recipient_key: recipient.clone(),
405            param: params.param.clone(),
406            num_namespaces: params.ns_table.len(),
407            namespaces: pieces,
408        })
409        .collect()
410}

412/// Group namespace indices into buckets whose payload sizes are each at least
413/// `threshold` bytes (except possibly the last), coalescing small namespaces so
414/// the disperser sends one message per bucket rather than one per namespace.
415///
416/// A `threshold` of 0 puts every namespace in its own bucket (no coalescing); a
417/// threshold larger than the whole payload yields a single bucket. A namespace
418/// at least `threshold` bytes on its own seals its bucket immediately, so large
419/// namespaces stay separate while small ones accumulate.
420fn bucketize(ns_table: &[Range<usize>], threshold: usize) -> Vec<Vec<usize>> {
421    let mut buckets = Vec::new();
422    let mut current = Vec::new();
423    let mut current_bytes = 0usize;
424    for (ns_index, range) in ns_table.iter().enumerate() {
425        current.push(ns_index);
426        current_bytes += range.len();
427        if current_bytes >= threshold {
428            buckets.push(mem::take(&mut current));
429            current_bytes = 0;
430        }
431    }
432    if !current.is_empty() {
433        buckets.push(current);
434    }
435    buckets
436}

441    Net(#[from] NetworkError),

448}

452        match self {
453            Self::Net(e) => e.is_critical(),

455        }
456    }

498    /// Build a contiguous namespace table from per-namespace byte lengths.
499    /// Only the lengths matter to `bucketize`; the offsets are incidental.
500    fn ns_ranges(lens: &[u8]) -> Vec<Range<usize>> {
501        let mut start = 0;
502        lens.iter()
503            .map(|&len| {
504                let range = start..start + usize::from(len);
505                start += usize::from(len);
506                range
507            })
508            .collect()
509    }

511    /// Total payload bytes of the namespaces at `indices`.
512    fn bytes(indices: &[usize], ns_table: &[Range<usize>]) -> usize {
513        indices.iter().map(|&i| ns_table[i].len()).sum()
514    }

516    quickcheck! {
517        /// The buckets partition the namespace indices: every index appears
518        /// exactly once, contiguously, and in namespace-table order. This pins
519        /// completeness, disjointness, and ordering in a single property.
520        fn prop_buckets_partition(lens: Vec<u8>, threshold: u16) -> bool {
521            let ns_table = ns_ranges(&lens);
522            bucketize(&ns_table, threshold.into())
523                .into_iter()
524                .flatten()
525                .eq(0..ns_table.len())
526        }
527
528        /// No bucket is ever empty.
529        fn prop_buckets_non_empty(lens: Vec<u8>, threshold: u16) -> bool {
530            let ns_table = ns_ranges(&lens);
531            bucketize(&ns_table, threshold.into())
532                .iter()
533                .all(|bucket| !bucket.is_empty())
534        }
535
536        /// Every bucket except the last reaches the threshold: small namespaces
537        /// are coalesced until the bucket is large enough.
538        fn prop_non_last_buckets_meet_threshold(lens: Vec<u8>, threshold: u16) -> bool {
539            let ns_table = ns_ranges(&lens);
540            let buckets = bucketize(&ns_table, threshold.into());
541            let non_last = buckets.len().saturating_sub(1);
542            buckets
543                .iter()
544                .take(non_last)
545                .all(|bucket| bytes(bucket, &ns_table) >= usize::from(threshold))
546        }
547
548        /// Buckets are minimal: dropping a non-last bucket's final namespace
549        /// leaves it below the threshold, so the disperser seals as soon as it
550        /// reaches the cutoff rather than over-coalescing. (Vacuous at
551        /// `threshold == 0`, where every namespace is its own bucket.)
552        fn prop_non_last_buckets_are_minimal(lens: Vec<u8>, threshold: u16) -> TestResult {
553            if threshold == 0 {
554                return TestResult::discard();
555            }
556            let ns_table = ns_ranges(&lens);
557            let buckets = bucketize(&ns_table, threshold.into());
558            let non_last = buckets.len().saturating_sub(1);
559            TestResult::from_bool(buckets.iter().take(non_last).all(|bucket| {
560                bytes(&bucket[..bucket.len() - 1], &ns_table) < usize::from(threshold)
561            }))

635        let commitment = built.payload_commitment().unwrap();

639    /// Runs [`disperse_payload`] with a `send` that checks each message is the leader's signed
640    /// fragment for its recipient, records the fragment, and returns `reply()`.
641    fn disperse_recording<R>(
642        dispersal: Dispersal<TestTypes>,
643        commitment: VidCommitment2,
644        reply: R,
645    ) -> (
646        Result<(), VidDisperseError>,
647        Vec<AvidmGf2DisperseShareFragment<TestTypes>>,
648    )
649    where
650        R: Fn() -> Result<(), NetworkError> + Clone + Send,
651    {
652        let (leader, private_key) = BLSPubKey::generated_from_seed_indexed([1; 32], 0);
653        let sent = Arc::new(Mutex::new(Vec::new()));
654        let send = {
655            let (leader, sent) = (leader, Arc::clone(&sent));
656            move |recipient: &BLSPubKey, message: &Message<TestTypes, Validated>| {
657                let MessageType::Consensus(ConsensusMessage::VidShareFragment(fragment)) =
658                    &message.message_type
659                else {
660                    panic!("the disperser sends only VID share fragments");
661                };
662                assert_eq!(message.sender, leader);
663                assert!(leader.validate(&fragment.signature, commitment.as_ref()));
664                assert_eq!(&fragment.data.recipient_key, recipient);
665                sent.lock().unwrap().push(fragment.data.clone());
666                reply()
667            }
668        };
669        let result = disperse_payload(
670            dispersal,
671            ViewNumber::new(7),
672            EpochNumber::new(3),
673            commitment,
674            &leader,
675            &private_key,
676            send,
677        );
678        let sent = mem::take(&mut *sent.lock().unwrap());
679        (result, sent)
680    }

682    /// Each recipient's fragments, ordered by their first namespace.
683    fn by_recipient(
684        sent: Vec<AvidmGf2DisperseShareFragment<TestTypes>>,
685    ) -> BTreeMap<BLSPubKey, Vec<AvidmGf2DisperseShareFragment<TestTypes>>> {
686        let mut by_recipient: BTreeMap<_, Vec<_>> = BTreeMap::new();
687        for fragment in sent {
688            by_recipient
689                .entry(fragment.recipient_key)
690                .or_default()
691                .push(fragment);
692        }
693        for fragments in by_recipient.values_mut() {
694            fragments.sort_by_key(|fragment| fragment.namespaces[0].ns_index);
695        }
696        by_recipient
697    }

699    /// The namespace indices each fragment carries: one list per bucket.
700    fn buckets(fragments: &[AvidmGf2DisperseShareFragment<TestTypes>]) -> Vec<Vec<usize>> {
701        fragments
702            .iter()
703            .map(|fragment| {
704                fragment
705                    .namespaces
706                    .iter()
707                    .map(|piece| piece.ns_index)
708                    .collect()
709            })
710            .collect()
711    }

714    fn disperse_payload_sends_one_fragment_per_recipient_and_bucket() {
715        let (payload, metadata) = large_payload();
716        let params =
717            AvidmGf2DisperseParams::new(&stake_table(), payload.clone().into(), &metadata).unwrap();
718        let (built, commitment) = built_dispersal((payload, metadata));
719
720        let (result, sent) = disperse_recording(Dispersal::Built(built), commitment, || Ok(()));
721        result.unwrap();
722        let sent = by_recipient(sent);
723        let (result, encoded) =
724            disperse_recording(Dispersal::Encode(params), commitment, || Ok(()));
725        result.unwrap();
726        assert_eq!(
727            sent,
728            by_recipient(encoded),
729            "a built dispersal sends what encoding while sending does"
730        );
731
732        let recipients: BTreeSet<_> = stake_table()
733            .iter()
734            .map(|peer| peer.stake_table_entry.public_key())
735            .collect();
736        assert_eq!(sent.keys().cloned().collect::<BTreeSet<_>>(), recipients);
737        let first = buckets(sent.values().next().unwrap());
738        assert!(
739            first.concat().into_iter().eq(0..3),
740            "the buckets hold every namespace once"
741        );
742        for fragments in sent.values() {
743            assert_eq!(
744                buckets(fragments),
745                first,
746                "every recipient gets each bucket once"
747            );
748            assert!(
749                fragments
750                    .iter()
751                    .all(|fragment| fragment.num_namespaces == 3)
752            );
753        }
754    }
