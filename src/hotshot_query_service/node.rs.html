13//! A node's view of a HotShot chain
14//!
15//! The node API provides a subjective view of the HotShot blockchain, from the perspective of
16//! one particular node. It provides access to information that the
17//! [availability](crate::availability) API does not, because this information depends on the
18//! perspective of the node observing it, and may be subject to eventual consistency. For example,
19//! `/node/block-height` may return smaller counts than expected, if the node being queried is not
20//! fully synced with the entire history of the chain. However, the node will _eventually_ sync and
21//! return the expected counts.
22
23pub(crate) mod data_source;
24pub(crate) mod query_data;
25pub use data_source::*;
26pub use hotshot_query_service_types::node::*;
27
28#[derive(Debug)]
29pub struct Options {
30    /// The maximum number of headers which can be loaded in a single `header/window` query.
31    pub window_limit: usize,
32}
33
34impl Default for Options {
35    fn default() -> Self {
36        Self { window_limit: 500 }
37    }
38}

60    #[test_log::test(tokio::test(flavor = "multi_thread"))]
61    async fn test_api() {
62        let window_limit = 78;
63
64        let mut node = ChainNode::<MockDataSource>::new().await;
65        node.push_empty(4).await;
66
67        let ds = node.data_source();
68        let block_height = ds.block_height().await.unwrap();
69
70        // We test these counters with non-trivial values in `data_source.rs`, here we just want to
71        // make sure the queries are working, so a response of 0 is fine.
72        assert_eq!(ds.count_transactions().await.unwrap(), 0);
73        assert_eq!(ds.payload_size().await.unwrap(), 0);
74
75        // Get VID share for each block but the tip, which is left to be the `next` of the header
76        // window below.
77        tracing::info!(block_height, "checking VID shares");
78        let (_tip, blocks) = node.chain().blocks().split_last().unwrap();
79        let mut headers = vec![];
80        for block in blocks {
81            let header = block.leaf.header().clone();
82            tracing::info!(height = header.block_number, "checking share");
83
84            let share = ds
85                .vid_share(BlockId::<MockTypes>::Number(header.block_number as usize))
86                .await
87                .unwrap();
88            assert_eq!(share, block.vid_shares[0]);
89
90            // Query various other ways.
91            assert_eq!(
92                share,
93                ds.vid_share(BlockId::<MockTypes>::Hash(header.commit()))
94                    .await
95                    .unwrap()
96            );
97            assert_eq!(
98                share,
99                ds.vid_share(BlockId::<MockTypes>::PayloadHash(header.payload_commitment))
100                    .await
101                    .unwrap()
102            );
103            headers.push(header);
104        }
105
106        // Check time window queries. The various edge cases are thoroughly tested for each
107        // individual data source. In this test, we just smoketest parameter handling.
108        let first_header = &headers[0];
109        let last_header = &headers.last().unwrap();
110        let window: TimeWindowQueryData<Header<MockTypes>> = ds
111            .get_header_window(
112                WindowStart::Time(first_header.timestamp),
113                last_header.timestamp + 1,
114                window_limit,
115            )
116            .await
117            .unwrap();
118        assert!(window.window.contains(first_header));
119        assert!(window.window.contains(last_header));
120        assert!(window.next.is_some());
121
122        // Query for the same window other ways.
123        assert_eq!(
124            window,
125            ds.get_header_window(
126                WindowStart::<MockTypes>::Height(0),
127                last_header.timestamp + 1,
128                window_limit
129            )
130            .await
131            .unwrap()
132        );
133        assert_eq!(
134            window,
135            ds.get_header_window(
136                WindowStart::<MockTypes>::Hash(first_header.commit()),
137                last_header.timestamp + 1,
138                window_limit
139            )
140            .await
141            .unwrap()
142        );
143
144        // In this simple test, the node should be fully synchronized.
145        let sync_status = ds.sync_status().await.unwrap();
146        assert!(sync_status.is_fully_synced(), "{sync_status:#?}");
147    }

149    #[apply(sql_data_sources)]
150    async fn test_aggregate_ranges<D: TestableDataSource>(_d: PhantomData<D>) {
151        let mut node = ChainNode::<D>::new().await;
152        let ds = node.data_source();
153
154        // Decide two transactions, each in its own block after some empty ones.
155        let mut tx_heights = vec![];
156        let mut tx_sizes = vec![];
157        for i in [1, 2] {
158            node.push_empty(2).await;
159            let block = node.push([mock_transaction(vec![0; i])]).await;
160            tx_heights.push(block.height() as usize);
161            tx_sizes.push(block.block.payload().encode().len());
162        }
163        tracing::info!(?tx_heights, ?tx_sizes, "transactions sequenced");
164
165        // Wait for the aggregator to process the inserted blocks.
166        while let Err(err) = ds
167            .count_transactions_in_range(0..=tx_heights[1], None)
168            .await
169        {
170            match err {
171                QueryError::NotFound | QueryError::Missing => {
172                    tracing::info!(?tx_heights, "waiting for aggregator");
173                    sleep(Duration::from_secs(1)).await;
174                },
175                err => panic!("unexpected error: {err:#}"),
176            }
177        }
178
179        // Range including empty blocks (genesis block) only
180        assert_eq!(
181            0,
182            ds.count_transactions_in_range(0..=0, None).await.unwrap()
183        );
184        assert_eq!(0, ds.payload_size_in_range(0..=0, None).await.unwrap());
185
186        // First transaction only
187        assert_eq!(
188            1,
189            ds.count_transactions_in_range(0..=tx_heights[0], None)
190                .await
191                .unwrap()
192        );
193        assert_eq!(
194            tx_sizes[0],
195            ds.payload_size_in_range(0..=tx_heights[0], None)
196                .await
197                .unwrap()
198        );
199
200        // Last transaction only
201        assert_eq!(
202            1,
203            ds.count_transactions_in_range(tx_heights[0] + 1..=tx_heights[1], None)
204                .await
205                .unwrap()
206        );
207        assert_eq!(
208            tx_sizes[1],
209            ds.payload_size_in_range(tx_heights[0] + 1..=tx_heights[1], None)
210                .await
211                .unwrap()
212        );
213
214        // All transactions
215        assert_eq!(2, ds.count_transactions().await.unwrap());
216        assert_eq!(tx_sizes[0] + tx_sizes[1], ds.payload_size().await.unwrap());
217    }
