52    async fn block_height(&mut self) -> QueryResult<usize> {
53        match query_as::<(Option<i64>,)>("SELECT max(height) FROM header")
54            .fetch_one(self.as_mut())
55            .await?
56        {
57            (Some(height),) => {
58                // The height of the block is the number of blocks below it, so the total number of
59                // blocks is one more than the height of the highest block.
60                Ok(height as usize + 1)
61            },
62            (None,) => {
63                // If there are no blocks yet, the height is 0.
64                Ok(0)
65            },
66        }
67    }

69    async fn count_transactions_in_range(
70        &mut self,
71        range: impl RangeBounds<usize> + Send,
72        namespace: Option<NamespaceId<Types>>,
73    ) -> QueryResult<usize> {
74        let namespace: i64 = namespace.map(|ns| ns.into()).unwrap_or(-1);
75        let Some((from, to)) = aggregate_range_bounds::<Types>(self, range).await? else {
76            return Ok(0);
77        };
78        let (mut count, _) = cumulative_totals(self, to, namespace).await?;
79        if from > 0 {
80            let (prev_count, _) = cumulative_totals(self, from - 1, namespace).await?;
81            count = count.saturating_sub(prev_count);
82        }
83
84        Ok(count)
85    }

105    async fn vid_share<ID>(&mut self, id: ID) -> QueryResult<VidShare>
106    where
107        ID: Into<BlockId<Types>> + Send + Sync,
108    {
109        let mut query = QueryBuilder::default();
110        let where_clause = query.header_where_clause(id.into())?;
111        // ORDER BY h.height ASC ensures that if there are duplicate blocks (this can happen when
112        // selecting by payload ID, as payloads are not unique), we return the first one.
113        let sql = format!(
114            "SELECT vid_share FROM header AS h
115              WHERE {where_clause}
116              ORDER BY h.height
117              LIMIT 1"
118        );
119        let (share_data,) = query
120            .query_as::<(Option<Vec<u8>>,)>(&sql)
121            .fetch_one(self.as_mut())
122            .await?;
123        let share_data = share_data.context(MissingSnafu)?;
124        let share = bincode::deserialize(&share_data).decode_error("malformed VID share")?;
125        Ok(share)
126    }

128    async fn sync_status_for_range(
129        &mut self,
130        from: usize,
131        to: usize,
132    ) -> QueryResult<SyncStatusQueryData> {
133        // A block can be missing if its corresponding header is missing or if the block's pyaload
134        // information is missing.
135        let blocks = self
136            .sync_status_ranges(
137                "header AS h JOIN payload AS p ON (h.payload_hash, h.ns_table) = (p.hash, \
138                 p.ns_table)",
139                "height",
140                from,
141                to,
142            )
143            .await?;
144
145        let leaves = if blocks.is_fully_synced() {
146            // A common special case is that there are no missing blocks. In this case, we already
147            // know there are no missing leaves either, since a block can only be present if we
148            // already have the corresponding leaf. Just return the fully-synced status for leaves
149            // without doing another expensive query.
150            blocks.clone()
151        } else {
152            // A leaf can only be missing if there is no row for it in the database (all its columns
153            // are non-nullable). We use `height` as an indicator for `NULL` rows in an inner join,
154            // which allows an index-only scan.
155            self.sync_status_ranges("leaf2", "height", from, to).await?
156        };
157
158        // VID common works just like blocks.
159        let vid_common = self
160            .sync_status_ranges(
161                "header AS h JOIN vid_common AS v ON h.payload_hash = v.hash",
162                "height",
163                from,
164                to,
165            )
166            .await?;
167
168        Ok(SyncStatusQueryData {
169            leaves,
170            blocks,
171            vid_common,
172            pruned_height: None,
173        })
174    }

176    async fn get_header_window(
177        &mut self,
178        start: impl Into<WindowStart<Types>> + Send + Sync,
179        end: u64,
180        limit: usize,
181    ) -> QueryResult<TimeWindowQueryData<Header<Types>>> {
182        // Find the specific block that starts the requested window.
183        let first_block = match start.into() {
184            WindowStart::Time(t) => {
185                // If the request is not to start from a specific block, but from a timestamp, we
186                // use a different method to find the window, as detecting whether we have
187                // sufficient data to answer the query is not as simple as just trying `load_header`
188                // for a specific block ID.
189                return self.time_window::<Types>(t, end, limit).await;
190            },
191            WindowStart::Height(h) => h,
192            WindowStart::Hash(h) => self.load_header::<Types>(h).await?.block_number(),
193        };
194
195        // Find all blocks starting from `first_block` with timestamps less than `end`. Block
196        // timestamps are monotonically increasing, so this query is guaranteed to return a
197        // contiguous range of blocks ordered by increasing height.
198        let sql = format!(
199            "SELECT {HEADER_COLUMNS}
200               FROM header AS h
201              WHERE h.height >= $1 AND h.timestamp < $2
202              ORDER BY h.height
203              LIMIT $3"
204        );
205        let rows = query(&sql)
206            .bind(first_block as i64)
207            .bind(end as i64)
208            .bind(limit as i64)
209            .fetch(self.as_mut());
210        let window = rows
211            .map(|row| parse_header::<Types, _>(row?))
212            .try_collect::<Vec<_>>()
213            .await?;
214
215        // Find the block just before the window.
216        let prev = if first_block > 0 {
217            Some(self.load_header::<Types>(first_block as usize - 1).await?)
218        } else {
219            None
220        };
221
222        let next = if window.len() < limit {
223            // If we are not limited, complete the window by finding the block just after the
224            // window. We order by timestamp _then_ height, because the timestamp order allows the
225            // query planner to use the index on timestamp to also efficiently solve the WHERE
226            // clause, but this process may turn up multiple results, due to the 1-second resolution
227            // of block timestamps. The final sort by height guarantees us a unique, deterministic
228            // result (the first block with a given timestamp). This sort may not be able to use an
229            // index, but it shouldn't be too expensive, since there will never be more than a
230            // handful of blocks with the same timestamp.
231            let sql = format!(
232                "SELECT {HEADER_COLUMNS}
233               FROM header AS h
234              WHERE h.timestamp >= $1
235              ORDER BY h.timestamp, h.height
236              LIMIT 1"
237            );
238            query(&sql)
239                .bind(end as i64)
240                .fetch_optional(self.as_mut())
241                .await?
242                .map(parse_header::<Types, _>)
243                .transpose()?
244        } else {
245            // If we have been limited, return a `null` next block indicating an incomplete window.
246            // The client will have to query again with an adjusted starting point to get subsequent
247            // results.
248            tracing::debug!(limit, "cutting off header window request due to limit");
249            None
250        };
251
252        Ok(TimeWindowQueryData { window, prev, next })
253    }

277    /// Characterize consecutive ranges of objects in the given `height`-indexed table by status.
278    ///
279    /// This function will find all ranges in `[0, block_height)`. If `pruned_height` is specified,
280    /// an initial range will be created with bounds `[0, pruned_height]` and status
281    /// [`SyncStatus::Pruned`]. Then only the range `[pruned_height + 1, block_height)` will
282    /// actually be searched.
283    ///
284    /// The search process uses an indexed outer self-join on `table`, which requires traversing
285    /// the table twice. Thus, it can be fairly expensive on large tables, but it is still linear in
286    /// the size of the table.
287    ///
288    /// The value of `indicator_column` in the outer join results is used to check for missing
289    /// objects (indicated by a `NULL` value). If `indicator_column` is a `NOT NULL` column, such as
290    /// `height`, then this function will only consider objects missing if there is no corresponding
291    /// row in the database at all. However, `indicator_column` may also be a nullable column (such
292    /// as `payload.data`, in which case objects are treated as missing if there is no corresponding
293    /// row _or_ if there is a row but it has an explicit `NULL` value for `indicator_column`).
294    #[instrument(skip(self))]
295    async fn sync_status_ranges(
296        &mut self,
297        table: &str,
298        indicator_column: &str,
299        start: usize,
300        end: usize,
301    ) -> QueryResult<ResourceSyncStatus> {
302        let mut ranges = vec![];
303        tracing::debug!("searching for missing ranges");
304
305        // Find every height in the range `[start, end)` which is the first height in a sequence of
306        // present objects (i.e. the object just before it is missing).
307        //
308        // We do this by outer joining the table with itself, with height shifted by one, to get a
309        // combined table where each row contains a successor object and its immediate predecessor,
310        // if present. If the predecessor is missing, its height will be `NULL` (which is impossible
311        // otherwise, because `height` is a `NOT NULL` column).
312        //
313        // For each table in the self-join, we _first_ sub-select just the range of interest (i.e.
314        // [start, end) for the successor table and [start - 1, end - 1) for the predecessor table).
315        // It is more efficient to do this first to reduce the number of rows involved in the join,
316        // which is the expensive part of the operation. In fact, due to the nature of the outer
317        // join, it is impossible to do this filtering after the join for the predecessor table,
318        // since at that point the table will not necessarily be indexed and will contain some rows
319        // with `NULL` height.
320        let query = format!(
321            "WITH range AS (SELECT height, {indicator_column} AS indicator FROM {table}
322                WHERE height >= $1 AND height < $2)
323            SELECT successor.height FROM range AS predecessor
324            RIGHT JOIN range AS successor
325            ON successor.height = predecessor.height + 1
326            WHERE successor.indicator IS NOT NULL
327              AND predecessor.indicator IS NULL
328            ORDER BY successor.height"
329        );
330        let range_starts = query_as::<(i64,)>(&query)
331            .bind(start as i64)
332            .bind(end as i64)
333            .fetch_all(self.as_mut())
334            .await?;
335        tracing::debug!(
336            ?range_starts,
337            "found {} starting heights for present ranges",
338            range_starts.len()
339        );
340
341        let range_ends = if range_starts.len() <= 10 {
342            // In the common case, where we are mostly or entirely synced and only missing a few
343            // objects, chopping the space into only a small number of present ranges, it is more
344            // efficient to pick out reach range's end individually with a specific, efficient,
345            // height-indexed query, rather than execute another very expensive query which is the
346            // mirror of the `range_starts` query to load all the range ends in bulk.
347            let mut ends = vec![];
348            for (i, &(start,)) in range_starts.iter().enumerate() {
349                // We can easily find the end of the range from the start by finding the maximum
350                // height which is still present between the start and the next range's start.
351                let query = format!(
352                    "SELECT height FROM {table}
353                      WHERE height < $1 AND {indicator_column} IS NOT NULL
354                      ORDER BY height DESC
355                      LIMIT 1"
356                );
357                let upper_bound = if i + 1 < range_starts.len() {
358                    range_starts[i + 1].0
359                } else {
360                    end as i64
361                };
362                let (end,) = query_as::<(i64,)>(&query)
363                    .bind(upper_bound)
364                    // This query is guaranteed to return one result, since `start` satisfies the
365                    // requirements even if nothing else does.
366                    .fetch_one(self.as_mut())
367                    .await?;
368                tracing::debug!(start, end, "found end for present range");
369                ends.push((end,));
370            }
371            ends
372        } else {
373            // When the number of distinct ranges becomes large, making many small queries to fetch
374            // each specific range end becomes inefficient because it is dominated by overhead. In
375            // this case, we fall back to fetching the range ends using a single moderately
376            // expensive query, which is the mirror image of the query we used to fetch the range
377            // starts.
378            let query = format!(
379                "WITH range AS (SELECT height, {indicator_column} AS indicator FROM {table}
380                    WHERE height >= $1 AND height < $2)
381                SELECT predecessor.height FROM range AS predecessor
382                LEFT JOIN range AS successor
383                ON successor.height = predecessor.height + 1
384                WHERE predecessor.indicator IS NOT NULL
385                  AND successor.indicator IS NULL
386                ORDER BY predecessor.height"
387            );
388            let ends = query_as::<(i64,)>(&query)
389                .bind(start as i64)
390                .bind(end as i64)
391                .fetch_all(self.as_mut())
392                .await?;
393            tracing::debug!(
394                ?ends,
395                "found {} ending heights for present ranges",
396                ends.len()
397            );
398            ends
399        };
400
401        // Sanity check: every range has a start and an end.
402        if range_starts.len() != range_ends.len() {
403            return Err(QueryError::Error {
404                message: format!(
405                    "number of present range starts ({}) does not match number of present range \
406                     ends ({})",
407                    range_starts.len(),
408                    range_ends.len(),
409                ),
410            });
411        }
412
413        // Now we can simply zip `range_starts` and `range_ends` to find the full sequence of
414        // [`SyncStatus::Present`] ranges. We can then interpolate [`SyncStatus::Missing`] ranges
415        // between each present range.
416        let mut prev = start;
417        for ((start,), (end,)) in range_starts.into_iter().zip(range_ends) {
418            let start = start as usize;
419            let end = end as usize;
420
421            // Sanity check range bounds.
422            if start < prev {
423                return Err(QueryError::Error {
424                    message: format!(
425                        "found present ranges out of order: range start {start} is before \
426                         previous range end {prev}"
427                    ),
428                });
429            }
430            if end < start {
431                return Err(QueryError::Error {
432                    message: format!("malformed range: start={start}, end={end}"),
433                });
434            }
435
436            if start != prev {
437                // There is a range in between this one and the previous one, which must correspond
438                // to missing objects.
439                tracing::debug!(start = prev, end = start, "found missing range");
440                ranges.push(SyncStatusRange {
441                    start: prev,
442                    end: start,
443                    status: SyncStatus::Missing,
444                });
445            }
446
447            ranges.push(SyncStatusRange {
448                start,
449                end: end + 1, // convert inclusive range to exclusive
450                status: SyncStatus::Present,
451            });
452            prev = end + 1;
453        }
454
455        // There is possibly one more missing range, between the final present range and the overall
456        // block height.
457        if prev != end {
458            tracing::debug!(start = prev, end, "found missing range");
459            ranges.push(SyncStatusRange {
460                start: prev,
461                end,
462                status: SyncStatus::Missing,
463            });
464        }
465
466        let missing = ranges
467            .iter()
468            .filter_map(|range| {
469                if range.status == SyncStatus::Missing {
470                    Some(range.end - range.start)
471                } else {
472                    None
473                }
474            })
475            .sum();
476        tracing::debug!(
477            missing,
478            "found missing objects in {} total ranges",
479            ranges.len()
480        );
481
482        Ok(ResourceSyncStatus { missing, ranges })
483    }
484}

486impl<Types, Mode: TransactionMode> AggregatesStorage<Types> for Transaction<Mode>
487where
488    Types: NodeType,
489    Header<Types>: QueryableHeader<Types>,
490{
491    async fn aggregates_height(&mut self) -> anyhow::Result<usize> {
492        let (height,): (i64,) = query_as("SELECT coalesce(max(height) + 1, 0) FROM aggregate")
493            .fetch_one(self.as_mut())
494            .await?;
495        Ok(height as usize)
496    }
497
498    async fn load_prev_aggregate(&mut self) -> anyhow::Result<Option<Aggregate<Types>>> {
499        // Get the maximum height for which we have stored aggregated results
500        // then query all the namespace info for that height
501        let res: (Option<i64>,) =
502            query_as("SELECT max(height) FROM aggregate WHERE namespace = -1")
503                .fetch_one(self.as_mut())
504                .await?;
505
506        let (Some(max_height),) = res else {
507            return Ok(None);
508        };
509
510        let rows: Vec<(i64, i64, i64)> = query_as(
511            r#"
512        SELECT namespace, num_transactions, payload_size from aggregate WHERE height = $1
513        "#,
514        )
515        .bind(max_height)
516        .fetch_all(self.as_mut())
517        .await?;
518
519        let mut num_transactions = HashMap::default();
520        let mut payload_size = HashMap::default();
521
522        for (namespace_id, num_tx, payload_sz) in rows {
523            // Null namespace is represented as - 1 in database
524            // as it is part of primary key and primary key can not be NULL
525            // This namespace represents the cumulative sum of all the namespaces
526            let key = if namespace_id == -1 {
527                None
528            } else {
529                Some(namespace_id.into())
530            };
531            num_transactions.insert(key, num_tx as usize);
532            payload_size.insert(key, payload_sz as usize);
533        }
534
535        Ok(Some(Aggregate {
536            height: max_height,
537            num_transactions,
538            payload_size,
539        }))
540    }
541}

543impl<Types: NodeType> UpdateAggregatesStorage<Types> for Transaction<Write>
544where
545    Header<Types>: QueryableHeader<Types>,
546{
547    async fn update_aggregates(
548        &mut self,
549        prev: Aggregate<Types>,
550        blocks: &[PayloadMetadata<Types>],
551    ) -> anyhow::Result<Aggregate<Types>> {
552        let height = blocks[0].height();
553        let (prev_tx_count, prev_size) = (prev.num_transactions, prev.payload_size);
554
555        let mut rows = Vec::new();
556
557        // Cumulatively sum up new statistics for each block in this chunk.
558        let aggregates = blocks
559            .iter()
560            .scan(
561                (height, prev_tx_count, prev_size),
562                |(height, tx_count, size), block| {
563                    if *height != block.height {
564                        return Some(Err(anyhow!(
565                            "blocks in update_aggregates are not sequential; expected {}, got {}",
566                            *height,
567                            block.height()
568                        )));
569                    }
570                    *height += 1;
571
572                    //  Update total global stats
573                    // `None` represents stats across all namespaces.
574                    // It is represented as -1 in database
575
576                    *tx_count.entry(None).or_insert(0) += block.num_transactions as usize;
577                    *size.entry(None).or_insert(0) += block.size as usize;
578
579                    // Add row for global cumulative stats (namespace = -1)
580
581                    rows.push((
582                        block.height as i64,
583                        -1,
584                        tx_count[&None] as i64,
585                        size[&None] as i64,
586                    ));
587
588                    // Update per-namespace cumulative stats
589                    for (&ns_id, info) in &block.namespaces {
590                        let key = Some(ns_id);
591
592                        *tx_count.entry(key).or_insert(0) += info.num_transactions as usize;
593                        *size.entry(key).or_insert(0) += info.size as usize;
594                    }
595
596                    //  Insert cumulative stats for all known namespaces
597                    // Even if a namespace wasn't present in this block,
598                    // we still insert its latest cumulative stats at this height.
599                    for ns_id in tx_count.keys().filter_map(|k| k.as_ref()) {
600                        let key = Some(*ns_id);
601                        rows.push((
602                            block.height as i64,
603                            (*ns_id).into(),
604                            tx_count[&key] as i64,
605                            size[&key] as i64,
606                        ));
607                    }
608
609                    Some(Ok((block.height as i64, tx_count.clone(), size.clone())))
610                },
611            )
612            .collect::<anyhow::Result<Vec<_>>>()?;
613        let last_aggregate = aggregates.last().cloned();
614
615        let (height, num_transactions, payload_size) =
616            last_aggregate.ok_or_else(|| anyhow!("no row"))?;
617
618        self.upsert(
619            "aggregate",
620            ["height", "namespace", "num_transactions", "payload_size"],
621            ["height", "namespace"],
622            rows,
623        )
624        .await?;
625        Ok(Aggregate {
626            height,
627            num_transactions,
628            payload_size,
629        })
630    }
631}

633impl<Mode: TransactionMode> Transaction<Mode> {
634    async fn time_window<Types: NodeType>(
635        &mut self,
636        start: u64,
637        end: u64,
638        limit: usize,
639    ) -> QueryResult<TimeWindowQueryData<Header<Types>>> {
640        // Find all blocks whose timestamps fall within the window [start, end). Block timestamps
641        // are monotonically increasing, so this query is guaranteed to return a contiguous range of
642        // blocks ordered by increasing height.
643        //
644        // We order by timestamp _then_ height, because the timestamp order allows the query planner
645        // to use the index on timestamp to also efficiently solve the WHERE clause, but this
646        // process may turn up multiple results, due to the 1-second resolution of block timestamps.
647        // The final sort by height guarantees us a unique, deterministic result (the first block
648        // with a given timestamp). This sort may not be able to use an index, but it shouldn't be
649        // too expensive, since there will never be more than a handful of blocks with the same
650        // timestamp.
651        let sql = format!(
652            "SELECT {HEADER_COLUMNS}
653               FROM header AS h
654              WHERE h.timestamp >= $1 AND h.timestamp < $2
655              ORDER BY h.timestamp, h.height
656              LIMIT $3"
657        );
658        let rows = query(&sql)
659            .bind(start as i64)
660            .bind(end as i64)
661            .bind(limit as i64)
662            .fetch(self.as_mut());
663        let window: Vec<_> = rows
664            .map(|row| parse_header::<Types, _>(row?))
665            .try_collect()
666            .await?;
667
668        let next = if window.len() < limit {
669            // If we are not limited, complete the window by finding the block just after.
670            let sql = format!(
671                "SELECT {HEADER_COLUMNS}
672               FROM header AS h
673              WHERE h.timestamp >= $1
674              ORDER BY h.timestamp, h.height
675              LIMIT 1"
676            );
677            query(&sql)
678                .bind(end as i64)
679                .fetch_optional(self.as_mut())
680                .await?
681                .map(parse_header::<Types, _>)
682                .transpose()?
683        } else {
684            // If we have been limited, return a `null` next block indicating an incomplete window.
685            // The client will have to query again with an adjusted starting point to get subsequent
686            // results.
687            tracing::debug!(limit, "cutting off header window request due to limit");
688            None
689        };
690
691        // If the `next` block exists, _or_ if any block in the window exists, we know we have
692        // enough information to definitively say at least where the window starts (we may or may
693        // not have where it ends, depending on how many blocks have thus far been produced).
694        // However, if we have neither a block in the window nor a block after it, we cannot say
695        // whether the next block produced will have a timestamp before or after the window start.
696        // In this case, we don't know what the `prev` field of the response should be, so we return
697        // an error: the caller must try again after more blocks have been produced.
698        if window.is_empty() && next.is_none() {
699            return Err(QueryError::NotFound);
700        }
701
702        // Find the block just before the window.
703        let sql = format!(
704            "SELECT {HEADER_COLUMNS}
705               FROM header AS h
706              WHERE h.timestamp < $1
707              ORDER BY h.timestamp DESC, h.height DESC
708              LIMIT 1"
709        );
710        let prev = query(&sql)
711            .bind(start as i64)
712            .fetch_optional(self.as_mut())
713            .await?
714            .map(parse_header::<Types, _>)
715            .transpose()?;
716
717        Ok(TimeWindowQueryData { window, prev, next })
718    }
719}

721/// Get inclusive start and end bounds for a range to pull aggregate statistics.
722///
723/// Returns [`None`] if there are no blocks in the given range, in which case the result should be
724/// the default value of the aggregate statistic.
725async fn aggregate_range_bounds<Types>(
726    tx: &mut Transaction<impl TransactionMode>,
727    range: impl RangeBounds<usize>,
728) -> QueryResult<Option<(usize, usize)>>
729where
730    Types: NodeType,
731    Header<Types>: QueryableHeader<Types>,
732{
733    let from = match range.start_bound() {
734        Bound::Included(from) => *from,
735        Bound::Excluded(from) => *from + 1,
736        Bound::Unbounded => 0,
737    };
738    let to = match range.end_bound() {
739        Bound::Included(to) => *to,
740        Bound::Excluded(0) => return Ok(None),
741        Bound::Excluded(to) => *to - 1,
742        Bound::Unbounded => {
743            let height = AggregatesStorage::<Types>::aggregates_height(tx)
744                .await
745                .map_err(|err| QueryError::Error {
746                    message: format!("{err:#}"),
747                })?;
748            if height == 0 {
749                return Ok(None);
750            }
751            if height < from {
752                return Ok(None);
753            }
754            height - 1
755        },
756    };
757    Ok(Some((from, to)))
758}

760/// The cumulative transaction count and payload size of `namespace` (-1 for all namespaces) at
761/// `height`.
762///
763/// Fails with [`QueryError::NotFound`] if `height` is not aggregated yet. A namespace has no rows
764/// below the first block that contains it, so there it counts as zero.
765async fn cumulative_totals(
766    tx: &mut Transaction<impl TransactionMode>,
767    height: usize,
768    namespace: i64,
769) -> QueryResult<(usize, usize)> {
770    let (count, size) = query_as::<(i64, i64)>(
771        "SELECT coalesce(ns.num_transactions, 0), coalesce(ns.payload_size, 0)
772           FROM aggregate AS total
773           LEFT JOIN aggregate AS ns ON ns.height = total.height AND ns.namespace = $2
774          WHERE total.height = $1 AND total.namespace = -1",
775    )
776    .bind(height as i64)
777    .bind(namespace)
778    .fetch_one(tx.as_mut())
779    .await?;
780    Ok((count as usize, size as usize))
781}
