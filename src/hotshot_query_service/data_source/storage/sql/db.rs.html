35/// The database a copy of the SQL storage runs on.
36///
37/// The SQL storage is compiled once per backend, with [`Db`] naming that backend's
38/// [`sqlx::Database`], and each backend implements this trait for its own [`Db`]. Code branches on
39/// [`EMBEDDED`](Self::EMBEDDED) where both branches compile for either database, and calls a
40/// method of this trait where a branch would only compile for one.
41///
42/// ### Design Choice
43/// The reason for taking this approach over sqlx's Any database is that we can support SQL types
44/// which are implemented for the two backends we care about (Postgres and SQLite) but not for _any_
45/// SQL database, such as MySQL. Crucially, JSON types fall in this category.
46///
47/// The reason for taking this approach rather than writing all of our code to be generic over the
48/// Database implementation is that sqlx does not have the necessary trait bounds on all of the
49/// associated types (e.g. Database::Connection does not implement Executor for all possible
50/// databases, the Executor impl lives on each concrete connection type) and Rust does not provide
51/// a good way of encapsulating a collection of trait bounds on associated types. Thus, our function
52/// signatures become untenably messy with bounds like
53///
54/// ```rust
55/// # use sqlx::{Database, Encode, Executor, IntoArguments, Type};
56/// fn foo<DB: Database>()
57/// where
58///     for<'a> &'a mut DB::Connection: Executor<'a>,
59///     for<'q> DB::Arguments<'q>: IntoArguments<'q, DB>,
60///     for<'a> i64: Type<DB> + Encode<'a, DB>,
61/// {}
62/// ```
63///
64/// Code matches on [`KIND`](Self::KIND) where both arms compile for either database, and calls a
65/// method of this trait where an arm would only compile for one. Each backend implements it in its
66/// own module.
67pub trait Backend: Database {
68    type ConnectOptions: ConnectOptions<Connection = <Self as Database>::Connection>
69        + Clone
70        + Debug
71        + FromStr<Err = sqlx::Error>;
72
73    const KIND: DbKind;
74
75    /// The SQL function returning the greater of its arguments.
76    const MAX_FN: &'static str;
77
78    /// The connection options [`Config::default`](super::Config) starts from.
79    fn default_connect_options() -> Self::ConnectOptions;
80
81    /// The migrations that build the default schema, in no particular order.
82    fn migrations() -> Vec<Migration>;
83
84    /// Delete the database file, for a backend that keeps the database in one.
85    fn delete_file(opts: &Self::ConnectOptions) -> std::io::Result<()>;
86
87    /// Hooks every pooled connection needs on this backend, for a database using `schema`.
88    fn session_hooks(pool: PoolOptions<Self>, schema: String) -> PoolOptions<Self>;
89
90    /// Log the sessions and locks behind a serialization conflict in `op`, in the background.
91    fn log_conflict_sessions(pool: Pool<Self>, op: &'static str);
92}

94/// Merklized state queries with a portable form and a faster form that only Postgres can run.
95///
96/// The provided methods are the portable form, written against [`Db`] rather than `Self` so they
97/// type check in every copy of the SQL storage. Postgres overrides them.
98pub(crate) trait StateQueries: Backend {
99    fn hashes_by_id(
100        conn: &mut <Db as Database>::Connection,
101        ids: Vec<i64>,
102    ) -> impl Future<Output = QueryResult<HashMap<i64, Vec<u8>>>> + Send {
103        async move {
104            let (query, sql) = super::build_where_in("SELECT id, value FROM hash", "id", ids)?;
105            Ok(query
106                .query_as(&sql)
107                .fetch(conn)
108                .try_collect::<HashMap<i64, Vec<u8>>>()
109                .await?)
110        }
111    }
112
113    fn insert_hashes(
114        tx: &mut Transaction<Write>,
115        hashes: Vec<Vec<u8>>,
116    ) -> impl Future<Output = QueryResult<HashMap<Vec<u8>, i64>>> + Send {
117        async move {
118            let mut hash_ids = HashMap::with_capacity(hashes.len());
119            for hash_chunk in hashes.chunks(20) {
120                let (query, sql) = build_hash_batch_insert(hash_chunk)?;
121                let chunk_ids: HashMap<Vec<u8>, i64> = query
122                    .query_as(&sql)
123                    .fetch(tx.as_mut())
124                    .try_collect()
125                    .await?;
126                hash_ids.extend(chunk_ids);
127            }
128            Ok(hash_ids)
129        }
130    }
131
132    fn upsert_nodes<Types, State, const ARITY: usize>(
133        nodes: Vec<Node>,
134        tx: &mut Transaction<Write>,
135    ) -> impl Future<Output = anyhow::Result<()>> + Send
136    where
137        Types: NodeType,
138        State: MerklizedState<Types, ARITY>,
139    {
140        async move {
141            let name = State::state_type();
142            for node_chunk in nodes.chunks(20) {
143                let rows: Vec<_> = node_chunk
144                    .iter()
145                    .map(|n| {
146                        let children_bitvec: Option<String> = n
147                            .children_bitvec
148                            .clone()
149                            .map(|b| b.iter().map(|bit| if bit { '1' } else { '0' }).collect());
150
151                        (
152                            n.path.clone(),
153                            n.created,
154                            n.hash_id,
155                            n.children.clone(),
156                            children_bitvec,
157                            n.idx.clone(),
158                            n.entry.clone(),
159                        )
160                    })
161                    .collect();
162
163                tx.upsert(
164                    name,
165                    [
166                        "path",
167                        "created",
168                        "hash_id",
169                        "children",
170                        "children_bitvec",
171                        "idx",
172                        "entry",
173                    ],
174                    ["path", "created"],
175                    rows,
176                )
177                .await?;
178            }
179            Ok(())
180        }
181    }
182
183    fn node_from_row(row: <Db as Database>::Row) -> Node {
184        let bit_string: Option<String> = row.get_unchecked("children_bitvec");
185        let children_bitvec = bit_string.map(|b| b.chars().map(|c| c == '1').collect());
186
187        Node {
188            path: row.get_unchecked("path"),
189            created: row.get_unchecked("created"),
190            hash_id: row.get_unchecked("hash_id"),
191            children: row.get_unchecked("children"),
192            children_bitvec,
193            idx: row.get_unchecked("idx"),
194            entry: row.get_unchecked("entry"),
195        }
196    }
197
198    /// Build a query selecting, for each of `paths`, the newest node at that path created at or
199    /// before `created`, ordered from the leaf to the root.
200    fn get_path_query<'q, Types, State, const ARITY: usize>(
201        paths: Vec<serde_json::Value>,
202        created: i64,
203    ) -> QueryResult<(QueryBuilder<'q>, String)>
204    where
205        Types: NodeType,
206        State: MerklizedState<Types, ARITY>,
207    {
208        let table = State::state_type();
209        let mut query = QueryBuilder::default();
210        let created_param = query.bind(created)?;
211        // SQLite has no native array binding, so we keep the UNION form. Each subquery is a point
212        // lookup by the composite (path, created) primary key.
213        let mut sub_queries = Vec::with_capacity(paths.len());
214        for path in paths {
215            let node_path = query.bind(path)?;
216            sub_queries.push(format!(
217                "SELECT * FROM (SELECT * FROM {table} WHERE path = {node_path} AND created <= \
218                 {created_param} ORDER BY path DESC, created DESC LIMIT 1) AS latest_node",
219            ));
220        }
221        let sql = format!(
222            "SELECT * FROM ({}) as t ORDER BY length(t.path) DESC",
223            sub_queries.join(" UNION ")
224        );
225        Ok((query, sql))
226    }
227}
